185void denoise_luma_plane_inplace(cv::Mat& y,
186 const config::LumaDenoiseConfig::WaveletConfig& cfg,
187 const cv::Mat& valid, const cv::Mat& protect) {
188 if (!cfg.enabled) return;
189 cv::Mat approximation = y.clone();
190 std::vector<cv::Mat> details;
191 const int levels = std::max(1, cfg.levels);
192 for (int lvl = 0; lvl < levels; ++lvl) {
193 const double sigma = std::pow(2.0, static_cast<double>(lvl)) * 0.75;
194 cv::Mat low;
195 low = masked_gaussian_blur(approximation, valid, sigma);
196 details.push_back(approximation - low);
197 approximation = low;
198 }
199 cv::Mat noise_mask = valid.clone();
200 if (!protect.empty()) noise_mask.setTo(0, protect > 0.1f);
201 cv::Mat reconstructed = approximation;
202 for (cv::Mat& detail : details) {
203 const float sigma_n = robust_sigma_mad_from_mat(detail, noise_mask);
204 const float tau = cfg.threshold_scale * sigma_n;
205 cv::Mat ratio;
206 cv::divide(tau, cv::abs(detail) + 1.0e-12f, ratio);
207 cv::pow(ratio, cfg.soft_k, ratio);
208 cv::Mat gain = 1.0f - ratio;
209 cv::max(gain, 0.0f, gain);
210 if (cfg.boost > 0.0f) {
211 // Only coefficients clearly above the noise floor (>= 2*tau) get any
212 // extra amplification; the ramp is 0 at 2*tau, reaching cfg.boost as
213 // |detail| grows well past it -- coefficients near tau stay at the
214 // plain denoised gain above, never boosted.
215 cv::Mat boost_ratio;
216 cv::divide(2.0f * tau, cv::abs(detail) + 1.0e-12f, boost_ratio);
217 cv::Mat extra = 1.0f - boost_ratio;
218 cv::max(extra, 0.0f, extra);
219 gain += extra * cfg.boost;
220 }
221 reconstructed += detail.mul(gain);
222 }
223 y = reconstructed;
224}

250LumaDenoiseStats luma_denoise_rgb_inplace(
251 Matrix2Df& r, Matrix2Df& g, Matrix2Df& b,
252 const config::LumaDenoiseConfig& cfg,
253 Matrix2Df* protection_mask_out,
254 const std::vector<std::uint8_t>* valid_mask,
255 LumaDenoiseDiagnostics* diagnostics) {
256 LumaDenoiseStats stats;
257 if (!cfg.enabled) return stats;
258 if (r.size() <= 0 || g.size() <= 0 || b.size() <= 0) return stats;
259 if (r.rows() != g.rows() || r.cols() != g.cols() ||
260 r.rows() != b.rows() || r.cols() != b.cols()) {
261 return stats;
262 }
263
264 cv::Mat R(r.rows(), r.cols(), CV_32F, r.data());
265 cv::Mat G(g.rows(), g.cols(), CV_32F, g.data());
266 cv::Mat B(b.rows(), b.cols(), CV_32F, b.data());
267 const cv::Mat valid = build_valid_mask(R, G, B, valid_mask);
268 const auto valid_pixels = static_cast<std::uint64_t>(cv::countNonZero(valid));
269 if (valid_pixels == 0) return stats;
270
271 cv::Mat Y = 0.25f * R + 0.5f * G + 0.25f * B;
272 stats.applied = true;
273 stats.valid_pixels = valid_pixels;
274 stats.input_luma_sigma = high_frequency_sigma(Y, valid);
275
276 cv::Mat protect = build_protection_mask(Y, valid, cfg, &stats, diagnostics);
277 if (protection_mask_out != nullptr) {
278 protection_mask_out->resize(r.rows(), r.cols());
279 cv::Mat out_view(r.rows(), r.cols(), CV_32F, protection_mask_out->data());
280 protect.copyTo(out_view);
281 }
282
283 cv::Mat Y_denoised = Y.clone();
284 denoise_luma_plane_inplace(Y_denoised, cfg.wavelet, valid, protect);
285 denoise_luma_bilateral_inplace(Y_denoised, cfg.bilateral, valid, protect);
286
287 cv::Mat amount_map(Y.size(), CV_32F, cv::Scalar(cfg.blend_amount));
288 amount_map = amount_map.mul(1.0f - cfg.luma_guard_strength * protect);
289 cv::min(amount_map, cfg.blend_amount, amount_map);
290 cv::max(amount_map, 0.0, amount_map);
291 stats.mean_denoise_fraction = cv::mean(amount_map, valid)[0];
292 if (diagnostics) copy_to_matrix(amount_map, diagnostics->effective_amount);
293
294 cv::Mat one_minus = 1.0f - amount_map;
295 cv::Mat Y_mix = Y.mul(one_minus) + Y_denoised.mul(amount_map);
296
297 // Preserve each pixel's original per-channel DIFFERENCE from luma (the
298 // chroma, R-Y/G-Y/B-Y) rather than its RATIO to luma. A ratio divides two
299 // independently-noisy, correlated quantities (R and Y share the same
300 // per-pixel noise realisation in each channel), which blows up exactly
301 // where noise is proportionally largest -- faint background, and the
302 // partially-protected PSF wings around a star where the blend fades in --
303 // producing visible dark-pixel speckle and chroma fringing instead of
304 // removing them. Adding the same smooth delta (Y_mix - Y) to every
305 // channel is mathematically exact (0.25+0.5+0.25 == 1, so the new
306 // weighted luma of R+delta/G+delta/B+delta is exactly Y_mix) and leaves
307 // every additive channel DIFFERENCE unchanged. This deliberately does not
308 // claim invariant RGB ratios or perceptual saturation: a large common offset
309 // changes both, so the blend and protection masks remain part of the
310 // scientific contract.
311 cv::Mat delta = Y_mix - Y;
312 cv::Mat R_new = R + delta;
313 cv::Mat G_new = G + delta;
314 cv::Mat B_new = B + delta;
315 R.copyTo(R_new, valid == 0);
316 G.copyTo(G_new, valid == 0);
317 B.copyTo(B_new, valid == 0);
318
319 R_new.copyTo(R);
320 G_new.copyTo(G);
321 B_new.copyTo(B);
322 return stats;
323}
