114inline int omp_effective_threads(int max_configured_workers, int work_items = 0) {
115#if defined(_OPENMP)
116 if (omp_in_parallel()) return 1;
117 int threads = std::max(1, max_configured_workers);
118 const int hw = static_cast<int>(std::thread::hardware_concurrency());
119 if (hw > 0) threads = std::min(threads, hw);
120 if (work_items > 0) threads = std::min(threads, work_items);
121 return std::max(1, threads);
122#else
123 (void)max_configured_workers;
124 (void)work_items;
125 return 1;
126#endif
127}
