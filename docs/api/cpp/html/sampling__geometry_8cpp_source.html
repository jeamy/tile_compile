34double percentile_double(std::vector<double> values, double pct) {
35 if (values.empty())
36 return 0.0;
37 std::sort(values.begin(), values.end());
38 const double idx = (pct / 100.0) * static_cast<double>(values.size() - 1);
39 const size_t lo = static_cast<size_t>(std::floor(idx));
40 const size_t hi = static_cast<size_t>(std::ceil(idx));
41 if (lo == hi)
42 return values[lo];
43 const double frac = idx - static_cast<double>(lo);
44 return values[lo] * (1.0 - frac) + values[hi] * frac;
45}

47// Exact float order statistics with bounded RAM. Positive finite float bit
48// patterns preserve numeric order. Four sequential radix passes select a rank;
49// no full-canvas n_eff vector or sort copy is resident.
50class DiskQuantile {
51 core::AtomicOutput spool{std::filesystem::temp_directory_path() /
52 "drizzle-quantile"};
53 struct CloseFile {
54 void operator()(FILE *f) const { std::fclose(f); }
55 };
56 std::unique_ptr<FILE, CloseFile> file{
57 std::fopen(spool.path().string().c_str(), "w+b")};
58 size_t count = 0;
59 float select(size_t rank) {
60 uint32_t prefix = 0, mask = 0;
61 std::array<uint32_t, 4096> buffer{};
62 for (int shift = 24; shift >= 0; shift -= 8) {
63 if (std::fflush(file.get()) || std::fseek(file.get(), 0, SEEK_SET))
64 throw std::runtime_error("COVERAGE_TEMP_IO");
65 std::array<size_t, 256> bins{};
66 size_t n;
67 while ((n = std::fread(buffer.data(), sizeof(uint32_t), buffer.size(),
68 file.get())))
69 for (size_t i = 0; i < n; ++i)
70 if ((buffer[i] & mask) == prefix)
71 ++bins[(buffer[i] >> shift) & 255];
72 if (std::ferror(file.get()))
73 throw std::runtime_error("COVERAGE_TEMP_IO");
74 int bin = 0;
75 while (bin < 255 && rank >= bins[bin]) {
76 rank -= bins[bin];
77 ++bin;
78 }
79 prefix |= static_cast<uint32_t>(bin) << shift;
80 mask |= 255u << shift;
81 }
82 return std::bit_cast<float>(prefix);
83 }
84
85public:
86 DiskQuantile() {
87 if (!file)
88 throw std::runtime_error("COVERAGE_TEMP_IO: cannot create spool");
89 }
90 void add(float v) {
91 if (!std::isfinite(v) || v < 0 ||
92 std::fwrite(&v, sizeof(v), 1, file.get()) != 1)
93 throw std::runtime_error(
94 "COVERAGE_TEMP_IO: invalid value or write failure");
95 ++count;
96 }
97 // Fold every value previously add()-ed to `src` into this quantile. The
98 // radix-select rank (select()) is a function of the multiset only, so the
99 // merge order across stripes does not change p10() bit-for-bit --- this is
100 // what lets O1 give each stripe worker its own spool.
101 void absorb(DiskQuantile &src) {
102 if (std::fflush(src.file.get()) ||
103 std::fseek(src.file.get(), 0, SEEK_SET))
104 throw std::runtime_error("COVERAGE_TEMP_IO");
105 std::array<float, 4096> buffer{};
106 size_t n;
107 while ((n = std::fread(buffer.data(), sizeof(float), buffer.size(),
108 src.file.get())))
109 for (size_t i = 0; i < n; ++i)
110 add(buffer[i]);
111 if (std::ferror(src.file.get()))
112 throw std::runtime_error("COVERAGE_TEMP_IO");
113 }
114 double p10() {
115 if (!count)
116 return 0;
117 const double rank = 0.1 * static_cast<double>(count - 1);
118 const size_t lo = static_cast<size_t>(std::floor(rank)),
119 hi = static_cast<size_t>(std::ceil(rank));
120 const double a = select(lo), b = lo == hi ? a : select(hi);
121 return a + (b - a) * (rank - lo);
122 }
123};

125// Connected components on two scanlines. Components touching outside the
126// independent reference footprint are boundary loss, not interior holes.
127class StripeHoles {
128 struct Node {
129 int parent;
130 long long area;
131 bool exterior;
132 };
133 int width;
134 std::vector<int> previous;
135 std::vector<uint8_t> previous_ref;
136 std::vector<Node> components;
137 long long largest = 0;
138
139public:
140 explicit StripeHoles(int w) : width(w), previous(w, -1), previous_ref(w, 0) {}
141 void row(const uint8_t *reference, const uint8_t *support) {
142 std::vector<Node> nodes = components;
143 const int old = static_cast<int>(nodes.size());
144 nodes.reserve(old + width);
145 std::vector<int> current(width, -1);
146 auto root = [&](int n) {
147 while (nodes[n].parent != n) {
148 nodes[n].parent = nodes[nodes[n].parent].parent;
149 n = nodes[n].parent;
150 }
151 return n;
152 };
153 auto join = [&](int a, int b) {
154 a = root(a);
155 b = root(b);
156 if (a != b) {
157 nodes[b].parent = a;
158 nodes[a].area += nodes[b].area;
159 nodes[a].exterior = nodes[a].exterior || nodes[b].exterior;
160 }
161 };
162 for (int x = 0; x < width; ++x) {
163 if (previous[x] >= 0 && !reference[x])
164 nodes[root(previous[x])].exterior = true;
165 if (!reference[x] || support[x])
166 continue;
167 int id = static_cast<int>(nodes.size());
168 nodes.push_back({id, 1,
169 x == 0 || x == width - 1 || !previous_ref[x] ||
170 (x > 0 && !reference[x - 1]) ||
171 (x + 1 < width && !reference[x + 1])});
172 current[x] = id;
173 if (x > 0 && current[x - 1] >= 0)
174 join(id, current[x - 1]);
175 if (previous[x] >= 0)
176 join(id, previous[x]);
177 }
178 std::vector<int> remap(nodes.size(), -1);
179 std::vector<Node> next;
180 next.reserve(width);
181 for (int x = 0; x < width; ++x)
182 if (current[x] >= 0) {
183 int r = root(current[x]);
184 if (remap[r] < 0) {
185 int id = static_cast<int>(next.size());
186 remap[r] = id;
187 next.push_back({id, nodes[r].area, nodes[r].exterior});
188 }
189 current[x] = remap[r];
190 }
191 for (int i = 0; i < static_cast<int>(nodes.size()); ++i)
192 if (root(i) == i && remap[i] < 0 && !nodes[i].exterior)
193 largest = std::max(largest, nodes[i].area);
194 previous = std::move(current);
195 components = std::move(next);
196 std::copy(reference, reference + width, previous_ref.begin());
197 }
198 long long result() const {
199 return largest;
200 } // active last-row components touch the exterior
201};

203// T6b: affine GPU gather for the coverage CFA pass. Returns true when the
204// per-cell B plane (channel-major, `channels * canvas_w * rows` doubles) was
205// produced on the device; false if the caller should fall back to the CPU
206// `rasterize_drizzle_stripe`. out_b[static_cast<size_t>(c) * n + i] holds the
207// same value the CPU path accumulates into B[c][i] --- the kernel scans each
208// cell's source neighbourhood in canonical (sy, sx) order, so the per-cell
209// sums are bit-identical. It moves only the B plane
210// (~canvas_w*rows*channels*8B) and uploads nothing (coverage is
211// geometry-only).
212//
213// Thread safety: a static mutex serializes the device call across OpenMP
214// workers; the caller's fold over out_b runs outside the lock (per-band
215// buffers). TC_COVERAGE_DISABLE_CUDA forces the CPU path (parity tests).
216bool gather_coverage_stripe_cuda_affine(
217 const RegistrationSamplingPlan &plan, const FrameSamplingTransform &f,
218 int internal_scale, float pixfrac, int y_begin, int rows, int canvas_w,
219 double *out_b, int channels) {
220 if (f.has_smooth_local_model) return false; // local-warp: CPU only
221 if (std::getenv("TC_COVERAGE_DISABLE_CUDA") != nullptr) return false;
222 const auto dev_mem = reconstruction::forward_drizzle_cuda_device_memory();
223 if (dev_mem.free_bytes == 0) return false; // no CUDA device
224
225 const auto &s2c = f.source_to_canvas;
226 WarpMatrix inv;
227 if (!invert_affine_2x3(s2c, 1e-12f, 1e12f, inv)) return false;
228
229 const double affine6[6] = {
230 static_cast<double>(s2c(0, 0)), static_cast<double>(s2c(0, 1)),
231 static_cast<double>(s2c(0, 2)), static_cast<double>(s2c(1, 0)),
232 static_cast<double>(s2c(1, 1)), static_cast<double>(s2c(1, 2))};
233 const double inv6[6] = {
234 static_cast<double>(inv(0, 0)), static_cast<double>(inv(0, 1)),
235 static_cast<double>(inv(0, 2)), static_cast<double>(inv(1, 0)),
236 static_cast<double>(inv(1, 1)), static_cast<double>(inv(1, 2))};
237 const bool mono = plan.color_mode == ColorMode::MONO;
238
239 static std::mutex cuda_mutex;
240 std::lock_guard<std::mutex> lk(cuda_mutex);
241 return reconstruction::forward_drizzle_cuda_affine_coverage_gather(
242 affine6, inv6, internal_scale, static_cast<double>(pixfrac) / 2.0,
243 /*target_x_begin=*/0, y_begin, canvas_w, rows, plan.source_width,
244 plan.source_height, static_cast<int>(plan.bayer_pattern),
245 plan.cfa_origin_x, plan.cfa_origin_y, mono, out_b);
246}

248// Conservative internal-canvas bounding box of a frame's mapped source
249// rectangle (the union P of all unshrunk pixel squares; every pixfrac droplet
250// stays inside P). Padded 2 cells so every contributing cell index is inside.
251// Local-warp and non-affine frames get the full canvas: never skipped.
252struct FrameCanvasBox {
253 int x0 = 0, y0 = 0, x1 = 0, y1 = 0; // half-open
254};
255FrameCanvasBox frame_canvas_box(const RegistrationSamplingPlan &plan,
256 const FrameSamplingTransform &f,
257 int internal_scale) {
258 const int W = plan.canvas_width_native * internal_scale;
259 const int H = plan.canvas_height_native * internal_scale;
260 const FrameCanvasBox full{0, 0, W, H};
261 if (f.has_smooth_local_model || !f.source_to_canvas_affine_valid)
262 return full;
263 const auto &a = f.source_to_canvas;
264 double xlo = std::numeric_limits<double>::infinity(), xhi = -xlo,
265 ylo = xlo, yhi = -xlo;
266 for (double sx : {0.0, static_cast<double>(plan.source_width)})
267 for (double sy : {0.0, static_cast<double>(plan.source_height)}) {
268 const double qx =
269 static_cast<double>(a(0, 0)) * sx + static_cast<double>(a(0, 1)) * sy +
270 static_cast<double>(a(0, 2));
271 const double qy =
272 static_cast<double>(a(1, 0)) * sx + static_cast<double>(a(1, 1)) * sy +
273 static_cast<double>(a(1, 2));
274 if (!std::isfinite(qx) || !std::isfinite(qy))
275 return full;
276 xlo = std::min(xlo, qx * internal_scale);
277 xhi = std::max(xhi, qx * internal_scale);
278 ylo = std::min(ylo, qy * internal_scale);
279 yhi = std::max(yhi, qy * internal_scale);
280 }
281 return {std::clamp(static_cast<int>(std::floor(xlo)) - 2, 0, W),
282 std::clamp(static_cast<int>(std::floor(ylo)) - 2, 0, H),
283 std::clamp(static_cast<int>(std::ceil(xhi)) + 2, 0, W),
284 std::clamp(static_cast<int>(std::ceil(yhi)) + 2, 0, H)};
285}

289// Interior-hole detection on `support_mask` (W x H, 4-connectivity):
290// 1. flood-fill from every unsupported border pixel -> "exterior"
291// 2. any unsupported pixel not reached is an interior-hole pixel
292// 3. connected-component-label interior-hole pixels, return the largest
293// component's area (0 if there are none)
294int largest_interior_hole_area(const std::vector<uint8_t> &support_mask, int W,
295 int H) {
296 if (W <= 0 || H <= 0)
297 return 0;
298 if (support_mask.size() != static_cast<size_t>(W) * H)
299 throw std::invalid_argument("COVERAGE_MASK_SHAPE_MISMATCH");
300 StripeHoles holes(W);
301 std::vector<uint8_t> reference(W, 1);
302 for (int y = 0; y < H; ++y)
303 holes.row(reference.data(),
304 support_mask.data() + static_cast<size_t>(y) * W);
305 return static_cast<int>(
306 std::min<long long>(holes.result(), std::numeric_limits<int>::max()));
307}

309// Plan section 9.3/8.x circular dither-spread diagnostic. Site coordinates
310// are native-canvas (canvas_to_source is defined directly on native-canvas
311// coordinates, plan section 7 --- no internal_scale involved, so this needs
312// no inversion and no internal-canvas rescale). Diagnostic only, never gates.
313DitherSpreadCircularDiagnostic compute_dither_spread_circular_diagnostic(
314 const RegistrationSamplingPlan &plan) {
315 DitherSpreadCircularDiagnostic result;
316 const double W = static_cast<double>(plan.canvas_width_native);
317 const double H = static_cast<double>(plan.canvas_height_native);
318 if (W <= 0.0 || H <= 0.0)
319 return result;
320
321 // center + 4 corners
322 const double site_x[5] = {W / 2.0, 0.0, W - 1.0, 0.0, W - 1.0};
323 const double site_y[5] = {H / 2.0, 0.0, 0.0, H - 1.0, H - 1.0};
324
325 std::vector<double> sigma_x_per_site, sigma_y_per_site;
326 sigma_x_per_site.reserve(5);
327 sigma_y_per_site.reserve(5);
328
329 for (int s = 0; s < 5; ++s) {
330 double sum_cos_x = 0.0, sum_sin_x = 0.0, sum_cos_y = 0.0, sum_sin_y = 0.0;
331 int n = 0;
332 for (const FrameSamplingTransform &f : plan.frames) {
333 if (!f.valid)
334 continue;
335 const WarpMatrix &c2s = f.canvas_to_source;
336 const double sx =
337 c2s(0, 0) * site_x[s] + c2s(0, 1) * site_y[s] + c2s(0, 2);
338 const double sy =
339 c2s(1, 0) * site_x[s] + c2s(1, 1) * site_y[s] + c2s(1, 2);
340 if (!std::isfinite(sx) || !std::isfinite(sy))
341 continue;
342 double mx = std::fmod(sx, 2.0);
343 if (mx < 0.0)
344 mx += 2.0;
345 double my = std::fmod(sy, 2.0);
346 if (my < 0.0)
347 my += 2.0;
348 const double theta_x = M_PI * mx;
349 const double theta_y = M_PI * my;
350 sum_cos_x += std::cos(theta_x);
351 sum_sin_x += std::sin(theta_x);
352 sum_cos_y += std::cos(theta_y);
353 sum_sin_y += std::sin(theta_y);
354 ++n;
355 }
356 if (n == 0) {
357 sigma_x_per_site.push_back(0.0);
358 sigma_y_per_site.push_back(0.0);
359 continue;
360 }
361 double Rx = std::sqrt(sum_cos_x * sum_cos_x + sum_sin_x * sum_sin_x) / n;
362 double Ry = std::sqrt(sum_cos_y * sum_cos_y + sum_sin_y * sum_sin_y) / n;
363 // R in [0,1] up to floating-point slop; clamp so ln() stays finite.
364 Rx = std::min(std::max(Rx, 1e-12), 1.0 - 1e-12);
365 Ry = std::min(std::max(Ry, 1e-12), 1.0 - 1e-12);
366 sigma_x_per_site.push_back(std::sqrt(-2.0 * std::log(Rx)) / M_PI);
367 sigma_y_per_site.push_back(std::sqrt(-2.0 * std::log(Ry)) / M_PI);
368 }
369
370 result.x_p10 = percentile_double(sigma_x_per_site, 10.0);
371 result.y_p10 = percentile_double(sigma_y_per_site, 10.0);
372 return result;
373}

375GeometricCoverageResult compute_geometric_coverage(
376 const RegistrationSamplingPlan &plan, int internal_scale, float pixfrac,
377 const config::ReconstructionCoverageGateConfig &gate_cfg, float fraction,
378 int num_workers, const config::ReconstructionDrizzleConfig &resources,

380 using namespace reconstruction;
381 // O1 (plan §30.72): stripes are independent --- disjoint canvas i-ranges,
382 // per-stripe accumulators, per-i frame-ordered w/w2 sums --- so a stripe-
383 // parallel run is bit-identical to the serial one. num_workers <= 1 keeps the
384 // exact serial path. Worker count NEVER changes memory.rows (see below): the
385 // chunk height, and therefore resolved_chunk_rows + every geometry hash, stay
386 // a function of the W=1 budget only.
387 const int req_workers = std::max(1, num_workers);
388 // §30.75: the dense-footprint pass classifies internal cells against the
389 // frame's mapped source rectangle instead of rasterizing every source pixel,
390 // and is byte-identical to the per-pixel rasterize by construction. The env
391 // override forces the exact reference path for the parity test.
392 const bool footprint_exact =
393 std::getenv("TC_COVERAGE_FOOTPRINT_EXACT") != nullptr;
394 if (!std::isfinite(fraction) || fraction <= 0 || fraction > 1)
395 throw std::invalid_argument("COVERAGE_INVALID_COMMON_FRACTION");

398 cfg.pixfrac = pixfrac;

415 retained + row_scratch, false,
416 [](const std::string &msg) {
417 std::cout << "[SAMPLING_GEOMETRY][warn] " << msg << std::endl;

419 const auto space =
420 std::filesystem::space(std::filesystem::temp_directory_path());
421 const size_t disk_needed =
422 pixels * channels * sizeof(float) + 64 * 1024 * 1024;
423 if (space.available < disk_needed)
424 throw std::runtime_error(
425 "COVERAGE_TEMP_SPACE: insufficient quantile spool space");

429 result.local_samples_discarded =

431 result.excluded_frames =

433 result.internal_width = memory.width;

436 result.reconstruction_support_mask.assign(pixels, 0);

440 nullptr, nullptr}

442 &result.support_count_g,
443 &result.support_count_b};
444 if (retain_channel_counts)
445 for (int c = 0; c < channels; ++c)
446 counts[c]->assign(pixels, 0);
447 const int N = prepared.frames.size();
448 const int required = std::max(
449 1, static_cast<int>(std::ceil(static_cast<double>(fraction) * N)));
450 {
451 int local_n = 0;
452 for (const auto *pf : prepared.frames)
453 if (pf->has_smooth_local_model)
454 ++local_n;
455 reconstruction::geomstats::stamp_context(
456 plan.source_width, plan.source_height, plan.canvas_width_native,
457 plan.canvas_height_native, internal_scale, memory.rows, memory.height,
458 N, local_n);

461 std::array<std::unique_ptr<StripeHoles>, 3> holes;
462 std::array<size_t, 3> supported{};

464 quantiles[c] = std::make_unique<DiskQuantile>();

472 // The hole detector (StripeHoles) is a sequential connected-components
473 // accumulator over the full canvas height, so it cannot run inside the
474 // parallel loop. Persist the per-channel support planes canvas-wide (disjoint
475 // band writes) and run one serial hole pass afterwards --- bit-identical row
476 // order, bit-identical inputs.
477 std::array<std::vector<uint8_t>, 3> support_full;
478 for (int c = 0; c < channels; ++c)
479 support_full[c].assign(pixels, 0);

481 // Parallel work partition. `rasterize_drizzle_stripe` and the per-cell
482 // reduction are correct for ANY row window, and the compute-invariance tests
483 // prove the output does not depend on the row-band height --- so for
484 // workers > 1 we cut the canvas into ~4x more bands than workers so
485 // schedule(dynamic,1) can balance the (heavily non-uniform) per-band load.
486 // band_rows never exceeds memory.rows, so a band's accumulators are never
487 // larger than the W=1 stripe the budget was resolved against.
488 int workers = std::max(1, req_workers);
489 int band_rows = memory.rows;
490 if (workers > 1) {
491 const long long target_bands =
492 static_cast<long long>(workers) * 4;

495 }

497 band_rows > 0 ? (memory.height + band_rows - 1) / band_rows : 0;

501 // (B/w/w2/count/support + footprint_count/touched). memory.rows /
502 // resolved_chunk_rows stay exactly as the W=1 budget resolved them.
503 const size_t per_band_bytes =
504 static_cast<size_t>(memory.width) * band_rows *
505 (static_cast<size_t>(channels) *
506 (3 * sizeof(double) + sizeof(uint32_t) + 1) +
507 sizeof(uint32_t) + 1);

509 const size_t support_full_bytes = static_cast<size_t>(channels) * pixels;
510 const size_t committed = retained + row_scratch + support_full_bytes +
511 per_band_bytes + 64ull * 1024 * 1024;
512 while (workers > 1 &&
513 committed + static_cast<size_t>(workers - 1) * per_band_bytes >
514 memory.budget_bytes)
515 --workers;
516 }
517 gate.workers_used = workers;

519 // geomstats::Registry is process-global and non-atomic; disable it for the
520 // parallel region (RAII restore). At workers == 1 (the default and the
521 // reference path) the ScopedVariant/ScopedGeometryTimer scopes below are
522 // byte-identical to before.

552 auto run_band = [&](int s) {
553 const int y = s * band_rows;
554 const int rows = std::min(band_rows, memory.height - y);
555 const int W = memory.width;
556 const size_t n = static_cast<size_t>(W) * rows,
557 offset = static_cast<size_t>(y) * W;
558 std::array<std::vector<double>, 3> B, w, w2;
559 std::array<std::vector<uint32_t>, 3> count;
560 std::array<std::vector<uint8_t>, 3> support;
561 for (int c = 0; c < channels; ++c) {
562 B[c].assign(n, 0);
563 w[c].assign(n, 0);
564 w2[c].assign(n, 0);
565 count[c].assign(n, 0);
566 support[c].assign(n, 0);

569 std::vector<uint8_t> touched(n, 0);
570 // Channel-major B plane filled by the CUDA coverage gather; per-frame.
571 std::vector<double> gathered(static_cast<size_t>(channels) * n);
572 for (size_t fi = 0; fi < prepared.frames.size(); ++fi) {
573 const auto *f = prepared.frames[fi];

576 continue; // frame footprint misses this band entirely
577 const int fx0 = std::max(fb.x0, 0);

584 reconstruction::geomstats::Variant::kCoverageCfa, pixfrac);

587 // available; serialized inside the helper. Bit-identical: the kernel
588 // scans each cell's source neighbourhood in canonical order.

594 // frame's canvas box (outside it every entry stays 0).
595 {
596 reconstruction::geomstats::ScopedVariant _v2(
597 reconstruction::geomstats::Variant::kCoverageAccumulatorReset,
598 pixfrac);

602 std::fill_n(B[c].begin() + static_cast<size_t>(r) * W + fx0,
603 static_cast<size_t>(win_w), 0.0);

606 // window-local, translated back to the full-width band index.
607 rasterize_drizzle_stripe(
608 plan, *f, internal_scale, pixfrac, y, rows,
609 [&](int, int, int c, int, size_t i, double k) {
610 const size_t r = i / static_cast<size_t>(win_w);
611 const size_t col = i % static_cast<size_t>(win_w);
612 B[c][r * static_cast<size_t>(W) + fx0 + col] += k;
613 },
614 {}, fx0, win_w);
615 }
616 }
617 // T6: accumulator reduction subtimer --- per-frame w/w2/count fold.
618 {
619 reconstruction::geomstats::ScopedVariant _v(
620 reconstruction::geomstats::Variant::kCoverageAccumulatorReduction,
621 pixfrac);
622 reconstruction::geomstats::ScopedGeometryTimer _t;
623 long long emitted = 0;
624 for (int c = 0; c < channels; ++c) {
625 const double *src =
626 cuda_used ? gathered.data() + static_cast<size_t>(c) * n
627 : B[c].data();
628 for (int r = 0; r < rows; ++r) {
629 const size_t i0 = static_cast<size_t>(r) * W + fx0;
630 for (size_t i = i0; i < i0 + static_cast<size_t>(win_w); ++i)
631 if (src[i] > 0) {
632 ++count[c][i];
633 w[c][i] += src[i];
634 w2[c][i] += src[i] * src[i];
635 emitted += cuda_used ? 1 : 0;
636 }
637 }
638 }
639 // Same accounting the CPU path gets inside the leaf enumerator: one
640 // emitted (channel, cell) pair per positive contribution. Same-channel
641 // droplets are disjoint for pixfrac <= 1, so a cell receives at most
642 // one record per channel --- the positive fold count equals the record
643 // count the old pipeline accumulated.
644 if (reconstruction::geomstats::registry().enabled)
645 reconstruction::geomstats::registry().cur().leaf_cells_emitted +=
646 static_cast<std::uint64_t>(emitted);

648 // Dense source pixel squares define full-frame footprints, independently
649 // of CFA colour and the shrunken reconstruction droplet. §30.75: cell
650 // classification against the mapped source rectangle, exact per-pixel
651 // fallback only for boundary cells; byte-identical to the rasterize.
652 {
653 reconstruction::geomstats::ScopedVariant _v(
654 reconstruction::geomstats::Variant::kCoverageFootprint, 1.0);
655 reconstruction::geomstats::ScopedGeometryTimer _t;
656 reconstruction::dense_footprint_touched_stripe(
657 plan, *f, internal_scale, y, rows, touched, footprint_exact);
658 }
659 for (int r = 0; r < rows; ++r) {
660 const size_t i0 = static_cast<size_t>(r) * W + fx0;
661 for (size_t i = i0; i < i0 + static_cast<size_t>(win_w); ++i)
662 footprint_count[i] += touched[i];
663 }
664 }
665 for (size_t i = 0; i < n; ++i) {
666 const bool reference =
667 footprint_count[i] >= static_cast<uint32_t>(required);
668 result.analysis_common_mask[offset + i] = reference;
669 if (reference)
670 ++band_analysis_px[static_cast<size_t>(s)];
671 bool all = true;
672 for (int c = 0; c < channels; ++c) {
673 support[c][i] = count[c][i] > 0;
674 all = all && support[c][i];
675 support_full[c][offset + i] = support[c][i];
676 if (retain_channel_counts)

678 if (reference) {

681 w2[c][i] > 0 ? static_cast<float>(w[c][i] * w[c][i] / w2[c][i])
682 : 0.0f);

774 gate.min_channel_n_eff_p10 = std::min(gate.min_channel_n_eff_p10, neff);
775 gate.largest_internal_hole_area_px =
776 std::max(gate.largest_internal_hole_area_px, static_cast<int>(hole));

778 gate.violations.push_back("coverage_gate.min_supported_fraction " + name +
779 ": " + std::to_string(coverage));
780 if (neff < required_neff)
781 gate.violations.push_back("coverage_gate.min_channel_n_eff " + name +
782 ": " + std::to_string(neff));
783 if (hole > gate_cfg.max_internal_hole_area_px)
784 gate.violations.push_back("coverage_gate.max_internal_hole_area_px " +

786 }
787 if (N < gate_cfg.min_frames)
788 gate.violations.push_back("coverage_gate.min_frames: " + std::to_string(N));

790 gate.violations.push_back(
791 "coverage_gate.min_analysis_pixels: insufficient_analysis_support");
792 gate.passed = gate.violations.empty();
793 result.dither_spread_circular =
794 compute_dither_spread_circular_diagnostic(plan);
795 return result;

798std::string
799compute_coverage_geometry_hash(const RegistrationSamplingPlan &plan,
800 const config::ReconstructionDrizzleConfig &cfg,
801 float common_fraction) {
802 const std::string prefix = "coverage:v2:edge-centers:exact-polygons:dense-"
803 "footprint:whole-sample-veto:" +
804 compute_plan_hash(plan) + ":" + cfg.kernel;
805 std::vector<uint8_t> bytes(prefix.begin(), prefix.end());
806 auto u32 = [&](uint32_t v) {
807 for (int i = 0; i < 4; ++i)
808 bytes.push_back(static_cast<uint8_t>(v >> (8 * i)));
809 };
810 auto f32 = [&](float v) { u32(std::bit_cast<uint32_t>(v)); };
811 u32(cfg.internal_scale);
812 f32(cfg.pixfrac);
813 f32(common_fraction);
814 reconstruction::ForwardDrizzleSubdivisionParams sub;
815 f32(sub.position_epsilon_internal_px);
816 f32(sub.area_relative_epsilon);
817 u32(sub.max_subdivision_depth);
818 f32(sub.per_frame_inversion_error_rate_max);
819 return core::sha256_bytes(bytes);
820}
