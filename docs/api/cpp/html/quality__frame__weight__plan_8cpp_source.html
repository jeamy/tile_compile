22void validate_weights(const QualityFrameWeightPlan &plan) {
23 std::set<std::string> ids;
24 if (plan.source_identity_hash.empty() || plan.sampling_plan_hash.empty() ||
25 plan.source_quality_config_hash.empty())
26 throw std::invalid_argument("quality_frame_weight_plan missing provenance");
27 for (const auto &f : plan.frames) {
28 if (f.frame_id.empty() || !ids.insert(f.frame_id).second)
29 throw std::invalid_argument("quality_frame_weight_plan duplicate/empty frame_id");
30 for (float v : {f.g_quality, f.model_prediction_factor,
31 f.registration_residual_factor, f.g_eff})
32 if (!std::isfinite(v) || v < 0 || v > 1)
33 throw std::invalid_argument("quality_frame_weight_plan invalid factor");
34 if (f.g_eff != f.g_quality * f.model_prediction_factor * f.registration_residual_factor)
35 throw std::invalid_argument("quality_frame_weight_plan inconsistent g_eff");
36 }
37}

77QualityFrameWeightPlan build_quality_frame_weight_plan(
78 const registration::RegistrationSamplingPlan& sampling_plan, const VectorXf& g_quality,
79 const std::string& source_quality_config_hash) {
80 if (static_cast<size_t>(g_quality.size()) != sampling_plan.frames.size()) {
81 throw std::invalid_argument(
82 "QUALITY_FRAME_WEIGHT_PLAN: g_quality size does not match sampling plan frame count");
83 }
84 QualityFrameWeightPlan plan;
85 plan.source_identity_hash = sampling_plan.source_identity_hash;
86 plan.sampling_plan_hash = sampling_plan.plan_hash;
87 plan.source_quality_config_hash = source_quality_config_hash;
88 plan.frames.reserve(sampling_plan.frames.size());
89 for (size_t i = 0; i < sampling_plan.frames.size(); ++i) {
90 const auto& sf = sampling_plan.frames[i];
91 QualityFrameWeight w;
92 w.frame_id = sf.frame_id;
93 w.g_quality = g_quality[static_cast<int>(i)];
94 // Read verbatim from the sampling plan --- never recomputed here.
95 w.model_prediction_factor = sf.model_prediction_factor;
96 w.registration_residual_factor = sf.registration_residual_factor;
97 w.g_eff = w.g_quality * w.model_prediction_factor * w.registration_residual_factor;
98 plan.frames.push_back(std::move(w));
99 }
100 validate_weights(plan);
101 plan.plan_hash = compute_quality_frame_weight_plan_hash(plan);
102 return plan;
103}

125bool parse_quality_frame_weight_plan(const std::string& text, QualityFrameWeightPlan& out,
126 std::string& error) {
127 try {
128 const json j = json::parse(text);
129 if (j.value("schema_version", -1) != QualityFrameWeightPlan::kSchemaVersion) {
130 error = "unsupported quality_frame_weight_plan schema_version";
131 return false;
132 }
133 QualityFrameWeightPlan plan;
134 plan.source_identity_hash = j.at("source_identity_hash").get<std::string>();
135 plan.sampling_plan_hash = j.at("sampling_plan_hash").get<std::string>();
136 plan.source_quality_config_hash = j.at("source_quality_config_hash").get<std::string>();
137 plan.plan_hash = j.at("plan_hash").get<std::string>();
138 if (!j.at("frames").is_array())
139 throw std::invalid_argument("quality_frame_weight_plan frames must be an array");
140 for (const auto& jf : j.at("frames")) {
141 QualityFrameWeight f;
142 f.frame_id = jf.at("frame_id").get<std::string>();
143 f.g_quality = jf.at("g_quality").get<float>();
144 f.model_prediction_factor = jf.at("model_prediction_factor").get<float>();
145 f.registration_residual_factor = jf.at("registration_residual_factor").get<float>();
146 f.g_eff = jf.at("g_eff").get<float>();
147 plan.frames.push_back(std::move(f));
148 }
149 // Fail closed on a tampered / truncated artifact: the stored hash must
150 // match a fresh recompute, and g_eff must be the exact product.
151 if (plan.plan_hash != compute_quality_frame_weight_plan_hash(plan)) {
152 error = "quality_frame_weight_plan hash mismatch";
153 return false;
154 }
155 for (const auto& f : plan.frames) {
156 const float expect = f.g_quality * f.model_prediction_factor * f.registration_residual_factor;
157 if (std::abs(expect - f.g_eff) > 1e-6f * (1.0f + std::abs(f.g_eff))) {
158 error = "quality_frame_weight_plan g_eff is not the product of its factors";
159 return false;
160 }
161 }
162 validate_weights(plan);
163 error.clear();
164 out = std::move(plan);
165 return true;
166 } catch (const std::exception& e) {
167 error = std::string("quality_frame_weight_plan parse error: ") + e.what();
168 return false;
169 }
170}
