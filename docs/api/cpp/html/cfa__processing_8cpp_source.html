32inline void fill_bayer_color_lut(BayerPattern pattern, uint8_t color_lut[4]) {
33 if (pattern == BayerPattern::UNKNOWN) {
34 pattern = BayerPattern::GBRG;
35 }
36
37 switch (pattern) {
38 case BayerPattern::RGGB:
39 color_lut[0] = static_cast<uint8_t>(CfaColor::Red);
40 color_lut[1] = static_cast<uint8_t>(CfaColor::Green);
41 color_lut[2] = static_cast<uint8_t>(CfaColor::Green);
42 color_lut[3] = static_cast<uint8_t>(CfaColor::Blue);
43 break;
44 case BayerPattern::BGGR:
45 color_lut[0] = static_cast<uint8_t>(CfaColor::Blue);
46 color_lut[1] = static_cast<uint8_t>(CfaColor::Green);
47 color_lut[2] = static_cast<uint8_t>(CfaColor::Green);
48 color_lut[3] = static_cast<uint8_t>(CfaColor::Red);
49 break;
50 case BayerPattern::GRBG:
51 color_lut[0] = static_cast<uint8_t>(CfaColor::Green);
52 color_lut[1] = static_cast<uint8_t>(CfaColor::Red);
53 color_lut[2] = static_cast<uint8_t>(CfaColor::Blue);
54 color_lut[3] = static_cast<uint8_t>(CfaColor::Green);
55 break;
56 case BayerPattern::GBRG:
57 default:
58 color_lut[0] = static_cast<uint8_t>(CfaColor::Green);
59 color_lut[1] = static_cast<uint8_t>(CfaColor::Blue);
60 color_lut[2] = static_cast<uint8_t>(CfaColor::Red);
61 color_lut[3] = static_cast<uint8_t>(CfaColor::Green);
62 break;
63 }

90inline float average_neighbors_of_color_strided(const float* data,
91 int h,
92 int w,
93 int stride,
94 int origin_y,
95 int origin_x,

97 const int* ys,
98 const int* xs,
99 int n,
100 CfaColor desired_color) {
101 float sum = 0.0f;
102 int count = 0;
103 const uint8_t desired = static_cast<uint8_t>(desired_color);
104 for (int i = 0; i < n; ++i) {
105 const int cy = clamp_index(ys[i], 0, h - 1);
106 const int cx = clamp_index(xs[i], 0, w - 1);
107 const int parity_idx = (((origin_y + cy) & 1) << 1) | ((origin_x + cx) & 1);
108 if (color_lut[parity_idx] != desired) {

110 }
111 const float v = data[static_cast<size_t>(cy) * static_cast<size_t>(stride) +
112 static_cast<size_t>(cx)];
113 if (!(std::isfinite(v) && v > 0.0f)) {
114 continue;
115 }
116 sum += v;
117 ++count;
118 }
119 if (count > 0) {
120 return sum / static_cast<float>(count);
121 }
122 const float fallback = sample_clamped_strided(data, h, w, stride, ys[0], xs[0]);

149void debayer_bilinear_core(const float* src, int h, int w, int stride,
150 BayerPattern pattern, int origin_x, int origin_y,
151 Matrix2Df& R_out, Matrix2Df& G_out,
152 Matrix2Df& B_out) {
153 R_out.resize(h, w);
154 G_out.resize(h, w);
155 B_out.resize(h, w);
156 R_out.setZero();
157 G_out.setZero();
158 B_out.setZero();
159
160 int r_row = 1;
161 if (pattern == BayerPattern::UNKNOWN) {
162 pattern = BayerPattern::GBRG;
163 } else if (pattern == BayerPattern::RGGB) {
164 r_row = 0;
165 } else if (pattern == BayerPattern::BGGR) {
166 r_row = 1;
167 } else if (pattern == BayerPattern::GRBG) {
168 r_row = 0;
169 }
170 uint8_t color_lut[4] = {0, 0, 0, 0};
171 fill_bayer_color_lut(pattern, color_lut);
172
173 for (int y = 0; y < h; ++y) {
174 for (int x = 0; x < w; ++x) {
175 const int parity_idx = (((origin_y + y) & 1) << 1) | ((origin_x + x) & 1);
176 const uint8_t color = color_lut[parity_idx];
177 float r_val = 0.0f;
178 float g_val = 0.0f;
179 float b_val = 0.0f;
180
181 if (color == static_cast<uint8_t>(CfaColor::Red)) {
182 r_val = sample_clamped_strided(src, h, w, stride, y, x);
183 const int gy[4] = {y - 1, y + 1, y, y};
184 const int gx[4] = {x, x, x - 1, x + 1};
185 g_val = average_neighbors_of_color_strided(
186 src, h, w, stride, origin_y, origin_x, color_lut, gy, gx,
187 4, CfaColor::Green);
188 const int by[4] = {y - 1, y - 1, y + 1, y + 1};
189 const int bx[4] = {x - 1, x + 1, x - 1, x + 1};
190 b_val = average_neighbors_of_color_strided(
191 src, h, w, stride, origin_y, origin_x, color_lut, by, bx,
192 4, CfaColor::Blue);
193 } else if (color == static_cast<uint8_t>(CfaColor::Blue)) {
194 b_val = sample_clamped_strided(src, h, w, stride, y, x);
195 const int gy[4] = {y - 1, y + 1, y, y};
196 const int gx[4] = {x, x, x - 1, x + 1};
197 g_val = average_neighbors_of_color_strided(
198 src, h, w, stride, origin_y, origin_x, color_lut, gy, gx,
199 4, CfaColor::Green);
200 const int ry[4] = {y - 1, y - 1, y + 1, y + 1};
201 const int rx[4] = {x - 1, x + 1, x - 1, x + 1};
202 r_val = average_neighbors_of_color_strided(
203 src, h, w, stride, origin_y, origin_x, color_lut, ry, rx,
204 4, CfaColor::Red);
205 } else {
206 g_val = sample_clamped_strided(src, h, w, stride, y, x);
207 const bool green_on_red_row = (((origin_y + y) & 1) == r_row);
208 if (green_on_red_row) {
209 const int ry[2] = {y, y};
210 const int rx[2] = {x - 1, x + 1};
211 r_val = average_neighbors_of_color_strided(
212 src, h, w, stride, origin_y, origin_x, color_lut,
213 ry, rx, 2, CfaColor::Red);
214 const int by[2] = {y - 1, y + 1};
215 const int bx[2] = {x, x};
216 b_val = average_neighbors_of_color_strided(
217 src, h, w, stride, origin_y, origin_x, color_lut,
218 by, bx, 2, CfaColor::Blue);
219 } else {
220 const int ry[2] = {y - 1, y + 1};
221 const int rx[2] = {x, x};
222 r_val = average_neighbors_of_color_strided(
223 src, h, w, stride, origin_y, origin_x, color_lut,
224 ry, rx, 2, CfaColor::Red);
225 const int by[2] = {y, y};
226 const int bx[2] = {x - 1, x + 1};
227 b_val = average_neighbors_of_color_strided(
228 src, h, w, stride, origin_y, origin_x, color_lut,
229 by, bx, 2, CfaColor::Blue);
230 }
231 }
232
233 R_out(y, x) = r_val;
234 G_out(y, x) = g_val;
235 B_out(y, x) = b_val;
236 }

242void debayer_nearest_neighbor_core(const float* src, int h, int w, int stride,
243 BayerPattern pattern, int origin_x, int origin_y,
244 Matrix2Df& R_out, Matrix2Df& G_out,
245 Matrix2Df& B_out) {
246 R_out.resize(h, w);
247 G_out.resize(h, w);
248 B_out.resize(h, w);
249
250 uint8_t color_lut[4] = {0, 0, 0, 0};
251 fill_bayer_color_lut(pattern, color_lut);
252
253 for (int y = 0; y < h; ++y) {
254 for (int x = 0; x < w; ++x) {
255 float r = 0, g = 0, b = 0;
256 int abs_y = origin_y + y;
257 int abs_x = origin_x + x;
258 int block_y = abs_y & ~1;
259 int block_x = abs_x & ~1;
260
261 for (int dy = 0; dy < 2; ++dy) {
262 for (int dx = 0; dx < 2; ++dx) {
263 int py = block_y + dy - origin_y;
264 int px = block_x + dx - origin_x;
265 float val = sample_clamped_strided(src, h, w, stride, py, px);
266 int parity = (dy << 1) | dx;
267 CfaColor c = static_cast<CfaColor>(color_lut[parity]);
268 if (c == CfaColor::Red) r = val;
269 else if (c == CfaColor::Blue) b = val;
270 else g = val;
271 }
272 }
273
274 float v = sample_clamped_strided(src, h, w, stride, y, x);
275 int current_parity = ((abs_y & 1) << 1) | (abs_x & 1);
276 CfaColor current_c = static_cast<CfaColor>(color_lut[current_parity]);
277 if (current_c == CfaColor::Red) r = v;
278 else if (current_c == CfaColor::Blue) b = v;
279 else g = v;
280
281 R_out(y, x) = r;
282 G_out(y, x) = g;
283 B_out(y, x) = b;
284 }
285 }

325// Green sites of the 2x2 CFA cell (former cfa_green_mask convention):
326// RGGB/BGGR put green at (0,1),(1,0) (odd y+x), GBRG/GRBG at (0,0),(1,1).
327bool cfa_green_on_odd_parity(const std::string& bayer_pattern) {
328 std::string bp = bayer_pattern;
329 std::transform(bp.begin(), bp.end(), bp.begin(), ::toupper);
330 if (bp.empty()) bp = "GBRG";
331 return bp == "RGGB" || bp == "BGGR";
332}

334// Green-proxy value of one mosaic pixel: the pixel itself on a green site,
335// else the mean of its green 4-neighbours with edge replication (the same
336// values the former padded-plane formulation produced; the neighbour sum is
337// evaluated in the same up, down, left, right order and the mask product
338// keeps the NaN * 0 propagation of the plane form).
339inline float cfa_green_proxy_at(const Matrix2Df& mosaic, int h, int w, int y,
340 int x, bool green_odd) {
341 auto gmask = [&](int yy, int xx) -> float {
342 const bool odd = ((yy + xx) & 1) != 0;
343 return (odd == green_odd) ? 1.0f : 0.0f;
344 };
345 if (gmask(y, x) > 0.5f) return mosaic(y, x);
346 const int yu = std::max(0, y - 1), yd = std::min(h - 1, y + 1);
347 const int xl = std::max(0, x - 1), xr = std::min(w - 1, x + 1);
348 const float mu = gmask(yu, x), md = gmask(yd, x);
349 const float ml = gmask(y, xl), mr = gmask(y, xr);
350 const float sum4 = mosaic(yu, x) * mu + mosaic(yd, x) * md +
351 mosaic(y, xl) * ml + mosaic(y, xr) * mr;
352 const float cnt4 = mu + md + ml + mr;
353 return (cnt4 > 0.5f) ? (sum4 / cnt4) : 0.0f;
354}

362Matrix2Df cfa_green_proxy(const Matrix2Df& mosaic, const std::string& bayer_pattern) {
363 const int h = mosaic.rows();
364 const int w = mosaic.cols();
365 const bool green_odd = cfa_green_on_odd_parity(bayer_pattern);
366 Matrix2Df out(h, w);
367 // Direct 4-neighbour stencil with clamped indices: no padded copies of
368 // the mosaic, mask and product planes. Serial on purpose: callers run
369 // this per frame inside std::thread worker pools.
370 for (int y = 0; y < h; ++y) {
371 for (int x = 0; x < w; ++x) {
372 out(y, x) = cfa_green_proxy_at(mosaic, h, w, y, x, green_odd);
373 }
374 }
375 return out;
376}

378Matrix2Df cfa_green_proxy_downsample2x2(const Matrix2Df& mosaic, const std::string& bayer_pattern) {
379 // Fused stencil + 2x2 mean straight to the half-resolution output; same
380 // shape contract and tap clamping as core::downsample2x2_mean, same
381 // per-pixel proxy values and the same 0.25f * (a + b + c + d) sum order.
382 const int h = mosaic.rows();
383 const int w = mosaic.cols();
384 if (h <= 0 || w <= 0) return Matrix2Df(0, 0);
385 const bool green_odd = cfa_green_on_odd_parity(bayer_pattern);
386 const int h2 = h - (h % 2);
387 const int w2 = w - (w % 2);
388 const int out_h = std::max(1, h2 / 2);
389 const int out_w = std::max(1, w2 / 2);
390 Matrix2Df out(out_h, out_w);
391 for (int oy = 0; oy < out_h; ++oy) {
392 const int sy = std::min(oy * 2, h - 1);
393 const int sy1 = std::min(sy + 1, h - 1);
394 for (int ox = 0; ox < out_w; ++ox) {
395 const int sx = std::min(ox * 2, w - 1);
396 const int sx1 = std::min(sx + 1, w - 1);
397 const float a = cfa_green_proxy_at(mosaic, h, w, sy, sx, green_odd);
398 const float b = cfa_green_proxy_at(mosaic, h, w, sy, sx1, green_odd);
399 const float c = cfa_green_proxy_at(mosaic, h, w, sy1, sx, green_odd);
400 const float d = cfa_green_proxy_at(mosaic, h, w, sy1, sx1, green_odd);
401 out(oy, ox) = 0.25f * (a + b + c + d);
402 }
403 }
404 return out;
405}

411Matrix2Df warp_cfa_mosaic_via_subplanes(
412 const Matrix2Df& mosaic,
413 const WarpMatrix& warp,
414 int out_height,
415 int out_width,
416 const std::string& border_mode,
417 const std::string& interpolation
418) {
419 const int h = mosaic.rows();
420 const int w = mosaic.cols();
421 const auto dims = tile_compile::core::compute_cfa_warp_dims(h, w, out_height, out_width);
422 auto sub = tile_compile::core::extract_cfa_subplanes(mosaic, dims);
423 const auto warps = tile_compile::core::make_all_cfa_subplane_warps(warp);
424
425 cv::Mat a_cv(dims.sub_h, dims.sub_w, CV_32F, sub.a.data());
426 cv::Mat b_cv(dims.sub_h, dims.sub_w, CV_32F, sub.b.data());
427 cv::Mat c_cv(dims.sub_h, dims.sub_w, CV_32F, sub.c.data());
428 cv::Mat d_cv(dims.sub_h, dims.sub_w, CV_32F, sub.d.data());
429
430 int interp_flag = cv::INTER_LINEAR;
431 if (interpolation == "nearest") {
432 interp_flag = cv::INTER_NEAREST;
433 } else if (interpolation == "cubic") {
434 interp_flag = cv::INTER_CUBIC;
435 } else if (interpolation == "lanczos4") {
436 interp_flag = cv::INTER_LANCZOS4;
437 }
438 int flags = interp_flag | cv::WARP_INVERSE_MAP;
439
440 int border_flag = cv::BORDER_CONSTANT;
441 if (border_mode == "replicate") {
442 border_flag = cv::BORDER_REPLICATE;
443 } else if (border_mode == "reflect") {
444 border_flag = cv::BORDER_REFLECT_101;
445 }
446
447 cv::Mat a_w, b_w, c_w, d_w;
448 cv::warpAffine(a_cv, a_w, warps.a, cv::Size(dims.out_w_sub, dims.out_h_sub), flags, border_flag);
449 cv::warpAffine(b_cv, b_w, warps.b, cv::Size(dims.out_w_sub, dims.out_h_sub), flags, border_flag);
450 cv::warpAffine(c_cv, c_w, warps.c, cv::Size(dims.out_w_sub, dims.out_h_sub), flags, border_flag);
451 cv::warpAffine(d_cv, d_w, warps.d, cv::Size(dims.out_w_sub, dims.out_h_sub), flags, border_flag);
452
453 return tile_compile::core::reassemble_cfa_subplanes(a_w, b_w, c_w, d_w, dims);
454}

460CFAChannels split_cfa_channels(const Matrix2Df& mosaic, const std::string& bayer_pattern) {
461 std::string bp = bayer_pattern;
462 std::transform(bp.begin(), bp.end(), bp.begin(), ::toupper);
463 if (bp.empty()) bp = "GBRG";
464
465 int h = mosaic.rows();
466 int w = mosaic.cols();
467 int h2 = h - (h % 2);
468 int w2 = w - (w % 2);
469 int sub_h = h2 / 2;
470 int sub_w = w2 / 2;
471
472 // Bayer pattern positions
473 int r_row, r_col, b_row, b_col, g1_row, g1_col, g2_row, g2_col;
474
475 if (bp == "RGGB") {
476 r_row = 0; r_col = 0; g1_row = 0; g1_col = 1; g2_row = 1; g2_col = 0; b_row = 1; b_col = 1;
477 } else if (bp == "BGGR") {
478 b_row = 0; b_col = 0; g1_row = 0; g1_col = 1; g2_row = 1; g2_col = 0; r_row = 1; r_col = 1;
479 } else if (bp == "GBRG") {
480 g1_row = 0; g1_col = 0; b_row = 0; b_col = 1; r_row = 1; r_col = 0; g2_row = 1; g2_col = 1;
481 } else { // GRBG
482 g1_row = 0; g1_col = 0; r_row = 0; r_col = 1; b_row = 1; b_col = 0; g2_row = 1; g2_col = 1;
483 }
484
485 CFAChannels channels;
486 channels.R = Matrix2Df(sub_h, sub_w);
487 channels.G = Matrix2Df(sub_h, sub_w);
488 channels.B = Matrix2Df(sub_h, sub_w);
489
490 for (int y = 0; y < sub_h; ++y) {
491 for (int x = 0; x < sub_w; ++x) {
492 channels.R(y, x) = mosaic(y * 2 + r_row, x * 2 + r_col);
493 channels.B(y, x) = mosaic(y * 2 + b_row, x * 2 + b_col);
494 float g1 = mosaic(y * 2 + g1_row, x * 2 + g1_col);
495 float g2 = mosaic(y * 2 + g2_row, x * 2 + g2_col);
496 channels.G(y, x) = 0.5f * (g1 + g2);
497 }
498 }
499
500 return channels;
501}

507Matrix2Df reassemble_cfa_mosaic(
508 const Matrix2Df& r_plane,
509 const Matrix2Df& g_plane,
510 const Matrix2Df& b_plane,
511 const std::string& bayer_pattern
512) {
513 std::string bp = bayer_pattern;
514 std::transform(bp.begin(), bp.end(), bp.begin(), ::toupper);
515 if (bp.empty()) bp = "GBRG";
516
517 int sub_h = r_plane.rows();
518 int sub_w = r_plane.cols();
519 int h = sub_h * 2;
520 int w = sub_w * 2;
521
522 int r_row, r_col, b_row, b_col, g1_row, g1_col, g2_row, g2_col;
523
524 if (bp == "RGGB") {
525 r_row = 0; r_col = 0; g1_row = 0; g1_col = 1; g2_row = 1; g2_col = 0; b_row = 1; b_col = 1;
526 } else if (bp == "BGGR") {
527 b_row = 0; b_col = 0; g1_row = 0; g1_col = 1; g2_row = 1; g2_col = 0; r_row = 1; r_col = 1;
528 } else if (bp == "GBRG") {
529 g1_row = 0; g1_col = 0; b_row = 0; b_col = 1; r_row = 1; r_col = 0; g2_row = 1; g2_col = 1;
530 } else { // GRBG
531 g1_row = 0; g1_col = 0; r_row = 0; r_col = 1; b_row = 1; b_col = 0; g2_row = 1; g2_col = 1;
532 }
533
534 Matrix2Df mosaic = Matrix2Df::Zero(h, w);
535
536 for (int y = 0; y < sub_h; ++y) {
537 for (int x = 0; x < sub_w; ++x) {
538 mosaic(y * 2 + r_row, x * 2 + r_col) = r_plane(y, x);
539 mosaic(y * 2 + b_row, x * 2 + b_col) = b_plane(y, x);
540 mosaic(y * 2 + g1_row, x * 2 + g1_col) = g_plane(y, x);
541 mosaic(y * 2 + g2_row, x * 2 + g2_col) = g_plane(y, x);
542 }
543 }
544
545 return mosaic;
546}

552void bayer_offsets(const std::string& bayer_pattern,
553 int& r_row, int& r_col, int& b_row, int& b_col) {
554 std::string bp = bayer_pattern;
555 std::transform(bp.begin(), bp.end(), bp.begin(), ::toupper);
556 r_row = 1; r_col = 0;
557 b_row = 0; b_col = 1;
558 if (bp == "RGGB") {
559 r_row = 0; r_col = 0;
560 b_row = 1; b_col = 1;
561 } else if (bp == "BGGR") {
562 r_row = 1; r_col = 1;

708static int opencv_bayer_code(BayerPattern pattern, int origin_x, int origin_y,
709 bool ahd) {
710 if (pattern == BayerPattern::UNKNOWN) {
711 pattern = BayerPattern::GBRG;
712 }
713 uint8_t lut[4] = {0, 0, 0, 0};
714 fill_bayer_color_lut(pattern, lut);
715 const int c00 = lut[((origin_y & 1) << 1) | (origin_x & 1)];
716 const int c01 = lut[((origin_y & 1) << 1) | ((origin_x + 1) & 1)];
717 // Empirically verified OpenCV code mapping (differs from the naive
718 // first-row reading): standard RGGB -> BayerBG, BGGR -> BayerRG,
719 // GRBG -> BayerGB, GBRG -> BayerGR.
720 if (c00 == static_cast<int>(CfaColor::Blue)) {
721 return ahd ? cv::COLOR_BayerRG2RGB_EA : cv::COLOR_BayerRG2RGB_VNG;
722 }
723 if (c00 == static_cast<int>(CfaColor::Green) &&
724 c01 == static_cast<int>(CfaColor::Blue)) {
725 return ahd ? cv::COLOR_BayerGR2RGB_EA : cv::COLOR_BayerGR2RGB_VNG;
726 }
727 if (c00 == static_cast<int>(CfaColor::Red)) {
728 return ahd ? cv::COLOR_BayerBG2RGB_EA : cv::COLOR_BayerBG2RGB_VNG;
729 }
730 return ahd ? cv::COLOR_BayerGB2RGB_EA : cv::COLOR_BayerGB2RGB_VNG;
731}

733void debayer_opencv_into(const Matrix2Df& mosaic,
734 BayerPattern pattern,
735 int origin_x,
736 int origin_y,
737 bool ahd,
738 Matrix2Df& R_out,
739 Matrix2Df& G_out,
740 Matrix2Df& B_out) {
741 const int h = static_cast<int>(mosaic.rows());
742 const int w = static_cast<int>(mosaic.cols());
743 if (h <= 0 || w <= 0) {
744 R_out.resize(0, 0);
745 G_out.resize(0, 0);
746 B_out.resize(0, 0);
747 return;
748 }
749 // cv::demosaicing supports only integer input: 16U for EA/AHD, 8U for
750 // VNG. Map the float mosaic linearly to the full integer range and back;
751 // the affine transform preserves channel ratios.
752 cv::Mat src(h, w, CV_32F, const_cast<float*>(mosaic.data()),
753 static_cast<size_t>(w) * sizeof(float));
754 double v_min = 0.0, v_max = 0.0;
755 cv::minMaxLoc(src, &v_min, &v_max);
756 const double range = v_max - v_min;
757 if (!(range > 0.0) || !std::isfinite(range)) {
758 R_out = Matrix2Df::Constant(h, w, static_cast<float>(v_min));
759 G_out = Matrix2Df::Constant(h, w, static_cast<float>(v_min));
760 B_out = Matrix2Df::Constant(h, w, static_cast<float>(v_min));
761 return;
762 }
763 cv::Mat rgb32;
764 if (ahd) {
765 cv::Mat src16, rgb16;
766 src.convertTo(src16, CV_16U, 65535.0 / range, -v_min * 65535.0 / range);
767 cv::demosaicing(src16, rgb16,
768 opencv_bayer_code(pattern, origin_x, origin_y, true));
769 rgb16.convertTo(rgb32, CV_32F, range / 65535.0, v_min);
770 } else {
771 cv::Mat src8, rgb8;
772 src.convertTo(src8, CV_8U, 255.0 / range, -v_min * 255.0 / range);
773 cv::demosaicing(src8, rgb8,
774 opencv_bayer_code(pattern, origin_x, origin_y, false));
775 rgb8.convertTo(rgb32, CV_32F, range / 255.0, v_min);
776 }
777 R_out.resize(h, w);
778 G_out.resize(h, w);
779 B_out.resize(h, w);
780 for (int y = 0; y < h; ++y) {
781 const cv::Vec3f* row = rgb32.ptr<cv::Vec3f>(y);
782 for (int x = 0; x < w; ++x) {
783 R_out(y, x) = row[x][0];
784 G_out(y, x) = row[x][1];
785 B_out(y, x) = row[x][2];
786 }
787 }
788}
