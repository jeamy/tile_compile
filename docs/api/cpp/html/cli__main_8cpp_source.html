100static json yaml_node_to_json(const YAML::Node& node) {
101 if (node.IsMap()) {
102 json out = json::object();
103 for (auto it = node.begin(); it != node.end(); ++it) {
104 out[it->first.as<std::string>()] = yaml_node_to_json(it->second);

143static std::string trim_trailing_zeros(std::string text) {
144 const auto dot = text.find('.');
145 if (dot == std::string::npos) return text;
146 while (!text.empty() && text.back() == '0') text.pop_back();
147 if (!text.empty() && text.back() == '.') text.pop_back();
148 if (text == "-0") return "0";
149 return text.empty() ? "0" : text;

156static std::string format_config_float_scalar(double value) {
157 if (!std::isfinite(value)) return "0";
158
159 const double rounded = std::round(value * 100.0) / 100.0;
160 if (rounded == 0.0 && value != 0.0 && std::fabs(value) < 0.01) {
161 std::ostringstream oss;
162 oss << std::scientific << std::setprecision(2) << value;
163 return oss.str();

165
166 std::ostringstream oss;
167 oss << std::fixed << std::setprecision(2) << rounded;
168 return trim_trailing_zeros(oss.str());

176 if (!node || node.IsNull()) return;

235static FitsHeaderInfo read_fits_header_info(const fs::path& path) {
236 FitsHeaderInfo info;
237 fitsfile* fptr = nullptr;
238 int status = 0;
239
240 if (fits_open_file(&fptr, path.string().c_str(), READONLY, &status)) {
241 info.read_error = true;
242 info.error_msg = "Cannot open FITS file";
243 return info;
244 }

246 int naxis = 0;

248 int bitpix = 0;
249
250 auto try_read_bayerpat = [&](int &bp_status) {
251 char bayerpat[FLEN_VALUE];
252 char comment[FLEN_COMMENT];
253 fits_read_key(fptr, TSTRING, const_cast<char*>("BAYERPAT"), bayerpat, comment, &bp_status);
254 if (bp_status == 0) {
255 std::string bp_str(bayerpat);
256 // Trim whitespace and quotes
257 bp_str.erase(0, bp_str.find_first_not_of(" \t\n\r'\""));
258 bp_str.erase(bp_str.find_last_not_of(" \t\n\r'\"") + 1);
259 if (!bp_str.empty()) {
260 info.has_bayerpat = true;
261 info.bayerpat = bp_str;
262 std::transform(info.bayerpat.begin(), info.bayerpat.end(), info.bayerpat.begin(), ::toupper);
263 }
264 }
265 };
266
267 // Primary HDU often carries BAYERPAT even if image data is in extension.
268 int bp_status = 0;
269 try_read_bayerpat(bp_status);
270
271 bool found_image_hdu = false;
272 status = 0;
273 fits_get_img_param(fptr, 3, &bitpix, &naxis, naxes, &status);
274 if (status == 0 && naxis >= 2) {
275 found_image_hdu = true;
276 }
277
278 if (!found_image_hdu) {
279 status = 0;
280 int nhdus = 0;

285 status = 0;

287 if (status) continue;
288 status = 0;
289 fits_get_img_param(fptr, 3, &bitpix, &naxis, naxes, &status);
290 if (status == 0 && naxis >= 2) {
291 found_image_hdu = true;
292 break;
293 }
294 }
295 }
296 }
297
298 if (!found_image_hdu) {
299 info.read_error = true;
300 info.error_msg = "Cannot read FITS image parameters";
301 fits_close_file(fptr, &status);
302 return info;
303 }
304
305 info.naxis = naxis;
306 info.naxis1 = static_cast<int>(naxes[0]);
307 info.naxis2 = static_cast<int>(naxes[1]);
308
309 // If not in primary, try current image HDU as fallback.
310 if (!info.has_bayerpat) {
311 int bp2_status = 0;
312 try_read_bayerpat(bp2_status);
313 }

315 // Read additional metadata keywords (best-effort, ignore errors)
316 auto read_double_key = [&](const char* key) -> double {
317 double val = 0.0;
318 int ks = 0;
319 fits_read_key(fptr, TDOUBLE, const_cast<char*>(key), &val, nullptr, &ks);
320 return (ks == 0) ? val : -1.0;
321 };
322 auto read_string_key = [&](const char* key) -> std::string {
323 char val[FLEN_VALUE];
324 int ks = 0;
325 fits_read_key(fptr, TSTRING, const_cast<char*>(key), val, nullptr, &ks);
326 if (ks != 0) return "";
327 std::string s(val);
328 s.erase(0, s.find_first_not_of(" \t\n\r'\""));
329 s.erase(s.find_last_not_of(" \t\n\r'\"") + 1);
330 return s;

333 // Move back to primary HDU for header keywords
334 {
335 int ks = 0;
336 fits_movabs_hdu(fptr, 1, nullptr, &ks);
337 }
338 double exp = read_double_key("EXPTIME");
339 if (exp < 0) exp = read_double_key("EXPOSURE");
340 info.exptime = exp;
341
342 double gain = read_double_key("GAIN");
343 if (gain < 0) gain = read_double_key("EGAIN");
344 info.gain = gain;
345
346 double temp = read_double_key("CCD-TEMP");
347 if (temp < -900) temp = read_double_key("SET-TEMP");
348 info.ccd_temp = temp;
349
350 info.object = read_string_key("OBJECT");
351 info.instrume = read_string_key("INSTRUME");
352 info.telescop = read_string_key("TELESCOP");
353 info.filter = read_string_key("FILTER");
354 info.date_obs = read_string_key("DATE-OBS");

356 // Read RA/DEC — try multiple keyword variants used by different capture software.
357 // Values may be in degrees (float) or in sexagesimal hours/degrees (string like "05 35 17.3").
358 auto read_ra_dec = [&]() {
359 // Try numeric RA/DEC (degrees) first
360 double ra_val = read_double_key("RA");
361 double dec_val = read_double_key("DEC");
362 if (ra_val >= 0 && dec_val > -900) {
363 info.ra_deg = ra_val;
364 info.dec_deg = dec_val;
365 return;
366 }
367 // Try OBJCTRA/OBJCTDEC (sexagesimal, "HH MM SS.S" / "DD MM SS.S")
368 std::string objctra = read_string_key("OBJCTRA");
369 std::string objctdec = read_string_key("OBJCTDEC");
370 if (!objctra.empty() && !objctdec.empty()) {
371 // Parse sexagesimal: split on whitespace or colons
372 auto parse_sex = [](const std::string& s, bool is_ra) -> double {
373 // Replace colons with spaces for uniform splitting
374 std::string cleaned = s;
375 std::replace(cleaned.begin(), cleaned.end(), ':', ' ');
376 std::istringstream iss(cleaned);
377 double h = 0, m = 0, sec = 0;
378 iss >> h >> m >> sec;
379 if (iss.fail()) return -999.0;
380 double deg = is_ra ? (h + m / 60.0 + sec / 3600.0) * 15.0
381 : (std::abs(h) + m / 60.0 + sec / 3600.0);
382 if (!is_ra && !s.empty() && s[0] == '-') deg = -deg;
383 return deg;
384 };
385 double ra = parse_sex(objctra, true);
386 double dec = parse_sex(objctdec, false);
387 if (ra >= 0 && ra < 360 && dec > -90 && dec < 90) {
388 info.ra_deg = ra;
389 info.dec_deg = dec;
390 return;
391 }
392 }
393 // Try SIT-RA/SIT-DEC (some mounts write telescope position)
394 double sit_ra = read_double_key("SIT-RA");
395 double sit_dec = read_double_key("SIT-DEC");
396 if (sit_ra >= 0 && sit_dec > -900) {
397 info.ra_deg = sit_ra;
398 info.dec_deg = sit_dec;
399 }
400 };
401 read_ra_dec();
402
403 fits_close_file(fptr, &status);
404 return info;
405}

411static json compute_fits_stats_buffer(const std::vector<float>& buf) {
412 double mean = 0.0;
413 double m2 = 0.0;
414 int64_t n = 0;
415
416 float min_v = std::numeric_limits<float>::infinity();
417 float max_v = -std::numeric_limits<float>::infinity();
418 int64_t n_nan = 0;
419 int64_t n_inf = 0;
420 int64_t n_zero = 0;
421
422 for (float v : buf) {
423 if (std::isnan(v)) {
424 n_nan++;
425 continue;
426 }
427 if (!std::isfinite(v)) {
428 n_inf++;
429 continue;
430 }
431
432 if (v == 0.0f) n_zero++;
433 if (v < min_v) min_v = v;
434 if (v > max_v) max_v = v;
435
436 n++;
437 const double x = static_cast<double>(v);
438 const double delta = x - mean;
439 mean += delta / static_cast<double>(n);
440 const double delta2 = x - mean;
441 m2 += delta * delta2;
442 }
443
444 json j;
445 j["count"] = n;
446 j["nan"] = n_nan;
447 j["inf"] = n_inf;
448 j["zero"] = n_zero;
449 j["min"] = (n > 0) ? min_v : 0.0f;
450 j["max"] = (n > 0) ? max_v : 0.0f;
451 j["mean"] = (n > 0) ? mean : 0.0;
452 j["stddev"] = (n > 1) ? std::sqrt(m2 / static_cast<double>(n - 1)) : 0.0;
453 return j;
454}

460static json fits_stats_file(const fs::path& path) {
461 json result;
462 result["path"] = path.string();
463
464 fitsfile* fptr = nullptr;
465 int status = 0;
466 if (fits_open_file(&fptr, path.string().c_str(), READONLY, &status)) {
467 result["ok"] = false;
468 result["error"] = "Cannot open FITS file";
469 return result;
470 }
471
472 int naxis = 0;
473 long naxes[3] = {0, 0, 0};
474 int bitpix = 0;
475 fits_get_img_param(fptr, 3, &bitpix, &naxis, naxes, &status);
476 if (status || naxis < 2) {
477 fits_close_file(fptr, &status);
478 result["ok"] = false;
479 result["error"] = "Cannot read FITS image parameters";
480 return result;
481 }
482
483 result["ok"] = true;
484 result["bitpix"] = bitpix;
485 result["naxis"] = naxis;
486 result["naxes"] = json::array();
487 for (int i = 0; i < naxis; ++i) result["naxes"].push_back(static_cast<int64_t>(naxes[i]));
488
489 long width = naxes[0];
490 long height = naxes[1];
491 long depth = (naxis >= 3) ? naxes[2] : 1;
492 const int64_t plane_pixels = static_cast<int64_t>(width) * static_cast<int64_t>(height);
493 const int64_t total_pixels = plane_pixels * static_cast<int64_t>(depth);
494 result["pixels"] = total_pixels;
495
496 std::vector<float> buffer(static_cast<size_t>(total_pixels));
497 long fpixel[3] = {1, 1, 1};
498 if (fits_read_pix(fptr, TFLOAT, fpixel, static_cast<long>(total_pixels), nullptr, buffer.data(), nullptr, &status)) {
499 fits_close_file(fptr, &status);
500 result["ok"] = false;
501 result["error"] = "Cannot read FITS pixel data";
502 return result;
503 }
504 fits_close_file(fptr, &status);
505
506 result["stats"] = compute_fits_stats_buffer(buffer);
507
508 if (depth > 1) {
509 result["per_plane_stats"] = json::array();
510 for (long z = 0; z < depth; ++z) {
511 const int64_t start = static_cast<int64_t>(z) * plane_pixels;
512 const int64_t end = start + plane_pixels;
513 std::vector<float> plane;
514 plane.reserve(static_cast<size_t>(plane_pixels));
515 for (int64_t i = start; i < end; ++i) plane.push_back(buffer[static_cast<size_t>(i)]);
516 json pj = compute_fits_stats_buffer(plane);
517 pj["plane"] = z;
518 result["per_plane_stats"].push_back(pj);
519 }
520 }
521
522 return result;
523}

763 return 1;
764 }
765 json r;
766 r["ok"] = true;
767 r["input"] = in_path;
768 r["output"] = out_path;
769 r["migration"] = json::parse(report.to_json_string());
770 print_json(r);
771 return 0;
772}

850int cmd_scan(const std::string& input_path, int frames_min, bool with_checksums) {
851 fs::path p(input_path);
852
853 json result;
854 result["ok"] = false;
855 result["input_path"] = input_path;
856 result["frames_detected"] = 0;
857 result["frames"] = json::array();
858 result["errors"] = json::array();
859 result["warnings"] = json::array();
860 result["color_mode"] = "UNKNOWN";
861 result["bayer_pattern"] = nullptr;
862 result["color_mode_candidates"] = json::array();
863 result["requires_user_confirmation"] = false;
864 result["image_width"] = 0;
865 result["image_height"] = 0;
866
867 if (!fs::exists(p)) {
868 json err;
869 err["severity"] = "error";
870 err["code"] = "input_path_not_found";
871 err["message"] = "Input path does not exist: " + input_path;
872 result["errors"].push_back(err);
873 print_json(result);
874 return 0;
875 }
876
877 if (!fs::is_directory(p)) {
878 json err;
879 err["severity"] = "error";
880 err["code"] = "input_path_not_directory";
881 err["message"] = "Input path is not a directory: " + input_path;
882 result["errors"].push_back(err);
883 print_json(result);
884 return 0;
885 }
886
887 auto files = find_fits_files(p);
888 result["frames_detected"] = static_cast<int>(files.size());
889
890 if (static_cast<int>(files.size()) < frames_min) {
891 json err;
892 err["severity"] = "error";
893 err["code"] = "too_few_frames";
894 err["message"] = "frames_detected (" + std::to_string(files.size()) +
895 ") < frames_min (" + std::to_string(frames_min) + ")";
896 result["errors"].push_back(err);
897 }
898
899 // Scan FITS headers
900 int image_width = 0;
901 int image_height = 0;
902 bool has_bayerpat = false;
903 bool has_rgb_cube = false;
904 std::string bayer_pattern;
905 bool bayer_pattern_inconsistent = false;
906 std::map<std::string, int> bayerpat_counts;
907
908 for (const auto& f : files) {
909 json frame;
910 frame["file_name"] = f.filename().string();
911 frame["abs_path"] = f.string();
912
913 if (with_checksums) {
914 frame["sha256"] = compute_sha256_file(f);
915 }
916
917 // Read FITS header
918 FitsHeaderInfo info = read_fits_header_info(f);
919
920 if (info.read_error) {
921 json err;
922 err["severity"] = "error";
923 err["code"] = "fits_read_error";
924 err["message"] = "Failed to read FITS header for " + f.filename().string() + ": " + info.error_msg;
925 result["errors"].push_back(err);
926 continue;
927 }
928
929 if (info.naxis1 <= 0 || info.naxis2 <= 0) {
930 json err;
931 err["severity"] = "error";
932 err["code"] = "fits_missing_axis";
933 err["message"] = "Missing or invalid NAXIS1/NAXIS2 in FITS header for " + f.filename().string();
934 result["errors"].push_back(err);
935 continue;
936 }
937
938 // Check dimension consistency
939 if (image_width == 0) {
940 image_width = info.naxis1;
941 image_height = info.naxis2;
942 } else {
943 if (info.naxis1 != image_width || info.naxis2 != image_height) {
944 json err;
945 err["severity"] = "error";
946 err["code"] = "inconsistent_image_dimensions";
947 err["message"] = "Inconsistent image size: expected " + std::to_string(image_width) + "x" +
948 std::to_string(image_height) + ", got " + std::to_string(info.naxis1) + "x" +
949 std::to_string(info.naxis2) + " in " + f.filename().string();
950 result["errors"].push_back(err);
951 }
952 }
953
954 if (info.naxis >= 3) {
955 has_rgb_cube = true;
956 }
957
958 // Track BAYERPAT
959 if (info.has_bayerpat) {
960 has_bayerpat = true;
961 const std::string& bp = info.bayerpat;
962
963 // Check if it's a valid Bayer pattern
964 if (bp == "RGGB" || bp == "BGGR" || bp == "GBRG" || bp == "GRBG") {
965 bayerpat_counts[bp]++;
966
967 if (bayer_pattern.empty()) {
968 bayer_pattern = bp;
969 } else if (bayer_pattern != bp) {
970 bayer_pattern_inconsistent = true;
971 }
972 }
973 }
974
975 if (info.exptime >= 0) frame["exposure_seconds"] = info.exptime;
976 if (info.gain >= 0) frame["gain"] = info.gain;
977 if (info.ccd_temp > -900) frame["temperature_c"] = info.ccd_temp;
978 if (!info.object.empty()) frame["target"] = info.object;
979 if (!info.instrume.empty()) frame["camera"] = info.instrume;
980 if (!info.telescop.empty()) frame["telescope"] = info.telescop;
981
982 result["frames"].push_back(frame);
983 }
984
985 result["image_width"] = image_width;
986 result["image_height"] = image_height;
987
988 // Determine color mode
989 bool requires_user_confirmation = false;
990 std::string color_mode = "UNKNOWN";
991 std::vector<std::string> candidates;
992
993 const bool has_readable_frames = (image_width > 0 && image_height > 0);
994
995 if (has_readable_frames) {
996 if (has_rgb_cube) {
997 candidates.push_back("RGB");
998 }
999 if (has_bayerpat) {
1000 candidates.push_back("OSC");
1001 }
1002 if (!has_rgb_cube && !has_bayerpat) {
1003 // No RGB cube and no CFA hint: default to MONO (same fallback philosophy
1004 // as pipeline SCAN_INPUT for hint-less FITS).
1005 candidates.push_back("MONO");
1006 }
1007
1008 if (candidates.empty()) {
1009 candidates.push_back("MONO");
1010 }
1011
1012 color_mode = candidates.front();
1013 }
1014
1015 if (has_readable_frames && candidates.size() > 1) {
1016 requires_user_confirmation = true;
1017 color_mode = "UNKNOWN";
1018 json warn;
1019 warn["severity"] = "warning";
1020 warn["code"] = "color_mode_ambiguous";
1021 warn["message"] = "Mixed color-mode hints across FITS headers (RGB/CFA); color_mode requires user confirmation";
1022 result["warnings"].push_back(warn);
1023 }
1024
1025 if (has_readable_frames && has_bayerpat && bayer_pattern_inconsistent) {
1026 requires_user_confirmation = true;
1027 json warn;
1028 warn["severity"] = "warning";
1029 warn["code"] = "bayer_pattern_inconsistent";
1030 warn["message"] = "BAYERPAT differs across frames; bayer_pattern requires user confirmation";
1031 result["warnings"].push_back(warn);
1032 }
1033
1034 result["color_mode"] = color_mode;
1035 if (bayer_pattern.empty()) {
1036 result["bayer_pattern"] = nullptr;
1037 } else {
1038 result["bayer_pattern"] = bayer_pattern;
1039 }
1040 result["color_mode_candidates"] = json(candidates);
1041 result["requires_user_confirmation"] = requires_user_confirmation;
1042
1043 // Check if we have any readable frames
1044 if (image_width == 0 || image_height == 0) {
1045 json err;
1046 err["severity"] = "error";
1047 err["code"] = "no_readable_frames";
1048 err["message"] = "No readable FITS frames found";
1049 result["errors"].push_back(err);
1050 }
1051
1052 if (result["errors"].empty()) {
1053 result["ok"] = true;
1054 }
1055
1056 print_json(result);
1057 return 0;
1058}

1068int cmd_scan_metrics(const std::string& input_path) {
1069 namespace io = tile_compile::io;
1070 namespace metrics = tile_compile::metrics;
1071
1072 fs::path p(input_path);
1073 json result;
1074 result["ok"] = false;
1075 result["input_path"] = input_path;
1076 result["frames"] = json::array();
1077 result["aggregate"] = json::object();
1078
1079 if (!fs::exists(p) || !fs::is_directory(p)) {
1080 result["error"] = "Input path does not exist or is not a directory";
1081 print_json(result);
1082 return 1;
1083 }
1084
1085 auto all_files = find_fits_files(p);
1086 const int total = static_cast<int>(all_files.size());
1087 result["frames_total"] = total;
1088
1089 if (total == 0) {
1090 result["error"] = "No FITS files found";
1091 print_json(result);
1092 return 1;
1093 }
1094
1095 std::vector<FitsHeaderInfo> header_infos;
1096 header_infos.reserve(all_files.size());
1097 for (const auto& file : all_files) header_infos.push_back(read_fits_header_info(file));
1098
1099 // All frames are processed — no sampling.
1100 std::vector<size_t> indices;
1101 indices.reserve(static_cast<size_t>(total));
1102 for (int i = 0; i < total; ++i) indices.push_back(static_cast<size_t>(i));
1103 const int sample_count = total;
1104 result["sample_count"] = sample_count;
1105
1106 // Per-frame results storage
1107 struct FrameResult {
1108 int frame_index = -1;
1109 std::string file_name;
1110 float background = 0.0f;
1111 float noise = 0.0f;
1112 float gradient_energy = 0.0f;
1113 float sky_gradient = 0.0f;
1114 float fwhm = 0.0f;
1115 float fwhm_x = 0.0f;
1116 float fwhm_y = 0.0f;
1117 float roundness = 0.0f;
1118 int star_count = 0;
1119 bool ok = false;
1120 std::string error;
1121 };
1122 std::vector<FrameResult> results(indices.size());
1123
1124 // Process frames in parallel
1125 const int hw_threads = std::max(1, static_cast<int>(std::thread::hardware_concurrency()));
1126 const int workers = std::min(hw_threads, static_cast<int>(indices.size()));
1127 std::atomic<size_t> next_idx{0};
1128 std::atomic<size_t> done_count{0};
1129
1130 auto worker_fn = [&]() {
1131 while (true) {
1132 const size_t wi = next_idx.fetch_add(1);
1133 if (wi >= indices.size()) break;
1134 const size_t frame_idx = indices[wi];
1135 FrameResult& fr = results[wi];
1136 fr.frame_index = static_cast<int>(frame_idx);
1137 fr.file_name = all_files[frame_idx].filename().string();
1138 try {
1139 tile_compile::Matrix2Df img = io::read_fits_pixels_float(all_files[frame_idx]);
1140 if (img.size() <= 0) {
1141 fr.error = "empty_image";
1142 done_count.fetch_add(1);
1143 continue;
1144 }
1145 auto fm = metrics::calculate_frame_metrics(img);
1146 fr.background = fm.background;
1147 fr.noise = fm.noise;
1148 fr.gradient_energy = fm.gradient_energy;
1149 fr.sky_gradient = fm.sky_gradient;
1150
1151 auto sm = metrics::measure_frame_stars(img, 0);
1152 fr.fwhm = sm.fwhm;
1153 fr.fwhm_x = sm.fwhm_x;
1154 fr.fwhm_y = sm.fwhm_y;
1155 fr.roundness = sm.roundness;
1156 fr.star_count = sm.star_count;
1157 fr.ok = true;
1158 } catch (const std::exception& e) {
1159 fr.error = e.what();
1160 }
1161 done_count.fetch_add(1);
1162 }
1163 };
1164
1165 std::cerr << "[scan-metrics] Analyzing all " << total
1166 << " frames using " << workers << " threads..." << std::endl;
1167
1168 std::atomic<bool> progress_done{false};
1169 std::thread progress_thread([&]() {
1170 while (!progress_done.load()) {
1171 std::this_thread::sleep_for(std::chrono::seconds(1));
1172 const size_t done = done_count.load();
1173 std::cerr << "[scan-metrics] " << done << " / " << total << " frames analyzed"
1174 << (done == static_cast<size_t>(total) ? " - done." : "...") << std::endl;
1175 }
1176 });
1177
1178 if (workers > 1) {
1179 std::vector<std::thread> threads;
1180 threads.reserve(static_cast<size_t>(workers));
1181 for (int w = 0; w < workers; ++w) threads.emplace_back(worker_fn);
1182 for (auto& t : threads) t.join();
1183 } else {
1184 worker_fn();
1185 }
1186 progress_done.store(true);
1187 progress_thread.join();
1188 std::cerr << "[scan-metrics] " << done_count.load() << " / " << total
1189 << " frames completed." << std::endl;
1190
1191 // Build output
1192 std::vector<float> all_bg, all_noise, all_grad, all_sky, all_fwhm, all_round;
1193 std::vector<int> all_stars;
1194 int ok_count = 0;
1195
1196 for (const auto& fr : results) {
1197 json fj;
1198 fj["frame_index"] = fr.frame_index;
1199 fj["file_name"] = fr.file_name;
1200 fj["ok"] = fr.ok;
1201 if (fr.frame_index >= 0 && static_cast<size_t>(fr.frame_index) < header_infos.size()) {
1202 const auto& hi = header_infos[static_cast<size_t>(fr.frame_index)];
1203 json hj = json::object();
1204 if (!hi.read_error) {
1205 if (hi.exptime >= 0) hj["exposure_seconds"] = hi.exptime;
1206 if (hi.gain >= 0) hj["gain"] = hi.gain;
1207 if (hi.ccd_temp > -900) hj["temperature_c"] = hi.ccd_temp;
1208 if (!hi.filter.empty()) hj["filter"] = hi.filter;
1209 if (!hi.object.empty()) hj["target"] = hi.object;
1210 if (!hi.instrume.empty()) hj["camera"] = hi.instrume;
1211 if (!hi.date_obs.empty()) hj["date_obs"] = hi.date_obs;
1212 if (hi.ra_deg > -900) hj["ra_deg"] = hi.ra_deg;
1213 if (hi.dec_deg > -900) hj["dec_deg"] = hi.dec_deg;
1214 } else {
1215 hj["error"] = hi.error_msg;
1216 }
1217 if (!hj.empty()) fj["header"] = std::move(hj);
1218 }
1219 if (fr.ok) {
1220 fj["background"] = fr.background;
1221 fj["noise"] = fr.noise;
1222 fj["gradient_energy"] = fr.gradient_energy;
1223 fj["sky_gradient"] = fr.sky_gradient;
1224 fj["fwhm"] = fr.fwhm;
1225 fj["roundness"] = fr.roundness;
1226 fj["star_count"] = fr.star_count;
1227 all_bg.push_back(fr.background);
1228 all_noise.push_back(fr.noise);
1229 all_grad.push_back(fr.gradient_energy);
1230 all_sky.push_back(fr.sky_gradient);
1231 if (fr.fwhm > 0) all_fwhm.push_back(fr.fwhm);
1232 if (fr.roundness > 0) all_round.push_back(fr.roundness);
1233 all_stars.push_back(fr.star_count);
1234 ++ok_count;
1235 } else {
1236 fj["error"] = fr.error;
1237 }
1238 result["frames"].push_back(fj);
1239 }
1240
1241 // Aggregate statistics
1242 auto agg_stats = [](std::vector<float>& vals) -> json {
1243 if (vals.empty()) return nullptr;
1244 std::sort(vals.begin(), vals.end());
1245 float sum = 0;
1246 for (float v : vals) sum += v;
1247 float mean = sum / static_cast<float>(vals.size());
1248 float median = vals[vals.size() / 2];
1249 float p10 = vals[std::min(vals.size()-1, static_cast<size_t>(vals.size() * 0.1))];
1250 float p90 = vals[std::min(vals.size()-1, static_cast<size_t>(vals.size() * 0.9))];
1251 return json{{"min", vals.front()}, {"max", vals.back()},
1252 {"mean", mean}, {"median", median},
1253 {"p10", p10}, {"p90", p90}, {"count", static_cast<int>(vals.size())}};
1254 };
1255
1256 json agg;
1257 agg["background"] = agg_stats(all_bg);
1258 agg["noise"] = agg_stats(all_noise);
1259 agg["gradient_energy"] = agg_stats(all_grad);
1260 agg["sky_gradient"] = agg_stats(all_sky);
1261 agg["fwhm"] = agg_stats(all_fwhm);
1262 agg["roundness"] = agg_stats(all_round);
1263 if (!all_stars.empty()) {
1264 std::sort(all_stars.begin(), all_stars.end());
1265 int sum = 0;
1266 for (int v : all_stars) sum += v;
1267 agg["star_count"] = json{{"min", all_stars.front()}, {"max", all_stars.back()},
1268 {"mean", static_cast<float>(sum) / static_cast<float>(all_stars.size())},
1269 {"median", all_stars[all_stars.size() / 2]},
1270 {"count", static_cast<int>(all_stars.size())}};
1271 }
1272 agg["frames_ok"] = ok_count;
1273 agg["frames_failed"] = static_cast<int>(results.size()) - ok_count;
1274
1275 // Session geometry: time span, RA/DEC, estimated field rotation
1276 {
1277 json geom = json::object();
1278 // Collect DATE-OBS timestamps
1279 std::string first_date, last_date;
1280 for (const auto& hi : header_infos) {
1281 if (!hi.date_obs.empty()) {
1282 if (first_date.empty() || hi.date_obs < first_date) first_date = hi.date_obs;
1283 if (last_date.empty() || hi.date_obs > last_date) last_date = hi.date_obs;
1284 }
1285 }
1286 if (!first_date.empty()) geom["first_date_obs"] = first_date;
1287 if (!last_date.empty()) geom["last_date_obs"] = last_date;
1288
1289 // Median RA/DEC (target coordinates)
1290 std::vector<double> ra_vals, dec_vals;
1291 for (const auto& hi : header_infos) {
1292 if (hi.ra_deg > -900 && hi.dec_deg > -900) {
1293 ra_vals.push_back(hi.ra_deg);
1294 dec_vals.push_back(hi.dec_deg);
1295 }
1296 }
1297 if (!ra_vals.empty()) {
1298 std::sort(ra_vals.begin(), ra_vals.end());
1299 std::sort(dec_vals.begin(), dec_vals.end());
1300 double med_ra = ra_vals[ra_vals.size() / 2];
1301 double med_dec = dec_vals[dec_vals.size() / 2];
1302 geom["target_ra_deg"] = med_ra;
1303 geom["target_dec_deg"] = med_dec;
1304 geom["ra_dec_available"] = true;
1305
1306 // Estimate field rotation if we have time span
1307 // Field rotation rate for Alt/Az mount: dθ/dt = |cos(lat) * cos(declination) * cos(H) / sin(altitude)|
1308 // where H is hour angle. For a rough estimate we use the parallactic angle formula.
1309 // Without latitude we assume mid-latitude (45°N) as default.
1310 // The total rotation over the session is approximately:
1311 // Δθ ≈ 15°/hr * cos(declination) * session_duration_hours * sin(parallactic_angle_factor)
1312 // For a conservative upper bound we use: Δθ ≈ 15°/hr * cos(dec) * duration_hours
1313 if (!first_date.empty() && !last_date.empty()) {
1314 // Parse ISO 8601 timestamps and compute duration
1315 auto parse_iso = [](const std::string& s) -> double {
1316 // Expected format: YYYY-MM-DDThh:mm:ss[.sss]Z or with timezone
1317 struct tm tm_val{};
1318 double frac = 0.0;
1319 char zone[16] = {0};
1320 int matched = sscanf(s.c_str(), "%d-%d-%dT%d:%d:%lf%15s",
1321 &tm_val.tm_year, &tm_val.tm_mon, &tm_val.tm_mday,
1322 &tm_val.tm_hour, &tm_val.tm_min, &frac, zone);
1323 if (matched < 6) return -1.0;
1324 tm_val.tm_year -= 1900;
1325 tm_val.tm_mon -= 1;
1326#ifdef _WIN32
1327 time_t t = _mkgmtime(&tm_val);
1328#else
1329 time_t t = timegm(&tm_val);
1330#endif
1331 return static_cast<double>(t) + frac;
1332 };
1333 double t0 = parse_iso(first_date);
1334 double t1 = parse_iso(last_date);
1335 if (t0 >= 0 && t1 > t0) {
1336 double duration_hours = (t1 - t0) / 3600.0;
1337 geom["session_duration_hours"] = duration_hours;
1338
1339 // Earth's rotation rate: 15°/hr
1340 // For Alt/Az mount, field rotation rate depends on parallactic angle.
1341 // Maximum rotation rate (worst case): 15°/hr * cos(dec)
1342 // Typical rate at 45°N latitude for dec=0: ~10.6°/hr
1343 // We provide both the theoretical max and a mid-latitude estimate.
1344 double dec_rad = med_dec * M_PI / 180.0;
1345 double max_rotation_rate = 15.0 * std::cos(std::abs(dec_rad)); // °/hr
1346 double max_rotation = max_rotation_rate * duration_hours;
1347 geom["estimated_max_field_rotation_deg"] = max_rotation;
1348 geom["estimated_field_rotation_note"] =
1349 "Maximum theoretical field rotation for Alt/Az mount. "
1350 "Actual rotation depends on latitude and parallactic angle. "
1351 "For equatorial mounts, field rotation is negligible.";
1352 }
1353 }
1354 } else {
1355 geom["ra_dec_available"] = false;
1356 }
1357 result["session_geometry"] = geom;
1358 }
1359
1360 result["aggregate"] = agg;
1361 result["ok"] = (ok_count > 0);
1362
1363 print_json(result);
1364 return 0;
1365}

1374int cmd_list_runs(const std::string& runs_dir) {

1581 result["wcs_file"] = wcs_path;
1582
1583 try {
1584 const std::string source_requested = source.empty() ? "auto" : to_lower_copy(source);
1585 result["source_requested"] = source_requested;
1586 result["catalog_dir"] = siril_catalog_dir.empty() ? astro::default_siril_gaia_catalog_dir() : siril_catalog_dir;
1587
1588 astro::WCS wcs = astro::parse_wcs_file(wcs_path);

1590 result["ok"] = false;
1591 result["error"] = "Invalid WCS solution";

1593 return 1;

1597 {"ra_deg", wcs.crval1},
1598 {"dec_deg", wcs.crval2},
1599 {"pixel_scale_arcsec", wcs.pixel_scale_arcsec()},
1600 {"image_width", wcs.naxis1},
1601 {"image_height", wcs.naxis2},
1602 };

1612 return false;

1615 if (!stars.empty()) {
1616 used_source = "siril";
1617 return true;
1618 }
1619 return false;
1620 };
1621
1622 auto try_vizier_gaia = [&]() -> bool {
1623 stars = astro::vizier_gaia_cone_search(wcs.crval1, wcs.crval2, search_r, config.mag_limit);

1690 result["estimated_fwhm_px"] = estimated_fwhm_px;

1696 result["residual_rms"] = pcc_result.residual_rms;
1697 result["determinant"] = pcc_result.determinant;
1698 result["condition_number"] = pcc_result.condition_number;
1699 result["apply_mode"] = pcc_result.apply_mode;
1700 result["apply_attenuation"] = resolved_config.apply_attenuation;

1710 pcc_result.matrix[r][0],
1711 pcc_result.matrix[r][1],
1712 pcc_result.matrix[r][2],
1713 }));
1714 }
1715 result["matrix"] = matrix_json;
1716
1717 if (!pcc_result.success) {
1718 result["ok"] = false;
1719 result["error"] = pcc_result.error_message;

1742 print_json(result);
1743 return 0;
1744 } catch (const std::exception& e) {
1745 result["ok"] = false;
1746 result["error"] = e.what();
1747 print_json(result);

1749 }
