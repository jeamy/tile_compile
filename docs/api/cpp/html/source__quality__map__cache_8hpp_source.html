3// Multi-stream source-space quality-map cache --- milestone M5 of the
4// CFA-forward-drizzle plan (docs/AQMH/aqmh_cfa_forward_drizzle_multiband_
5// implementierungsplan_de.md, sections 13.3-13.5).
6//
7// Layout (plan 13.4):
8// cache/source_quality_maps/
9// metadata.json <- sole commit point
10// composite/source_quality_composite_NNNNNN.bin
11// scale_0/ .. scale_3/ source_quality_sK_NNNNNN.bin
12// artifact/source_quality_artifact_NNNNNN.bin
13//
14// Each .bin is a small header + row-major uint16 on the storage grid
15// (source geometry downsampled by storage_divisor). Value semantics: a
16// reserved code 0 = veto (Q=0 or unsupported); positive quality in (0,1] maps
17// to [1,65535]. This guarantees an exact zero-veto never decodes positive and
18// no nonzero code decodes to 0 (plan 13.5).

110 // Store one source-geometry map (values in [0,1], NaN/<=0 = veto) for one
111 // frame into one stream ("composite", "scale_0".."scale_3", "artifact").

139// Reader. Fail-closed: usable() is true only when metadata.json parses, its
140// schema/identity/config hashes match what is expected, its declared
141// source_quality_cache_hash recomputes, and every listed .bin file exists
142// with the expected size (T1: trusted run, no content hash).
143//
144// T2: an unordered_map keyed by (stream, source_index) is built once in the
145// constructor so has() and file_path() are O(1) lookups instead of linear
146// scans over meta_.files. The sorted meta_.files order is preserved for
147// deterministic metadata identity (cache_manifest_hash).
148struct SourceQualityFileKey {
149 std::string stream;
150 std::size_t source_index = 0;
151 bool operator==(const SourceQualityFileKey &o) const {
152 return source_index == o.source_index && stream == o.stream;
153 }
154};
155struct SourceQualityFileKeyHash {
156 std::size_t operator()(const SourceQualityFileKey &k) const noexcept {
157 std::size_t h = std::hash<std::size_t>{}(k.source_index);
158 h ^= std::hash<std::string>{}(k.stream) + 0x9e3779b9u + (h << 6) + (h >> 2);
159 return h;
160 }
161};
162// Raw storage-grid window of one (stream, frame) without float expansion:
163// the uint16 cell values and veto flags covering the requested source rect,
164// plus the absolute storage-grid origin/dims/divisor needed to decode a
165// source pixel by (x / divisor - storage_x_begin, y / divisor -
166// storage_y_begin). The source_* fields record the clamped requested rect.
167struct SourceQualityPackedWindow {
168 int storage_x_begin = 0;
169 int storage_y_begin = 0;
170 int storage_width = 0;
171 int storage_height = 0;
172 int storage_divisor = 1;
173 int source_x_begin = 0;
174 int source_y_begin = 0;
175 int source_width = 0;
176 int source_height = 0;
177 std::vector<std::uint16_t> cells; // storage_width*storage_height
178 std::vector<std::uint8_t> veto; // same extent
179};

203
204 // Rectangle read (§30.81 step 3a-2): decode only the source rectangle
205 // [y0, y1) x [x0, x1). Returns a (y1 - y0) x (x1 - x0) map whose element
206 // (y - y0, x - x0) equals read_full()(y, x) for every (y, x) in the rect ---
207 // the storage-grid cell is picked from the ABSOLUTE (y, x), so the
208 // edge-clamp against the last storage row/column is unchanged. Ranges are
209 // clamped to the source geometry; an empty rect returns Matrix2Df(0, 0).

211 int y0, int y1, int x0, int x1) const;

213 // Compact rectangle read: identical coverage/counters as read_rect
214 // (bin_loads, bin_cells_decoded) but returns the raw window cells/veto
215 // without materialising the float map --- expanded_floats is NOT
216 // incremented. Expanding the returned window per source pixel reproduces
217 // read_rect() exactly. An empty rect returns a zero-sized window.
218 SourceQualityPackedWindow read_packed_rect(const std::string &stream,
219 std::size_t source_index,

223 // Same contract as read_packed_rect, but fills a reusable caller window:
224 // metadata fields are reset and cells/veto are resized (capacity kept).
225 // Storage rows are read straight into the output spans --- no BinWindow,
226 // row scratch buffer, or returned-vector copy. Counters identical.
227 void read_packed_rect_into(const std::string &stream,
228 std::size_t source_index, int y0, int y1,
229 int x0, int x1,
230 SourceQualityPackedWindow &out) const;
231
232 // Tranche 8: aligned sample read for the canonical ragged affine path.
233 // `spans` is the canonical ascending-source_y list of half-open row
234 // intervals (non-decreasing storage rows required). The .bin is opened
235 // once; for each touched storage row the union x interval covering every
236 // needed cell is seek-read exactly once into `scratch` (divisor-2
237 // duplicate source rows share the same storage row without a second
238 // read). `cells`/`veto` are filled 1:1 with the active samples in span
239 // order: sample (sx, sy) gets the cell at (sy/divisor, sx/divisor).
240 // expanded_floats stays untouched; bin_cells_decoded counts the unique
241 // storage cells read, not the aligned output length. All buffers keep
242 // capacity (pre-reserve for an allocation-free hot path).
243 void read_packed_samples_into(

246 std::vector<std::uint16_t> &cells, std::vector<std::uint8_t> &veto,
247 SourceQualityPackedWindow &scratch) const;

249 // §30.81 step 3a-2 / 3a-2b: I/O accounting for the read_rect path.

252 // covering the rect, vs storage_w*storage_h whole-grid)

261 }
