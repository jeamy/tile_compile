20double median_of(std::vector<float> v) {
21 if (v.empty()) return 0.0;
22 const std::size_t mid = v.size() / 2;
23 std::nth_element(v.begin(), v.begin() + mid, v.end());
24 double m = v[mid];
25 if (v.size() % 2 == 0) {
26 const double lo = *std::max_element(v.begin(), v.begin() + mid);
27 m = 0.5 * (m + lo);
28 }
29 return m;
30}

32double percentile_of(std::vector<float> v, double p) {
33 if (v.empty()) return 0.0;
34 const std::size_t k = std::min(
35 v.size() - 1, static_cast<std::size_t>(p * static_cast<double>(v.size() - 1)));
36 std::nth_element(v.begin(), v.begin() + k, v.end());
37 return v[k];

40// Fills the cells NOT marked ok in-place, from nearby ok cells only (a few rounds of a small masked
41// blur, i.e. a short diffusion/push-outward from the known values), so the median filter right after
42// this never sees a value pulled from the WHOLE image. A single global constant is wrong here: for a
43// large excluded zone (e.g. a bright star's full sigma_px-scaled margin, which can span several whole
44// coarse cells with no real coverage left at all) it pulls those cells toward the whole image's median
45// regardless of the true local brightness nearby -- genuinely brighter nebula, or the excluded star's
46// own immediate surroundings -- and that then leaks into the 5x5 median filter and shows up as a dark
47// disc right where the star was (2026-09-28/29, real IC434 data). A handful of iterations is enough to
48// reach across any real star's margin; a cell still unreached after that (e.g. deep inside a much
49// larger masked-out region such as the frame border) keeps the global median as a last resort.
50// `grounded` (if given) marks every cell this diffusion actually reached with real signal --
51// the original cell_ok cells plus every cell the 8 rounds below pulled a value into -- as opposed
52// to a cell that only got the last-resort `global_fill` because nothing real was reachable at all
53// (e.g. deep inside the image border). Callers that need to know which cells now carry a
54// trustworthy value, not just a filled-in placeholder, should use `grounded`, not `cell_ok`: a
55// dense cluster of overlapping star-exclusion margins (e.g. a bright star with several fainter
56// companions nearby) can leave a combined excluded area wider than the fixed-scale second-stage
57// blur below can reach on its own, even though THIS diffusion -- which iterates outward rather
58// than using one single fixed-radius blur -- already reached across it (2026-09-29, real IC434
59// data: Alnitak plus nearby companions left a dark, wrong-shaped hole in the boost right where the
60// combined margin was too wide for the old single-pass reach test, even after the per-star margin
61// and the cell_ok-drop fixes).
62void fill_from_neighbours(cv::Mat &small, const cv::Mat &cell_ok, float global_fill, cv::Mat *grounded = nullptr) {
63 cv::Mat weight;
64 cell_ok.convertTo(weight, CV_32F, 1.0 / 255.0);
65 cv::Mat value = small.mul(weight);
66 for (int iter = 0; iter < 8; ++iter) {
67 cv::Mat value_blur, weight_blur;
68 cv::GaussianBlur(value, value_blur, cv::Size(0, 0), 2.0, 2.0, cv::BORDER_REPLICATE);
69 cv::GaussianBlur(weight, weight_blur, cv::Size(0, 0), 2.0, 2.0, cv::BORDER_REPLICATE);
70 for (int y = 0; y < small.rows; ++y)
71 for (int x = 0; x < small.cols; ++x) {
72 if (weight.at<float>(y, x) >= 0.999f) continue; // already fully known, do not dilute it
73 const float w = weight_blur.at<float>(y, x);
74 if (w > 1e-6f) {
75 value.at<float>(y, x) = value_blur.at<float>(y, x);
76 weight.at<float>(y, x) = std::min(1.0f, w);
77 }
78 }
79 }
80 if (grounded) *grounded = cv::Mat(small.rows, small.cols, CV_8U, cv::Scalar(0));
81 for (int y = 0; y < small.rows; ++y)
82 for (int x = 0; x < small.cols; ++x)
83 if (!cell_ok.at<std::uint8_t>(y, x)) {
84 const float w = weight.at<float>(y, x);
85 const bool reached = w > 1e-3f;
86 small.at<float>(y, x) = reached ? value.at<float>(y, x) / w : global_fill;
87 if (grounded) grounded->at<std::uint8_t>(y, x) = reached ? 255 : 0;
88 } else if (grounded) {
89 grounded->at<std::uint8_t>(y, x) = 255;
90 }
91}

93// True for every cell whose not-cell_ok connected component (4-connectivity) never touches the
94// grid's outer border. A bright-star exclusion (including several overlapping ones merged into one
95// blob) is always an ISLAND fully surrounded by real coverage; a genuinely invalid area -- the
96// image's own edge, or an external mask cutting into the frame -- always touches the grid border
97// (it has no "far side" to be surrounded by). This tells the diffusion fallback below which excluded
98// cells it may fill in (islands) and which it must leave alone (border-touching), so a real border
99// still gets a hard, unextrapolated edge instead of a spurious boost bleeding in from extrapolated
100// values (caught by a unit test, 2026-09-29).
101cv::Mat interior_islands(const cv::Mat &cell_ok) {
102 cv::Mat not_ok = cell_ok == 0;
103 cv::Mat labels;
104 const int n = cv::connectedComponents(not_ok, labels, 4, CV_32S);
105 std::vector<std::uint8_t> touches_border(static_cast<std::size_t>(n), 0);
106 const int rows = labels.rows, cols = labels.cols;
107 for (int x = 0; x < cols; ++x) {
108 touches_border[static_cast<std::size_t>(labels.at<std::int32_t>(0, x))] = 1;
109 touches_border[static_cast<std::size_t>(labels.at<std::int32_t>(rows - 1, x))] = 1;
110 }
111 for (int y = 0; y < rows; ++y) {
112 touches_border[static_cast<std::size_t>(labels.at<std::int32_t>(y, 0))] = 1;
113 touches_border[static_cast<std::size_t>(labels.at<std::int32_t>(y, cols - 1))] = 1;
114 }
115 cv::Mat island(rows, cols, CV_8U, cv::Scalar(0));
116 for (int y = 0; y < rows; ++y)
117 for (int x = 0; x < cols; ++x) {
118 const std::int32_t lbl = labels.at<std::int32_t>(y, x);
119 if (lbl != 0 && !touches_border[static_cast<std::size_t>(lbl)]) island.at<std::uint8_t>(y, x) = 1;
120 }
121 return island;
122}

124CoarseMap build_map(const cv::Mat &plane, const cv::Mat &valid, int factor, float sigma_px) {
125 CoarseMap out;
126 const int rows = std::max(1, plane.rows / factor), cols = std::max(1, plane.cols / factor);
127 out.rows = rows;
128 out.cols = cols;
129 cv::Mat num, den, valid_f;
130 valid.convertTo(valid_f, CV_32F);
131 cv::multiply(plane, valid_f, num);
132 cv::Mat num_s, den_s;
133 cv::resize(num, num_s, cv::Size(cols, rows), 0, 0, cv::INTER_AREA);
134 cv::resize(valid_f, den_s, cv::Size(cols, rows), 0, 0, cv::INTER_AREA);
135 // A LOW bar on purpose: a cell only needs a little real coverage to compute its own local average
136 // from it. Originally 0.9 ("mostly inside the valid area"), which was fine when the only excluded
137 // region was one contiguous border strip -- every cell was either fully in or fully out. With
138 // bright_source_mask's small, scattered per-star exclusions, most affected coarse cells keep the
139 // bulk of their area from ordinary sky; forcing them to the >90 % bar instead marked them "not ok"
140 // and pulled them onto the GLOBAL median fill below, which then leaked into the 5x5 median filter's
141 // neighbourhood at nearby REAL cells and showed up as small dark boxes around faint stars near a
142 // brighter area (2026-09-28, real IC434 data). A cell is only truly unusable when it has next to no
143 // real coverage left, which for a small dilated star exclusion is rare.
144 cv::Mat cell_ok = den_s > 0.15f;
145 cv::Mat small(rows, cols, CV_32F, cv::Scalar(0));
146 for (int y = 0; y < rows; ++y)
147 for (int x = 0; x < cols; ++x)
148 if (cell_ok.at<std::uint8_t>(y, x)) small.at<float>(y, x) = num_s.at<float>(y, x) / den_s.at<float>(y, x);
149 std::vector<float> usable;
150 for (int y = 0; y < rows; ++y)
151 for (int x = 0; x < cols; ++x)
152 if (cell_ok.at<std::uint8_t>(y, x)) usable.push_back(small.at<float>(y, x));
153 const float global_fill = static_cast<float>(median_of(usable));
154 cv::Mat grounded;
155 fill_from_neighbours(small, cell_ok, global_fill, &grounded);
156 // Fallback eligibility (see interior_islands): only an interior excluded island may use the
157 // diffusion fallback below; a border-touching excluded area keeps the strict cell_ok-weighted
158 // Gaussian result even where that leaves it unmarked.
159 cv::Mat fallback_ok;
160 cv::bitwise_and(grounded, interior_islands(cell_ok), fallback_ok);
161 cv::Mat med;
162 if (rows >= 5 && cols >= 5) cv::medianBlur(small, med, 5); // float median: ksize <= 5
163 else med = small;
164 // Masked (normalised) Gaussian: border cells are averaged only over usable neighbours. The weight
165 // here is `cell_ok` (real, directly-measured coverage only), deliberately NOT `grounded`: a
166 // diffusion-filled cell (e.g. deep inside a genuinely invalid image border) carries an
167 // extrapolated, not measured, value, and letting the smoothing Gaussian pull that value back
168 // into a real, fully-covered cell nearby is a small but measurable leak into otherwise flat sky
169 // right next to the border (caught by a unit test, 2026-09-29). `grounded` is used below only as
170 // a fallback for cells this Gaussian itself cannot reach.
171 const double sigma = std::max(0.5, static_cast<double>(sigma_px) / factor);
172 cv::Mat ok_f, med_ok, blur_num, blur_den;
173 cell_ok.convertTo(ok_f, CV_32F, 1.0 / 255.0);
174 cv::multiply(med, ok_f, med_ok);
175 cv::GaussianBlur(med_ok, blur_num, cv::Size(0, 0), sigma, sigma, cv::BORDER_CONSTANT);
176 cv::GaussianBlur(ok_f, blur_den, cv::Size(0, 0), sigma, sigma, cv::BORDER_CONSTANT);
177 out.map = cv::Mat(rows, cols, CV_32F, cv::Scalar(0));
178 out.ok = cv::Mat(rows, cols, CV_8U, cv::Scalar(0));
179 // Primary source: the cell_ok-weighted Gaussian above, wherever it has enough real coverage in
180 // reach (blur_den > 0.5) -- this is the properly smoothed estimate and is preferred whenever it's
181 // available. Fallback: `fallback_ok` (grounded AND an interior island, see interior_islands and
182 // fill_from_neighbours) for a cell this Gaussian's fixed reach cannot cover on its own -- typically
183 // a coarse cell deep inside a WIDE excluded area made of several overlapping bright_source_mask
184 // margins (e.g. a bright star with nearby fainter companions), which can be wider than this
185 // Gaussian's sigma even though the diffusion in fill_from_neighbours, which iterates outward rather
186 // than using one fixed-radius blur, already reached across it and left a real value in `med`.
187 // Without this fallback, such a cell got no value at all (out.map/out.ok stayed at their initial
188 // zero), producing a sharp, correctly-shaped but visibly dark hole exactly the size of the combined
189 // exclusion area (2026-09-29, real IC434 data, Alnitak plus nearby companions) -- the opposite
190 // failure from a global-median leak: not a wrong value, but no value at all. The `interior_islands`
191 // restriction keeps this from also applying to a genuinely invalid, border-touching area (e.g. the
192 // image edge): the diffusion in fill_from_neighbours has no "far side" to be surrounded by there and
193 // still marks it `grounded` after enough iterations, which would otherwise leak an extrapolated
194 // value into real sky right next to the border (caught by a unit test, 2026-09-29).
195 for (int y = 0; y < rows; ++y)
196 for (int x = 0; x < cols; ++x) {
197 const float d = blur_den.at<float>(y, x);
198 if (d > 0.5f) {
199 out.map.at<float>(y, x) = blur_num.at<float>(y, x) / d;
200 out.ok.at<std::uint8_t>(y, x) = 1;
201 } else if (fallback_ok.at<std::uint8_t>(y, x)) {
202 out.map.at<float>(y, x) = med.at<float>(y, x);
203 out.ok.at<std::uint8_t>(y, x) = 1;
204 }
205 }
206 return out;
207}

209// Bright-source (star) mask on the FULL-resolution luminance: POINT-LIKE bright pixels, and a margin
210// around them, are excluded from the coarse-map construction below (never from the final boost itself
211// -- a star still receives whatever value its surrounding sky gets). A bright, extended star's wings
212// can span several coarse cells at typical sigma_px (e.g. a factor-8 grid), so they survive the coarse
213// 5x5 median unless removed earlier; without this, the star's own bump is smoothed by the Gaussian and
214// added back as a soft halo -- exactly the "large-scale structure" this stage boosts, applied to a star
215// instead of a nebula. Detection is deliberately a LOCAL-contrast test (residual against a small-sigma
216// blur), not a plain "brighter than the image's global sky level" one: a real nebula core can be
217// brighter than the sky by far more than any fixed multiple of the noise, but it varies smoothly across
218// many multiples of the local blur scale, so it leaves almost no residual here, while a star -- much
219// sharper than the local blur -- does. The margin AROUND a detected core, though, must scale with
220// sigma_px: real-data measurement on a very bright star (Alnitak, IC434, 2026-09-28) found its
221// above-sky excess still present at 1 % of sky at r=40 px (sigma_px 48 there) and gone by r=60-80,
222// i.e. comparable to sigma_px itself, not a small fixed handful of pixels -- a star's smooth WING is
223// exactly the kind of gently-varying brightness this stage's own detector is designed to leave alone
224// (that is the point, for nebula), so the wing is not caught by the point-source test and needs this
225// separate, sigma_px-scaled margin instead. The local sigma, threshold multiple and sigma floor stay
226// fixed constants; only the margin depends on sigma_px, and even that is not user-facing -- both exist
227// to make the exclusion this header already promises ("hot pixels/stars ... removed") actually hold for
228// real stars, not to be a user-tunable star-detector.
229cv::Mat bright_source_mask(const cv::Mat &lum, const cv::Mat &valid, float sigma_px) {
230 const int rows = lum.rows, cols = lum.cols;
231 cv::Mat smooth;
232 cv::GaussianBlur(lum, smooth, cv::Size(0, 0), 4.0, 4.0, cv::BORDER_REPLICATE);
233 cv::Mat residual = lum - smooth;
234 const std::size_t total = static_cast<std::size_t>(rows) * cols;
235 const std::size_t stride = std::max<std::size_t>(1, total / 500000u);
236 std::vector<float> sample;
237 sample.reserve(std::min<std::size_t>(total, 500000u));
238 std::size_t linear = 0;
239 for (int y = 0; y < rows; ++y)
240 for (int x = 0; x < cols; ++x, ++linear)
241 if (linear % stride == 0 && valid.at<std::uint8_t>(y, x)) sample.push_back(residual.at<float>(y, x));
242 cv::Mat mask(rows, cols, CV_8U, cv::Scalar(0));
243 if (sample.empty()) return mask;
244 const double med = median_of(sample);
245 std::vector<float> dev;
246 dev.reserve(sample.size());
247 for (float v : sample) dev.push_back(std::fabs(v - static_cast<float>(med)));
248 // A floor well below any real per-pixel noise, so a noise-free (or near flat) region does not make
249 // the threshold pathologically sensitive to numerical-precision-level residual.
250 const double sigma = std::max(1.4826 * median_of(dev), 1e-4);
251 const double threshold = med + 8.0 * sigma;
252 for (int y = 0; y < rows; ++y)
253 for (int x = 0; x < cols; ++x)
254 if (valid.at<std::uint8_t>(y, x) && residual.at<float>(y, x) > threshold) mask.at<std::uint8_t>(y, x) = 255;
255 cv::Mat dilated;
256 const int margin_radius = std::clamp(static_cast<int>(std::lround(sigma_px * 0.75)), 5, 200);
257 const int kernel = 2 * margin_radius + 1;
258 cv::dilate(mask, dilated, cv::getStructuringElement(cv::MORPH_ELLIPSE, cv::Size(kernel, kernel)));
259 return dilated;
260}

262// Robust radial fit in r^2 (degree 3) over the usable cells; returns the fitted
263// value at every cell. IRLS with a 2-sigma soft clip keeps a bright nebula
264// from driving the vignette estimate.
265cv::Mat radial_fit(const CoarseMap &m) {
266 const int rows = m.rows, cols = m.cols;
267 double cy = 0, cx = 0;
268 std::size_t n = 0;
269 for (int y = 0; y < rows; ++y)
270 for (int x = 0; x < cols; ++x)
271 if (m.ok.at<std::uint8_t>(y, x)) { cy += y; cx += x; ++n; }
272 cv::Mat fit(rows, cols, CV_32F, cv::Scalar(0));
273 if (n < 16) return fit;
274 cy /= n; cx /= n;
275 const double scale = std::max(cy, cx) * std::max(cy, cx) + 1e-9;
276 constexpr int kDeg = 3;
277 auto basis = [&](int y, int x, double *b) {
278 const double r2 = ((y - cy) * (y - cy) + (x - cx) * (x - cx)) / scale;
279 double p = 1.0;
280 for (int k = 0; k <= kDeg; ++k) { b[k] = p; p *= r2; }
281 };
282 std::vector<double> w(n, 1.0);
283 double coef[kDeg + 1] = {0, 0, 0, 0};
284 for (int iter = 0; iter < 5; ++iter) {
285 double ata[kDeg + 1][kDeg + 1] = {}, atb[kDeg + 1] = {};
286 std::size_t i = 0;
287 for (int y = 0; y < rows; ++y)
288 for (int x = 0; x < cols; ++x) {
289 if (!m.ok.at<std::uint8_t>(y, x)) continue;
290 double b[kDeg + 1];
291 basis(y, x, b);
292 const double v = m.map.at<float>(y, x);
293 for (int r = 0; r <= kDeg; ++r) {
294 atb[r] += w[i] * b[r] * v;
295 for (int c = 0; c <= kDeg; ++c) ata[r][c] += w[i] * b[r] * b[c];
296 }
297 ++i;
298 }
299 // Gaussian elimination on the 4x4 normal equations.
300 double a[kDeg + 1][kDeg + 2];
301 for (int r = 0; r <= kDeg; ++r) {
302 for (int c = 0; c <= kDeg; ++c) a[r][c] = ata[r][c] + (r == c ? 1e-12 : 0.0);
303 a[r][kDeg + 1] = atb[r];
304 }
305 for (int col = 0; col <= kDeg; ++col) {
306 int piv = col;
307 for (int r = col + 1; r <= kDeg; ++r)
308 if (std::fabs(a[r][col]) > std::fabs(a[piv][col])) piv = r;
309 if (std::fabs(a[piv][col]) < 1e-14) return fit;
310 for (int c = 0; c <= kDeg + 1; ++c) std::swap(a[col][c], a[piv][c]);
311 for (int r = col + 1; r <= kDeg; ++r) {
312 const double f = a[r][col] / a[col][col];
313 for (int c = col; c <= kDeg + 1; ++c) a[r][c] -= f * a[col][c];
314 }
315 }
316 for (int r = kDeg; r >= 0; --r) {
317 double s = a[r][kDeg + 1];
318 for (int c = r + 1; c <= kDeg; ++c) s -= a[r][c] * coef[c];
319 coef[r] = s / a[r][r];
320 }
321 std::vector<float> res;
322 res.reserve(n);
323 i = 0;
324 std::vector<double> resid(n);
325 for (int y = 0; y < rows; ++y)
326 for (int x = 0; x < cols; ++x) {
327 if (!m.ok.at<std::uint8_t>(y, x)) continue;
328 double b[kDeg + 1];
329 basis(y, x, b);
330 double f = 0;
331 for (int k = 0; k <= kDeg; ++k) f += coef[k] * b[k];
332 resid[i] = m.map.at<float>(y, x) - f;
333 res.push_back(static_cast<float>(resid[i]));
334 ++i;
335 }
336 const double med = median_of(res);
337 std::vector<float> dev;
338 dev.reserve(n);
339 for (float r : res) dev.push_back(std::fabs(r - static_cast<float>(med)));
340 const double s = 1.4826 * median_of(dev) + 1e-9;
341 for (std::size_t j = 0; j < n; ++j) w[j] = 1.0 / std::max(1.0, std::fabs(resid[j]) / (2.0 * s));
342 }
343 for (int y = 0; y < rows; ++y)
344 for (int x = 0; x < cols; ++x) {
345 double b[kDeg + 1];
346 basis(y, x, b);
347 double f = 0;
348 for (int k = 0; k <= kDeg; ++k) f += coef[k] * b[k];
349 fit.at<float>(y, x) = static_cast<float>(f);
350 }
351 return fit;
352}

354// Deviation map of a plane from its own sky level (median over usable cells),
355// vignette component removed if asked; zero outside the usable cells.
356cv::Mat deviation_map(const cv::Mat &plane, const cv::Mat &valid, int factor, float sigma_px,
357 bool remove_vignette, CoarseMap *map_out, double *reference_out) {
358 CoarseMap m = build_map(plane, valid, factor, sigma_px);
359 std::vector<float> usable;
360 for (int y = 0; y < m.rows; ++y)
361 for (int x = 0; x < m.cols; ++x)
362 if (m.ok.at<std::uint8_t>(y, x)) usable.push_back(m.map.at<float>(y, x));
363 const double ref = median_of(usable);
364 cv::Mat dev(m.rows, m.cols, CV_32F, cv::Scalar(0));
365 cv::Mat radial;
366 if (remove_vignette) radial = radial_fit(m);
367 for (int y = 0; y < m.rows; ++y)
368 for (int x = 0; x < m.cols; ++x) {
369 if (!m.ok.at<std::uint8_t>(y, x)) continue;
370 float d = m.map.at<float>(y, x) - static_cast<float>(ref);
371 if (remove_vignette) d -= radial.at<float>(y, x) - static_cast<float>(ref);
372 dev.at<float>(y, x) = d;
373 }
374 if (map_out) *map_out = m;
375 if (reference_out) *reference_out = ref;
376 return dev;
377}

379double span_p5_p95(const cv::Mat &dev, const cv::Mat &ok) {
380 std::vector<float> v;
381 for (int y = 0; y < dev.rows; ++y)
382 for (int x = 0; x < dev.cols; ++x)
383 if (ok.at<std::uint8_t>(y, x)) v.push_back(dev.at<float>(y, x));
384 return percentile_of(v, 0.95) - percentile_of(v, 0.05);

393LargeScaleContrastResult apply_large_scale_contrast(
394 Matrix2Df &R, Matrix2Df &G, Matrix2Df &B, const LargeScaleContrastConfig &cfg,
395 const std::vector<std::uint8_t> *valid_mask) {
396 LargeScaleContrastResult res;
397 if (!cfg.enabled) return res;
398 const int rows = static_cast<int>(R.rows()), cols = static_cast<int>(R.cols());
399 if (rows < 32 || cols < 32 || G.rows() != R.rows() || G.cols() != R.cols() || B.rows() != R.rows() || B.cols() != R.cols()) {
400 res.status = "error";
401 res.error_message = "channels must have the same size and at least 32x32 pixels";
402 return res;
403 }
404 if (!(cfg.amount >= 0.0f && cfg.chroma_amount >= 0.0f && cfg.sigma_px > 0.0f)) {
405 res.status = "error";
406 res.error_message = "amount and chroma_amount must be >= 0 and sigma_px > 0";
407 return res;
408 }
409 if (cfg.amount == 0.0f && cfg.chroma_amount == 0.0f) {
410 res.status = "not_needed";
411 return res;
412 }
413 if (valid_mask && valid_mask->size() != static_cast<std::size_t>(rows) * cols) {
414 res.status = "error";
415 res.error_message = "valid mask size does not match the image";
416 return res;
417 }
418 cv::Mat valid(rows, cols, CV_8U, cv::Scalar(0));
419 std::size_t n_valid = 0;
420 for (int y = 0; y < rows; ++y)
421 for (int x = 0; x < cols; ++x) {
422 const std::size_t i = static_cast<std::size_t>(y) * cols + x;
423 bool v;
424 if (valid_mask) {
425 v = (*valid_mask)[i] != 0;
426 } else {
427 const float r = R(y, x), g = G(y, x), b = B(y, x);
428 v = std::isfinite(r) && std::isfinite(g) && std::isfinite(b) && (r != 0.0f || g != 0.0f || b != 0.0f);
429 }
430 if (v) { valid.at<std::uint8_t>(y, x) = 1; ++n_valid; }
431 }
432 if (n_valid < static_cast<std::size_t>(rows) * cols / 20) { // less than 5 % of the frame
433 res.status = "too_few_valid_pixels";
434 return res;
435 }
436 // Coarse grid of about sigma/6 pixels per cell (at least 1, at most 16).
437 const int factor = std::clamp(static_cast<int>(std::floor(cfg.sigma_px / 6.0f)), 1, 16);
438 res.downsample_factor = factor;
439
440 Matrix2Df lum(rows, cols);
441 for (int y = 0; y < rows; ++y)
442 for (int x = 0; x < cols; ++x)
443 lum(y, x) = valid.at<std::uint8_t>(y, x) ? (R(y, x) + G(y, x) + B(y, x)) / 3.0f : 0.0f;
444
445 // Per-channel ceilings: the boost never pushes a channel beyond its own input maximum.
446 float max_r = 0, max_g = 0, max_b = 0;
447 for (int y = 0; y < rows; ++y)
448 for (int x = 0; x < cols; ++x)
449 if (valid.at<std::uint8_t>(y, x)) {
450 max_r = std::max(max_r, R(y, x));
451 max_g = std::max(max_g, G(y, x));
452 max_b = std::max(max_b, B(y, x));
453 }
454
455 // Excluded only from the coarse-map construction (deviation_map calls below); the boost is still
456 // applied to every valid pixel, stars included, via `valid` in the final loop further down.
457 cv::Mat clean_valid = valid.clone();
458 clean_valid.setTo(0, bright_source_mask(wrap_const(lum), valid, cfg.sigma_px));
459
460 cv::Mat delta_l, delta_rg, delta_bg;
461 CoarseMap lum_map;
462 double ref = 0.0;
463 cv::Mat dev_l = deviation_map(wrap_const(lum), clean_valid, factor, cfg.sigma_px, cfg.remove_vignette, &lum_map, &ref);
464 res.sky_reference = ref;
465 res.vignette_removed = cfg.remove_vignette;
466 res.span_before = span_p5_p95(dev_l, lum_map.ok);
467 if (cfg.amount > 0.0f) {
468 cv::Mat scaled = dev_l * cfg.amount;
469 cv::resize(scaled, delta_l, cv::Size(cols, rows), 0, 0, cv::INTER_LINEAR);
470 res.span_after = span_p5_p95(scaled, lum_map.ok);
471 } else {
472 res.span_after = res.span_before;
473 }
474 if (cfg.chroma_amount > 0.0f) {
475 Matrix2Df rg(rows, cols), bg(rows, cols);
476 for (int y = 0; y < rows; ++y)
477 for (int x = 0; x < cols; ++x) {
478 rg(y, x) = valid.at<std::uint8_t>(y, x) ? R(y, x) - G(y, x) : 0.0f;
479 bg(y, x) = valid.at<std::uint8_t>(y, x) ? B(y, x) - G(y, x) : 0.0f;
480 }
481 cv::Mat d_rg = deviation_map(wrap_const(rg), clean_valid, factor, cfg.sigma_px, cfg.remove_vignette, nullptr, nullptr) * cfg.chroma_amount;
482 cv::Mat d_bg = deviation_map(wrap_const(bg), clean_valid, factor, cfg.sigma_px, cfg.remove_vignette, nullptr, nullptr) * cfg.chroma_amount;
483 cv::resize(d_rg, delta_rg, cv::Size(cols, rows), 0, 0, cv::INTER_LINEAR);
484 cv::resize(d_bg, delta_bg, cv::Size(cols, rows), 0, 0, cv::INTER_LINEAR);
485 }
486 for (int y = 0; y < rows; ++y)
487 for (int x = 0; x < cols; ++x) {
488 if (!valid.at<std::uint8_t>(y, x)) continue;
489 const float dl = delta_l.empty() ? 0.0f : delta_l.at<float>(y, x);
490 const float dr = delta_rg.empty() ? 0.0f : delta_rg.at<float>(y, x);
491 const float db = delta_bg.empty() ? 0.0f : delta_bg.at<float>(y, x);
492 R(y, x) = std::clamp(R(y, x) + dl + dr, 0.0f, std::max(max_r, R(y, x)));
493 G(y, x) = std::clamp(G(y, x) + dl, 0.0f, std::max(max_g, G(y, x)));
494 B(y, x) = std::clamp(B(y, x) + dl + db, 0.0f, std::max(max_b, B(y, x)));
495 }
496 res.applied = true;
497 res.status = "applied";
498 return res;
499}
