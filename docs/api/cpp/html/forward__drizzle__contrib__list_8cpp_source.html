32StripeGeom stripe_geom(const RegistrationSamplingPlan &plan,
33 const config::ReconstructionDrizzleConfig &cfg,
34 int y_begin, int rows) {
35 StripeGeom g;
36 g.scale = cfg.internal_scale;
37 g.W = plan.canvas_width_native * g.scale;
38 g.H = plan.canvas_height_native * g.scale;
39 g.channels = plan.color_mode == ColorMode::MONO ? 1 : 3;
40 if (g.scale < 1 || g.W <= 0 || rows <= 0 || y_begin < 0 ||
41 y_begin + rows > g.H)
42 throw std::invalid_argument("DRIZZLE_CONTRIB_LIST_BAD_STRIPE");
43 return g;
44}

46// Materialise one frame's contributions for this stripe. `src` covers the
47// source rectangle [src_y_origin, src_y_origin + src.rows()) x
48// [src_x_origin, src_x_origin + src.cols()) --- a full frame for (0, 0)
49// origins, or a banded rect-provider read (A1). It must contain the stripe's
50// scan box. E1: single-pass fill --- the old two-pass form rasterized the
51// whole stripe twice (count, then fill). The running bound enforces the same
52// DRIZZLE_CONTRIB_LIST_BUDGET contract before `out` can exceed it, and every
53// emitted record is stored, so the former predicted-count drift check is
54// subsumed. NOT sorted here --- the caller sorts (once globally, or per
55// frame).
56std::vector<DrizzleContrib> build_frame_records(
57 const RegistrationSamplingPlan &plan,
58 const registration::FrameSamplingTransform &f, std::uint32_t frame_order,
59 const Matrix2Df &src, int src_y_origin, int src_x_origin,
60 const config::ReconstructionDrizzleConfig &cfg,
61 const StripeGeom &g, int y_begin, int rows,
62 const ForwardDrizzleSubdivisionParams &sub, std::size_t mem_budget_bytes,
63 int x_begin = 0, int cols = -1) {
64 // The buffer must cover exactly what the rasterizer will scan. For a
65 // full-frame buffer (origins 0) this is the historical shape check; for a
66 // banded rect it is the coverage check.
67 const auto scan_box = drizzle_source_scan_box(plan, f, g.scale, y_begin,
68 rows, x_begin, cols);
69 if (src_y_origin > scan_box.y0 || src_x_origin > scan_box.x0 ||
70 src_y_origin + src.rows() < scan_box.y1 ||
71 src_x_origin + src.cols() < scan_box.x1)
72 throw std::invalid_argument("DRIZZLE_SOURCE_SHAPE_MISMATCH");
73
74 // §30.81 step 3a: the rasterizer sink indexes the window row-major (`win_w`
75 // wide, origin `xb`); decode target_x/y back to ABSOLUTE canvas coords.
76 const int xb = std::clamp(x_begin, 0, g.W);
77 const int win_w = cols < 0 ? g.W : std::clamp(x_begin + cols, xb, g.W) - xb;
78 if (win_w <= 0) return {};
79
80 // E1: the two-pass budget contract, enforced incrementally. The old form
81 // counted first and rejected when `count >= kCountCeiling ||
82 // count * sizeof(DrizzleContrib) > mem_budget_bytes`; the single-pass form
83 // therefore permits at most
84 // min(kCountCeiling - 1, mem_budget_bytes / sizeof(DrizzleContrib))
85 // records and throws on the next one --- the identical bound.
86 const std::size_t kRecordLimit = static_cast<std::size_t>(std::min(
87 std::numeric_limits<unsigned long long>::max() /
88 static_cast<unsigned long long>(sizeof(DrizzleContrib)) -
89 1,
90 static_cast<unsigned long long>(
91 mem_budget_bytes / sizeof(DrizzleContrib))));
92
93 std::vector<DrizzleContrib> out;
94 const auto Wwin = static_cast<std::size_t>(win_w);
95 const auto xb_u = static_cast<std::uint32_t>(xb);
96 geomstats::ScopedVariant _v(geomstats::Variant::kContribFill, cfg.pixfrac);
97 geomstats::ScopedGeometryTimer _t;
98 rasterize_drizzle_stripe(
99 plan, f, g.scale, cfg.pixfrac, y_begin, rows,
100 [&](int sx, int sy, int c, int leaf, std::size_t i, double k) {
101 const double v = static_cast<double>(
102 src(sy - src_y_origin, sx - src_x_origin));
103 if (!std::isfinite(v)) return;
104 if (out.size() >= kRecordLimit)
105 throw std::runtime_error("DRIZZLE_CONTRIB_LIST_BUDGET");
106 DrizzleContrib rec;
107 rec.key.frame_order = frame_order;
108 rec.key.channel = static_cast<std::uint32_t>(c);
109 rec.key.target_y = static_cast<std::uint32_t>(i / Wwin);
110 rec.key.target_x = xb_u + static_cast<std::uint32_t>(i % Wwin);
111 rec.key.source_y = static_cast<std::uint32_t>(sy);
112 rec.key.source_x = static_cast<std::uint32_t>(sx);
113 rec.key.leaf_order = static_cast<std::uint32_t>(leaf);
114 rec.area = k;
115 rec.value = v;
116 out.push_back(rec);
117 },
118 sub, xb, win_w);
119 return out;
120}

122// Segment-reduce an already-canonically-sorted record range [begin, end) into
123// wx/w/w2: for each maximal run of equal (frame, channel, target_y, target_x),
124// one double accumulator pair in record order, then wx += A / w += B / w2 += B*B.
125void reduce_sorted_range(const DrizzleContrib *begin, const DrizzleContrib *end,
126 int W, DrizzleUniformAccum &acc) {
127 const DrizzleContrib *s = begin;
128 while (s < end) {
129 const DrizzleContrib *e = s + 1;
130 const auto &k0 = s->key;
131 while (e < end) {
132 const auto &k = e->key;
133 if (k.frame_order != k0.frame_order || k.channel != k0.channel ||
134 k.target_y != k0.target_y || k.target_x != k0.target_x)
135 break;
136 ++e;
137 }
138 double A = 0.0, B = 0.0;
139 for (const DrizzleContrib *t = s; t < e; ++t) {
140 A += t->area * t->value;
141 B += t->area;
142 }
143 const std::size_t i = static_cast<std::size_t>(k0.target_y) *
144 static_cast<std::size_t>(W) +
145 static_cast<std::size_t>(k0.target_x);
146 const int c = static_cast<int>(k0.channel);
147 acc.wx[c][i] += A;
148 acc.w[c][i] += B;
149 acc.w2[c][i] += B * B;
150 s = e;
151 }
152}

154DrizzleUniformAccum make_accum(const StripeGeom &g, int rows) {
155 DrizzleUniformAccum acc;
156 acc.width = g.W;
157 acc.rows = rows;
158 acc.channels = g.channels;
159 const std::size_t n =
160 static_cast<std::size_t>(g.W) * static_cast<std::size_t>(rows);
161 for (int c = 0; c < g.channels; ++c) {
162 acc.wx[c].assign(n, 0.0);
163 acc.w[c].assign(n, 0.0);
164 acc.w2[c].assign(n, 0.0);
165 }
166 return acc;
167}

175DrizzleContribList build_uniform_contrib_list(
176 const RegistrationSamplingPlan &plan, const SourceImageProvider &source_of,
177 const config::ReconstructionDrizzleConfig &cfg, int y_begin, int rows,
178 const ForwardDrizzleSubdivisionParams &subdivision,
179 std::size_t mem_budget_bytes) {
180 const StripeGeom g = stripe_geom(plan, cfg, y_begin, rows);
181 const auto prepared = prepare_drizzle_frames(plan, cfg, subdivision);
182
183 DrizzleContribList list;
184 list.width = g.W;
185 list.rows = rows;
186 list.channels = g.channels;
187 list.y_begin = y_begin;
188
189 for (std::size_t fo = 0; fo < prepared.frames.size(); ++fo) {
190 const auto &f = *prepared.frames[fo];
191 auto recs = build_frame_records(
192 plan, f, static_cast<std::uint32_t>(fo), source_of(f.source_index), 0,
193 0, cfg, g, y_begin, rows, subdivision, mem_budget_bytes);
194 // The running total must also respect the budget (whole-stripe form).
195 if ((list.records.size() + recs.size()) * sizeof(DrizzleContrib) >
196 mem_budget_bytes)
197 throw std::runtime_error("DRIZZLE_CONTRIB_LIST_BUDGET");
198 list.records.insert(list.records.end(), recs.begin(), recs.end());
199 }
200 list.predicted_count = list.records.size();
201
202 std::sort(list.records.begin(), list.records.end(),
203 [](const DrizzleContrib &a, const DrizzleContrib &b) {
204 return contrib_key_less(a.key, b.key);
205 });
206 return list;
207}

209DrizzleUniformAccum reduce_uniform_contrib_list(const DrizzleContribList &list) {
210 DrizzleUniformAccum acc;
211 acc.width = list.width;
212 acc.rows = list.rows;
213 acc.channels = list.channels;
214 const std::size_t n =
215 static_cast<std::size_t>(list.width) * static_cast<std::size_t>(list.rows);
216 for (int c = 0; c < list.channels; ++c) {
217 acc.wx[c].assign(n, 0.0);
218 acc.w[c].assign(n, 0.0);
219 acc.w2[c].assign(n, 0.0);
220 }
221 if (!list.records.empty())
222 reduce_sorted_range(list.records.data(),
223 list.records.data() + list.records.size(), list.width,
224 acc);
225 return acc;
226}

228DrizzleUniformAccum accumulate_uniform_by_frame(
229 const RegistrationSamplingPlan &plan, const SourceImageProvider &source_of,
230 const config::ReconstructionDrizzleConfig &cfg, int y_begin, int rows,
231 const ForwardDrizzleSubdivisionParams &subdivision,
232 std::size_t per_frame_mem_budget_bytes) {
233 const StripeGeom g = stripe_geom(plan, cfg, y_begin, rows);
234 const auto prepared = prepare_drizzle_frames(plan, cfg, subdivision);
235 DrizzleUniformAccum acc = make_accum(g, rows);
236
237 // frame_order leads the canonical key, so a frame's records form a
238 // self-contained block: build -> sort -> reduce one frame at a time, in
239 // prepared-frame order. Bit-identical to the whole-stripe build+sort+reduce.
240 for (std::size_t fo = 0; fo < prepared.frames.size(); ++fo) {
241 const auto &f = *prepared.frames[fo];
242 auto recs = build_frame_records(
243 plan, f, static_cast<std::uint32_t>(fo), source_of(f.source_index), 0,
244 0, cfg, g, y_begin, rows, subdivision, per_frame_mem_budget_bytes);
245 std::sort(recs.begin(), recs.end(),
246 [](const DrizzleContrib &a, const DrizzleContrib &b) {
247 return contrib_key_less(a.key, b.key);
248 });
249 if (!recs.empty())
250 reduce_sorted_range(recs.data(), recs.data() + recs.size(), g.W, acc);
251 }
252 return acc;
253}
