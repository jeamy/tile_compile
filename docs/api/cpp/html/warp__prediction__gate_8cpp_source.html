14float wrap_to_pi(float a) {
15 while (a > kPi) {

24float median_of(std::vector<float> v) {
25 if (v.empty()) {
26 return 0.0f;
27 }
28 const size_t mid = v.size() / 2;
29 std::nth_element(v.begin(), v.begin() + static_cast<long>(mid), v.end());
30 if (v.size() % 2 == 1) {
31 return v[mid];
32 }
33 const float hi = v[mid];
34 std::nth_element(v.begin(), v.begin() + static_cast<long>(mid - 1), v.end());

40WarpPredictionGateResult evaluate_warp_prediction(
41 float fi, float pred_ang_rad, float pred_tx, float pred_ty,
42 std::span<const WarpPredictionAnchor> anchors) {
43 WarpPredictionGateResult result;
44 const int n = static_cast<int>(anchors.size());
45
46 int l = -1;
47 int r = -1;
48 for (int i = 0; i < n; ++i) {
49 if (anchors[static_cast<size_t>(i)].fi <= fi) {
50 l = i;
51 }
52 if (anchors[static_cast<size_t>(i)].fi > fi) {
53 r = i;
54 break;
55 }
56 }
57 if (l < 0 && r < 0) {
58 result.reason = WarpPredictionGateReason::no_anchor;
59 return result;
60 }
61
62 // Per-consecutive-pair rates over the full anchor sequence.
63 std::vector<float> rate_ang;
64 std::vector<float> rate_shift;
65 rate_ang.reserve(static_cast<size_t>(std::max(0, n - 1)));
66 rate_shift.reserve(static_cast<size_t>(std::max(0, n - 1)));
67 for (int i = 0; i + 1 < n; ++i) {
68 const auto &a = anchors[static_cast<size_t>(i)];
69 const auto &b = anchors[static_cast<size_t>(i + 1)];
70 const float dfi = std::max(1.0f, b.fi - a.fi);
71 rate_ang.push_back(std::fabs(b.ang_rad - a.ang_rad) / dfi);
72 rate_shift.push_back(std::hypot(b.tx - a.tx, b.ty - a.ty) / dfi);
73 }
74 const float global_rate_ang = median_of(rate_ang);
75 const float global_rate_shift = median_of(rate_shift);
76
77 // Local rate on one side of an anchor: max over the up-to
78 // kWarpGateLocalRatePairs consecutive pairs adjacent to the anchor on that
79 // side (pairs ending at `anchor_idx` for a left anchor, pairs starting at
80 // `anchor_idx` for a right anchor). Falls back to the global rate.
81 auto side_rates = [&](int anchor_idx, bool left_side, float &ra,
82 float &rs) {
83 float la = 0.0f;
84 float ls = 0.0f;
85 bool any = false;
86 const int lo = left_side ? std::max(0, anchor_idx - kWarpGateLocalRatePairs)
87 : anchor_idx;
88 const int hi = left_side ? anchor_idx - 1
89 : std::min(n - 2, anchor_idx +
90 kWarpGateLocalRatePairs - 1);
91 for (int i = lo; i <= hi; ++i) {
92 la = std::max(la, rate_ang[static_cast<size_t>(i)]);
93 ls = std::max(ls, rate_shift[static_cast<size_t>(i)]);
94 any = true;
95 }
96 ra = std::max(global_rate_ang, any ? la : 0.0f);
97 rs = std::max(global_rate_shift, any ? ls : 0.0f);
98 };
99
100 // Single-sided: extrapolation past the anchor span.
101 if (l < 0 || r < 0) {
102 const int anchor_idx = (l >= 0) ? l : r;
103 const float dist = std::fabs(fi - anchors[static_cast<size_t>(anchor_idx)].fi);
104 result.anchor_distance = static_cast<int>(std::lround(dist));
105 if (dist > static_cast<float>(kWarpGateMaxExtrapolationFrames)) {
106 result.reason = WarpPredictionGateReason::extrapolation_too_long;
107 return result;
108 }
109 }

111 struct SideEval {
112 int anchor_idx = -1;
113 float angle_dev_deg = 0.0f;
114 float angle_allowed_deg = 0.0f;
115 float shift_dev_px = 0.0f;
116 float shift_allowed_px = 0.0f;
117 float dist = 0.0f;
118 float angle_ratio = 0.0f;
119 float shift_ratio = 0.0f;
120 WarpPredictionGateReason fail = WarpPredictionGateReason::ok;
121 };
122
123 auto eval_side = [&](int anchor_idx, bool left_side) -> SideEval {
124 SideEval e;
125 e.anchor_idx = anchor_idx;
126 const auto &a = anchors[static_cast<size_t>(anchor_idx)];
127 e.dist = std::fabs(fi - a.fi);
128 float ra = 0.0f;
129 float rs = 0.0f;
130 side_rates(anchor_idx, left_side, ra, rs);
131 e.angle_allowed_deg =
132 std::min(kWarpGateAngleHardCapDeg,
133 kWarpGateAngleAbsTolDeg +
134 kWarpGateRateMult * ra * kRadToDeg * e.dist);
135 e.angle_dev_deg =
136 std::fabs(wrap_to_pi(pred_ang_rad - a.ang_rad)) * kRadToDeg;
137 e.angle_ratio = e.angle_dev_deg / std::max(1.0e-6f, e.angle_allowed_deg);
138 if (e.angle_dev_deg > e.angle_allowed_deg) {
139 e.fail = WarpPredictionGateReason::angle_discontinuity;
140 return e;
141 }
142 e.shift_allowed_px =
143 std::min(kWarpGateShiftHardCapPx,
144 kWarpGateShiftAbsTolPx + kWarpGateRateMult * rs * e.dist);
145 e.shift_dev_px = std::hypot(pred_tx - a.tx, pred_ty - a.ty);
146 e.shift_ratio = e.shift_dev_px / std::max(1.0e-6f, e.shift_allowed_px);
147 if (e.shift_dev_px > e.shift_allowed_px) {
148 e.fail = WarpPredictionGateReason::shift_discontinuity;
149 }
150 return e;
151 };
152
153 auto fill_from = [&](const SideEval &e) {
154 result.angle_dev_deg = e.angle_dev_deg;
155 result.angle_allowed_deg = e.angle_allowed_deg;
156 result.shift_dev_px = e.shift_dev_px;
157 result.shift_allowed_px = e.shift_allowed_px;
158 result.anchor_distance = static_cast<int>(std::lround(e.dist));
159 };
160
161 // Angle check has priority over shift check; check l before r.
162 SideEval el;
163 SideEval er;
164 if (l >= 0) {
165 el = eval_side(l, true);
166 if (el.fail != WarpPredictionGateReason::ok) {
167 result.reason = el.fail;
168 fill_from(el);
169 return result;
170 }
171 }
172 if (r >= 0) {
173 er = eval_side(r, false);
174 if (er.fail != WarpPredictionGateReason::ok) {
175 result.reason = er.fail;
176 fill_from(er);
177 return result;
178 }
179 }
180
181 result.ok = true;
182 result.reason = WarpPredictionGateReason::ok;
183 if (l >= 0 && r >= 0) {
184 fill_from((el.angle_ratio + el.shift_ratio >=
185 er.angle_ratio + er.shift_ratio)
186 ? el
187 : er);
188 } else if (l >= 0) {
189 fill_from(el);
190 } else {
191 fill_from(er);
192 }
193 return result;
194}
