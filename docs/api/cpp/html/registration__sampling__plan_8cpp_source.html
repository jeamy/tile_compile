41bool invert_affine_2x3(const WarpMatrix& canvas_to_source,
42 float det_min, float det_max,
43 WarpMatrix& out) {
44 const float a = canvas_to_source(0, 0);
45 const float b = canvas_to_source(0, 1);
46 const float c = canvas_to_source(0, 2);
47 const float d = canvas_to_source(1, 0);
48 const float e = canvas_to_source(1, 1);
49 const float f = canvas_to_source(1, 2);
50
51 for (float v : {a, b, c, d, e, f}) {
52 if (!std::isfinite(v)) return false;
53 }
54
55 const float det = a * e - b * d;
56 if (!std::isfinite(det)) return false;
57 // Orientation-preserving only: registration warps for stacking never reflect,
58 // so a non-positive determinant or one outside the scale-reject bounds
59 // (det_min = reject_scale_min^2, det_max = reject_scale_max^2) is rejected.
60 if (det < det_min || det > det_max) return false;
61
62 const float inv_det = 1.0f / det;
63 const float ai = e * inv_det;
64 const float bi = -b * inv_det;
65 const float di = -d * inv_det;
66 const float ei = a * inv_det;
67 const float ci = -(ai * c + bi * f);
68 const float fi = -(di * c + ei * f);
69
70 for (float v : {ai, bi, ci, di, ei, fi}) {
71 if (!std::isfinite(v)) return false;
72 }
73
74 out(0, 0) = ai; out(0, 1) = bi; out(0, 2) = ci;
75 out(1, 0) = di; out(1, 1) = ei; out(1, 2) = fi;
76 return true;
77}

85// d(q) in native-canvas render units. Mirrors render_smooth_local_displacement:
86// model_coordinate = (q - offset) * scale ; displacement_render = d_model / scale
87// Returns {0,0} when there is no active local model (caller should use the
88// affine inverse in that case).
89bool local_displacement_render_units(const FrameSamplingTransform& frame,
90 float qx, float qy,
91 float& out_dx, float& out_dy) {
92 out_dx = 0.0f;
93 out_dy = 0.0f;
94 if (frame.has_smooth_local_model && !frame.smooth_local_model.valid) return false;
95 if (!frame.has_smooth_local_model) {
96 return true; // no correction here == using the model, not a silent fallback
97 }
98 const float scale = frame.model_coordinate_scale;
99 if (!(scale > 0.0f) || !std::isfinite(scale)) {
100 return false; // a local model without a valid coordinate scale is broken
101 }
102 const float qmx = (qx - frame.model_offset_x) * scale;
103 const float qmy = (qy - frame.model_offset_y) * scale;
104 const cv::Point2f dm =
105 evaluate_smooth_local_displacement(frame.smooth_local_model, qmx, qmy);
106 const float inv_scale = 1.0f / scale;
107 out_dx = dm.x * inv_scale;
108 out_dy = dm.y * inv_scale;
109 return std::isfinite(out_dx) && std::isfinite(out_dy);
110}

114bool invert_local_source_to_canvas(const FrameSamplingTransform& frame,
115 float sx, float sy,
116 int canvas_width_native,
117 int canvas_height_native,
118 const LocalInversionParams& params,
119 float& out_qx, float& out_qy) {
120 if (!frame.source_to_canvas_affine_valid) return false;
121
122 // u = inverse(W_global)(s) == affine source_to_canvas applied to (sx, sy).
123 const WarpMatrix& s2c = frame.source_to_canvas;
124 const float ux = s2c(0, 0) * sx + s2c(0, 1) * sy + s2c(0, 2);
125 const float uy = s2c(1, 0) * sx + s2c(1, 1) * sy + s2c(1, 2);
126 if (!std::isfinite(ux) || !std::isfinite(uy)) return false;
127
128 const float margin = params.safety_margin_px;
129 auto out_of_bounds = [&](float x, float y) {
130 return x < -margin || y < -margin ||
131 x > static_cast<float>(canvas_width_native) + margin ||
132 y > static_cast<float>(canvas_height_native) + margin;
133 };
134 if (out_of_bounds(ux, uy)) return false;
135
136 float qx = ux;
137 float qy = uy;
138 bool converged = false;
139 const int max_iter = params.max_iter > 0 ? params.max_iter : 1;
140 const bool instrument = reconstruction::geomstats::registry().enabled;
141 if (instrument)
142 ++reconstruction::geomstats::registry().cur().invert_calls;
143 for (int n = 0; n < max_iter; ++n) {
144 if (instrument)
145 ++reconstruction::geomstats::registry().cur().invert_iterations;
146 float dx = 0.0f;
147 float dy = 0.0f;
148 if (!local_displacement_render_units(frame, qx, qy, dx, dy)) return false;
149 const float qx_new = ux - dx;
150 const float qy_new = uy - dy;
151 if (!std::isfinite(qx_new) || !std::isfinite(qy_new)) return false;
152 if (out_of_bounds(qx_new, qy_new)) return false;
153 const float step = std::max(std::abs(qx_new - qx), std::abs(qy_new - qy));
154 qx = qx_new;
155 qy = qy_new;
156 if (step < params.tol_px) {
157 converged = true;
158 break;
159 }
160 }
161 if (!converged) return false;
162 out_qx = qx;
163 out_qy = qy;
164 return true;
165}

345 f.chain_depth = jf.value("chain_depth", 0);
346 f.provenance = jf.value("provenance", std::string());

415std::string compute_plan_hash(const RegistrationSamplingPlan& plan) {
416 ByteSink s;
417 // A fixed tag + schema version so an encoding change is itself a hash change.
418 s.str("registration_sampling_plan:v2:edge-centers");
419 s.str(plan.source_identity_hash);
420 s.i32(RegistrationSamplingPlan::kSchemaVersion);
421 s.i32(static_cast<int32_t>(plan.convention));
422
423 s.i32(plan.source_width);
424 s.i32(plan.source_height);
425 s.i32(plan.canvas_width_native);
426 s.i32(plan.canvas_height_native);
427 s.i32(plan.canvas_offset_x_native);
428 s.i32(plan.canvas_offset_y_native);
429 s.i32(static_cast<int32_t>(plan.color_mode));
430 s.i32(static_cast<int32_t>(plan.bayer_pattern));
431 s.i32(plan.cfa_origin_x);
432 s.i32(plan.cfa_origin_y);
433
434 // NOTE: internal_scale / output_scale are deliberately excluded (drizzle
435 // geometry hash domain, plan section 18.3). chain_depth, model_predicted and
436 // provenance are excluded as diagnostic-only (their weight effect lives in
437 // model_prediction_factor).
438 s.u64(plan.frames.size());
439 for (const auto& f : plan.frames) {
440 s.str(f.frame_id);
441 s.u64(f.source_index);
442 s.b(f.valid);
443 write_warp(s, f.canvas_to_source);
444 write_warp(s, f.source_to_canvas);
445 s.b(f.source_to_canvas_affine_valid);
446 s.b(f.has_smooth_local_model);
447 s.b(f.smooth_local_model.valid);
448 s.i32(f.smooth_local_model.image_rows);
449 s.i32(f.smooth_local_model.image_cols);
450 for (int i = 0; i < f.smooth_local_model.coeff_x.size(); ++i)
451 s.f32(f.smooth_local_model.coeff_x[i]);
452 for (int i = 0; i < f.smooth_local_model.coeff_y.size(); ++i)
453 s.f32(f.smooth_local_model.coeff_y[i]);
454 s.f32(f.model_coordinate_scale);
455 s.f32(f.model_offset_x);
456 s.f32(f.model_offset_y);
457 s.f32(f.registration_residual_factor);
458 s.b(f.residual_applicable);
459 s.f32(f.model_prediction_factor);
460 }
461 return core::sha256_bytes(s.bytes);
462}
