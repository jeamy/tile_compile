45// Shared CFA-channel classification (plan section 11.4): the colour of a
46// source sample is determined *exclusively* from its integer cache
47// coordinates, the Bayer pattern and the persisted CFA origin --- never from
48// canvas offset, rotation or dither. This is the single implementation; do
49// not duplicate the parity arithmetic elsewhere (M1's sampling_geometry.cpp
50// and M2's forward_drizzle.cpp both call this).
51enum class CfaChannel { R, G, B, L };
52
53inline CfaChannel cfa_channel_for_source_pixel(int sx, int sy, BayerPattern bayer,
54 int cfa_origin_x, int cfa_origin_y) {
55 const BayerOffsets off = get_bayer_offsets(bayer);
56 const int px = (sx + cfa_origin_x) & 1;
57 const int py = (sy + cfa_origin_y) & 1;
58 if (py == off.r_row && px == off.r_col) return CfaChannel::R;
59 if (py == off.b_row && px == off.b_col) return CfaChannel::B;
60 return CfaChannel::G;
61}

73inline BayerPattern string_to_bayer_pattern(const std::string& s) {
74 std::string norm = s;
75 auto not_space = [](unsigned char c) { return !std::isspace(c); };
76 norm.erase(norm.begin(),
77 std::find_if(norm.begin(), norm.end(), not_space));
78 norm.erase(std::find_if(norm.rbegin(), norm.rend(), not_space).base(),
79 norm.end());
80 std::transform(norm.begin(), norm.end(), norm.begin(),
81 [](unsigned char c) { return static_cast<char>(std::toupper(c)); });
82
83 if (norm == "RGGB") return BayerPattern::RGGB;
84 if (norm == "BGGR") return BayerPattern::BGGR;
85 if (norm == "GRBG") return BayerPattern::GRBG;
86 if (norm == "GBRG") return BayerPattern::GBRG;
87 return BayerPattern::UNKNOWN;
88}
