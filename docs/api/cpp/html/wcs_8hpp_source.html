8// Simple WCS (World Coordinate System) for TAN projection
9// Supports pixel <-> sky (RA/Dec) coordinate conversion
10// Parses ASTAP .wcs output files
11struct WCS {
12 // Reference pixel (1-indexed, FITS convention)
13 double crpix1 = 0.0;
14 double crpix2 = 0.0;
15
16 // Reference sky coordinates (degrees)
17 double crval1 = 0.0; // RA
18 double crval2 = 0.0; // Dec
19
20 // CD matrix (degrees/pixel) — encodes scale + rotation
21 double cd1_1 = 0.0;
22 double cd1_2 = 0.0;
23 double cd2_1 = 0.0;
24 double cd2_2 = 0.0;
25
26 // Image dimensions
27 int naxis1 = 0;
28 int naxis2 = 0;

30 // Convenience: pixel scale (arcsec/pixel) and rotation (degrees)
31 double pixel_scale_arcsec() const {
32 double s1 = std::sqrt(cd1_1 * cd1_1 + cd2_1 * cd2_1);
33 double s2 = std::sqrt(cd1_2 * cd1_2 + cd2_2 * cd2_2);
34 return 0.5 * (s1 + s2) * 3600.0;
35 }
36
37 double rotation_deg() const {
38 return std::atan2(cd2_1, cd1_1) * 180.0 / M_PI;
39 }
40
41 double fov_width_deg() const {
42 return naxis1 * std::sqrt(cd1_1 * cd1_1 + cd2_1 * cd2_1);
43 }

49 // Pixel (0-indexed) -> sky (RA, Dec in degrees)
50 // Uses TAN (gnomonic) projection
51 void pixel_to_sky(double px, double py, double &ra_deg, double &dec_deg) const {
52 // Convert to 1-indexed for FITS convention
53 double dx = (px + 1.0) - crpix1;
54 double dy = (py + 1.0) - crpix2;
55
56 // Intermediate world coordinates (degrees)
57 double xi = cd1_1 * dx + cd1_2 * dy;
58 double eta = cd2_1 * dx + cd2_2 * dy;
59
60 // Convert to radians
61 constexpr double D2R = M_PI / 180.0;
62 double xi_r = xi * D2R;
63 double eta_r = eta * D2R;
64 double ra0_r = crval1 * D2R;
65 double dec0_r = crval2 * D2R;
66
67 // TAN (gnomonic) deprojection
68 double sin_dec0 = std::sin(dec0_r);
69 double cos_dec0 = std::cos(dec0_r);
70 double denom = cos_dec0 - eta_r * sin_dec0;
71
72 ra_deg = std::atan2(xi_r, denom) / D2R + crval1;
73 dec_deg = std::atan2((sin_dec0 + eta_r * cos_dec0) * std::cos(ra_deg * D2R - ra0_r), denom) / D2R;
74
75 // Normalize RA to [0, 360)
76 while (ra_deg < 0.0) ra_deg += 360.0;
77 while (ra_deg >= 360.0) ra_deg -= 360.0;
78 }

80 // Sky (RA, Dec in degrees) -> pixel (0-indexed)
81 // Returns false if point is behind the projection
82 bool sky_to_pixel(double ra_deg, double dec_deg, double &px, double &py) const {
83 constexpr double D2R = M_PI / 180.0;
84 double ra_r = ra_deg * D2R;
85 double dec_r = dec_deg * D2R;
86 double ra0_r = crval1 * D2R;
87 double dec0_r = crval2 * D2R;
88
89 double sin_dec = std::sin(dec_r);
90 double cos_dec = std::cos(dec_r);
91 double sin_dec0 = std::sin(dec0_r);
92 double cos_dec0 = std::cos(dec0_r);
93 double delta_ra = ra_r - ra0_r;
94 double cos_dra = std::cos(delta_ra);
95
96 // TAN projection
97 double denom = sin_dec * sin_dec0 + cos_dec * cos_dec0 * cos_dra;
98 if (denom <= 0.0) return false; // behind projection
99
100 double xi_r = (cos_dec * std::sin(delta_ra)) / denom;
101 double eta_r = (sin_dec * cos_dec0 - cos_dec * sin_dec0 * cos_dra) / denom;
102
103 // Convert from radians to degrees
104 double xi = xi_r / D2R;
105 double eta = eta_r / D2R;
106
107 // Invert CD matrix: [dx, dy] = CD^-1 * [xi, eta]
108 double det = cd1_1 * cd2_2 - cd1_2 * cd2_1;
109 if (std::abs(det) < 1e-30) return false;
110
111 double dx = ( cd2_2 * xi - cd1_2 * eta) / det;
112 double dy = (-cd2_1 * xi + cd1_1 * eta) / det;
113
114 // Convert back to 0-indexed pixel
115 px = dx + crpix1 - 1.0;
116 py = dy + crpix2 - 1.0;
117 return true;
118 }
119
120 // Check if a sky coordinate falls within the image
121 bool contains(double ra_deg, double dec_deg) const {
122 double px, py;
123 if (!sky_to_pixel(ra_deg, dec_deg, px, py)) return false;
124 return px >= 0 && px < naxis1 && py >= 0 && py < naxis2;
125 }
126
127 // Search radius (degrees) that covers the entire image diagonal
128 double search_radius_deg() const {
129 double diag_px = std::sqrt(double(naxis1) * naxis1 + double(naxis2) * naxis2);
130 return 0.6 * diag_px * pixel_scale_arcsec() / 3600.0; // slight margin
131 }
132
133 bool valid() const {
134 return naxis1 > 0 && naxis2 > 0 &&
135 (std::abs(cd1_1) > 0 || std::abs(cd1_2) > 0);
136 }
