78std::vector<fs::path> discover_frames(const fs::path& input_dir, const std::string& pattern) {
79 std::vector<fs::path> frames;
80
81 if (!fs::exists(input_dir) || !fs::is_directory(input_dir)) {
82 return frames;
83 }
84
85 std::vector<std::string> patterns;
86 {
87 std::string token;
88 for (char c : pattern) {
89 if (c == ';' || c == ',') {
90 // trim token
91 size_t b = token.find_first_not_of(" \t");
92 if (b != std::string::npos) {
93 size_t e = token.find_last_not_of(" \t");
94 patterns.push_back(token.substr(b, e - b + 1));
95 }
96 token.clear();
97 } else {
98 token.push_back(c);
99 }
100 }
101 size_t b = token.find_first_not_of(" \t");
102 if (b != std::string::npos) {
103 size_t e = token.find_last_not_of(" \t");
104 patterns.push_back(token.substr(b, e - b + 1));
105 }
106 }
107 if (patterns.empty()) {
108 patterns.push_back(pattern.empty() ? "*" : pattern);
109 }
110
111 for (const auto& entry : fs::directory_iterator(input_dir)) {
112 if (entry.is_regular_file()) {
113 std::string filename = entry.path().filename().string();
114 for (const auto& pat : patterns) {
115 if (glob_match(pat, filename)) {
116 frames.push_back(entry.path());
117 break;
118 }
119 }
120 }
121 }
122
123 std::sort(frames.begin(), frames.end());
124 return frames;
125}

280 if (data.size() == 0) return 0.0f;

320float compute_percentile(const VectorXf& data, float percentile) {
321 if (data.size() == 0) return 0.0f;
322
323 std::vector<float> sorted(data.data(), data.data() + data.size());
324 std::sort(sorted.begin(), sorted.end());
325
326 float idx = percentile / 100.0f * (sorted.size() - 1);
327 size_t lower = static_cast<size_t>(idx);
328 size_t upper = std::min(lower + 1, sorted.size() - 1);
329 float frac = idx - lower;
330
331 return sorted[lower] * (1.0f - frac) + sorted[upper] * frac;
332}

424float median_of(std::vector<float>& v) {
425 if (v.empty()) return 0.0f;
426 const size_t n = v.size();
427 const size_t mid = n / 2;
428 std::nth_element(v.begin(), v.begin() + static_cast<std::ptrdiff_t>(mid), v.end());
429 const float hi = v[mid];
430 if ((n % 2) == 1) return hi;
431 std::nth_element(v.begin(), v.begin() + static_cast<std::ptrdiff_t>(mid - 1), v.end());
432 const float lo = v[mid - 1];
433 return 0.5f * (lo + hi);
434}

440float stddev_of(const std::vector<float>& v) {
441 if (v.size() < 2) return 0.0f;
442 double sum = 0.0;
443 for (float x : v) sum += static_cast<double>(x);
444 const double mean = sum / static_cast<double>(v.size());
445 double var = 0.0;
446 for (float x : v) {
447 const double d = static_cast<double>(x) - mean;
448 var += d * d;
449 }
450 var /= static_cast<double>(v.size());
451 return (var > 0.0) ? static_cast<float>(std::sqrt(var)) : 0.0f;
452}

483float estimate_background_sigma_clip(std::vector<float> pixels) {
484 if (pixels.empty()) return 0.0f;
485 for (int iter = 0; iter < 5; ++iter) {
486 float mu = median_of(pixels);
487 float sigma = stddev_of(pixels);
488 if (!(sigma > 0.0f)) break;
489
490 std::vector<float> clipped;
491 clipped.reserve(pixels.size());
492 const float thr = 3.0f * sigma;
493 for (float x : pixels) {
494 if (std::fabs(x - mu) < thr) {
495 clipped.push_back(x);
496 }
497 }
498 if (clipped.size() == pixels.size() || clipped.empty()) break;
499 pixels.swap(clipped);
500 }
501 return median_of(pixels);
502}

508std::vector<size_t> sample_indices(size_t count, int max_samples) {
509 std::vector<size_t> out;
510 if (count == 0 || max_samples <= 0) return out;
511 size_t n = std::min(count, static_cast<size_t>(max_samples));
512 if (n == 1) {
513 out.push_back(0);
514 return out;
515 }
516 out.reserve(n);
517 for (size_t i = 0; i < n; ++i) {
518 float t = static_cast<float>(i) / static_cast<float>(n - 1);
519 size_t idx = static_cast<size_t>(std::round(t * static_cast<float>(count - 1)));
520 if (out.empty() || out.back() != idx) {
521 out.push_back(idx);
522 }
523 }
524 return out;
525}

531void robust_zscore(const std::vector<float>& v, std::vector<float>& out) {
532 out.assign(v.size(), 0.0f);
533 if (v.empty())
534 return;
535 std::vector<float> tmp = v;
536 float med = median_of(tmp);
537 for (float& x : tmp)
538 x = std::fabs(x - med);
539 float mad = median_of(tmp);
540 float sigma = 1.4826f * mad;
541 if (!(sigma > 0.0f)) {
542 std::fill(out.begin(), out.end(), 0.0f);
543 return;
544 }
545 for (size_t i = 0; i < v.size(); ++i) {
546 out[i] = (v[i] - med) / sigma;
547 }
548}

586StretchResult stretch_to_u16_linear_from_zero_inplace(Matrix2Df& img) {
587 StretchResult result;
588 float max_value = 0.0f;
589 size_t sample_count = 0;
590 for (Eigen::Index i = 0; i < img.size(); ++i) {
591 const float v = img.data()[i];
592 if (!std::isfinite(v) || v < 0.0f) continue;
593 max_value = std::max(max_value, v);
594 ++sample_count;
595 }
596
597 result.sample_count = sample_count;
598 result.low = 0.0f;
599 result.high = max_value;
600 if (!(max_value > 1.0e-6f)) return result;
601
602 const float scale = 65535.0f / max_value;
603 for (Eigen::Index i = 0; i < img.size(); ++i) {
604 const float v = img.data()[i];
605 if (std::isfinite(v) && v >= 0.0f) {
606 img.data()[i] = std::clamp(v * scale, 0.0f, 65535.0f);
607 } else {
608 img.data()[i] = 0.0f;
609 }
610 }
611
612 result.applied = true;
613 return result;
614}
