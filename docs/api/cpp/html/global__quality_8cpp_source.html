17// The exact, documented bounding transform: w/(1+w) = sigmoid(k*Q). See
18// global_quality.hpp for why this is necessary and why it is the minimal fix.
19// Applied identically in both compute_global_quality_weights and
20// compute_global_quality_weights_from_metrics.
21VectorXf apply_sigmoid_bounding(const VectorXf& raw) {
22 VectorXf out(raw.size());
23 for (long i = 0; i < raw.size(); ++i) {
24 const float w = raw[i];
25 out[i] = w / (1.0f + w);
26 }
27 return out;
28}

46 auto one_frame = [&](size_t i, const Matrix2Df &src) {
47 auto proxy = compute_source_quality_proxy_v1(src, color_mode, bayer_pattern,
48 cfa_origin_x, cfa_origin_y);
49 frame_metrics[i] = metrics::calculate_frame_metrics(proxy.proxy_full);
50 star_metrics[i] = metrics::measure_frame_stars(proxy.proxy_full,

52 cfg.star_max_corners,
53 cfg.star_patch_radius);
54 };
55
56 // Frame 0 first: it fixes ref_star_count for every other frame (legacy
57 // convention). Its own measure_frame_stars is called with ref_star_count 0.
58 one_frame(0, source_of(0));
59 ref_star_count = star_metrics[0].star_count;
60
61 const int nw =
62 static_cast<int>(std::min<size_t>(std::max(1, workers), n));
63#ifdef _OPENMP
64 if (nw > 1 && make_thread_provider) {
65 std::atomic<bool> failed{false};
66 std::exception_ptr err;
67#pragma omp parallel num_threads(nw)
68 {
69 SourceImageProvider tp;
70 try {
71 tp = make_thread_provider();
72 } catch (...) {
73#pragma omp critical(gq_err)
74 if (!err) err = std::current_exception();
75 failed.store(true, std::memory_order_relaxed);
76 }
77#pragma omp for schedule(dynamic, 1)
78 for (long long i = 1; i < static_cast<long long>(n); ++i) {
79 if (failed.load(std::memory_order_relaxed) || !tp)
80 continue;
81 try {
82 one_frame(static_cast<size_t>(i), tp(static_cast<size_t>(i)));
83 } catch (...) {
84#pragma omp critical(gq_err)
85 if (!err) err = std::current_exception();
86 failed.store(true, std::memory_order_relaxed);
87 }
88 }
89 }
90 if (err) std::rethrow_exception(err);
91 } else
92#endif
93 {
94 (void)make_thread_provider;
95 for (size_t i = 1; i < n; ++i)
96 one_frame(i, source_of(i));
97 }

99 const VectorXf raw = metrics::calculate_global_weights_with_stars(
