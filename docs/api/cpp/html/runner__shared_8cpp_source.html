74size_t sample_average_file_bytes(const std::vector<fs::path> &paths,
75 size_t max_samples) {
76 if (paths.empty() || max_samples == 0) {
77 return 0;
78 }
79 const size_t sample_count = std::min(max_samples, paths.size());
80 const size_t stride = std::max<size_t>(1, paths.size() / sample_count);
81 uint64_t total = 0;
82 size_t used = 0;
83 for (size_t i = 0; i < paths.size() && used < sample_count; i += stride) {
84 std::error_code ec;
85 const auto sz = fs::file_size(paths[i], ec);
86 if (ec || sz <= 0) {
87 continue;
88 }
89 total += static_cast<uint64_t>(sz);
90 ++used;
91 }
92 if (used == 0 && !paths.empty()) {
93 std::error_code ec;
94 const auto sz = fs::file_size(paths.front(), ec);
95 if (!ec && sz > 0) {
96 return static_cast<size_t>(sz);
97 }
98 }
99 return (used > 0) ? static_cast<size_t>(total / used) : 0;
100}

149bool invert_affine_warp(const WarpMatrix &w, WarpMatrix &inv) {
150 const float a = w(0, 0);
151 const float b = w(0, 1);
152 const float c = w(1, 0);
153 const float d = w(1, 1);
154 const float tx = w(0, 2);
155 const float ty = w(1, 2);
156 const float det = a * d - b * c;
157 if (std::fabs(det) < 1.0e-12f) {
158 return false;
159 }
160 const float inv_det = 1.0f / det;
161 inv(0, 0) = d * inv_det;
162 inv(0, 1) = -b * inv_det;
163 inv(1, 0) = -c * inv_det;
164 inv(1, 1) = a * inv_det;
165 inv(0, 2) = -(inv(0, 0) * tx + inv(0, 1) * ty);
166 inv(1, 2) = -(inv(1, 0) * tx + inv(1, 1) * ty);
167 return true;
168}

174WarpBounds compute_warps_bounds(int width, int height,
175 const std::vector<WarpMatrix> &warps) {
176 WarpBounds b;
177 if (width <= 0 || height <= 0 || warps.empty()) {
178 b.max_x = std::max(0, width);
179 b.max_y = std::max(0, height);
180 return b;
181 }
182
183 const float corners_x[4] = {0.0f, static_cast<float>(width), 0.0f,
184 static_cast<float>(width)};
185 const float corners_y[4] = {0.0f, 0.0f, static_cast<float>(height),
186 static_cast<float>(height)};
187
188 bool init = false;
189 float min_xf = 0.0f;
190 float min_yf = 0.0f;
191 float max_xf = 0.0f;
192 float max_yf = 0.0f;
193 for (const auto &w : warps) {
194 WarpMatrix fwd;
195 if (!invert_affine_warp(w, fwd)) {
196 continue;
197 }
198 for (int i = 0; i < 4; ++i) {
199 const float x = corners_x[i];
200 const float y = corners_y[i];
201 const float tx = fwd(0, 0) * x + fwd(0, 1) * y + fwd(0, 2);
202 const float ty = fwd(1, 0) * x + fwd(1, 1) * y + fwd(1, 2);
203 if (!init) {
204 min_xf = max_xf = tx;
205 min_yf = max_yf = ty;
206 init = true;
207 } else {
208 min_xf = std::min(min_xf, tx);
209 min_yf = std::min(min_yf, ty);
210 max_xf = std::max(max_xf, tx);
211 max_yf = std::max(max_yf, ty);
212 }
213 }
214 }
215
216 if (!init) {
217 std::cout << "[compute_warps_bounds] WARNING: all " << warps.size()
218 << " warp(s) are singular — falling back to identity canvas "
219 << width << "x" << height << std::endl;
220 b.max_x = std::max(0, width);
221 b.max_y = std::max(0, height);
222 return b;
223 }
224
225 b.min_x = static_cast<int>(std::floor(min_xf));
226 b.min_y = static_cast<int>(std::floor(min_yf));
227 b.max_x = static_cast<int>(std::ceil(max_xf));
228 b.max_y = static_cast<int>(std::ceil(max_yf));
229 return b;
230}

320int compute_adaptive_worker_count(
321 const config::Config &cfg, size_t task_count,
322 const std::vector<std::filesystem::path> &frames,
323 WorkerParallelProfile profile) {
324 int workers = cfg.runtime_limits.parallel_workers;
325 if (workers < 1) {
326 workers = 1;
327 }
328 const int cpu_cores = static_cast<int>(std::thread::hardware_concurrency());
329 if (cpu_cores > 0) {
330 workers = std::min(workers, cpu_cores);
331 }
332 if (task_count > 0) {
333 workers =
334 std::min(workers, static_cast<int>(std::max<size_t>(1, task_count)));
335 }
336 workers = std::max(1, workers);
337 if (workers <= 1 || profile == WorkerParallelProfile::CpuBound ||
338 frames.empty()) {
339 return workers;
340 }
341
342 const size_t avg_frame_bytes = sample_average_file_bytes(frames, 24);
343 if (avg_frame_bytes == 0) {
344 return workers;
345 }
346
347 const int io_cap =
348 cap_workers_for_io_profile(avg_frame_bytes, task_count, profile);
349 if (io_cap <= 0 || io_cap == std::numeric_limits<int>::max()) {
350 return workers;
351 }
352 return std::max(1, std::min(workers, io_cap));
353}

372 std::vector<uint8_t> &out_mask,

379 error_out = "missing canvas mask: " + mask_path.string();

395 for (int x = 0; x < cols; ++x) {
396 if (mask_img(y, x) > 0.5f) {
397 out_mask[static_cast<size_t>(y * cols + x)] = 1;

399 }
400 }

403 error_out = "canvas mask contains zero valid pixels";
404 return false;
405 }
406 return true;
407 } catch (const std::exception &e) {
408 error_out = std::string("cannot read canvas mask: ") + e.what();

438CropBox compute_nonzero_data_bbox(const Matrix2Df &luma, const Matrix2Df *r,
439 const Matrix2Df *g, const Matrix2Df *b) {
440 constexpr float kCropNonZeroEps = 1.0e-12f;
441 if (luma.rows() <= 0 || luma.cols() <= 0) {
442 return {};
443 }
444
445 const int rows = static_cast<int>(luma.rows());
446 const int cols = static_cast<int>(luma.cols());
447 const bool have_rgb =
448 (r != nullptr && g != nullptr && b != nullptr &&
449 r->rows() == luma.rows() && r->cols() == luma.cols() &&
450 g->rows() == luma.rows() && g->cols() == luma.cols() &&
451 b->rows() == luma.rows() && b->cols() == luma.cols());
452 auto is_valid_value = [](float v) {
453 return std::isfinite(v) && std::fabs(v) > kCropNonZeroEps;
454 };
455
456 int min_x = cols;
457 int min_y = rows;
458 int max_x = -1;
459 int max_y = -1;
460 for (int y = 0; y < rows; ++y) {
461 for (int x = 0; x < cols; ++x) {
462 bool has_data = is_valid_value(luma(y, x));
463 if (!has_data && have_rgb) {
464 has_data = is_valid_value((*r)(y, x)) || is_valid_value((*g)(y, x)) ||
465 is_valid_value((*b)(y, x));
466 }
467 if (!has_data) {
468 continue;
469 }
470 min_x = std::min(min_x, x);
471 min_y = std::min(min_y, y);
472 max_x = std::max(max_x, x);
473 max_y = std::max(max_y, y);
474 }
475 }
476
477 if (max_x < min_x || max_y < min_y) {
478 return {};
479 }
480 return CropBox{min_x, min_y, max_x - min_x + 1, max_y - min_y + 1};
481}

487CropBox compute_largest_valid_crop_box(const Matrix2Df &luma,
488 const std::vector<uint8_t> &common_valid_mask,
489 int mask_rows, int mask_cols,
490 const Matrix2Df *r,
491 const Matrix2Df *g,
492 const Matrix2Df *b) {
493 const CropBox data_box = compute_nonzero_data_bbox(luma, r, g, b);
494 if (!data_box.valid()) {
495 return {};
496 }
497
498 const size_t expected_mask_size =
499 static_cast<size_t>(std::max(0, mask_rows)) *
500 static_cast<size_t>(std::max(0, mask_cols));
501 if (mask_rows <= 0 || mask_cols <= 0 ||
502 common_valid_mask.size() != expected_mask_size) {
503 return data_box;
504 }
505
506 const int x0 = std::clamp(data_box.x, 0, mask_cols - 1);
507 const int y0 = std::clamp(data_box.y, 0, mask_rows - 1);
508 const int x1 =
509 std::clamp(data_box.x + data_box.width - 1, x0, mask_cols - 1);
510 const int y1 =
511 std::clamp(data_box.y + data_box.height - 1, y0, mask_rows - 1);
512 const int search_width = x1 - x0 + 1;
513 const int search_height = y1 - y0 + 1;
514 if (search_width <= 0 || search_height <= 0) {
515 return data_box;
516 }
517
518 std::vector<int> heights(static_cast<size_t>(search_width), 0);
519 CropBox best_box = data_box;
520 int best_area = 0;
521
522 for (int y = y0; y <= y1; ++y) {
523 const size_t row_off = static_cast<size_t>(y) * static_cast<size_t>(mask_cols);
524 for (int local_x = 0; local_x < search_width; ++local_x) {
525 const int gx = x0 + local_x;
526 const size_t idx = row_off + static_cast<size_t>(gx);
527 if (idx < common_valid_mask.size() && common_valid_mask[idx] != 0) {
528 heights[static_cast<size_t>(local_x)] += 1;
529 } else {
530 heights[static_cast<size_t>(local_x)] = 0;
531 }
532 }
533
534 std::vector<std::pair<int, int>> stack;
535 stack.reserve(static_cast<size_t>(search_width + 1));
536 for (int i = 0; i <= search_width; ++i) {
537 const int current_height =
538 (i < search_width) ? heights[static_cast<size_t>(i)] : 0;
539 int start = i;
540 while (!stack.empty() && stack.back().second > current_height) {
541 const auto [left, height] = stack.back();
542 stack.pop_back();
543 const int width = i - left;
544 const int area = height * width;
545 if (area > best_area && height > 0 && width > 0) {
546 best_area = area;
547 best_box = CropBox{ x0 + left, y - height + 1, width, height };
548 }
549 start = left;
550 }
551 if (stack.empty() || stack.back().second < current_height) {
552 stack.emplace_back(start, current_height);
553 }
554 }
555 }
556
557 return (best_area > 0) ? best_box : data_box;
558}

991void DiskCacheFrameStore::store(size_t fi, const Matrix2Df &frame) {
992 if (fi >= has_data_.size()) {
993 return;
994 }
995 if (frame.rows() != rows_ || frame.cols() != cols_) {
996 std::cout << "[DiskCache] Frame " << fi << " size mismatch: got "
997 << frame.rows() << "x" << frame.cols() << ", expected "
998 << rows_ << "x" << cols_ << std::endl;
999 has_data_[fi] = static_cast<uint8_t>(0);
1000 return;
1001 }
1002 // Drop a stale mapping before rewriting the file.
1003 invalidate_mapping(fi);
1004 fs::path p = frame_path(fi);
1005#ifdef _WIN32
1006 HANDLE hFile = CreateFileW(p.c_str(), GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
1007 if (hFile == INVALID_HANDLE_VALUE) {
1008 has_data_[fi] = static_cast<uint8_t>(0);
1009 return;
1010 }
1011 DWORD written = 0;
1012 const char *src = reinterpret_cast<const char *>(frame.data());
1013 WriteFile(hFile, src, static_cast<DWORD>(frame_bytes_), &written, NULL);
1014 CloseHandle(hFile);
1015 if (written == frame_bytes_) {
1016 has_data_[fi] = static_cast<uint8_t>(1);
1017 } else {
1018 has_data_[fi] = static_cast<uint8_t>(0);
1019 }
1020#else
1021 int fd = ::open(p.c_str(), O_WRONLY | O_CREAT | O_TRUNC, 0600);
1022 if (fd < 0) {
1023 has_data_[fi] = static_cast<uint8_t>(0);
1024 return;
1025 }
1026 size_t written = 0;
1027 const char *src = reinterpret_cast<const char *>(frame.data());
1028 while (written < frame_bytes_) {
1029 ssize_t n = ::write(fd, src + written, frame_bytes_ - written);
1030 if (n <= 0)
1031 break;
1032 written += static_cast<size_t>(n);
1033 }
1034 ::close(fd);
1035 if (written == frame_bytes_) {
1036 has_data_[fi] = static_cast<uint8_t>(1);
1037 } else {
1038 has_data_[fi] = static_cast<uint8_t>(0);
1039 }
1040#endif
1041}

1082bool DiskCacheFrameStore::extract_tile_into(size_t fi, const Tile &t,
1083 Matrix2Df &out, int offset_x,
1084 int offset_y) const {
1085 const float *src = mapped_frame_ptr(fi);
1086 if (src == nullptr) {
1087 out.resize(0, 0);
1088 return false;
1089 }
1090 int x0 = std::max(0, t.x + offset_x);
1091 int y0 = std::max(0, t.y + offset_y);
1092 int tw = t.width;
1093 int th = t.height;
1094 if (x0 + tw > cols_)
1095 tw = cols_ - x0;
1096 if (y0 + th > rows_)
1097 th = rows_ - y0;
1098 if (tw <= 0 || th <= 0) {
1099 out.resize(0, 0);
1100 return false;
1101 }
1102
1103 if (out.rows() != th || out.cols() != tw) {
1104 out.resize(th, tw);
1105 }
1106 for (int r = 0; r < th; ++r) {
1107 const float *row_src = src + static_cast<size_t>(y0 + r) *
1108 static_cast<size_t>(cols_) +
1109 static_cast<size_t>(x0);
1110 float *row_dst =
1111 out.data() + static_cast<size_t>(r) * static_cast<size_t>(tw);
1112 std::memcpy(row_dst, row_src, static_cast<size_t>(tw) * sizeof(float));
1113 }
1114 return true;
1115}

1121const float *DiskCacheFrameStore::mapped_frame_ptr(size_t fi) const {
1122 if (fi >= has_data_.size() || has_data_[fi] == 0) {
1123 return nullptr;
1124 }
1125
1126 {
1127 std::lock_guard<std::mutex> lock(mapped_mutex_);
1128 if (fi < mapped_views_.size() && mapped_views_[fi] != nullptr) {
1129 return static_cast<const float *>(mapped_views_[fi]);
1130 }
1131 }
1132
1133 fs::path p = frame_path(fi);
1134 void *new_view = nullptr;
1135#ifdef _WIN32
1136 HANDLE hFile = CreateFileW(p.c_str(), GENERIC_READ, 0, NULL, OPEN_EXISTING,
1137 FILE_ATTRIBUTE_NORMAL, NULL);
1138 if (hFile == INVALID_HANDLE_VALUE) {
1139 return nullptr;
1140 }
1141 HANDLE hMapping = CreateFileMappingW(hFile, NULL, PAGE_READONLY, 0, 0, NULL);
1142 if (!hMapping) {
1143 CloseHandle(hFile);
1144 return nullptr;
1145 }
1146 new_view = MapViewOfFile(hMapping, FILE_MAP_READ, 0, 0, frame_bytes_);
1147 CloseHandle(hMapping);
1148 CloseHandle(hFile);
1149 if (!new_view) {
1150 return nullptr;
1151 }
1152#else
1153 int fd = ::open(p.c_str(), O_RDONLY);
1154 if (fd < 0) {
1155 return nullptr;
1156 }
1157 new_view = ::mmap(nullptr, frame_bytes_, PROT_READ, MAP_PRIVATE, fd, 0);
1158 ::close(fd);
1159 if (new_view == MAP_FAILED) {
1160 return nullptr;
1161 }
1162#endif
1163
1164 void *existing_view = nullptr;
1165 {
1166 std::lock_guard<std::mutex> lock(mapped_mutex_);
1167 if (fi >= mapped_views_.size()) {
1168 unmap_view(new_view, frame_bytes_);
1169 return nullptr;
1170 }
1171 if (mapped_views_[fi] == nullptr) {
1172 mapped_views_[fi] = new_view;
1173 return static_cast<const float *>(new_view);
1174 }
1175 existing_view = mapped_views_[fi];
1176 }
1177
1178 // Another worker installed the mapping first.
1179 unmap_view(new_view, frame_bytes_);
1180 return static_cast<const float *>(existing_view);
1181}
