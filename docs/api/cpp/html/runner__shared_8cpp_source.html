92size_t sample_average_file_bytes(const std::vector<fs::path> &paths,
93 size_t max_samples) {
94 if (paths.empty() || max_samples == 0) {
95 return 0;
96 }
97 const size_t sample_count = std::min(max_samples, paths.size());
98 const size_t stride = std::max<size_t>(1, paths.size() / sample_count);
99 uint64_t total = 0;
100 size_t used = 0;
101 for (size_t i = 0; i < paths.size() && used < sample_count; i += stride) {
102 std::error_code ec;
103 const auto sz = fs::file_size(paths[i], ec);
104 if (ec || sz <= 0) {
105 continue;
106 }
107 total += static_cast<uint64_t>(sz);
108 ++used;
109 }
110 if (used == 0 && !paths.empty()) {
111 std::error_code ec;
112 const auto sz = fs::file_size(paths.front(), ec);
113 if (!ec && sz > 0) {
114 return static_cast<size_t>(sz);
115 }
116 }
117 return (used > 0) ? static_cast<size_t>(total / used) : 0;
118}

231bool invert_affine_warp(const WarpMatrix &w, WarpMatrix &inv) {
232 const float a = w(0, 0);
233 const float b = w(0, 1);
234 const float c = w(1, 0);
235 const float d = w(1, 1);
236 const float tx = w(0, 2);
237 const float ty = w(1, 2);
238 const float det = a * d - b * c;
239 if (std::fabs(det) < 1.0e-12f) {
240 return false;
241 }
242 const float inv_det = 1.0f / det;
243 inv(0, 0) = d * inv_det;
244 inv(0, 1) = -b * inv_det;
245 inv(1, 0) = -c * inv_det;
246 inv(1, 1) = a * inv_det;
247 inv(0, 2) = -(inv(0, 0) * tx + inv(0, 1) * ty);
248 inv(1, 2) = -(inv(1, 0) * tx + inv(1, 1) * ty);
249 return true;
250}

263bool project_tan_arcsec(double ra_deg, double dec_deg, double center_ra_deg,
264 double center_dec_deg, double &xi_arcsec,
265 double &eta_arcsec) {
266 const double ra = (center_ra_deg + normalized_ra_delta_deg(ra_deg, center_ra_deg)) *
267 kRadiansPerDegree;
268 const double dec = dec_deg * kRadiansPerDegree;
269 const double ra0 = center_ra_deg * kRadiansPerDegree;
270 const double dec0 = center_dec_deg * kRadiansPerDegree;
271 const double cos_delta_ra = std::cos(ra - ra0);
272 const double denominator = std::sin(dec) * std::sin(dec0) +
273 std::cos(dec) * std::cos(dec0) * cos_delta_ra;
274 if (!std::isfinite(denominator) || denominator <= 1.0e-9) {
275 return false;
276 }
277 const double xi = std::cos(dec) * std::sin(ra - ra0) / denominator;
278 const double eta = (std::sin(dec) * std::cos(dec0) -
279 std::cos(dec) * std::sin(dec0) * cos_delta_ra) / denominator;
280 xi_arcsec = xi / kRadiansPerDegree * 3600.0;
281 eta_arcsec = eta / kRadiansPerDegree * 3600.0;
282 return std::isfinite(xi_arcsec) && std::isfinite(eta_arcsec);
283}

311bool refine_similarity_transform(const std::vector<registration::StarPoint> &image_stars,
312 const std::vector<registration::StarPoint> &catalog_stars,
313 WarpMatrix &image_to_catalog,
314 float max_residual_px) {
315 struct Pair {
316 float x, y, u, v;
317 };
318 std::vector<Pair> pairs;
319 const float max_residual_sq = max_residual_px * max_residual_px;
320 for (const auto &image_star : image_stars) {
321 const float x = image_to_catalog(0, 0) * image_star.x +
322 image_to_catalog(0, 1) * image_star.y +
323 image_to_catalog(0, 2);
324 const float y = image_to_catalog(1, 0) * image_star.x +
325 image_to_catalog(1, 1) * image_star.y +
326 image_to_catalog(1, 2);
327 const registration::StarPoint *nearest = nullptr;
328 float best_sq = max_residual_sq;
329 for (const auto &catalog_star : catalog_stars) {
330 const float dx = x - catalog_star.x;
331 const float dy = y - catalog_star.y;
332 const float residual_sq = dx * dx + dy * dy;
333 if (residual_sq <= best_sq) {
334 best_sq = residual_sq;
335 nearest = &catalog_star;
336 }
337 }
338 if (nearest) pairs.push_back({image_star.x, image_star.y, nearest->x, nearest->y});
339 }
340 if (pairs.size() < 12) return false;
341
342 double mean_x = 0.0, mean_y = 0.0, mean_u = 0.0, mean_v = 0.0;
343 for (const auto &pair : pairs) {
344 mean_x += pair.x;
345 mean_y += pair.y;
346 mean_u += pair.u;
347 mean_v += pair.v;
348 }
349 const double count = static_cast<double>(pairs.size());
350 mean_x /= count;
351 mean_y /= count;
352 mean_u /= count;
353 mean_v /= count;
354
355 double denominator = 0.0, a_numerator = 0.0, b_numerator = 0.0;
356 for (const auto &pair : pairs) {
357 const double dx = pair.x - mean_x;
358 const double dy = pair.y - mean_y;
359 const double du = pair.u - mean_u;
360 const double dv = pair.v - mean_v;
361 denominator += dx * dx + dy * dy;
362 a_numerator += dx * du + dy * dv;
363 b_numerator += dx * dv - dy * du;
364 }
365 if (!std::isfinite(denominator) || denominator <= 1.0e-6) return false;
366 const double a = a_numerator / denominator;
367 const double b = b_numerator / denominator;
368 if (!std::isfinite(a) || !std::isfinite(b) || std::hypot(a, b) < 0.8 ||
369 std::hypot(a, b) > 1.2) {
370 return false;
371 }
372 image_to_catalog(0, 0) = static_cast<float>(a);
373 image_to_catalog(0, 1) = static_cast<float>(-b);
374 image_to_catalog(1, 0) = static_cast<float>(b);
375 image_to_catalog(1, 1) = static_cast<float>(a);
376 image_to_catalog(0, 2) = static_cast<float>(mean_u - a * mean_x + b * mean_y);
377 image_to_catalog(1, 2) = static_cast<float>(mean_v - b * mean_x - a * mean_y);
378 return true;
379}

381bool pair_vote_similarity_match(const std::vector<registration::StarPoint> &image_stars,
382 const std::vector<registration::StarPoint> &catalog_stars,
383 WarpMatrix &best_transform, int &best_inliers) {
384 struct StarPair {
385 int first = 0;
386 int second = 0;
387 float distance = 0.0f;
388 };
389 struct Vote {
390 int count = 0;
391 double a = 0.0, b = 0.0, tx = 0.0, ty = 0.0;
392 };
393 constexpr int kCandidateStars = 160;
394 constexpr float kMinPairDistance = 80.0f;
395 constexpr float kMinScale = 0.85f;
396 constexpr float kMaxScale = 1.15f;
397 constexpr float kAngleBinRad = 0.5f * static_cast<float>(kRadiansPerDegree);
398 constexpr float kTranslationBinPx = 24.0f;
399
400 const int image_count = std::min(kCandidateStars, static_cast<int>(image_stars.size()));
401 const int catalog_count = std::min(kCandidateStars, static_cast<int>(catalog_stars.size()));
402 if (image_count < 16 || catalog_count < 16) return false;
403 auto build_pairs = [](const std::vector<registration::StarPoint> &stars, int count) {
404 std::vector<StarPair> pairs;
405 pairs.reserve(static_cast<size_t>(count * (count - 1) / 2));
406 for (int first = 0; first < count; ++first) {
407 for (int second = first + 1; second < count; ++second) {
408 const float dx = stars[second].x - stars[first].x;
409 const float dy = stars[second].y - stars[first].y;
410 const float distance = std::hypot(dx, dy);
411 if (distance >= kMinPairDistance) pairs.push_back({first, second, distance});
412 }
413 }
414 return pairs;
415 };
416 const auto image_pairs = build_pairs(image_stars, image_count);
417 auto catalog_pairs = build_pairs(catalog_stars, catalog_count);
418 if (image_pairs.empty() || catalog_pairs.empty()) return false;
419 std::sort(catalog_pairs.begin(), catalog_pairs.end(),
420 [](const StarPair &lhs, const StarPair &rhs) { return lhs.distance < rhs.distance; });
421
422 std::unordered_map<uint64_t, Vote> votes;
423 votes.reserve(image_pairs.size() * 4);
424 auto add_vote = [&](const registration::StarPoint &image_first,
425 const registration::StarPoint &image_second,
426 const registration::StarPoint &catalog_first,
427 const registration::StarPoint &catalog_second) {
428 const float image_dx = image_second.x - image_first.x;
429 const float image_dy = image_second.y - image_first.y;
430 const float catalog_dx = catalog_second.x - catalog_first.x;
431 const float catalog_dy = catalog_second.y - catalog_first.y;
432 const float image_distance = std::hypot(image_dx, image_dy);
433 const float catalog_distance = std::hypot(catalog_dx, catalog_dy);
434 const float scale = catalog_distance / image_distance;
435 if (scale < kMinScale || scale > kMaxScale) return;
436 const float angle = std::atan2(catalog_dy, catalog_dx) -
437 std::atan2(image_dy, image_dx);
438 const float a = scale * std::cos(angle);
439 const float b = scale * std::sin(angle);
440 const float tx = catalog_first.x - a * image_first.x + b * image_first.y;
441 const float ty = catalog_first.y - b * image_first.x - a * image_first.y;
442 const int angle_bin = static_cast<int>(std::lround(angle / kAngleBinRad));
443 const int tx_bin = static_cast<int>(std::lround(tx / kTranslationBinPx));
444 const int ty_bin = static_cast<int>(std::lround(ty / kTranslationBinPx));
445 const uint64_t key =
446 (static_cast<uint64_t>(static_cast<uint32_t>(angle_bin + 32768) & 0xffffU) << 32U) |
447 (static_cast<uint64_t>(static_cast<uint32_t>(tx_bin + 32768) & 0xffffU) << 16U) |
448 static_cast<uint64_t>(static_cast<uint32_t>(ty_bin + 32768) & 0xffffU);
449 auto &vote = votes[key];
450 ++vote.count;
451 vote.a += a;
452 vote.b += b;
453 vote.tx += tx;
454 vote.ty += ty;
455 };
456
457 for (const auto &image_pair : image_pairs) {
458 const float lower_distance = image_pair.distance * kMinScale;
459 const float upper_distance = image_pair.distance * kMaxScale;
460 const auto begin = std::lower_bound(
461 catalog_pairs.begin(), catalog_pairs.end(), lower_distance,
462 [](const StarPair &pair, float distance) { return pair.distance < distance; });
463 const auto end = std::upper_bound(
464 begin, catalog_pairs.end(), upper_distance,
465 [](float distance, const StarPair &pair) { return distance < pair.distance; });
466 for (auto it = begin; it != end; ++it) {
467 add_vote(image_stars[image_pair.first], image_stars[image_pair.second],
468 catalog_stars[it->first], catalog_stars[it->second]);
469 add_vote(image_stars[image_pair.first], image_stars[image_pair.second],
470 catalog_stars[it->second], catalog_stars[it->first]);
471 }
472 }
473 std::vector<Vote> ranked_votes;
474 ranked_votes.reserve(votes.size());
475 for (const auto &[key, vote] : votes) {
476 (void)key;
477 if (vote.count >= 3) ranked_votes.push_back(vote);
478 }
479 std::sort(ranked_votes.begin(), ranked_votes.end(),
480 [](const Vote &lhs, const Vote &rhs) { return lhs.count > rhs.count; });
481
482 best_inliers = 0;
483 const size_t attempts = std::min<size_t>(24, ranked_votes.size());
484 for (size_t index = 0; index < attempts; ++index) {
485 const auto &vote = ranked_votes[index];
486 const double count = static_cast<double>(vote.count);
487 WarpMatrix candidate;
488 candidate << static_cast<float>(vote.a / count), static_cast<float>(-vote.b / count),
489 static_cast<float>(vote.tx / count), static_cast<float>(vote.b / count),
490 static_cast<float>(vote.a / count), static_cast<float>(vote.ty / count);
491 if (!refine_similarity_transform(image_stars, catalog_stars, candidate, 14.0f)) continue;
492 refine_similarity_transform(image_stars, catalog_stars, candidate, 6.0f);
493 const int inliers = count_catalog_inliers(image_stars, catalog_stars, candidate, 4.0f);
494 if (inliers > best_inliers) {
495 best_inliers = inliers;
496 best_transform = candidate;
497 }
498 }
499 return best_inliers >= 16;
500}

581 const std::string catalog_dir = astrometry::default_siril_gaia_catalog_dir();
582 if (!astrometry::is_siril_gaia_catalog_available(catalog_dir)) {
583 out.error_message = "local_gaia_catalog_unavailable";
584 return out;
585 }
586
587 Matrix2Df luma = (rgb.R + rgb.G + rgb.B) / 3.0f;
588 auto image_stars = registration::detect_stars_simple(luma, 2000, true);
589 out.image_stars = static_cast<int>(image_stars.size());
590 if (image_stars.size() < 24) {
591 out.error_message = "too_few_image_stars";
592 return out;
593 }
594
595 const double diagonal_radius_deg = 0.5 * std::hypot(width, height) *
596 nominal_scale_arcsec / 3600.0 + 0.5;
597 auto gaia_stars = astrometry::siril_gaia_cone_search(
598 catalog_dir, center_ra_deg, *dec_opt, diagonal_radius_deg, 12.5);
599 std::sort(gaia_stars.begin(), gaia_stars.end(),
600 [](const astrometry::GaiaStar &lhs, const astrometry::GaiaStar &rhs) {
601 return lhs.mag < rhs.mag;
602 });
603 if (gaia_stars.size() > 2000) gaia_stars.resize(2000);
604 out.catalog_stars = static_cast<int>(gaia_stars.size());
605 if (gaia_stars.size() < 24) {
606 out.error_message = "too_few_local_gaia_stars";
607 return out;
608 }
609
610 const float center_x = 0.5f * static_cast<float>(width - 1);
611 const float center_y = 0.5f * static_cast<float>(height - 1);
612 for (const bool reflected : {false, true}) {
613 std::vector<registration::StarPoint> catalog_points;
614 catalog_points.reserve(gaia_stars.size());
615 for (const auto &gaia_star : gaia_stars) {
616 double xi_arcsec = 0.0;
617 double eta_arcsec = 0.0;
618 if (!project_tan_arcsec(gaia_star.ra, gaia_star.dec, center_ra_deg, *dec_opt,
619 xi_arcsec, eta_arcsec)) {
620 continue;
621 }
622 const float x = center_x + static_cast<float>(xi_arcsec / nominal_scale_arcsec);
623 const float y_north_down = center_y -
624 static_cast<float>(eta_arcsec / nominal_scale_arcsec);
625 const float y = reflected ? static_cast<float>(height - 1) - y_north_down
626 : y_north_down;
627 if (x < -0.25f * width || x > 1.25f * width || y < -0.25f * height ||
628 y > 1.25f * height) {
629 continue;
630 }
631 catalog_points.push_back({x, y, -gaia_star.mag});
632 }
633 if (catalog_points.size() < 24) continue;
634
635 const auto registration_result = registration::triangle_star_matching(
636 luma, luma, true, 2000, 8, 4.0f, "similarity", true,
637 static_cast<float>(std::hypot(width, height)), &image_stars, &catalog_points);
638 WarpMatrix image_to_catalog;
639 int inliers = 0;
640 if (registration_result.success &&
641 invert_affine_warp(registration_result.warp, image_to_catalog)) {
642 inliers = count_catalog_inliers(image_stars, catalog_points,
643 image_to_catalog, 5.0f);
644 }
645 // The generic triangle matcher deliberately limits itself to the 60
646 // brightest stars. A stacked wide-field image can order detections and
647 // Gaia magnitudes differently, so independently vote for a similarity
648 // transform from pair distances across 160 stars before giving up.
649 WarpMatrix pair_vote_transform;
650 int pair_vote_inliers = 0;
651 if (pair_vote_similarity_match(image_stars, catalog_points,
652 pair_vote_transform, pair_vote_inliers) &&
653 pair_vote_inliers > inliers) {
654 image_to_catalog = pair_vote_transform;
655 inliers = pair_vote_inliers;
656 }
657 if (inliers < 16 || inliers <= out.inlier_stars) continue;
658
659 astrometry::WCS candidate_wcs;
660 if (!wcs_from_catalog_transform(image_to_catalog, center_ra_deg, *dec_opt,
661 nominal_scale_arcsec, reflected, width, height,
662 candidate_wcs)) {
663 continue;
664 }
665 out.success = true;
666 out.wcs = candidate_wcs;
667 out.inlier_stars = inliers;
668 out.reflected_solution = reflected;
669 out.error_message.clear();
670 }
671
672 if (!out.success) {
673 out.error_message = "local_gaia_match_failed";
674 }
675 return out;
676}

706WarpBounds compute_warps_bounds(int width, int height,
707 const std::vector<WarpMatrix> &warps) {
708 WarpBounds b;
709 if (width <= 0 || height <= 0 || warps.empty()) {
710 b.max_x = std::max(0, width);
711 b.max_y = std::max(0, height);
712 return b;
713 }
714
715 const float corners_x[4] = {0.0f, static_cast<float>(width), 0.0f,
716 static_cast<float>(width)};
717 const float corners_y[4] = {0.0f, 0.0f, static_cast<float>(height),
718 static_cast<float>(height)};
719
720 bool init = false;
721 float min_xf = 0.0f;
722 float min_yf = 0.0f;
723 float max_xf = 0.0f;
724 float max_yf = 0.0f;
725 for (const auto &w : warps) {
726 WarpMatrix fwd;
727 if (!invert_affine_warp(w, fwd)) {
728 continue;
729 }
730 for (int i = 0; i < 4; ++i) {
731 const float x = corners_x[i];
732 const float y = corners_y[i];
733 const float tx = fwd(0, 0) * x + fwd(0, 1) * y + fwd(0, 2);
734 const float ty = fwd(1, 0) * x + fwd(1, 1) * y + fwd(1, 2);
735 if (!init) {
736 min_xf = max_xf = tx;
737 min_yf = max_yf = ty;
738 init = true;
739 } else {
740 min_xf = std::min(min_xf, tx);
741 min_yf = std::min(min_yf, ty);
742 max_xf = std::max(max_xf, tx);
743 max_yf = std::max(max_yf, ty);
744 }
745 }
746 }
747
748 if (!init) {
749 std::cout << "[compute_warps_bounds] WARNING: all " << warps.size()
750 << " warp(s) are singular — falling back to identity canvas "
751 << width << "x" << height << std::endl;
752 b.max_x = std::max(0, width);
753 b.max_y = std::max(0, height);
754 return b;
755 }
756
757 b.min_x = static_cast<int>(std::floor(min_xf));
758 b.min_y = static_cast<int>(std::floor(min_yf));
759 b.max_x = static_cast<int>(std::ceil(max_xf));
760 b.max_y = static_cast<int>(std::ceil(max_yf));
761 return b;
762}

852int compute_adaptive_worker_count(
853 const config::Config &cfg, size_t task_count,
854 const std::vector<std::filesystem::path> &frames,
855 WorkerParallelProfile profile) {
856 int workers = cfg.runtime_limits.parallel_workers;
857 if (workers < 1) {
858 workers = 1;
859 }
860 const int cpu_cores = static_cast<int>(std::thread::hardware_concurrency());
861 if (cpu_cores > 0) {
862 workers = std::min(workers, cpu_cores);
863 }
864 if (task_count > 0) {
865 workers =
866 std::min(workers, static_cast<int>(std::max<size_t>(1, task_count)));
867 }
868 workers = std::max(1, workers);
869 if (workers <= 1 || frames.empty()) {
870 return workers;
871 }
872
873 const size_t avg_frame_bytes = sample_average_file_bytes(frames, 24);
874 if (avg_frame_bytes == 0) {
875 return workers;
876 }
877
878 const int io_cap =
879 cap_workers_for_io_profile(avg_frame_bytes, task_count, profile);
880 if (io_cap <= 0 || io_cap == std::numeric_limits<int>::max()) {
881 return workers;
882 }
883 return std::max(1, std::min(workers, io_cap));
884}

886FrameSubBatchPlan compute_memory_capped_frame_sub_batch(
887 size_t frame_count,
888 size_t pixels_per_worker,
889 int channels,
890 int requested_workers,
891 int memory_budget_mb) {
892 FrameSubBatchPlan plan;
893 plan.frame_sub_batch_size = frame_count;
894 plan.effective_workers = std::max(1, requested_workers);
895 plan.memory_budget_bytes =
896 static_cast<uint64_t>(std::max(1, memory_budget_mb)) * 1024ull * 1024ull;
897 plan.bytes_per_frame_per_worker =
898 static_cast<uint64_t>(pixels_per_worker) *
899 static_cast<uint64_t>(std::max(1, channels)) * sizeof(float);
900
901 if (frame_count == 0 || pixels_per_worker == 0 ||
902 plan.bytes_per_frame_per_worker == 0) {
903 plan.frame_sub_batch_size = 0;
904 return plan;
905 }
906
907 const uint64_t denominator =
908 plan.bytes_per_frame_per_worker *
909 static_cast<uint64_t>(std::max(1, plan.effective_workers));
910 if (denominator == 0) return plan;
911
912 const size_t total_frame_budget =
913 static_cast<size_t>((plan.memory_budget_bytes * 8ull / 10ull) / denominator);
914 if (total_frame_budget < 1) {
915 plan.effective_workers = 1;
916 plan.frame_sub_batch_size = frame_count;
917 plan.budget_too_small_for_requested_workers = true;
918 return plan;
919 }
920
921 if (total_frame_budget < frame_count) {
922 plan.frame_sub_batch_size = total_frame_budget;
923 plan.sub_batch_limited = true;
924 }
925 return plan;
926}

1028void apply_autobge_exclusion_polygons(
1029 const config::BGEConfig &src, int rows, int cols,
1030 image::BGEConfig &dst) {
1031 dst.sampling_valid_mask.clear();
1032 dst.sampling_mask_rows = 0;
1033 dst.sampling_mask_cols = 0;
1034 if (rows <= 0 || cols <= 0 || src.autobge.exclusion_polygons.empty())
1035 return;
1036 dst.sampling_valid_mask.assign(static_cast<size_t>(rows * cols), 1u);
1037 dst.sampling_mask_rows = rows;
1038 dst.sampling_mask_cols = cols;
1039 for (const auto &polygon : src.autobge.exclusion_polygons) {
1040 if (polygon.size() < 3) continue;
1041 for (int y = 0; y < rows; ++y) {
1042 const float py = (static_cast<float>(y) + 0.5f) / static_cast<float>(rows);
1043 for (int x = 0; x < cols; ++x) {
1044 const float px = (static_cast<float>(x) + 0.5f) / static_cast<float>(cols);
1045 bool inside = false;
1046 for (size_t i = 0, j = polygon.size() - 1; i < polygon.size(); j = i++) {
1047 const float xi = polygon[i][0], yi = polygon[i][1];
1048 const float xj = polygon[j][0], yj = polygon[j][1];
1049 if (((yi > py) != (yj > py)) &&
1050 px < (xj - xi) * (py - yi) / ((yj - yi) + 1e-20f) + xi)
1051 inside = !inside;
1052 }
1053 if (inside)
1054 dst.sampling_valid_mask[static_cast<size_t>(y * cols + x)] = 0u;
1055 }
1056 }
1057 }

1480void DiskCacheFrameStore::store(size_t fi, const Matrix2Df &frame) {
1481 if (fi >= has_data_.size()) {
1482 return;
1483 }
1484 if (frame.rows() != rows_ || frame.cols() != cols_) {
1485 std::cout << "[DiskCache] Frame " << fi << " size mismatch: got "
1486 << frame.rows() << "x" << frame.cols() << ", expected "
1487 << rows_ << "x" << cols_ << std::endl;
1488 has_data_[fi] = static_cast<uint8_t>(0);
1489 return;
1490 }
1491 // Drop a stale mapping before rewriting the file.
1492 invalidate_mapping(fi);
1493 fs::path p = frame_path(fi);
1494#ifdef _WIN32
1495 HANDLE hFile = CreateFileW(p.c_str(), GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
1496 if (hFile == INVALID_HANDLE_VALUE) {
1497 has_data_[fi] = static_cast<uint8_t>(0);
1498 return;
1499 }
1500 DWORD written = 0;
1501 const char *src = reinterpret_cast<const char *>(frame.data());
1502 WriteFile(hFile, src, static_cast<DWORD>(frame_bytes_), &written, NULL);
1503 CloseHandle(hFile);
1504 if (written == frame_bytes_) {
1505 has_data_[fi] = static_cast<uint8_t>(1);
1506 } else {
1507 has_data_[fi] = static_cast<uint8_t>(0);
1508 }
1509#else
1510 int fd = ::open(p.c_str(), O_WRONLY | O_CREAT | O_TRUNC, 0600);
1511 if (fd < 0) {
1512 has_data_[fi] = static_cast<uint8_t>(0);
1513 return;
1514 }
1515 size_t written = 0;
1516 const char *src = reinterpret_cast<const char *>(frame.data());
1517 while (written < frame_bytes_) {
1518 ssize_t n = ::write(fd, src + written, frame_bytes_ - written);
1519 if (n <= 0)
1520 break;
1521 written += static_cast<size_t>(n);
1522 }
1523 ::close(fd);
1524 if (written == frame_bytes_) {
1525 has_data_[fi] = static_cast<uint8_t>(1);
1526 } else {
1527 has_data_[fi] = static_cast<uint8_t>(0);
1528 }
1529#endif
1530}

1571bool DiskCacheFrameStore::extract_tile_into(size_t fi, const Tile &t,
1572 Matrix2Df &out, int offset_x,
1573 int offset_y) const {
1574 const float *src = mapped_frame_ptr(fi);
1575 if (src == nullptr) {
1576 out.resize(0, 0);
1577 return false;
1578 }
1579 int x0 = std::max(0, t.x + offset_x);
1580 int y0 = std::max(0, t.y + offset_y);
1581 int tw = t.width;
1582 int th = t.height;
1583 if (x0 + tw > cols_)
1584 tw = cols_ - x0;
1585 if (y0 + th > rows_)
1586 th = rows_ - y0;
1587 if (tw <= 0 || th <= 0) {
1588 out.resize(0, 0);
1589 return false;
1590 }
1591
1592 if (out.rows() != th || out.cols() != tw) {
1593 out.resize(th, tw);
1594 }
1595 for (int r = 0; r < th; ++r) {
1596 const float *row_src = src + static_cast<size_t>(y0 + r) *
1597 static_cast<size_t>(cols_) +
1598 static_cast<size_t>(x0);
1599 float *row_dst =
1600 out.data() + static_cast<size_t>(r) * static_cast<size_t>(tw);
1601 std::memcpy(row_dst, row_src, static_cast<size_t>(tw) * sizeof(float));
1602 }
1603 return true;
1604}

1610const float *DiskCacheFrameStore::mapped_frame_ptr(size_t fi) const {
1611 if (fi >= has_data_.size() || has_data_[fi] == 0) {
1612 return nullptr;
1613 }
1614
1615 {
1616 std::lock_guard<std::mutex> lock(mapped_mutex_);
1617 if (fi < mapped_views_.size() && mapped_views_[fi] != nullptr) {
1618 return static_cast<const float *>(mapped_views_[fi]);
1619 }
1620 }
1621
1622 fs::path p = frame_path(fi);
1623 void *new_view = nullptr;
1624#ifdef _WIN32
1625 HANDLE hFile = CreateFileW(p.c_str(), GENERIC_READ, 0, NULL, OPEN_EXISTING,
1626 FILE_ATTRIBUTE_NORMAL, NULL);
1627 if (hFile == INVALID_HANDLE_VALUE) {
1628 return nullptr;
1629 }
1630 HANDLE hMapping = CreateFileMappingW(hFile, NULL, PAGE_READONLY, 0, 0, NULL);
1631 if (!hMapping) {
1632 CloseHandle(hFile);
1633 return nullptr;
1634 }
1635 new_view = MapViewOfFile(hMapping, FILE_MAP_READ, 0, 0, frame_bytes_);
1636 CloseHandle(hMapping);
1637 CloseHandle(hFile);
1638 if (!new_view) {
1639 return nullptr;
1640 }
1641#else
1642 int fd = ::open(p.c_str(), O_RDONLY);
1643 if (fd < 0) {
1644 return nullptr;
1645 }
1646 new_view = ::mmap(nullptr, frame_bytes_, PROT_READ, MAP_PRIVATE, fd, 0);
1647 ::close(fd);
1648 if (new_view == MAP_FAILED) {
1649 return nullptr;
1650 }
1651#endif
1652
1653 void *existing_view = nullptr;
1654 {
1655 std::lock_guard<std::mutex> lock(mapped_mutex_);
1656 if (fi >= mapped_views_.size()) {
1657 unmap_view(new_view, frame_bytes_);
1658 return nullptr;
1659 }
1660 if (mapped_views_[fi] == nullptr) {
1661 mapped_views_[fi] = new_view;
1662 return static_cast<const float *>(new_view);
1663 }
1664 existing_view = mapped_views_[fi];
1665 }
1666
1667 // Another worker installed the mapping first.
1668 unmap_view(new_view, frame_bytes_);
1669 return static_cast<const float *>(existing_view);
1670}

1948std::optional<std::string> popen_first_line(const std::string &cmd) {
1949 FILE *fp = popen(cmd.c_str(), "r");
1950 if (!fp) return std::nullopt;
1951 char buf[512] = {};
1952 if (fgets(buf, sizeof(buf), fp)) {
1953 std::string found(buf);
1954 while (!found.empty() &&
1955 (found.back() == '\n' || found.back() == '\r' || found.back() == ' '))
1956 found.pop_back();
1957 pclose(fp);
1958 if (!found.empty()) return found;
1959 } else {
1960 pclose(fp);
1961 }
1962 return std::nullopt;

1970fs::path resolve_astap_binary_path(const std::string &astap_bin_cfg,
1971 const std::string &astap_data_dir) {
1972 // 1. Explicitly configured path
1973 if (!astap_bin_cfg.empty()) {
1974 fs::path explicit_path(astap_bin_cfg);
1975 if (astap_binary_exists(explicit_path)) return explicit_path;
1976 }
1977
1978 // 2. Inside the configured data directory
1979 if (!astap_data_dir.empty()) {
1980 fs::path data_dir_path(astap_data_dir);
1981 for (const char *name : {"astap_cli", "astap"}) {
1982 fs::path candidate = data_dir_path / name;
1983 if (astap_binary_exists(candidate)) return candidate;
1984 }
1985 }
1986
1987 // 3. PATH lookup
1988 for (const char *name : {"astap_cli", "astap"}) {
1989#ifdef _WIN32
1990 std::string cmd = std::string("where ") + name + " 2>nul";
1991#else
1992 std::string cmd = std::string("which ") + name + " 2>/dev/null";
1993#endif
1994 auto found = popen_first_line(cmd);
1995 if (found && astap_binary_exists(*found)) return fs::path(*found);
1996 }
1997
1998 // 4. Common Windows install locations
1999#ifdef _WIN32
2000 auto probe_common_dirs = [](const std::vector<fs::path> &roots) -> fs::path {
2001 for (const auto &root : roots) {
2002 if (root.empty()) continue;
2003 for (const char *name : {"astap_cli.exe", "astap.exe"}) {
2004 fs::path candidate = root / "astap" / name;
2005 if (astap_binary_exists(candidate)) return candidate;
2006 candidate = root / name;
2007 if (astap_binary_exists(candidate)) return candidate;
2008 }
2009 }
2010 return {};
2011 };
2012 std::vector<fs::path> roots;
2013 if (const char *pf = std::getenv("ProgramFiles")) roots.emplace_back(pf);
2014 if (const char *pf = std::getenv("ProgramFiles(x86)")) roots.emplace_back(pf);
2015 if (const char *la = std::getenv("LOCALAPPDATA")) roots.emplace_back(la);
2016 fs::path common = probe_common_dirs(roots);
2017 if (!common.empty()) return common;
2018#endif
2019
2020 return {};
2021}
