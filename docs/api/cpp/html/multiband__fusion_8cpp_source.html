36MultibandChannelResult fuse_multiband_channel(
37 const MultibandChannelInput &in, int width, int height,
38 const MultibandFusionParams &params,
39 const std::vector<std::vector<float>> &alpha_by_band) {
40 const int L = params.levels;
41 if (L < 1 || L > 4) throw std::invalid_argument("MULTIBAND_LEVELS_RANGE");
42 if (!in.uniform || !in.raw)
43 throw std::invalid_argument("MULTIBAND_MISSING_U_OR_R");
44 if (!in.fine) throw std::invalid_argument("MULTIBAND_MISSING_FINE");
45 if (L >= 2 && !in.medium)
46 throw std::invalid_argument("MULTIBAND_MISSING_MEDIUM");
47 const std::size_t n = static_cast<std::size_t>(width) * height;
48 if (!alpha_by_band.empty() &&
49 static_cast<int>(alpha_by_band.size()) != L)
50 throw std::invalid_argument("MULTIBAND_ALPHA_BAND_COUNT");
51 for (const auto &a : alpha_by_band)
52 if (!a.empty() && a.size() != n)
53 throw std::invalid_argument("MULTIBAND_ALPHA_MAP_SIZE");
54
55 const auto du = decompose_plane(*in.uniform, width, height, L, "uniform");
56 const auto dr = decompose_plane(*in.raw, width, height, L, "raw");
57 const auto df = decompose_plane(*in.fine, width, height, L, "fine");
58 const bool have_m = L >= 2;
59 const AtrousDecomposition dm =
60 have_m ? decompose_plane(*in.medium, width, height, L, "medium")
61 : AtrousDecomposition{};
62 return fuse_multiband_channel(
63 MultibandChannelDecompositions{&du, &dr, &df, have_m ? &dm : nullptr},
64 width, height, params, alpha_by_band);
65}

67MultibandChannelResult fuse_multiband_channel(
68 const MultibandChannelDecompositions &dec, int width, int height,
69 const MultibandFusionParams &params,
70 const std::vector<std::vector<float>> &alpha_by_band) {
71 const int L = params.levels;
72 if (L < 1 || L > 4) throw std::invalid_argument("MULTIBAND_LEVELS_RANGE");
73 if (!dec.uniform || !dec.raw)
74 throw std::invalid_argument("MULTIBAND_MISSING_U_OR_R");
75 if (!dec.fine) throw std::invalid_argument("MULTIBAND_MISSING_FINE");
76 if (L >= 2 && !dec.medium)
77 throw std::invalid_argument("MULTIBAND_MISSING_MEDIUM");
78 const std::size_t n = static_cast<std::size_t>(width) * height;
79 if (!alpha_by_band.empty() &&
80 static_cast<int>(alpha_by_band.size()) != L)
81 throw std::invalid_argument("MULTIBAND_ALPHA_BAND_COUNT");
82 for (const auto &a : alpha_by_band)
83 if (!a.empty() && a.size() != n)
84 throw std::invalid_argument("MULTIBAND_ALPHA_MAP_SIZE");
85 const AtrousDecomposition &du = *dec.uniform;
86 const AtrousDecomposition &dr = *dec.raw;
87 const AtrousDecomposition &df = *dec.fine;
88 static const AtrousDecomposition kNoMedium{};
89 const AtrousDecomposition &dm = dec.medium ? *dec.medium : kNoMedium;
90 for (const AtrousDecomposition *d : {&du, &dr, &df})
91 if (d->width != width || d->height != height || d->levels != L ||
92 d->coarse.size() != n || d->coarse_support.size() != n ||
93 d->bands.size() != static_cast<std::size_t>(L))
94 throw std::invalid_argument("MULTIBAND_DECOMPOSITION_GEOMETRY");
95 if (dec.medium &&
96 (dm.width != width || dm.height != height || dm.levels != L ||
97 dm.bands.size() != static_cast<std::size_t>(L)))
98 throw std::invalid_argument("MULTIBAND_DECOMPOSITION_GEOMETRY");
99
100 MultibandChannelResult out;
101 out.width = width;
102 out.height = height;
103 out.value.assign(n, nanf_());
104 out.support.assign(n, 0u);
105
106 auto alpha_at = [&](int band_index /*0-based*/, std::size_t i) -> double {
107 if (alpha_by_band.empty()) return 1.0;
108 const auto &a = alpha_by_band[static_cast<std::size_t>(band_index)];
109 if (a.empty()) return 1.0;
110 const double v = a[i];
111 return std::isfinite(v) ? std::clamp(v, 0.0, 1.0) : 0.0;
112 };
113
114 // Hoisted out of the per-pixel loop (redundant-reload analysis C3):
115 // band_source(j, L) and the fine/medium profile choice depend only on
116 // (j, L), which are invariant across pixels.
117 std::array<BandSource, 4> band_src_by_level{};
118 std::array<const AtrousDecomposition *, 4> band_profile_by_level{};
119 for (int j = 1; j <= L; ++j) {
120 band_src_by_level[j - 1] = band_source(j, L);
121 band_profile_by_level[j - 1] =
122 band_src_by_level[j - 1] == BandSource::kFine ? &df : &dm;
123 }
124
125 for (std::size_t i = 0; i < n; ++i) {
126 // Missing Raw coarse or any Raw band => whole multi-band pixel invalid
127 // (plan 14.2). Uniform coarse must also be valid (it supplies C_U,L).
128 if (!du.coarse_support[i] || !dr.coarse_support[i]) continue;
129 bool raw_bands_ok = true;
130 for (const auto &b : dr.bands)
131 if (!b.support[i]) { raw_bands_ok = false; break; }
132 if (!raw_bands_ok) continue;
133
134 double x = static_cast<double>(du.coarse[i]);
135 for (int j = 1; j <= L; ++j) {
136 const std::size_t bj = static_cast<std::size_t>(j - 1);
137 const double d_r = dr.bands[bj].detail[i];
138 const BandSource src = band_src_by_level[bj];
139 if (src == BandSource::kRaw) {
140 x += d_r; // alpha ineffective for R bands (plan 14.3)
141 continue;
142 }
143 const AtrousDecomposition &pd = *band_profile_by_level[bj];
144 double alpha = alpha_at(j - 1, i);
145 double d_p = d_r; // default: invalid detail profile => alpha 0
146 if (pd.bands[bj].support[i] && std::isfinite(pd.bands[bj].detail[i]))
147 d_p = pd.bands[bj].detail[i];
148 else
149 alpha = 0.0;
150 x += d_r + alpha * (d_p - d_r);
151 }
152
153 out.value[i] = static_cast<float>(x);
154 out.support[i] = 1u;
155 ++out.pixels_supported;
156 }
157 return out;
158}

170// B2: decompose each profile's channels ONCE per fusion call; the per-band
171// luminance combination below then only gathers band `band0` from the cached
172// decompositions instead of re-running atrous_decompose for every band.
173std::array<AtrousDecomposition, 3> decompose_channels(
174 const ForwardDrizzleUniformResult &profile, ColorMode mode, int w, int h,
175 int levels) {
176 std::array<AtrousDecomposition, 3> dec;
177 const auto ch = chans(profile, mode);
178 const int nch = mode == ColorMode::MONO ? 1 : 3;
179 for (int c = 0; c < nch; ++c)
180 dec[c] = decompose_plane(*ch[c], w, h, levels, "profile");
181 return dec;
182}
183
184// Luminance-combined detail for band j (0-based) of pre-decomposed channel
185// decompositions: MONO -> D_L,j, OSC -> 0.25 D_R,j + 0.5 D_G,j + 0.25 D_B,j
186// (plan 14.5). Valid only where every contributing channel band is valid.
187void luma_band(const std::array<AtrousDecomposition, 3> &dec, ColorMode mode,
188 int w, int h, int band0, std::vector<float> &out_detail,
189 std::vector<uint8_t> &out_support) {
190 const std::size_t n = static_cast<std::size_t>(w) * h;
191 out_detail.assign(n, std::numeric_limits<float>::quiet_NaN());
192 out_support.assign(n, 0u);
193 const int nch = mode == ColorMode::MONO ? 1 : 3;
194 const double wgt[3] = {mode == ColorMode::MONO ? 1.0 : kWorkingLumaWeightsOsc[0],
195 kWorkingLumaWeightsOsc[1], kWorkingLumaWeightsOsc[2]};
196 for (std::size_t i = 0; i < n; ++i) {
197 double acc = 0.0;
198 bool ok = true;
199 for (int c = 0; c < nch; ++c) {
200 const auto &b = dec[c].bands[static_cast<std::size_t>(band0)];
201 if (!b.support[i] || !std::isfinite(b.detail[i])) { ok = false; break; }
202 acc += wgt[c] * b.detail[i];
203 }
204 if (ok) {
205 out_detail[i] = static_cast<float>(acc);
206 out_support[i] = 1u;
207 }
208 }
209}

213MultibandResult fuse_multiband(
214 const ForwardDrizzleUniformResult &uniform,
215 const ForwardDrizzleUniformResult &raw,
216 const ForwardDrizzleUniformResult &fine,
217 const ForwardDrizzleUniformResult &medium, ColorMode mode, int width,
218 int height, const config::ReconstructionMultibandConfig &mb_cfg,
219 const AdaptiveAlphaParams &alpha_params,
220 const EnergyGuardParams &guard_params,
221 const std::vector<float> &a_separation, const std::vector<float> &a_artifact,
222 const std::vector<float> &a_registration,
223 const std::vector<double> &background_band_floor) {
224 const int L = mb_cfg.levels;
225 if (L < 1 || L > 4) throw std::invalid_argument("MULTIBAND_LEVELS_RANGE");
226 if (!background_band_floor.empty() &&
227 static_cast<int>(background_band_floor.size()) != L)
228 throw std::invalid_argument("MULTIBAND_FLOOR_BAND_COUNT");
229
230 // 1. adaptive alpha (F/M bands only; Raw bands empty).
231 auto alpha = compute_adaptive_alpha(uniform, fine, medium, mode, width, height,
232 L, alpha_params, a_separation, a_artifact,
233 a_registration);
234
235 // 2. energy guard + 3. B3 smoothing, per Fine/Medium band, on luma.
236 // B2: decompose each profile once; the band loop below only combines and
237 // the per-channel blend in step 4 reuses the same decompositions.
238 const auto uniform_dec = decompose_channels(uniform, mode, width, height, L);
239 const auto raw_dec = decompose_channels(raw, mode, width, height, L);
240 const auto fine_dec = decompose_channels(fine, mode, width, height, L);
241 const auto medium_dec =
242 L >= 2 ? decompose_channels(medium, mode, width, height, L)
243 : std::array<AtrousDecomposition, 3>{};
244 MultibandResult out;
245 out.alpha_final.assign(static_cast<std::size_t>(L), {});
246 for (int j = 1; j <= L; ++j) {
247 const std::size_t bj = static_cast<std::size_t>(j - 1);
248 if (alpha[bj].empty()) continue; // Raw-sourced band
249 const auto &profile_dec = (j == 1) ? fine_dec : medium_dec;
250 std::vector<float> dr_luma, dp_luma;
251 std::vector<uint8_t> sr, sp;
252 luma_band(raw_dec, mode, width, height, j - 1, dr_luma, sr);
253 luma_band(profile_dec, mode, width, height, j - 1, dp_luma, sp);
254 std::vector<uint8_t> support(sr.size(), 0u);
255 for (std::size_t i = 0; i < support.size(); ++i)
256 support[i] = (sr[i] && sp[i]) ? 1u : 0u;
257 const double floor_j =
258 background_band_floor.empty() ? 0.0 : background_band_floor[bj];
259 auto guarded = apply_energy_guard(alpha[bj], dr_luma, dp_luma, support,
260 width, height,
261 energy_guard_window_radius(j), floor_j,
262 guard_params);
263 out.alpha_final[bj] =
264 smooth_alpha_b3(guarded, support, width, height);
265 }
266
267 // 4. per-channel band blend with the shared, guarded, smoothed alpha.
268 MultibandFusionParams fp{L};
269 const int nch = mode == ColorMode::MONO ? 1 : 3;
270 std::vector<float> *outv[3] = {&out.R, &out.G, &out.B};
271 std::vector<uint8_t> *outs[3] = {&out.support_R, &out.support_G,
272 &out.support_B};
273 if (mode == ColorMode::MONO) { outv[0] = &out.L; outs[0] = &out.support_L; }
274 for (int c = 0; c < nch; ++c) {
275 const MultibandChannelDecompositions in{
276 &uniform_dec[c], &raw_dec[c], &fine_dec[c],
277 L >= 2 ? &medium_dec[c] : nullptr};
278 auto r = fuse_multiband_channel(in, width, height, fp, out.alpha_final);
279 *outv[c] = std::move(r.value);
280 *outs[c] = std::move(r.support);
281 out.pixels_supported += r.pixels_supported;
282 }
283 out.width = width;
284 out.height = height;
285 out.mode = mode;
286 return out;
287}

293ProfilePlane slice_plane(const ProfilePlane &src, int w, int ys, int ye) {
294 ProfilePlane p;
295 if (src.empty()) return p;
296 const int sub_h = ye - ys;
297 p.width = w;
298 p.height = sub_h;
299 const std::size_t off = static_cast<std::size_t>(ys) * w;
300 const std::size_t cnt = static_cast<std::size_t>(sub_h) * w;
301 auto cut = [&](const auto &v, auto &dst) {
302 if (v.empty()) return;
303 dst.assign(v.begin() + off, v.begin() + off + cnt);
304 };
305 cut(src.value, p.value);
306 cut(src.weight_sum, p.weight_sum);
307 cut(src.n_eff, p.n_eff);
308 cut(src.support, p.support);
309 return p;
310}

339int multiband_fusion_halo_rows(int levels) {
340 if (levels < 1) levels = 1;
341 if (levels > 4) levels = 4;
342 const int atrous_reach = 2 * ((1 << levels) - 1);
343 return atrous_reach + energy_guard_window_radius(levels) + 2;

346MultibandResult fuse_multiband_streamed(
347 const ForwardDrizzleUniformResult &uniform,
348 const ForwardDrizzleUniformResult &raw,
349 const ForwardDrizzleUniformResult &fine,
350 const ForwardDrizzleUniformResult &medium, ColorMode mode, int width,

353 const EnergyGuardParams &guard_params,

355 const std::vector<float> &a_registration,

359 if (width <= 0 || height <= 0)

362 return fuse_multiband(uniform, raw, fine, medium, mode, width, height,

365 }

367 const std::size_t n = static_cast<std::size_t>(width) * height;
368
369 MultibandResult out;
370 out.width = width;
371 out.height = height;
372 out.mode = mode;
373 std::vector<float> *outv[3];
374 std::vector<uint8_t> *outs[3];
375 int nch;
376 if (mode == ColorMode::MONO) {
377 out.L.assign(n, nanf_());

379 outv[0] = &out.L;

389 outv[0] = &out.R; outv[1] = &out.G; outv[2] = &out.B;
390 outs[0] = &out.support_R; outs[1] = &out.support_G; outs[2] = &out.support_B;
391 nch = 3;
392 }
393 out.alpha_final.assign(static_cast<std::size_t>(L), {});
394
395 for (int y0 = 0; y0 < height; y0 += chunk_rows) {
396 const int y1 = std::min(height, y0 + chunk_rows);
397 const int ys = std::max(0, y0 - halo);
398 const int ye = std::min(height, y1 + halo);
399 const int sub_h = ye - ys;
400
401 const auto su = slice_profiles(uniform, mode, width, ys, ye);
402 const auto sr = slice_profiles(raw, mode, width, ys, ye);
403 const auto sf = slice_profiles(fine, mode, width, ys, ye);
404 const auto sm = L >= 2 ? slice_profiles(medium, mode, width, ys, ye)

410 slice_vec(a_artifact, width, ys, ye),

412

414 const std::size_t src_off = static_cast<std::size_t>(y0 - ys) * width;
415 const std::size_t dst_off = static_cast<std::size_t>(y0) * width;
416
417 std::vector<float> *sv[3];
418 std::vector<uint8_t> *ss[3];
419 if (mode == ColorMode::MONO) { sv[0] = &stripe.L; ss[0] = &stripe.support_L; }
420 else {
421 sv[0] = &stripe.R; sv[1] = &stripe.G; sv[2] = &stripe.B;
422 ss[0] = &stripe.support_R; ss[1] = &stripe.support_G; ss[2] = &stripe.support_B;
423 }
424 for (int c = 0; c < nch; ++c) {
425 std::copy(sv[c]->begin() + src_off, sv[c]->begin() + src_off + core,
426 outv[c]->begin() + dst_off);
427 std::copy(ss[c]->begin() + src_off, ss[c]->begin() + src_off + core,
428 outs[c]->begin() + dst_off);
429 }
430 for (int b = 0; b < L; ++b) {
431 if (stripe.alpha_final[static_cast<std::size_t>(b)].empty()) continue;
432 auto &dst = out.alpha_final[static_cast<std::size_t>(b)];
433 if (dst.empty()) dst.assign(n, nanf_());
434 const auto &s = stripe.alpha_final[static_cast<std::size_t>(b)];
435 std::copy(s.begin() + src_off, s.begin() + src_off + core,
436 dst.begin() + dst_off);
437 }
438 }
439
440 for (int c = 0; c < nch; ++c)
441 for (std::size_t i = 0; i < n; ++i)
442 if ((*outs[c])[i]) ++out.pixels_supported;
443 return out;
444}
