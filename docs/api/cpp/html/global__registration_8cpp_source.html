40// Estimate rotation (deg) between ref and mov using log-polar phase correlation
41// on magnitude spectrum.
46float estimate_rotation_logpolar(const cv::Mat &ref, const cv::Mat &mov) {
47 cv::Mat ref_dft, mov_dft;
48 cv::dft(ref, ref_dft, cv::DFT_COMPLEX_OUTPUT);
49 cv::dft(mov, mov_dft, cv::DFT_COMPLEX_OUTPUT);
50
51 std::vector<cv::Mat> planes_ref, planes_mov;
52 cv::split(ref_dft, planes_ref);
53 cv::split(mov_dft, planes_mov);
54 cv::Mat mag_ref, mag_mov;
55 cv::magnitude(planes_ref[0], planes_ref[1], mag_ref);
56 cv::magnitude(planes_mov[0], planes_mov[1], mag_mov);
57
58 mag_ref += 1.0e-9f;
59 mag_mov += 1.0e-9f;
60
61 cv::Mat lp_ref, lp_mov;
62 const cv::Point2f center(static_cast<float>(ref.cols) / 2.0f,
63 static_cast<float>(ref.rows) / 2.0f);
64 const double M = ref.cols;
65 cv::warpPolar(mag_ref, lp_ref, mag_ref.size(), center, M,
66 static_cast<int>(cv::WARP_FILL_OUTLIERS) | cv::WARP_POLAR_LOG);
67 cv::warpPolar(mag_mov, lp_mov, mag_mov.size(), center, M,
68 static_cast<int>(cv::WARP_FILL_OUTLIERS) | cv::WARP_POLAR_LOG);
69
70 cv::Point2d shift = cv::phaseCorrelate(lp_mov, lp_ref);
71 double rotation_deg = -shift.y * 360.0 / static_cast<double>(lp_ref.rows);
72 return static_cast<float>(rotation_deg);
73}

79cv::Mat to_uint8_stretch(const Matrix2Df &src) {
80 cv::Mat f(src.rows(), src.cols(), CV_32F, const_cast<float *>(src.data()));
81 std::vector<float> vals;
82 vals.reserve(static_cast<size_t>(src.size()));
83 for (int r = 0; r < f.rows; ++r) {
84 const float *p = f.ptr<float>(r);
85 vals.insert(vals.end(), p, p + f.cols);
86 }
87 if (vals.empty())
88 return cv::Mat();
89 const size_t n = vals.size();
90 auto nth = [&](size_t k) {
91 std::nth_element(vals.begin(), vals.begin() + k, vals.end());
92 return vals[k];
93 };
94 float lo = nth(static_cast<size_t>(0.01 * n));
95 float hi = nth(static_cast<size_t>(0.99 * n));
96 if (hi <= lo)
97 hi = lo + 1.0f;
98 cv::Mat out;
99 f.convertTo(out, CV_8U, 255.0 / (hi - lo), -255.0 * lo / (hi - lo));
100 return out;
101}

125static bool invert_forward_affine_to_warp(const cv::Mat &A,
126 bool allow_rotation, WarpMatrix &warp,
127 std::string *error_message = nullptr) {
128 if (A.empty()) {
129 if (error_message) {
130 *error_message = "transform_fail";
131 }
132 return false;
133 }
134
135 float a00_fw = static_cast<float>(A.at<double>(0, 0));
136 float a01_fw = static_cast<float>(A.at<double>(0, 1));
137 float a10_fw = static_cast<float>(A.at<double>(1, 0));
138 float a11_fw = static_cast<float>(A.at<double>(1, 1));
139 float tx_fw = static_cast<float>(A.at<double>(0, 2));
140 float ty_fw = static_cast<float>(A.at<double>(1, 2));
141
142 if (!allow_rotation) {
143 a00_fw = 1.0f;
144 a01_fw = 0.0f;
145 a10_fw = 0.0f;
146 a11_fw = 1.0f;
147 }
148
149 const float det = a00_fw * a11_fw - a01_fw * a10_fw;
150 if (std::fabs(det) < 1e-8f) {
151 if (error_message) {
152 *error_message = "singular_matrix";
153 }
154 return false;
155 }
156 const float inv_det = 1.0f / det;
157 const float a00_inv = a11_fw * inv_det;
158 const float a01_inv = -a01_fw * inv_det;
159 const float a10_inv = -a10_fw * inv_det;
160 const float a11_inv = a00_fw * inv_det;
161 const float tx_inv = -(a00_inv * tx_fw + a01_inv * ty_fw);
162 const float ty_inv = -(a10_inv * tx_fw + a11_inv * ty_fw);
163 warp << a00_inv, a01_inv, tx_inv, a10_inv, a11_inv, ty_inv;
164 return true;
165}

171RegistrationResult feature_registration_similarity(const Matrix2Df &mov,
172 const Matrix2Df &ref,
173 bool allow_rotation,
174 const std::string &transform_model) {
175 cv::Mat ref_cv = to_uint8_stretch(ref);
176 cv::Mat mov_cv = to_uint8_stretch(mov);
177
178 RegistrationResult res;
179 res.warp = identity_warp();
180 res.correlation = 0.0f;
181 res.success = false;
182
183 if (ref_cv.empty() || mov_cv.empty()) {
184 res.error_message = "empty image";
185 return res;
186 }
187
188 cv::Ptr<cv::ORB> orb = cv::ORB::create();
189 std::vector<cv::KeyPoint> kps_ref, kps_mov;
190 cv::Mat desc_ref, desc_mov;
191 orb->detectAndCompute(ref_cv, cv::noArray(), kps_ref, desc_ref);
192 orb->detectAndCompute(mov_cv, cv::noArray(), kps_mov, desc_mov);
193
194 if (desc_ref.empty() || desc_mov.empty()) {
195 res.error_message = "no features";
196 return res;
197 }
198
199 cv::BFMatcher matcher(cv::NORM_HAMMING, true);
200 std::vector<cv::DMatch> matches;
201 matcher.match(desc_mov, desc_ref, matches);
202 if (matches.size() < 8) {
203 res.error_message = "few matches";
204 return res;
205 }
206
207 std::sort(matches.begin(), matches.end(),
208 [](const cv::DMatch &a, const cv::DMatch &b) {
209 return a.distance < b.distance;
210 });
211 const int keep = std::max<size_t>(15, matches.size() * 0.3);
212 matches.resize(std::min<size_t>(static_cast<size_t>(keep), matches.size()));
213
214 std::vector<cv::Point2f> pts_mov, pts_ref;
215 pts_mov.reserve(matches.size());
216 pts_ref.reserve(matches.size());
217 for (const auto &m : matches) {
218 pts_mov.push_back(kps_mov[m.queryIdx].pt);
219 pts_ref.push_back(kps_ref[m.trainIdx].pt);
220 }
221
222 cv::Mat inliers;
223 cv::Mat A = estimate_affine_family_transform(pts_mov, pts_ref,
224 transform_model, inliers);
225 if (A.empty()) {
226 res.error_message = "transform_fail";
227 return res;
228 }
229 if (!invert_forward_affine_to_warp(A, allow_rotation, res.warp,
230 &res.error_message)) {
231 return res;
232 }
233 int inl = inliers.empty() ? 0 : cv::countNonZero(inliers);
234 res.correlation = matches.empty() ? 0.0f
235 : static_cast<float>(inl) /
236 static_cast<float>(matches.size());
237 res.success = res.correlation > 0.1f;
238 return res;
239}

246static std::vector<StarPoint> detect_stars_with_threshold(
247 const Matrix2Df &img, int topk, float sigma_multiplier,
248 float med, float sigma) {
249 const int h = img.rows();
250 const int w = img.cols();
251 const float thresh = med + sigma_multiplier * sigma;
252
253 std::vector<StarPoint> stars;
254 stars.reserve(static_cast<size_t>(topk) * 2);
255 for (int y = 1; y < h - 1; ++y) {
256 const float *row = img.data() + static_cast<size_t>(y) * w;
257 for (int x = 1; x < w - 1; ++x) {
258 const float v = row[x];
259 if (v < thresh)
260 continue;
261 if (v <= row[x - 1] || v <= row[x + 1])
262 continue;
263
264 bool is_max = true;
265 for (int dy = -1; dy <= 1 && is_max; ++dy) {
266 if (dy == 0) continue; // Already checked horizontal neighbors
267 const float *r2 = img.data() + static_cast<size_t>(y + dy) * w;
268 if (r2[x - 1] >= v || r2[x] >= v || r2[x + 1] >= v) {
269 is_max = false;
270 }
271 }
272 if (!is_max)
273 continue;
274
275 // Compute flux-weighted centroid and hot pixel / elongation metrics
276 // in a 5×5 neighborhood for better discrimination
277 const int hw = 2; // half-width for analysis
278 float wsum = 0.0f;
279 float xs = 0.0f;
280 float ys = 0.0f;
281 float central_flux = row[x] - med;
282 if (central_flux < 0.0f)
283 central_flux = 0.0f;
284 float Ixx = 0.0f, Iyy = 0.0f, Ixy = 0.0f;
285 for (int dy = -hw; dy <= hw; ++dy) {
286 if (y + dy < 0 || y + dy >= h)
287 continue;
288 const float *r2 = img.data() + static_cast<size_t>(y + dy) * w;
289 for (int dx = -hw; dx <= hw; ++dx) {
290 if (x + dx < 0 || x + dx >= w)
291 continue;
292 const float val = r2[x + dx] - med;
293 if (val <= 0.0f)
294 continue;
295 wsum += val;
296 xs += (static_cast<float>(x + dx) * val);
297 ys += (static_cast<float>(y + dy) * val);
298 Ixx += static_cast<float>(dx * dx) * val;
299 Iyy += static_cast<float>(dy * dy) * val;
300 Ixy += static_cast<float>(dx * dy) * val;
301 }
302 }
303 if (wsum <= 0.0f)
304 continue;
305
306 // Hot pixel rejection: real stars spread flux over multiple pixels.
307 // Hot pixels concentrate >80% in the central pixel.
308 const float concentration = central_flux / wsum;
309 if (concentration > 0.8f)
310 continue;
311
312 // Roundness filter via second moments: reject elongated sources
313 if (Ixx + Iyy > 1e-6f) {
314 float trace = Ixx + Iyy;
315 float det2 = Ixx * Iyy - Ixy * Ixy;
316 float disc = trace * trace - 4.0f * det2;
317 if (disc < 0.0f) disc = 0.0f;
318 float sqrt_disc = std::sqrt(disc);
319 float lam1 = (trace + sqrt_disc) * 0.5f;
320 float lam2 = (trace - sqrt_disc) * 0.5f;
321 float roundness = (lam1 > 1e-6f) ? (lam2 / lam1) : 0.0f;
322 if (roundness < 0.15f)
323 continue; // too elongated (trail or artifact)
324 }
325
326 StarPoint s;
327 s.x = xs / wsum;
328 s.y = ys / wsum;
329 s.flux = wsum;
330 stars.push_back(s);
331 }
332 }
333 std::sort(
334 stars.begin(), stars.end(),
335 [](const StarPoint &a, const StarPoint &b) { return a.flux > b.flux; });
336 if (static_cast<int>(stars.size()) > topk) {
337 stars.resize(static_cast<size_t>(topk));
338 }
339 return stars;
340}

346std::vector<StarPoint> detect_stars_simple(const Matrix2Df &img, int topk,
347 bool enable_local_background_subtraction) {
348 const int h = img.rows();
349 const int w = img.cols();
350 if (h < 5 || w < 5)
351 return {};
352
353 // §4.4, §8.D — Lokale Hintergrundsubtraktion bei Gradienten/Mondlicht
354 Matrix2Df processed_img;
355 const Matrix2Df *img_ptr = &img;
356 if (enable_local_background_subtraction) {
357 // Lokale Hintergrundschätzung mit Box-Blur (31x31 Pixel)
358 processed_img = img; // Kopie
359 const int kernel_size = 31;
360 const int half_k = kernel_size / 2;
361 // Einfacher Box-Blur für Hintergrundschätzung
362 for (int y = 0; y < h; ++y) {
363 for (int x = 0; x < w; ++x) {
364 float sum = 0.0f;
365 int count = 0;
366 for (int dy = -half_k; dy <= half_k; ++dy) {
367 int py = std::clamp(y + dy, 0, h - 1);
368 for (int dx = -half_k; dx <= half_k; ++dx) {
369 int px = std::clamp(x + dx, 0, w - 1);
370 sum += img(py, px);
371 ++count;
372 }
373 }
374 float background = sum / count;
375 processed_img(y, x) = std::max(0.0f, img(y, x) - background);
376 }
377 }
378 img_ptr = &processed_img;
379 }
380
381 std::vector<float> pixels;
382 pixels.reserve(static_cast<size_t>(img_ptr->size()));
383 for (int y = 0; y < h; ++y) {
384 const float *row = img_ptr->data() + static_cast<size_t>(y) * w;
385 pixels.insert(pixels.end(), row, row + w);
386 }
387 float med = core::median_of(pixels);
388 float sigma = core::robust_sigma_mad(pixels);
389 if (sigma < 1.0e-6f)
390 sigma = 1.0f;
391
392 // Try standard threshold (3.5σ) — mit optionaler Hintergrundsubtraktion
393 std::vector<StarPoint> stars = detect_stars_with_threshold(*img_ptr, topk, 3.5f, med, sigma);
394
395 // Adaptive fallback: if we found very few stars, retry with lower threshold
396 // This helps with clouds/nebula where stars appear fainter
397 const int min_expected = std::max(4, topk / 2);
398 if (static_cast<int>(stars.size()) < min_expected) {
399 std::vector<StarPoint> stars_relaxed = detect_stars_with_threshold(*img_ptr, topk, 2.5f, med, sigma);
400 if (stars_relaxed.size() > stars.size()) {
401 stars = std::move(stars_relaxed);
402 }
403 }
404
405 return stars;
406}

415std::vector<MutualStarMatch> find_mutual_star_matches(
416 const std::vector<StarPoint> &ref_stars,
417 const std::vector<StarPoint> &warped_stars, float match_radius_px) {
418 std::vector<MutualStarMatch> matches;
419 if (match_radius_px <= 0.0f) {
420 return matches;
421 }
422 matches.reserve(std::min(ref_stars.size(), warped_stars.size()));
423 const float radius_sq = match_radius_px * match_radius_px;
424 constexpr float kAmbiguityRatioSq = 0.8f * 0.8f;
425 for (size_t warped_idx = 0; warped_idx < warped_stars.size(); ++warped_idx) {
426 float best_d2 = radius_sq;
427 float second_d2 = std::numeric_limits<float>::max();
428 int best_ref = -1;
429 for (size_t ref_idx = 0; ref_idx < ref_stars.size(); ++ref_idx) {
430 const float dx = warped_stars[warped_idx].x - ref_stars[ref_idx].x;
431 const float dy = warped_stars[warped_idx].y - ref_stars[ref_idx].y;
432 const float d2 = dx * dx + dy * dy;
433 if (d2 < best_d2) {
434 second_d2 = best_d2;
435 best_d2 = d2;
436 best_ref = static_cast<int>(ref_idx);
437 } else if (d2 < second_d2) {
438 second_d2 = d2;
439 }
440 }
441 if (best_ref < 0 ||
442 (std::isfinite(second_d2) &&
443 best_d2 > kAmbiguityRatioSq * second_d2)) {
444 continue;
445 }
446
447 int reverse_best = -1;
448 float reverse_best_d2 = radius_sq;
449 for (size_t candidate_idx = 0; candidate_idx < warped_stars.size();
450 ++candidate_idx) {
451 const float dx = warped_stars[candidate_idx].x -
452 ref_stars[static_cast<size_t>(best_ref)].x;
453 const float dy = warped_stars[candidate_idx].y -
454 ref_stars[static_cast<size_t>(best_ref)].y;
455 const float d2 = dx * dx + dy * dy;
456 if (d2 < reverse_best_d2) {
457 reverse_best_d2 = d2;
458 reverse_best = static_cast<int>(candidate_idx);
459 }
460 }
461 if (reverse_best == static_cast<int>(warped_idx)) {
462 matches.push_back({warped_idx, static_cast<size_t>(best_ref)});
463 }
464 }
465 return matches;
466}

499ResidualMetrics summarize_residuals(std::vector<float> values) {
500 ResidualMetrics out;
501 if (values.empty()) {
502 return out;
503 }
504 double sum_sq = 0.0;
505 for (const float value : values) {
506 sum_sq += static_cast<double>(value) * value;
507 }
508 std::sort(values.begin(), values.end());
509 const auto quantile = [&](float q) {
510 const float position = std::clamp(q, 0.0f, 1.0f) *
511 static_cast<float>(values.size() - 1);
512 const size_t lo = static_cast<size_t>(std::floor(position));
513 const size_t hi = std::min(values.size() - 1, lo + 1);
514 const float fraction = position - static_cast<float>(lo);
515 return values[lo] * (1.0f - fraction) + values[hi] * fraction;
516 };
517 out.median = quantile(0.5f);
518 out.p90 = quantile(0.9f);
519 out.rms = static_cast<float>(
520 std::sqrt(sum_sq / static_cast<double>(values.size())));
521 return out;
522}

524std::pair<float, float> singular_values_2x2(const Eigen::Matrix2f &matrix) {
525 const float a = matrix.col(0).squaredNorm();
526 const float b = matrix.col(0).dot(matrix.col(1));
527 const float d = matrix.col(1).squaredNorm();
528 const float discriminant =
529 std::sqrt(std::max(0.0f, (a - d) * (a - d) + 4.0f * b * b));
530 const float max_eigenvalue = std::max(0.0f, 0.5f * (a + d + discriminant));
531 const float min_eigenvalue = std::max(0.0f, 0.5f * (a + d - discriminant));
532 return {std::sqrt(min_eigenvalue), std::sqrt(max_eigenvalue)};
533}

535struct SmoothLocalGeometry {
536 float max_displacement = 0.0f;
537 float min_determinant = 1.0f;
538 float max_determinant = 1.0f;
539 float min_scale = 1.0f;
540 float max_scale = 1.0f;
541};
542
543SmoothLocalGeometry measure_smooth_local_geometry(
544 const cv::Mat &displacement_x, const cv::Mat &displacement_y) {
545 SmoothLocalGeometry out;
546 if (displacement_x.empty() || displacement_y.empty() ||
547 displacement_x.size() != displacement_y.size()) {
548 return out;
549 }
550 for (int y = 0; y < displacement_x.rows; ++y) {
551 const float *dx_row = displacement_x.ptr<float>(y);
552 const float *dy_row = displacement_y.ptr<float>(y);
553 for (int x = 0; x < displacement_x.cols; ++x) {
554 out.max_displacement =
555 std::max(out.max_displacement, std::hypot(dx_row[x], dy_row[x]));
556 }
557 }
558
559 const int geometry_step =
560 std::max(1, std::min(displacement_x.rows, displacement_x.cols) / 256);
561 for (int y = 0; y < displacement_x.rows; y += geometry_step) {
562 const int y_lo = std::max(0, y - 1);
563 const int y_hi = std::min(displacement_x.rows - 1, y + 1);
564 const float y_denominator =
565 static_cast<float>(std::max(1, y_hi - y_lo));
566 for (int x = 0; x < displacement_x.cols; x += geometry_step) {
567 const int x_lo = std::max(0, x - 1);
568 const int x_hi = std::min(displacement_x.cols - 1, x + 1);
569 const float x_denominator =
570 static_cast<float>(std::max(1, x_hi - x_lo));
571 Eigen::Matrix2f jacobian;
572 jacobian <<
573 1.0f + (displacement_x.at<float>(y, x_hi) -
574 displacement_x.at<float>(y, x_lo)) /
575 x_denominator,
576 (displacement_x.at<float>(y_hi, x) -
577 displacement_x.at<float>(y_lo, x)) /
578 y_denominator,
579 (displacement_y.at<float>(y, x_hi) -
580 displacement_y.at<float>(y, x_lo)) /
581 x_denominator,
582 1.0f + (displacement_y.at<float>(y_hi, x) -
583 displacement_y.at<float>(y_lo, x)) /
584 y_denominator;
585 const float determinant = jacobian.determinant();
586 const auto [min_scale, max_scale] = singular_values_2x2(jacobian);
587 out.min_determinant = std::min(out.min_determinant, determinant);
588 out.max_determinant = std::max(out.max_determinant, determinant);
589 out.min_scale = std::min(out.min_scale, min_scale);
590 out.max_scale = std::max(out.max_scale, max_scale);
591 }
592 }
593 return out;
594}

606AffineStarRefinementResult estimate_affine_star_refinement(
607 const std::vector<StarPoint> &ref_stars,

611 out.correction_warp = identity_warp();
612 out.rejection_reason = "too_few_stars";
613 if (ref_stars.size() < 24 || warped_stars.size() < 24 || image_rows <= 0 ||
614 image_cols <= 0 || match_radius_px <= 0.0f) {
615 return out;
616 }
617
618 const auto matches =
619 find_mutual_star_matches(ref_stars, warped_stars, match_radius_px);
620
621 out.matched_stars = static_cast<int>(matches.size());
622 if (matches.size() < 24) {
623 out.rejection_reason = "too_few_mutual_matches";
624 return out;
625 }
626
627 std::vector<cv::Point2f> points_warped;
628 std::vector<cv::Point2f> points_ref;
629 points_warped.reserve(matches.size());
630 points_ref.reserve(matches.size());
631 for (const MutualStarMatch &match : matches) {
632 const auto &warped = warped_stars[match.warped_idx];
633 const auto &ref = ref_stars[match.ref_idx];
634 points_warped.emplace_back(warped.x, warped.y);
635 points_ref.emplace_back(ref.x, ref.y);
636 }
637
638 cv::Mat inlier_mask;
639 cv::Mat forward = estimate_affine_family_transform(
640 points_warped, points_ref, "affine", inlier_mask, 0.75, 3000, 0.995);
641 if (forward.empty()) {
642 out.rejection_reason = "ransac_failed";
643 return out;
644 }
645 out.inlier_stars = inlier_mask.empty() ? 0 : cv::countNonZero(inlier_mask);
646 out.inlier_ratio = static_cast<float>(out.inlier_stars) /
647 static_cast<float>(matches.size());
648 if (out.inlier_stars < 18 || out.inlier_ratio < 0.60f) {
649 out.rejection_reason = "insufficient_ransac_consensus";
650 return out;
651 }
652
653 std::vector<cv::Point2f> inlier_ref_points;
654 inlier_ref_points.reserve(static_cast<size_t>(out.inlier_stars));
655 for (size_t i = 0; i < points_ref.size(); ++i) {
656 if (inlier_mask.at<uint8_t>(static_cast<int>(i)) != 0) {
657 inlier_ref_points.push_back(points_ref[i]);
658 }
659 }
660 const SpatialCoverage coverage =
661 measure_spatial_coverage(inlier_ref_points, image_rows, image_cols);
662 out.spatial_coverage = coverage.area_fraction;
663 if (out.spatial_coverage < 0.12f || coverage.x_span < 0.35f ||
664 coverage.y_span < 0.35f) {
665 out.rejection_reason = "insufficient_spatial_coverage";
666 return out;
667 }
668
669 const double a00 = forward.at<double>(0, 0);
670 const double a01 = forward.at<double>(0, 1);
671 const double tx = forward.at<double>(0, 2);
672 const double a10 = forward.at<double>(1, 0);
673 const double a11 = forward.at<double>(1, 1);
674 const double ty = forward.at<double>(1, 2);
675 if (!std::isfinite(a00) || !std::isfinite(a01) || !std::isfinite(tx) ||
676 !std::isfinite(a10) || !std::isfinite(a11) || !std::isfinite(ty)) {
677 out.rejection_reason = "non_finite_transform";
678 return out;
679 }
680 const double determinant = a00 * a11 - a01 * a10;
681 if (determinant <= 0.0) {
682 out.rejection_reason = "invalid_determinant";
683 return out;
684 }
685
686 cv::Mat linear = (cv::Mat_<double>(2, 2) << a00, a01, a10, a11);
687 cv::SVD svd(linear, cv::SVD::NO_UV);
688 out.max_scale = static_cast<float>(svd.w.at<double>(0));
689 out.min_scale = static_cast<float>(svd.w.at<double>(1));
690 if (out.min_scale < 0.99f || out.max_scale > 1.01f ||
691 out.max_scale / out.min_scale > 1.01f) {
692 out.rejection_reason = "scale_or_shear_out_of_bounds";
693 return out;
694 }
695
696 out.rotation_deg = static_cast<float>(
697 std::atan2(a10 - a01, a00 + a11) * 180.0 / CV_PI);
698 if (std::fabs(out.rotation_deg) > 0.5f) {
699 out.rejection_reason = "rotation_out_of_bounds";
700 return out;
701 }
702 const double center_x = 0.5 * static_cast<double>(image_cols - 1);
703 const double center_y = 0.5 * static_cast<double>(image_rows - 1);
704 const double center_dx = a00 * center_x + a01 * center_y + tx - center_x;
705 const double center_dy = a10 * center_x + a11 * center_y + ty - center_y;
706 out.center_displacement_px =
707 static_cast<float>(std::hypot(center_dx, center_dy));
708 if (out.center_displacement_px > 2.0f) {
709 out.rejection_reason = "center_displacement_out_of_bounds";
710 return out;
711 }
712
713 std::vector<float> residuals_before;
714 std::vector<float> residuals_after;
715 residuals_before.reserve(matches.size());
716 residuals_after.reserve(matches.size());
717 for (size_t i = 0; i < points_ref.size(); ++i) {
718 const double before_dx = points_warped[i].x - points_ref[i].x;
719 const double before_dy = points_warped[i].y - points_ref[i].y;
720 const float before = static_cast<float>(std::hypot(before_dx, before_dy));
721 const double predicted_x =
722 a00 * points_warped[i].x + a01 * points_warped[i].y + tx;
723 const double predicted_y =
724 a10 * points_warped[i].x + a11 * points_warped[i].y + ty;
725 const float after = static_cast<float>(
726 std::hypot(predicted_x - points_ref[i].x,
727 predicted_y - points_ref[i].y));
728 residuals_before.push_back(before);
729 residuals_after.push_back(after);
730 }
731 const ResidualMetrics before_metrics =
732 summarize_residuals(std::move(residuals_before));
733 const ResidualMetrics after_metrics =
734 summarize_residuals(std::move(residuals_after));
735 out.median_before_px = before_metrics.median;
736 out.p90_before_px = before_metrics.p90;
737 out.rms_before_px = before_metrics.rms;
738 out.median_after_px = after_metrics.median;
739 out.p90_after_px = after_metrics.p90;
740 out.rms_after_px = after_metrics.rms;
741
742 const float required_median_gain =
743 std::max(0.01f, 0.05f * out.median_before_px);
744 const float required_p90_gain = std::max(0.03f, 0.05f * out.p90_before_px);
745 if (out.median_after_px > out.median_before_px - required_median_gain) {
746 out.rejection_reason = "median_not_improved";
747 return out;
748 }
749 if (out.p90_after_px > out.p90_before_px - required_p90_gain) {
750 out.rejection_reason = "p90_not_improved";
751 return out;
752 }
753 if (out.rms_after_px > out.rms_before_px + 1.0e-4f) {
754 out.rejection_reason = "rms_regressed";
755 return out;
756 }
757
758 if (!invert_forward_affine_to_warp(forward, true, out.correction_warp,
759 &out.rejection_reason)) {
760 return out;
761 }
762 out.valid = true;
763 out.rejection_reason = "accepted";
764 return out;
765}

773float smoothstep01(float value) {
774 const float t = std::clamp(value, 0.0f, 1.0f);
775 return t * t * (3.0f - 2.0f * t);
776}

778SmoothLocalWarpModel::Coefficients smooth_local_basis(
779 float x, float y, int image_rows, int image_cols) {
780 SmoothLocalWarpModel::Coefficients basis =
781 SmoothLocalWarpModel::Coefficients::Zero();
782 if (image_rows <= 1 || image_cols <= 1 || x < 0.0f || y < 0.0f ||
783 x > static_cast<float>(image_cols - 1) ||
784 y > static_cast<float>(image_rows - 1)) {
785 return basis;
786 }
787
788 const float nx = x / static_cast<float>(image_cols - 1);
789 const float ny = y / static_cast<float>(image_rows - 1);
790 constexpr float kBoundaryTaperFraction = 0.08f;
791 const float edge_distance =
792 std::min({nx, 1.0f - nx, ny, 1.0f - ny});
793 const float taper = smoothstep01(edge_distance / kBoundaryTaperFraction);
794 if (taper <= 0.0f) {
795 return basis;
796 }
797
798 constexpr float kSigma = 0.28f;
799 constexpr float kInvTwoSigmaSq = 1.0f / (2.0f * kSigma * kSigma);
800 float sum = 0.0f;
801 int index = 0;
802 for (int gy = 0; gy < kSmoothLocalGridSize; ++gy) {
803 const float cy = static_cast<float>(gy) /
804 static_cast<float>(kSmoothLocalGridSize - 1);
805 for (int gx = 0; gx < kSmoothLocalGridSize; ++gx, ++index) {
806 const float cx = static_cast<float>(gx) /
807 static_cast<float>(kSmoothLocalGridSize - 1);
808 const float dx = nx - cx;
809 const float dy = ny - cy;
810 const float value = std::exp(-(dx * dx + dy * dy) * kInvTwoSigmaSq);
811 basis[index] = value;
812 sum += value;
813 }
814 }
815 if (sum > 1.0e-8f) {
816 basis *= taper / sum;
817 }
818 return basis;
819}

821float sample_displacement(const cv::Mat &field, float x, float y) {
822 if (field.empty()) {
823 return 0.0f;
824 }
825 const float clamped_x =
826 std::clamp(x, 0.0f, static_cast<float>(field.cols - 1));
827 const float clamped_y =
828 std::clamp(y, 0.0f, static_cast<float>(field.rows - 1));
829 const int x0 = static_cast<int>(std::floor(clamped_x));
830 const int y0 = static_cast<int>(std::floor(clamped_y));
831 const int x1 = std::min(field.cols - 1, x0 + 1);
832 const int y1 = std::min(field.rows - 1, y0 + 1);
833 const float tx = clamped_x - static_cast<float>(x0);
834 const float ty = clamped_y - static_cast<float>(y0);
835 const float top = field.at<float>(y0, x0) * (1.0f - tx) +
836 field.at<float>(y0, x1) * tx;
837 const float bottom = field.at<float>(y1, x0) * (1.0f - tx) +
838 field.at<float>(y1, x1) * tx;
839 return top * (1.0f - ty) + bottom * ty;
840}

857SmoothLocalRefinementResult estimate_smooth_local_star_refinement(
858 const std::vector<StarPoint> &ref_stars,
859 const std::vector<StarPoint> &warped_stars, int image_rows, int image_cols,
860 float match_radius_px) {
861 SmoothLocalRefinementResult out;
862 out.rejection_reason = "too_few_stars";
863 if (ref_stars.size() < 32 || warped_stars.size() < 32 || image_rows <= 1 ||
864 image_cols <= 1 || match_radius_px <= 0.0f) {
865 return out;
866 }
867
868 struct Match {
869 cv::Point2f ref;
870 cv::Point2f warped;
871 bool validation = false;
872 };
873 const auto mutual_matches =
874 find_mutual_star_matches(ref_stars, warped_stars, match_radius_px);
875 std::vector<Match> matches;
876 matches.reserve(mutual_matches.size());
877 for (const MutualStarMatch &match : mutual_matches) {
878 matches.push_back({
879 {ref_stars[match.ref_idx].x, ref_stars[match.ref_idx].y},
880 {warped_stars[match.warped_idx].x,
881 warped_stars[match.warped_idx].y},
882 false});
883 }
884
885 out.matched_stars = static_cast<int>(matches.size());
886 if (matches.size() < 32) {
887 out.rejection_reason = "too_few_mutual_matches";
888 return out;
889 }
890
891 std::sort(matches.begin(), matches.end(), [](const Match &a, const Match &b) {
892 if (a.ref.y != b.ref.y) {
893 return a.ref.y < b.ref.y;
894 }
895 return a.ref.x < b.ref.x;
896 });
897 std::vector<cv::Point2f> ref_points;
898 ref_points.reserve(matches.size());
899 for (size_t i = 0; i < matches.size(); ++i) {
900 matches[i].validation = (i % 4 == 1);
901 ref_points.push_back(matches[i].ref);
902 if (matches[i].validation) {
903 ++out.validation_stars;
904 } else {
905 ++out.training_stars;
906 }
907 }
908 if (out.training_stars < 24 || out.validation_stars < 8) {
909 out.rejection_reason = "insufficient_holdout_split";
910 return out;
911 }
912
913 const SpatialCoverage coverage =
914 measure_spatial_coverage(ref_points, image_rows, image_cols);
915 out.spatial_coverage = coverage.area_fraction;
916 if (out.spatial_coverage < 0.15f || coverage.x_span < 0.40f ||
917 coverage.y_span < 0.40f) {
918 out.rejection_reason = "insufficient_spatial_coverage";
919 return out;
920 }
921
922 Eigen::MatrixXf design(out.training_stars, kSmoothLocalBasisCount);
923 VectorXf target_x(out.training_stars);
924 VectorXf target_y(out.training_stars);
925 int row = 0;
926 for (const Match &match : matches) {
927 if (match.validation) {
928 continue;
929 }
930 design.row(row) =
931 smooth_local_basis(match.ref.x, match.ref.y, image_rows, image_cols)
932 .transpose();
933 target_x[row] = match.warped.x - match.ref.x;
934 target_y[row] = match.warped.y - match.ref.y;
935 ++row;
936 }
937
938 VectorXf weights = VectorXf::Ones(out.training_stars);
939 SmoothLocalWarpModel::Coefficients coeff_x =
940 SmoothLocalWarpModel::Coefficients::Zero();
941 SmoothLocalWarpModel::Coefficients coeff_y =
942 SmoothLocalWarpModel::Coefficients::Zero();
943 Eigen::MatrixXf targets(out.training_stars, 2);
944 targets.col(0) = target_x;
945 targets.col(1) = target_y;
946 constexpr float kRidgeLambda = 0.08f;
947 constexpr float kHuberDeltaPx = 0.35f;
948 for (int iteration = 0; iteration < 4; ++iteration) {
949 Eigen::MatrixXf weighted_design = design;
950 Eigen::MatrixXf weighted_targets = targets;
951 for (int i = 0; i < out.training_stars; ++i) {
952 const float sqrt_weight = std::sqrt(std::max(0.0f, weights[i]));
953 weighted_design.row(i) *= sqrt_weight;
954 weighted_targets.row(i) *= sqrt_weight;
955 }
956 Eigen::Matrix<float, kSmoothLocalBasisCount, kSmoothLocalBasisCount>
957 normal = weighted_design.transpose() * weighted_design;
958 normal.diagonal().array() += kRidgeLambda;
959 const Eigen::LDLT<decltype(normal)> solver(normal);
960 if (solver.info() != Eigen::Success) {
961 out.rejection_reason = "fit_failed";
962 return out;
963 }
964 const Eigen::Matrix<float, kSmoothLocalBasisCount, 2> coefficients =
965 solver.solve(weighted_design.transpose() * weighted_targets);
966 coeff_x = coefficients.col(0);
967 coeff_y = coefficients.col(1);
968 if (!coefficients.allFinite()) {
969 out.rejection_reason = "non_finite_fit";
970 return out;
971 }
972 const VectorXf residual_x = design * coeff_x - target_x;
973 const VectorXf residual_y = design * coeff_y - target_y;
974 for (int i = 0; i < out.training_stars; ++i) {
975 const float magnitude =
976 std::hypot(residual_x[i], residual_y[i]);
977 weights[i] = magnitude <= kHuberDeltaPx
978 ? 1.0f
979 : kHuberDeltaPx / std::max(magnitude, 1.0e-6f);
980 }
981 }
982
983 out.model.valid = true;
984 out.model.image_rows = image_rows;
985 out.model.image_cols = image_cols;
986 out.model.coeff_x = coeff_x;
987 out.model.coeff_y = coeff_y;
988
989 cv::Mat rendered_displacement_x;
990 cv::Mat rendered_displacement_y;
991 render_smooth_local_displacement(
992 out.model, image_rows, image_cols, 1.0f, 0.0f, 0.0f,
993 rendered_displacement_x, rendered_displacement_y);
994
995 std::vector<float> before_all;
996 std::vector<float> after_all;
997 std::vector<float> before_validation;
998 std::vector<float> after_validation;
999 before_all.reserve(matches.size());
1000 after_all.reserve(matches.size());
1001 before_validation.reserve(static_cast<size_t>(out.validation_stars));
1002 after_validation.reserve(static_cast<size_t>(out.validation_stars));
1003 for (const Match &match : matches) {
1004 const cv::Point2f displacement{
1005 sample_displacement(rendered_displacement_x, match.ref.x, match.ref.y),
1006 sample_displacement(rendered_displacement_y, match.ref.x,
1007 match.ref.y)};
1008 const float before =
1009 std::hypot(match.warped.x - match.ref.x,
1010 match.warped.y - match.ref.y);
1011 const float after =
1012 std::hypot(match.warped.x - (match.ref.x + displacement.x),
1013 match.warped.y - (match.ref.y + displacement.y));
1014 before_all.push_back(before);
1015 after_all.push_back(after);
1016 if (match.validation) {
1017 before_validation.push_back(before);
1018 after_validation.push_back(after);
1019 }
1020 }
1021 const ResidualMetrics all_before =
1022 summarize_residuals(std::move(before_all));
1023 const ResidualMetrics all_after =
1024 summarize_residuals(std::move(after_all));
1025 const ResidualMetrics validation_before =
1026 summarize_residuals(std::move(before_validation));
1027 const ResidualMetrics validation_after =
1028 summarize_residuals(std::move(after_validation));
1029 out.median_before_px = all_before.median;
1030 out.p90_before_px = all_before.p90;
1031 out.rms_before_px = all_before.rms;
1032 out.median_after_px = all_after.median;
1033 out.p90_after_px = all_after.p90;
1034 out.rms_after_px = all_after.rms;
1035 out.validation_median_before_px = validation_before.median;
1036 out.validation_p90_before_px = validation_before.p90;
1037 out.validation_rms_before_px = validation_before.rms;
1038 out.validation_median_after_px = validation_after.median;
1039 out.validation_p90_after_px = validation_after.p90;
1040 out.validation_rms_after_px = validation_after.rms;
1041
1042 const SmoothLocalGeometry geometry = measure_smooth_local_geometry(
1043 rendered_displacement_x, rendered_displacement_y);
1044 out.max_displacement_px = geometry.max_displacement;
1045 out.min_jacobian_determinant = geometry.min_determinant;
1046 out.max_jacobian_determinant = geometry.max_determinant;
1047 out.min_local_scale = geometry.min_scale;
1048 out.max_local_scale = geometry.max_scale;
1049
1050 if (out.max_displacement_px > 1.5f) {
1051 out.rejection_reason = "displacement_out_of_bounds";
1052 out.model.valid = false;
1053 return out;
1054 }
1055 if (out.min_jacobian_determinant < 0.94f ||
1056 out.max_jacobian_determinant > 1.06f || out.min_local_scale < 0.96f ||
1057 out.max_local_scale > 1.04f) {
1058 out.rejection_reason = "jacobian_out_of_bounds";
1059 out.model.valid = false;
1060 return out;
1061 }
1062
1063 const float all_median_gain =
1064 std::max(0.015f, 0.04f * out.median_before_px);
1065 const float all_p90_gain = std::max(0.04f, 0.05f * out.p90_before_px);
1066 const float validation_median_gain =
1067 std::max(0.015f, 0.04f * out.validation_median_before_px);
1068 const float validation_p90_gain =
1069 std::max(0.04f, 0.05f * out.validation_p90_before_px);
1070 if (out.median_after_px > out.median_before_px - all_median_gain) {
1071 out.rejection_reason = "median_not_improved";
1072 out.model.valid = false;
1073 return out;
1074 }
1075 if (out.p90_after_px > out.p90_before_px - all_p90_gain) {
1076 out.rejection_reason = "p90_not_improved";
1077 out.model.valid = false;
1078 return out;
1079 }
1080 if (out.rms_after_px > out.rms_before_px + 1.0e-4f) {
1081 out.rejection_reason = "rms_regressed";
1082 out.model.valid = false;
1083 return out;
1084 }
1085 if (out.validation_median_after_px >
1086 out.validation_median_before_px - validation_median_gain) {
1087 out.rejection_reason = "validation_median_not_improved";
1088 out.model.valid = false;
1089 return out;
1090 }
1091 if (out.validation_p90_after_px >
1092 out.validation_p90_before_px - validation_p90_gain) {
1093 out.rejection_reason = "validation_p90_not_improved";
1094 out.model.valid = false;
1095 return out;
1096 }
1097 if (out.validation_rms_after_px >
1098 out.validation_rms_before_px + 1.0e-4f) {
1099 out.rejection_reason = "validation_rms_regressed";
1100 out.model.valid = false;
1101 return out;
1102 }
1103
1104 out.valid = true;
1105 out.rejection_reason = "accepted";
1106 return out;
1107}

1152 int hi = 0;
1153 float fraction = 0.0f;
1154 bool valid = false;
1155 };
1156 const auto make_axis_samples = [](int output_size, float offset, float scale,
1157 int model_size, int coarse_size) {
1158 std::vector<AxisSample> samples(static_cast<size_t>(output_size));
1159 const float coarse_scale = static_cast<float>(coarse_size - 1) /
1160 static_cast<float>(model_size - 1);
1161 for (int output = 0; output < output_size; ++output) {
1162 const float model_coordinate =
1163 (static_cast<float>(output) - offset) * scale;
1164 if (model_coordinate < 0.0f ||
1165 model_coordinate > static_cast<float>(model_size - 1)) {
1166 continue;
1167 }
1168 const float coarse_coordinate = model_coordinate * coarse_scale;
1169 AxisSample &sample = samples[static_cast<size_t>(output)];
1170 sample.lo = static_cast<int>(std::floor(coarse_coordinate));
1171 sample.hi = std::min(coarse_size - 1, sample.lo + 1);
1172 sample.fraction = coarse_coordinate - static_cast<float>(sample.lo);
1173 sample.valid = true;
1174 }
1175 return samples;
1176 };
1177 const std::vector<AxisSample> x_samples = make_axis_samples(
1178 output_cols, model_offset_x, model_coordinate_scale, model.image_cols,
1179 coarse_cols);
1180 const std::vector<AxisSample> y_samples = make_axis_samples(
1181 output_rows, model_offset_y, model_coordinate_scale, model.image_rows,
1182 coarse_rows);
1183 const float inverse_scale = 1.0f / model_coordinate_scale;
1184 for (int y = 0; y < output_rows; ++y) {
1185 const AxisSample &ys = y_samples[static_cast<size_t>(y)];
1186 if (!ys.valid) {
1187 continue;
1188 }
1189 float *output_x_row = displacement_x.ptr<float>(y);
1190 float *output_y_row = displacement_y.ptr<float>(y);
1191 const float *coarse_x_top = coarse_x.ptr<float>(ys.lo);
1192 const float *coarse_x_bottom = coarse_x.ptr<float>(ys.hi);
1193 const float *coarse_y_top = coarse_y.ptr<float>(ys.lo);
1194 const float *coarse_y_bottom = coarse_y.ptr<float>(ys.hi);
1195 for (int x = 0; x < output_cols; ++x) {
1196 const AxisSample &xs = x_samples[static_cast<size_t>(x)];
1197 if (!xs.valid) {
1198 continue;
1199 }
1200 const float top_x = coarse_x_top[xs.lo] * (1.0f - xs.fraction) +
1201 coarse_x_top[xs.hi] * xs.fraction;
1202 const float bottom_x =
1203 coarse_x_bottom[xs.lo] * (1.0f - xs.fraction) +
1204 coarse_x_bottom[xs.hi] * xs.fraction;
1205 const float top_y = coarse_y_top[xs.lo] * (1.0f - xs.fraction) +
1206 coarse_y_top[xs.hi] * xs.fraction;
1207 const float bottom_y =
1208 coarse_y_bottom[xs.lo] * (1.0f - xs.fraction) +
1209 coarse_y_bottom[xs.hi] * xs.fraction;
1210 output_x_row[x] =
1211 (top_x * (1.0f - ys.fraction) + bottom_x * ys.fraction) *
1212 inverse_scale;
1213 output_y_row[x] =
1214 (top_y * (1.0f - ys.fraction) + bottom_y * ys.fraction) *
1215 inverse_scale;
1216 }
1217 }

1371struct SimilarityResult {
1372 bool ok = false;
1373 float scale = 1.0f;
1374 float theta = 0.0f; // radians
1375 Eigen::Vector2f t{0.0f, 0.0f};
1376 int inliers = 0;
1377 float mean_err = 1.0e9f;
1378};
1379
1384SimilarityResult score_similarity(const std::vector<StarPoint> &mov,
1385 const std::vector<StarPoint> &ref,
1386 float scale, float theta,
1387 const Eigen::Vector2f &t,
1388 float inlier_tol_px) {
1389 if (mov.empty() || ref.empty())
1390 return {};
1391 const float ct = std::cos(theta);
1392 const float st = std::sin(theta);
1393 int inl = 0;
1394 float err_sum = 0.0f;
1395 for (const auto &m : mov) {
1396 const float xr = scale * (ct * m.x - st * m.y) + t.x();
1397 const float yr = scale * (st * m.x + ct * m.y) + t.y();
1398 float best = std::numeric_limits<float>::max();
1399 for (const auto &r : ref) {
1400 const float dx = xr - r.x;
1401 const float dy = yr - r.y;
1402 const float d = std::sqrt(dx * dx + dy * dy);
1403 if (d < best)
1404 best = d;
1405 }
1406 if (best < inlier_tol_px) {
1407 ++inl;
1408 err_sum += best;
1409 }
1410 }
1411 SimilarityResult res;
1412 res.ok = inl > 0;
1413 res.inliers = inl;
1414 res.mean_err = (inl > 0) ? (err_sum / static_cast<float>(inl)) : res.mean_err;
1415 res.scale = scale;
1416 res.theta = theta;
1417 res.t = t;
1418 return res;
1419}

1421static std::vector<std::pair<cv::Point2f, cv::Point2f>>
1426build_similarity_consensus_pairs(const std::vector<StarPoint> &mov,
1427 const std::vector<StarPoint> &ref,
1428 const SimilarityResult &best,
1429 float inlier_tol_px) {
1430 struct Candidate {
1431 int mov_idx = -1;
1432 int ref_idx = -1;
1433 float dist = 0.0f;
1434 };
1435
1436 std::vector<Candidate> candidates;
1437 candidates.reserve(mov.size());
1438 const float ct = std::cos(best.theta);
1439 const float st = std::sin(best.theta);
1440 for (size_t mi = 0; mi < mov.size(); ++mi) {
1441 const auto &m = mov[mi];
1442 const float xr = best.scale * (ct * m.x - st * m.y) + best.t.x();
1443 const float yr = best.scale * (st * m.x + ct * m.y) + best.t.y();
1444 int best_ref_idx = -1;
1445 float best_dist = std::numeric_limits<float>::max();
1446 for (size_t ri = 0; ri < ref.size(); ++ri) {
1447 const float dx = xr - ref[ri].x;
1448 const float dy = yr - ref[ri].y;
1449 const float d = std::sqrt(dx * dx + dy * dy);
1450 if (d < best_dist) {
1451 best_dist = d;
1452 best_ref_idx = static_cast<int>(ri);
1453 }
1454 }
1455 if (best_ref_idx >= 0 && best_dist < inlier_tol_px) {
1456 candidates.push_back(
1457 {static_cast<int>(mi), best_ref_idx, best_dist});
1458 }
1459 }
1460
1461 std::sort(candidates.begin(), candidates.end(),
1462 [](const Candidate &a, const Candidate &b) {
1463 return a.dist < b.dist;
1464 });
1465 std::vector<char> ref_used(ref.size(), 0);
1466 std::vector<std::pair<cv::Point2f, cv::Point2f>> pairs;
1467 pairs.reserve(candidates.size());
1468 for (const Candidate &c : candidates) {
1469 if (c.ref_idx < 0 || c.ref_idx >= static_cast<int>(ref.size()) ||
1470 ref_used[static_cast<size_t>(c.ref_idx)] != 0) {
1471 continue;
1472 }
1473 ref_used[static_cast<size_t>(c.ref_idx)] = 1;
1474 pairs.emplace_back(cv::Point2f(mov[static_cast<size_t>(c.mov_idx)].x,
1475 mov[static_cast<size_t>(c.mov_idx)].y),
1476 cv::Point2f(ref[static_cast<size_t>(c.ref_idx)].x,
1477 ref[static_cast<size_t>(c.ref_idx)].y));
1478 }
1479 return pairs;
1480}

1488 const SimilarityResult &best, bool allow_rotation, float inlier_tol_px,

1538bool similarity_from_pairs(const Eigen::Vector2f &m1, const Eigen::Vector2f &m2,
1539 const Eigen::Vector2f &r1, const Eigen::Vector2f &r2,
1540 bool allow_rotation, float &scale, float &theta,
1541 Eigen::Vector2f &t) {
1542 Eigen::Vector2f v_m = m2 - m1;
1543 Eigen::Vector2f v_r = r2 - r1;
1544 const float len_m = v_m.norm();
1545 const float len_r = v_r.norm();
1546 if (len_m < 1.0e-3f || len_r < 1.0e-3f)
1547 return false;
1548 scale = len_r / len_m;
1549 if (!std::isfinite(scale) || scale <= 0.0f)
1550 return false;
1551 theta = allow_rotation
1552 ? (std::atan2(v_r.y(), v_r.x()) - std::atan2(v_m.y(), v_m.x()))
1553 : 0.0f;
1554 const float ct = std::cos(theta);
1555 const float st = std::sin(theta);
1556 Eigen::Matrix2f R;
1557 R << ct, -st, st, ct;
1558 t = r1 - scale * (R * m1);
1559 return true;
1560}

1562// =====================================================================
1563// Robust multi-scale Phase+ECC with gradient pre-processing
1564// Handles nebula/cloud gradients and large rotations better.
1565// =====================================================================
1566
1571static Matrix2Df gradient_preprocess(const Matrix2Df &img) {
1572 cv::Mat f(img.rows(), img.cols(), CV_32F,
1573 const_cast<float *>(img.data()));
1574 // Laplacian of Gaussian: removes low-frequency gradients (nebula, clouds)
1575 // while preserving high-frequency structure (stars, edges)
1576 cv::Mat blurred;
1577 cv::GaussianBlur(f, blurred, cv::Size(0, 0), 2.0);
1578 cv::Mat laplacian;
1579 cv::Laplacian(blurred, laplacian, CV_32F, 3);
1580
1581 // Normalize to [0,1]
1582 double minV, maxV;
1583 cv::minMaxLoc(laplacian, &minV, &maxV);
1584 if (maxV - minV < 1e-10)
1585 maxV = minV + 1.0;
1586 cv::Mat norm = (laplacian - minV) / (maxV - minV);
1587
1588 Matrix2Df result(img.rows(), img.cols());
1589 std::memcpy(result.data(), norm.data,
1590 static_cast<size_t>(img.size()) * sizeof(float));
1591 return result;
1592}

1598RegistrationResult robust_phase_ecc(const Matrix2Df &mov,
1599 const Matrix2Df &ref,
1600 bool allow_rotation) {
1601 RegistrationResult res;
1602 res.warp = identity_warp();
1603 res.success = false;
1604 res.correlation = 0.0f;
1605
1606 // Gradient pre-processing: removes nebula/cloud gradients
1607 Matrix2Df mov_grad = gradient_preprocess(mov);
1608 Matrix2Df ref_grad = gradient_preprocess(ref);
1609
1610 // Multi-scale pyramid: coarse-to-fine for large displacements/rotations
1611 // Level 2 (4x downsampled) → Level 1 (2x) → Level 0 (original)
1612 std::vector<Matrix2Df> mov_pyr = {mov_grad};
1613 std::vector<Matrix2Df> ref_pyr = {ref_grad};
1614
1615 // Build 2 pyramid levels
1616 for (int level = 0; level < 2; ++level) {
1617 mov_pyr.push_back(core::downsample2x2_mean(mov_pyr.back()));
1618 ref_pyr.push_back(core::downsample2x2_mean(ref_pyr.back()));
1619 }
1620
1621 // Start from coarsest level
1622 WarpMatrix current_warp = identity_warp();
1623
1624 for (int level = static_cast<int>(mov_pyr.size()) - 1; level >= 0; --level) {
1625 const Matrix2Df &m = mov_pyr[static_cast<size_t>(level)];
1626 const Matrix2Df &r = ref_pyr[static_cast<size_t>(level)];
1627
1628 Matrix2Df m_ecc = prepare_ecc_image(m);
1629 Matrix2Df r_ecc = prepare_ecc_image(r);
1630
1631 if (level == static_cast<int>(mov_pyr.size()) - 1) {
1632 // Coarsest level: estimate initial translation + rotation
1633 auto [dx, dy] = phasecorr_translation(m_ecc, r_ecc);
1634 // phaseCorrelate reports the forward content displacement. ECC and
1635 // apply_warp() consume the inverse-map translation, so negate it for
1636 // the initial warp seed.
1637 dx = -dx;
1638 dy = -dy;
1639 current_warp(0, 2) = dx;
1640 current_warp(1, 2) = dy;
1641
1642 if (allow_rotation) {
1643 cv::Mat r_cv(r_ecc.rows(), r_ecc.cols(), CV_32F,
1644 const_cast<float *>(r_ecc.data()));
1645 cv::Mat m_cv(m_ecc.rows(), m_ecc.cols(), CV_32F,
1646 const_cast<float *>(m_ecc.data()));
1647 float rot = estimate_rotation_logpolar(r_cv, m_cv);
1648 float th = rot * 3.14159265f / 180.0f;
1649 float ct = std::cos(th);
1650 float st = std::sin(th);
1651 float cx = static_cast<float>(m_ecc.cols()) * 0.5f;
1652 float cy = static_cast<float>(m_ecc.rows()) * 0.5f;
1653 float tx_rot = cx * (1.0f - ct) + cy * st;
1654 float ty_rot = cy * (1.0f - ct) - cx * st;
1655 current_warp << ct, -st, dx + tx_rot, st, ct, dy + ty_rot;
1656 }
1657 } else {
1658 // Scale warp from previous (coarser) level to current level
1659 current_warp(0, 2) *= 2.0f;
1660 current_warp(1, 2) *= 2.0f;
1661 }
1662
1663 // ECC refinement at this level
1664 int iters = (level == 0) ? 200 : 100;
1665 float eps = (level == 0) ? 1e-6f : 1e-4f;
1666 RegistrationResult level_res =
1667 ecc_warp(m_ecc, r_ecc, allow_rotation, current_warp, iters, eps);
1668 if (level_res.success) {
1669 current_warp = level_res.warp;
1670 res = level_res;
1671 }
1672 // If ECC fails at coarse level, still try finer levels with current seed
1673 }
1674
1675 // NOTE: OpenCV findTransformECC returns a warp that is intended to be used
1676 // directly with warpAffine(..., WARP_INVERSE_MAP) to align moving→reference.
1677 // Our apply_warp() also uses WARP_INVERSE_MAP, so we must NOT invert here.
1678 return res;
1679}

1698 std::vector<Matrix2Df> ref_pyr = {ref_grad};
1699 for (int level = 0; level < 2; ++level) {
1700 mov_pyr.push_back(core::downsample2x2_mean(mov_pyr.back()));

1716

1723 const float eps = (level == 0) ? 1e-6f : 1e-4f;

1727 current_warp = level_res.warp;
1728 res = level_res;
1729 }
1730 }

1853 res.error_message)) {
1854 return res;

1858 float th_fw = best.theta;
1859 float tx_fw = best.t.x();

1863 float sn_fw = std::sin(th_fw);

1867 // Scale_inv = 1/s
1868 // Rot_inv = -theta
1869 // T_inv = - (1/s) * R^T * t

1873 float sn_inv = -sn_fw; // sin(-th) = -sin(th)
1874

1877 float a10 = s_inv * sn_inv;
1878 float a11 = s_inv * c_inv;
1879
1880 // t_inv = - S_inv * t_fw = - [a00 a01; a10 a11] * [tx; ty]
1881 float tx_inv = -(a00 * tx_fw + a01 * ty_fw);
1882 float ty_inv = -(a10 * tx_fw + a11 * ty_fw);
1883
1884 res.warp << a00, a01, tx_inv, a10, a11, ty_inv;
1885 } else {
1886 res.error_message = "no_consensus";
1887 }
1888 return res;
1889}

1892// Triangle-based asterism matching (astroalign-style, rotation-invariant)
1893// =====================================================================
1894
1895struct Triangle {
1896 int i, j, k; // star indices
1897 float sides[3]; // sorted side lengths (ascending)
1898 float perimeter;
1899 float ratios[2]; // sides[0]/sides[2], sides[1]/sides[2] — invariants
1900 float signed_area2; // orientation-preserving triangle discriminator
1901};
1902
1903static std::vector<Triangle>
1908build_triangles(const std::vector<StarPoint> &stars, int max_triangles) {
1909 const int n = static_cast<int>(stars.size());
1910 std::vector<Triangle> tris;
1911 if (n < 3)
1912 return tris;
1913
1914 // Limit combinatorial explosion: use a wider star subset. The previous
1915 // hard cap of 30 caused dense fields to collapse into many ambiguous
1916 // asterisms, even though plenty of stars were available.
1917 const int limit = std::min(n, 60);
1918 tris.reserve(static_cast<size_t>(limit * (limit - 1) * (limit - 2) / 6));
1919
1920 for (int i = 0; i < limit; ++i) {
1921 for (int j = i + 1; j < limit; ++j) {
1922 for (int k = j + 1; k < limit; ++k) {
1923 float dx_ij = stars[j].x - stars[i].x;
1924 float dy_ij = stars[j].y - stars[i].y;
1925 float dx_ik = stars[k].x - stars[i].x;
1926 float dy_ik = stars[k].y - stars[i].y;
1927 float dx_jk = stars[k].x - stars[j].x;
1928 float dy_jk = stars[k].y - stars[j].y;
1929
1930 float d_ij = std::sqrt(dx_ij * dx_ij + dy_ij * dy_ij);
1931 float d_ik = std::sqrt(dx_ik * dx_ik + dy_ik * dy_ik);
1932 float d_jk = std::sqrt(dx_jk * dx_jk + dy_jk * dy_jk);
1933 const float signed_area2 = dx_ij * dy_ik - dy_ij * dx_ik;
1934
1935 // Skip degenerate triangles
1936 if (d_ij < 3.0f || d_ik < 3.0f || d_jk < 3.0f)
1937 continue;
1938 if (std::fabs(signed_area2) < 9.0f)
1939 continue;
1940
1941 Triangle t;
1942 t.i = i;
1943 t.j = j;
1944 t.k = k;
1945
1946 // Sort sides ascending
1947 float s[3] = {d_ij, d_ik, d_jk};
1948 if (s[0] > s[1]) std::swap(s[0], s[1]);
1949 if (s[1] > s[2]) std::swap(s[1], s[2]);
1950 if (s[0] > s[1]) std::swap(s[0], s[1]);
1951
1952 t.sides[0] = s[0];
1953 t.sides[1] = s[1];
1954 t.sides[2] = s[2];
1955 t.perimeter = s[0] + s[1] + s[2];
1956
1957 // Invariant ratios (scale + rotation invariant)
1958 t.ratios[0] = s[0] / s[2];
1959 t.ratios[1] = s[1] / s[2];
1960 t.signed_area2 = signed_area2;
1961
1962 // Near-equilateral and near-isosceles triangles are highly ambiguous
1963 // under ratio-only matching and poison the downstream affine fit.
1964 if (t.ratios[1] > 0.97f || (t.ratios[1] - t.ratios[0]) < 0.02f) {
1965 continue;
1966 }
1967
1968 tris.push_back(t);
1969 }
1970 }
1971 }
1972
1973 // Keep largest triangles (more robust)
1974 if (static_cast<int>(tris.size()) > max_triangles) {
1975 std::sort(tris.begin(), tris.end(),
1976 [](const Triangle &a, const Triangle &b) {
1977 return a.perimeter > b.perimeter;
1978 });
1979 tris.resize(static_cast<size_t>(max_triangles));
1980 }
1981
1982 return tris;
1983}

2032 struct PairVote {

2039 pair_votes.reserve(static_cast<size_t>(max_tris));

2043 float best_dist = ratio_tol * 2.0f;
2044 float second_best_dist = ratio_tol * 2.0f;
2045 const Triangle *best_rt = nullptr;

2047 for (const auto &rt : ref_tris) {
2048 if ((mt.signed_area2 > 0.0f) != (rt.signed_area2 > 0.0f)) {
2049 continue;

2052 float dr1 = mt.ratios[1] - rt.ratios[1];

2056 best_dist = d;
2057 best_rt = &rt;
2058 } else if (d < second_best_dist) {
2059 second_best_dist = d;

2070 // Determine vertex correspondence by matching sorted side lengths
2071 // The vertices opposite to the shortest/medium/longest sides correspond
2072 // mov triangle vertices: i, j, k
2073 // We need to figure out which vertex is opposite which side
2074 // Side ij is between i,j — opposite vertex is k
2075 // Side ik is between i,k — opposite vertex is j
2076 // Side jk is between j,k — opposite vertex is i
2077 auto vertex_order = [](const std::vector<StarPoint> &stars,
2078 const Triangle &t) -> std::array<int, 3> {
2079 float d_ij = 0, d_ik = 0, d_jk = 0;
2080 {
2081 float dx = stars[t.j].x - stars[t.i].x;
2082 float dy = stars[t.j].y - stars[t.i].y;
2083 d_ij = std::sqrt(dx * dx + dy * dy);

2086 float dx = stars[t.k].x - stars[t.i].x;
2087 float dy = stars[t.k].y - stars[t.i].y;

2091 float dx = stars[t.k].x - stars[t.j].x;

2097 float len;
2098 int opposite;
2099 };

2101 if (sv[0].len > sv[1].len) std::swap(sv[0], sv[1]);
2102 if (sv[1].len > sv[2].len) std::swap(sv[1], sv[2]);
2103 if (sv[0].len > sv[1].len) std::swap(sv[0], sv[1]);
2104 return {sv[0].opposite, sv[1].opposite, sv[2].opposite};
2105 };

2108 auto ref_order = vertex_order(ref_stars, *best_rt);
2109
2110 for (int v = 0; v < 3; ++v) {
2111 const int mov_idx = mov_order[v];
2112 const int ref_idx = ref_order[v];

2114 (static_cast<std::uint64_t>(static_cast<std::uint32_t>(mov_idx))
2115 << 32) |
2116 static_cast<std::uint32_t>(ref_idx);
2117 auto &vote = pair_votes[key];
2118 vote.mov_idx = mov_idx;
2119 vote.ref_idx = ref_idx;
2120 vote.votes += 1;
2121 vote.score_sum += best_dist;
2122 }
2123
2124 if (matches_found > 200)
2125 break;
2126 }

2129 ranked_pairs.reserve(pair_votes.size());

2131 ranked_pairs.push_back(entry.second);
2132 }
2133 std::sort(ranked_pairs.begin(), ranked_pairs.end(),
2134 [](const PairVote &a, const PairVote &b) {
2135 if (a.votes != b.votes) {
2136 return a.votes > b.votes;
2137 }
2138 return a.score_sum < b.score_sum;
2139 });
2140

2142 // (dx,dy). Compute per-pair support = sum of votes of other pairs whose
2143 // implied shift is within shift_radius px. The pair with highest support
2144 // is the shift-cluster anchor; keep only pairs within shift_radius of it.
2145 // This eliminates false matches that dominate vote counts in crowded fields.

2149 float dx, dy;

2159 mov_stars[static_cast<size_t>(vote.mov_idx)].x;
2160 float dy = ref_stars[static_cast<size_t>(vote.ref_idx)].y -
2161 mov_stars[static_cast<size_t>(vote.mov_idx)].y;

2192 float ddx = p.dx - anchor_dx;
2193 float ddy = p.dy - anchor_dy;

2196 cand_pairs.end());

2199 std::sort(cand_pairs.begin(), cand_pairs.end(),
2200 [](const PairShift &a, const PairShift &b) {
2201 return a.votes > b.votes;
2202 });

2205 pts_mov.reserve(cand_pairs.size());
2206 pts_ref.reserve(cand_pairs.size());

2208 std::vector<uint8_t> ref_used(ref_stars.size(), 0);

2211 ref_used[static_cast<size_t>(p.ref_idx)] != 0) {
2212 continue;
2213 }
2214 mov_used[static_cast<size_t>(p.mov_idx)] = 1;
2215 ref_used[static_cast<size_t>(p.ref_idx)] = 1;
2216 pts_mov.push_back(cv::Point2f(mov_stars[static_cast<size_t>(p.mov_idx)].x,
2217 mov_stars[static_cast<size_t>(p.mov_idx)].y));
2218 pts_ref.push_back(cv::Point2f(ref_stars[static_cast<size_t>(p.ref_idx)].x,
2219 ref_stars[static_cast<size_t>(p.ref_idx)].y));
2220 }
2221
2222 if (pts_mov.size() < static_cast<size_t>(std::max(3, min_inliers))) {
2223 res.error_message = "few_triangle_matches";
2224 return res;
2225 }
2226
2227 // Use RANSAC to find the best similarity transform from correspondences
2228 cv::Mat inliers;
2229 cv::Mat A = estimate_affine_family_transform(pts_mov, pts_ref,
2230 transform_model, inliers);
2231 if (A.empty()) {
2232 res.error_message = "transform_fail";
2233 return res;
2234 }
2235 if (!invert_forward_affine_to_warp(A, allow_rotation, res.warp,
2236 &res.error_message)) {
2237 return res;
2238 }
2239 int inl = inliers.empty() ? 0 : cv::countNonZero(inliers);
2240 res.correlation =
2241 pts_mov.empty()
2242 ? 0.0f
2243 : static_cast<float>(inl) / static_cast<float>(pts_mov.size());
2244 res.success = inl >= min_inliers;
2245 return res;
2246}

2309static float compute_ncc(const Matrix2Df &a, const Matrix2Df &b) {
2310 const int n = a.size();
2311 if (n <= 0 || n != b.size())
2312 return 0.0f;
2313 const float *da = a.data();
2314 const float *db = b.data();
2315 double ma = 0, mb = 0;
2316 for (int i = 0; i < n; ++i) {
2317 ma += da[i];
2318 mb += db[i];
2319 }
2320 ma /= n;
2321 mb /= n;
2322 double sab = 0, saa = 0, sbb = 0;
2323 for (int i = 0; i < n; ++i) {
2324 double va = da[i] - ma;
2325 double vb = db[i] - mb;
2326 sab += va * vb;
2327 saa += va * va;
2328 sbb += vb * vb;
2329 }
2330 double den = std::sqrt(saa * sbb);
2331 return (den > 1e-10) ? static_cast<float>(sab / den) : 0.0f;
2332}

2357float compute_ncc_masked(const Matrix2Df &a, const Matrix2Df &b,
2358 const cv::Mat &mask, int *used_pixels) {
2359 if (a.size() <= 0 || a.size() != b.size() || mask.empty() ||
2360 mask.rows != a.rows() || mask.cols != a.cols()) {
2361 if (used_pixels) {
2362 *used_pixels = 0;
2363 }
2364 return 0.0f;
2365 }
2366
2367 const float *da = a.data();

2370 double mb = 0.0;
2371 int n = 0;
2372 for (int y = 0; y < mask.rows; ++y) {
2373 const float *pm = mask.ptr<float>(y);
2374 const int row_off = y * mask.cols;
2375 for (int x = 0; x < mask.cols; ++x) {
2376 if (pm[x] <= 0.5f) {
2377 continue;
2378 }
2379 const int idx = row_off + x;
2380 ma += da[idx];
2381 mb += db[idx];

2383 }
2384 }
2385 if (used_pixels) {
2386 *used_pixels = n;
2387 }
2388 if (n <= 1) {
2389 return 0.0f;
2390 }
2391
2392 ma /= static_cast<double>(n);
2393 mb /= static_cast<double>(n);
2394 double sab = 0.0;
2395 double saa = 0.0;
2396 double sbb = 0.0;
2397 for (int y = 0; y < mask.rows; ++y) {
2398 const float *pm = mask.ptr<float>(y);
2399 const int row_off = y * mask.cols;
2400 for (int x = 0; x < mask.cols; ++x) {
2401 if (pm[x] <= 0.5f) {
2402 continue;
2403 }
2404 const int idx = row_off + x;
2405 const double va = static_cast<double>(da[idx]) - ma;
2406 const double vb = static_cast<double>(db[idx]) - mb;
2407 sab += va * vb;
2408 saa += va * va;
2409 sbb += vb * vb;
2410 }
2411 }
2412 const double den = std::sqrt(saa * sbb);
2413 return (den > 1e-10) ? static_cast<float>(sab / den) : 0.0f;

2450 << out.ncc_identity
2451 << " threshold=" << (1.0f - identity_gate_margin)
2452 << std::endl;

2458 auto stars_ref = detect_stars_simple(ref, rcfg.star_topk, rcfg.enable_local_background_subtraction);

2469 // Blur before NCC: makes the quality metric robust against hot pixels and
2470 // sharp point sources that cause extreme NCC sensitivity to sub-pixel shifts.
2471 // sigma=1.5 matches the ECC pre-processing blur (prepare_ecc_image).
2472 // Important: clamp to zero first so that negative background-subtracted
2473 // values do not bleed into star peaks via Gaussian blur.
2474 auto blur_for_ncc = [](const Matrix2Df &img) -> Matrix2Df {
2475 cv::Mat m(img.rows(), img.cols(), CV_32F,
2476 const_cast<float *>(img.data()));

2478 cv::max(m, 0.0f, clamped); // remove negative background artefacts
2479 cv::GaussianBlur(clamped, clamped, cv::Size(0, 0), 1.5);

2481 std::memcpy(result.data(), clamped.data,
2482 static_cast<size_t>(img.size()) * sizeof(float));
2483 return result;
2484 };

2486 // call (lazily, so the identity/early-exit paths pay nothing) instead of
2487 // once per candidate. Same blur on the same input: NCC values unchanged.
2488 Matrix2Df mov_b_cache, ref_b_cache;
2489 bool ncc_inputs_blurred = false;

2492 mov_b_cache = blur_for_ncc(mov);
2493 ref_b_cache = blur_for_ncc(ref);
2494 ncc_inputs_blurred = true;

2498 // Try a cascade method and validate with NCC
2499 auto try_method = [&](RegistrationResult rr,
2500 const std::string &method) -> bool {
2501 if (!rr.success) {
2502 if (diag) {

2504 << " FAIL: " << rr.error_message << std::endl;

2509 const cv::Mat valid_mask = warp_valid_mask(mov, rr.warp);
2510 int overlap_pixels = 0;
2511 blurred_inputs();

2513 const Matrix2Df warped_b = blur_for_ncc(warped);
2514 const Matrix2Df &ref_b = ref_b_cache;

2516 compute_ncc_masked(mov_b, ref_b, valid_mask, &overlap_pixels);

2518 if (diag) {

2529 << " a10=" << rr.warp(1,0) << " a11=" << rr.warp(1,1)

2536 // metrics are not penalised by interpolation blur.

2545 // aligned. A drop > 0.02 below blurred-identity indicates a false warp

2549 (ncc >= ncc_identity_overlap - 0.02f) &&
2550 (out.ncc_identity > 0.7f);

2552 (!near_identity && ncc < ncc_identity_overlap + min_ncc_improvement))
2553 return false; // warp doesn't improve alignment — reject
2554 out.reg = rr;
2555 // For near-identity warps use the uninterpolated NCC so downstream
2556 // quality metrics are not degraded by interpolation blur.

2558 out.method_used = method;
2559 out.ncc_warped = out.reg.correlation;
