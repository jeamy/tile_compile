37validate_linearity_frame(const Matrix2Df &img,
38 const std::string &strictness) {
39 LinearityFrameResult out;
40 if (img.size() <= 0)
41 return out;
42
43 cv::Mat cv_img(img.rows(), img.cols(), CV_32F,
44 const_cast<float *>(img.data()));
45 cv::Mat small = cv_img;
46
47 const int max_dim = 256;
48 if (cv_img.rows > max_dim || cv_img.cols > max_dim) {
49 float scale = static_cast<float>(max_dim) /
50 static_cast<float>(std::max(cv_img.rows, cv_img.cols));
51 cv::resize(cv_img, small, cv::Size(), scale, scale, cv::INTER_AREA);
52 }
53
54 std::vector<float> values;
55 values.reserve(static_cast<size_t>(small.rows) *
56 static_cast<size_t>(small.cols));
57 for (int y = 0; y < small.rows; ++y) {
58 const float *row = small.ptr<float>(y);
59 for (int x = 0; x < small.cols; ++x) {
60 float v = row[x];
61 if (std::isfinite(v))
62 values.push_back(v);
63 }
64 }
65
66 if (values.empty()) {
67 return out;
68 }
69
70 double mean = 0.0;
71 double m2 = 0.0;
72 for (size_t i = 0; i < values.size(); ++i) {
73 double x = static_cast<double>(values[i]);
74 double delta = x - mean;
75 mean += delta / static_cast<double>(i + 1);
76 double delta2 = x - mean;
77 m2 += delta * delta2;
78 }
79 double var =
80 (values.size() > 1) ? (m2 / static_cast<double>(values.size() - 1)) : 0.0;
81 double stddev = std::sqrt(std::max(0.0, var));
82
83 std::vector<float> sorted = values;
84 std::sort(sorted.begin(), sorted.end());
85 float p0 = sorted.front();
86 float p1 = core::percentile_from_sorted(sorted, 1.0f);
87 float p5 = core::percentile_from_sorted(sorted, 5.0f);
88 float p50 = core::percentile_from_sorted(sorted, 50.0f);
89 float p95 = core::percentile_from_sorted(sorted, 95.0f);
90 float p99 = core::percentile_from_sorted(sorted, 99.0f);
91 float p100 = sorted.back();
92
93 // Use robust percentiles for skewness/kurtosis that exclude stars.
94 // The previous formula (p99-p50)/(p50-p1) was dominated by bright stars
95 // in astro images, giving skewness~18 regardless of actual linearity.
96 // Trim to p25-p75 interquartile range for the moment-based checks.
97 float p25 = core::percentile_from_sorted(sorted, 25.0f);
98 float p75 = core::percentile_from_sorted(sorted, 75.0f);
99 float iqr = (p75 - p25) + 1.0e-12f;
100 out.skewness = (p75 - p50) / iqr;
101 // Kurtosis: inverted tail-to-body ratio. Low values = heavy-tailed
102 // (linear astro with sharp star peaks), high values = light-tailed
103 // (non-linear stretched, compressed dynamic range).
104 out.kurtosis = (p75 - p25) / ((p95 - p5) + 1.0e-12f);
105 // Variance coefficient: normalize by pixel range (not |mean|), because
106 // calibrated frames have mean≈0 which would make this diverge.
107 float pixel_range = (p99 - p1) + 1.0e-6f;
108 out.variance_coeff = static_cast<float>(stddev / static_cast<double>(pixel_range));
109
110 cv::Mat gx, gy, mag;
111 cv::Sobel(small, gx, CV_32F, 1, 0, 3);
112 cv::Sobel(small, gy, CV_32F, 0, 1, 3);
113 cv::magnitude(gx, gy, mag);
114 double mean_grad = cv::mean(mag)[0];
115 // Normalize gradient by pixel range (p99-p1) instead of mean, because
116 // background-subtracted frames have mean≈0 which would make this diverge.
117 out.gradient_consistency =
118 static_cast<float>(mean_grad / static_cast<double>(pixel_range));
119
120 out.energy_ratio = 0.0f;
121 if (small.rows >= 8 && small.cols >= 8) {
122 cv::Mat dft_img;
123 cv::dft(small, dft_img, cv::DFT_COMPLEX_OUTPUT);
124 std::vector<cv::Mat> planes;
125 cv::split(dft_img, planes);
126 cv::Mat mag2 = planes[0].mul(planes[0]) + planes[1].mul(planes[1]);
127 // Exclude DC bin (0,0) from both numerator and denominator: the DC
128 // is just |sum(pixels)|^2 and carries no linearity information.
129 // For uncalibrated frames it dominates and inflates energy_ratio;
130 // for calibrated frames (mean≈0) it's near-zero. Excluding it
131 // makes the ratio measure actual AC spectral shape.
132 double dc_energy = static_cast<double>(mag2.at<float>(0, 0));
133 double total_energy = cv::sum(mag2)[0] - dc_energy;
134 // Only the top-left corner is the true low-frequency region in
135 // standard (non-shifted) DFT layout. The other 3 corners are Nyquist/
136 // high-frequency aliases and must NOT be counted as low-frequency.
137 int r = std::max(1, std::min(mag2.rows, mag2.cols) / 8);
138 double low_energy = cv::sum(mag2(cv::Rect(0, 0, r, r)))[0] - dc_energy;
139 if (total_energy > 0.0) {
140 out.energy_ratio = static_cast<float>(low_energy / total_energy);
141 }
142 }
143
144 LinearityThresholds th = linearity_thresholds_for(strictness);
145 const float clip_eps =
146 std::max(1.0e-3f, (p99 - p1 + 1.0e-6f) * 1.0e-5f);
147 size_t low_clip_count = 0;
148 size_t high_clip_count = 0;
149 for (float v : values) {
150 if (v <= p0 + clip_eps) ++low_clip_count;
151 if (v >= p100 - clip_eps) ++high_clip_count;
152 }
153 const float low_clip_frac =
154 static_cast<float>(low_clip_count) / static_cast<float>(values.size());
155 const float high_clip_frac =
156 static_cast<float>(high_clip_count) / static_cast<float>(values.size());
157 float clip_fraction_max = 0.02f;
158 if (strictness == "moderate") {
159 clip_fraction_max = 0.05f;
160 } else if (strictness == "permissive") {
161 clip_fraction_max = 0.10f;
162 }
163 const bool low_hard_clip =
164 low_clip_frac > clip_fraction_max && p0 <= 1.0f + clip_eps;
165 const bool high_hard_clip =
166 high_clip_frac > clip_fraction_max && p100 >= 4095.0f - clip_eps;
167 const bool clipping_ok = !low_hard_clip && !high_hard_clip;
168
169 // A single light frame cannot prove camera-response linearity for every
170 // possible object. Keep the hard verdict deliberately conservative and
171 // object-agnostic: robust distribution shape and obvious clipping/compression
172 // are hard checks; variance, gradients and spectral layout remain diagnostics
173 // because they depend strongly on field content (empty sky, star clusters,
174 // nebulosity, galaxy cores, CFA texture).
175 out.moment_ok = (std::fabs(out.skewness) < th.skewness_max) &&
176 (std::fabs(out.kurtosis) < th.kurtosis_max) && clipping_ok;
177 out.spectral_ok = (out.energy_ratio >= th.energy_ratio_min);
178 out.spatial_ok = (out.gradient_consistency < th.gradient_consistency_max);
179
180 out.score =
181 (static_cast<float>(out.moment_ok) + static_cast<float>(out.spectral_ok) +
182 static_cast<float>(out.spatial_ok)) /
183 3.0f;
184 out.is_linear = out.moment_ok;
185 return out;
186}
