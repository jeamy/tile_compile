572float weighted_mean(const std::vector<float>& values,
573 const std::vector<float>& weights) {
574 double sum = 0.0;
575 double wsum = 0.0;
576 for (size_t i = 0; i < values.size(); ++i) {
577 const float v = values[i];
578 const float w = i < weights.size() ? weights[i] : 1.0f;
579 if (!std::isfinite(v) || !std::isfinite(w) || w <= 0.0f) continue;
580 sum += static_cast<double>(v) * static_cast<double>(w);
581 wsum += static_cast<double>(w);
582 }
583 return wsum > 0.0 ? static_cast<float>(sum / wsum) : 0.0f;
584}

586float median_value(std::vector<float> values) {
587 values.erase(std::remove_if(values.begin(), values.end(), [](float v) {
588 return !std::isfinite(v);
589 }), values.end());
590 if (values.empty()) return 0.0f;
591 std::sort(values.begin(), values.end());
592 const size_t mid = values.size() / 2;
593 if (values.size() % 2 == 1) return values[mid];
594 return 0.5f * (values[mid - 1] + values[mid]);
595}

597std::pair<float, float> mean_stddev(const std::vector<float>& values) {
598 std::vector<float> finite;
599 finite.reserve(values.size());
600 for (float v : values) if (std::isfinite(v)) finite.push_back(v);
601 if (finite.empty()) return {0.0f, 0.0f};
602 const double mean = std::accumulate(finite.begin(), finite.end(), 0.0) /
603 static_cast<double>(finite.size());
604 double var = 0.0;
605 for (float v : finite) {
606 const double d = static_cast<double>(v) - mean;
607 var += d * d;
608 }
609 var /= static_cast<double>(finite.size());
610 return {static_cast<float>(mean), static_cast<float>(std::sqrt(var))};
611}

613float reduce_pixel(std::vector<float>& values,
614 const std::vector<float>& weights,
615 const prep::RejectionConfig& rejection,
616 int& rejected_samples,
617 std::vector<float>& kept_values,
618 std::vector<float>& kept_weights,
619 std::vector<float>& next_values,
620 std::vector<float>& next_weights) {
621 if (values.empty()) return 0.0f;
622 const std::string method = rejection.method;
623 if (method == "median") {
624 return median_value(values);
625 }
626
627 if (method == "winsor") {
628 auto [mean, sigma] = mean_stddev(values);
629 if (sigma > 0.0f) {
630 const float lo = mean - rejection.low * sigma;
631 const float hi = mean + rejection.high * sigma;
632 for (float& v : values) {
633 if (!std::isfinite(v)) continue;
634 if (v < lo) {
635 v = lo;
636 ++rejected_samples;
637 } else if (v > hi) {
638 v = hi;
639 ++rejected_samples;
640 }
641 }
642 }
643 return weighted_mean(values, weights);
644 }
645
646 kept_values.clear();
647 kept_weights.clear();
648 kept_values.reserve(values.size());
649 kept_weights.reserve(weights.size());
650 for (size_t i = 0; i < values.size(); ++i) {
651 const float v = values[i];
652 if (!std::isfinite(v)) continue;
653 kept_values.push_back(v);
654 kept_weights.push_back(i < weights.size() ? weights[i] : 1.0f);
655 }
656 const size_t min_keep = static_cast<size_t>(
657 std::ceil(static_cast<float>(kept_values.size()) * rejection.min_fraction));
658 for (int iter = 0; iter < std::max(1, rejection.max_iters); ++iter) {
659 if (kept_values.size() <= std::max<size_t>(2, min_keep)) break;
660 auto [mean, sigma] = mean_stddev(kept_values);
661 if (sigma <= 0.0f || !std::isfinite(sigma)) break;
662 const float lo = mean - rejection.low * sigma;
663 const float hi = mean + rejection.high * sigma;
664 next_values.clear();
665 next_weights.clear();
666 next_values.reserve(kept_values.size());
667 next_weights.reserve(kept_weights.size());
668 int iter_rejected = 0;
669 for (size_t i = 0; i < kept_values.size(); ++i) {
670 const float v = kept_values[i];
671 if (v < lo || v > hi) {
672 ++iter_rejected;
673 continue;
674 }
675 next_values.push_back(v);
676 next_weights.push_back(i < kept_weights.size() ? kept_weights[i] : 1.0f);
677 }
678 if (iter_rejected == 0 || next_values.empty() || next_values.size() < min_keep) break;
679 rejected_samples += iter_rejected;
680 kept_values.swap(next_values);
681 kept_weights.swap(next_weights);
682 }
683 if (kept_values.empty()) return weighted_mean(values, weights);
684 return weighted_mean(kept_values, kept_weights);
685}

687Matrix2Df stack_frame_planes(const std::vector<Matrix2Df>& frames,
688 const std::vector<float>& weights,
689 const prep::RejectionConfig& rejection,
690 int workers,
691 int& rejected_samples,
692 int64_t& total_samples,
693 const std::function<void(int, int)>& row_progress) {
694 if (frames.empty()) return {};
695 const int rows = static_cast<int>(frames.front().rows());
696 const int cols = static_cast<int>(frames.front().cols());
697 Matrix2Df stacked(rows, cols);
698 std::atomic<int> rejected_total{0};
699 std::atomic<int64_t> sample_total{0};
700 std::mutex progress_mutex;
701 parallel_for_indices(static_cast<size_t>(rows),
702 default_parallel_workers(static_cast<size_t>(rows), workers),
703 [&](size_t row_idx) {
704 const int y = static_cast<int>(row_idx);
705 std::vector<float> values;
706 std::vector<float> kept_values;
707 std::vector<float> kept_weights;
708 std::vector<float> next_values;
709 std::vector<float> next_weights;
710 values.reserve(frames.size());
711 kept_values.reserve(frames.size());
712 kept_weights.reserve(frames.size());
713 next_values.reserve(frames.size());
714 next_weights.reserve(frames.size());
715 int rejected_local = 0;
716 int64_t total_local = 0;
717 for (int x = 0; x < cols; ++x) {
718 values.clear();
719 for (const auto& frame : frames) values.push_back(frame(y, x));
720 stacked(y, x) = reduce_pixel(values, weights, rejection, rejected_local,
721 kept_values, kept_weights, next_values, next_weights);
722 total_local += static_cast<int64_t>(frames.size());
723 }
724 rejected_total.fetch_add(rejected_local, std::memory_order_relaxed);
725 sample_total.fetch_add(total_local, std::memory_order_relaxed);
726 if (row_progress) {
727 std::lock_guard<std::mutex> lock(progress_mutex);
728 row_progress(y, rows);
729 }
730 });
731 rejected_samples += rejected_total.load(std::memory_order_relaxed);
732 total_samples += sample_total.load(std::memory_order_relaxed);
733 return stacked;
734}

814void accumulate_batch_stack(const Matrix2Df& batch,
815 float batch_weight,
816 Matrix2Df& accum,
817 Matrix2Df& wsum) {
818 if (batch.size() <= 0) return;
819 if (accum.size() <= 0) {
820 accum = Matrix2Df::Zero(batch.rows(), batch.cols());
821 wsum = Matrix2Df::Zero(batch.rows(), batch.cols());
822 }
823 for (int y = 0; y < batch.rows(); ++y) {
824 for (int x = 0; x < batch.cols(); ++x) {
825 const float v = batch(y, x);
826 if (std::isfinite(v)) {
827 accum(y, x) += v * batch_weight;
828 wsum(y, x) += batch_weight;
829 }
830 }
831 }
832}
833
834Matrix2Df finalize_batch_accum(const Matrix2Df& accum, const Matrix2Df& wsum) {
835 if (accum.size() <= 0 || wsum.size() != accum.size()) return {};
836 Matrix2Df out(accum.rows(), accum.cols());
837 for (int y = 0; y < accum.rows(); ++y) {
838 for (int x = 0; x < accum.cols(); ++x) {
839 out(y, x) = wsum(y, x) > 0.0f ? accum(y, x) / wsum(y, x)
840 : std::numeric_limits<float>::quiet_NaN();
841 }
842 }
843 return out;
844}

846PreprocessStackResult run_preprocess_stacking(
847 const std::string& run_id,
848 const prep::Config& cfg,
849 const runner::PreprocessPipelineContext& pp,
850 const runner::QualityAnalysisContext& qa,
851 const fs::path& run_dir,
852 core::EventEmitter& emitter,
853 std::ostream& event_out) {
854 PreprocessStackResult result;
855 result.method = cfg.rejection.method;
856 result.weighting = cfg.stacking.weighting;
857 result.rejected_input_frames = static_cast<int>(qa.rejected_indices.size());
858
859 emitter.phase_start(run_id, prep::phase_to_string(prep::Phase::STACKING), event_out);
860 const fs::path output_dir = run_dir / "outputs";
861 const fs::path artifact_dir = run_dir / "artifacts" / "preprocess";
862 fs::create_directories(output_dir);
863 fs::create_directories(artifact_dir);
864
865 std::vector<int> accepted;
866 accepted.reserve(qa.accepted_indices.size());
867 for (int idx : qa.accepted_indices) {
868 if (idx >= 0 && static_cast<size_t>(idx) < pp.prewarped_frames.size() &&
869 pp.prewarped_frames.has_data(static_cast<size_t>(idx))) {
870 accepted.push_back(idx);
871 }
872 }
873 if (accepted.empty()) {
874 emitter.phase_end(run_id, prep::phase_to_string(prep::Phase::STACKING), "error",
875 {{"error", "no accepted registered frames to stack"}}, event_out);
876 emitter.run_end(run_id, false, "error", event_out,
877 {{"message", "no accepted registered frames to stack"}});
878 throw std::runtime_error("no accepted registered frames to stack");
879 }
880
881 const int rows = pp.prewarped_frames.rows();
882 const int cols = pp.prewarped_frames.cols();
883 const std::vector<float> weights = stack_weights_for_accepted(cfg, qa);
884
885 const auto linear_batch_plan =
886 raw_stack_sub_batch_plan(rows, cols, 1, accepted.size(), cfg.runtime_limits);
887 const size_t linear_sub_batch =
888 linear_batch_plan.frame_sub_batch_size > 0 ? linear_batch_plan.frame_sub_batch_size
889 : accepted.size();
890 Matrix2Df linear_accum;
891 Matrix2Df linear_wsum;
892 for (size_t batch_start = 0; batch_start < accepted.size(); batch_start += linear_sub_batch) {
893 const size_t batch_end = std::min(batch_start + linear_sub_batch, accepted.size());
894 const size_t batch_size = batch_end - batch_start;
895 std::vector<const float*> batch_frames;
896 std::vector<float> batch_weights;
897 batch_frames.reserve(batch_size);
898 batch_weights.reserve(batch_size);
899 for (size_t i = batch_start; i < batch_end; ++i) {
900 const float* frame_ptr =
901 pp.prewarped_frames.frame_data(static_cast<size_t>(accepted[i]));
902 if (frame_ptr == nullptr) continue;
903 batch_frames.push_back(frame_ptr);
904 batch_weights.push_back(i < weights.size() ? weights[i] : 1.0f);
905 }
906 if (batch_frames.empty()) continue;
907 int batch_rejected = 0;
908 int64_t batch_total = 0;
909 Matrix2Df batch_stack = stack_frame_ptrs(batch_frames, rows, cols, batch_weights,
910 cfg.rejection,
911 cfg.runtime_limits.parallel_workers,
912 batch_rejected, batch_total);
913 result.rejected_samples += batch_rejected;
914 result.total_samples += batch_total;
915 accumulate_batch_stack(batch_stack, static_cast<float>(batch_frames.size()),
916 linear_accum, linear_wsum);
917 pp.prewarped_frames.clear_mappings();
918 emitter.phase_progress(run_id, prep::phase_to_string(prep::Phase::STACKING),
919 0.45f * static_cast<float>(batch_end) /
920 static_cast<float>(accepted.size()),
921 "stacked linear batch " + std::to_string(batch_end) +
922 "/" + std::to_string(accepted.size()),
923 event_out);
924 }
925 Matrix2Df stacked = finalize_batch_accum(linear_accum, linear_wsum);
926 if (stacked.size() <= 0) {
927 emitter.phase_end(run_id, prep::phase_to_string(prep::Phase::STACKING), "error",
928 {{"error", "linear stack produced no output"}}, event_out);
929 emitter.run_end(run_id, false, "error", event_out,
930 {{"message", "linear stack produced no output"}});
931 throw std::runtime_error("linear stack produced no output");
932 }
933
934 // Restore output scaling: the prewarped frames are stored after normalization
935 // (background subtracted, scaled). We need to undo this by applying the
936 // median background + scale back to the stacked output.
937 {
938 std::vector<float> P_mono, P_r, P_g, P_b;
939 std::vector<float> B_mono, B_r, B_g, B_b;
940 for (const auto& s : pp.norm_scales) {
941 if (std::isfinite(s.scale_mono) && s.scale_mono > 0.0f) P_mono.push_back(s.scale_mono);
942 if (std::isfinite(s.scale_r) && s.scale_r > 0.0f) P_r.push_back(s.scale_r);
943 if (std::isfinite(s.scale_g) && s.scale_g > 0.0f) P_g.push_back(s.scale_g);
944 if (std::isfinite(s.scale_b) && s.scale_b > 0.0f) P_b.push_back(s.scale_b);
945 if (std::isfinite(s.background_mono)) B_mono.push_back(s.background_mono);
946 if (std::isfinite(s.background_r)) B_r.push_back(s.background_r);
947 if (std::isfinite(s.background_g)) B_g.push_back(s.background_g);
948 if (std::isfinite(s.background_b)) B_b.push_back(s.background_b);
949 }
950 const float out_scale_mono = core::median_finite_positive(P_mono, 1.0f);
951 const float out_scale_r = core::median_finite_positive(P_r, 1.0f);
952 const float out_scale_g = core::median_finite_positive(P_g, 1.0f);
953 const float out_scale_b = core::median_finite_positive(P_b, 1.0f);
954 const float out_bg_mono = core::median_finite(B_mono, 0.0f);
955 const float out_bg_r = core::median_finite(B_r, 0.0f);
956 const float out_bg_g = core::median_finite(B_g, 0.0f);
957 const float out_bg_b = core::median_finite(B_b, 0.0f);
958 image::apply_output_scaling_inplace(stacked, 0, 0,
959 pp.color_mode, pp.bayer_pattern,
960 out_scale_mono, out_scale_r, out_scale_g, out_scale_b,
961 out_bg_mono, out_bg_r, out_bg_g, out_bg_b, 0.0f);
962 }
963
964 if (cfg.stacking.cosmetic_correction) {
965 stacked = image::cosmetic_correction(
966 stacked, cfg.stacking.cosmetic_correction_sigma, true);
967 }
968
969 io::FitsHeader header;
970 try {
971 if (!pp.effective_frames.empty()) header = io::read_fits_header(pp.effective_frames.front());
972 } catch (...) {}
973 header.set("TC_PRE", std::string("LINEAR_PRESTACK"));
974 header.set("TC_STKFR", static_cast<double>(accepted.size()));
975 header.set("TC_REJ", cfg.rejection.method);
976 header.set("TC_WEIGHT", cfg.stacking.weighting);
977
978 result.stacked_linear_path = output_dir / "stacked_linear.fits";
979 io::write_fits_float(result.stacked_linear_path, stacked, header);
980 result.frames_used = static_cast<int>(accepted.size());
981
982 if (pp.color_mode == tile_compile::ColorMode::OSC) {
983 const auto pattern = tile_compile::string_to_bayer_pattern(pp.bayer_pattern);
984 if (pattern != tile_compile::BayerPattern::UNKNOWN) {
985 int rgb_rejected_samples = 0;
986 int64_t rgb_total_samples = 0;
987 const auto rgb_batch_plan =
988 raw_stack_sub_batch_plan(rows, cols, 3, accepted.size(), cfg.runtime_limits);
989 const size_t rgb_sub_batch =
990 rgb_batch_plan.frame_sub_batch_size > 0 ? rgb_batch_plan.frame_sub_batch_size
991 : accepted.size();
992 Matrix2Df accum_R;
993 Matrix2Df accum_G;
994 Matrix2Df accum_B;
995 Matrix2Df wsum_R;
996 Matrix2Df wsum_G;
997 Matrix2Df wsum_B;
998 for (size_t batch_start = 0; batch_start < accepted.size(); batch_start += rgb_sub_batch) {
999 const size_t batch_end = std::min(batch_start + rgb_sub_batch, accepted.size());
1000 std::vector<Matrix2Df> r_frames;
1001 std::vector<Matrix2Df> g_frames;
1002 std::vector<Matrix2Df> b_frames;
1003 std::vector<float> batch_weights;
1004 r_frames.reserve(batch_end - batch_start);
1005 g_frames.reserve(batch_end - batch_start);
1006 b_frames.reserve(batch_end - batch_start);
1007 batch_weights.reserve(batch_end - batch_start);
1008 Matrix2Df deb_r;
1009 Matrix2Df deb_g;
1010 Matrix2Df deb_b;
1011 for (size_t i = batch_start; i < batch_end; ++i) {
1012 const float* frame_ptr = pp.prewarped_frames.frame_data(static_cast<size_t>(accepted[i]));
1013 if (frame_ptr == nullptr) continue;
1014 image::debayer_nearest_neighbor_strided_into(
1015 frame_ptr, rows, cols, cols, pattern, 0, 0, deb_r, deb_g, deb_b);
1016 if (deb_r.rows() == rows && deb_r.cols() == cols &&
1017 deb_g.rows() == rows && deb_g.cols() == cols &&
1018 deb_b.rows() == rows && deb_b.cols() == cols) {
1019 r_frames.push_back(std::move(deb_r));
1020 g_frames.push_back(std::move(deb_g));
1021 b_frames.push_back(std::move(deb_b));
1022 batch_weights.push_back(i < weights.size() ? weights[i] : 1.0f);
1023 }
1024 }
1025 if (g_frames.empty()) continue;
1026 int batch_rejected = 0;
1027 int64_t batch_total = 0;
1028 Matrix2Df batch_R = stack_frame_planes(r_frames, batch_weights, cfg.rejection,
1029 cfg.runtime_limits.parallel_workers,
1030 batch_rejected, batch_total, nullptr);
1031 Matrix2Df batch_G = stack_frame_planes(g_frames, batch_weights, cfg.rejection,
1032 cfg.runtime_limits.parallel_workers,
1033 batch_rejected, batch_total, nullptr);
1034 Matrix2Df batch_B = stack_frame_planes(b_frames, batch_weights, cfg.rejection,
1035 cfg.runtime_limits.parallel_workers,
1036 batch_rejected, batch_total, nullptr);
1037 rgb_rejected_samples += batch_rejected;
1038 rgb_total_samples += batch_total;
1039 const float batch_weight = static_cast<float>(g_frames.size());
1040 accumulate_batch_stack(batch_R, batch_weight, accum_R, wsum_R);
1041 accumulate_batch_stack(batch_G, batch_weight, accum_G, wsum_G);
1042 accumulate_batch_stack(batch_B, batch_weight, accum_B, wsum_B);
1043 pp.prewarped_frames.clear_mappings();
1044 emitter.phase_progress(run_id, prep::phase_to_string(prep::Phase::STACKING),
1045 0.45f + 0.55f * static_cast<float>(batch_end) /
1046 static_cast<float>(accepted.size()),
1047 "stacked RGB batch " + std::to_string(batch_end) +
1048 "/" + std::to_string(accepted.size()),
1049 event_out);
1050 }
1051 Matrix2Df R = finalize_batch_accum(accum_R, wsum_R);
1052 Matrix2Df G = finalize_batch_accum(accum_G, wsum_G);
1053 Matrix2Df B = finalize_batch_accum(accum_B, wsum_B);
1054 if (R.size() <= 0 || G.size() <= 0 || B.size() <= 0) {
1055 emitter.phase_end(run_id, prep::phase_to_string(prep::Phase::STACKING), "error",
1056 {{"error", "RGB stack produced no output"}}, event_out);
1057 emitter.run_end(run_id, false, "error", event_out,
1058 {{"message", "RGB stack produced no output"}});
1059 throw std::runtime_error("RGB stack produced no output");
1060 }
1061 // Restore output scaling for R, G, B channels separately.
1062 {
1063 std::vector<float> P_r, P_g, P_b, B_r, B_g, B_b;
1064 for (const auto& s : pp.norm_scales) {
1065 if (std::isfinite(s.scale_r) && s.scale_r > 0.0f) P_r.push_back(s.scale_r);
1066 if (std::isfinite(s.scale_g) && s.scale_g > 0.0f) P_g.push_back(s.scale_g);
1067 if (std::isfinite(s.scale_b) && s.scale_b > 0.0f) P_b.push_back(s.scale_b);
1068 if (std::isfinite(s.background_r)) B_r.push_back(s.background_r);
1069 if (std::isfinite(s.background_g)) B_g.push_back(s.background_g);
1070 if (std::isfinite(s.background_b)) B_b.push_back(s.background_b);
1071 }
1072 const float os_r = core::median_finite_positive(P_r, 1.0f);
1073 const float os_g = core::median_finite_positive(P_g, 1.0f);
1074 const float os_b = core::median_finite_positive(P_b, 1.0f);
1075 const float ob_r = core::median_finite(B_r, 0.0f);
1076 const float ob_g = core::median_finite(B_g, 0.0f);
1077 const float ob_b = core::median_finite(B_b, 0.0f);
1078 // R, G, B are already debayered full-res channels (no CFA offset needed)
1079 R.array() = R.array() * os_r + ob_r;
1080 G.array() = G.array() * os_g + ob_g;
1081 B.array() = B.array() * os_b + ob_b;
1082 }
1083 if (cfg.stacking.cosmetic_correction) {
1084 R = image::cosmetic_correction(R, cfg.stacking.cosmetic_correction_sigma, true);
1085 G = image::cosmetic_correction(G, cfg.stacking.cosmetic_correction_sigma, true);
1086 B = image::cosmetic_correction(B, cfg.stacking.cosmetic_correction_sigma, true);
1087 }
1088 result.rejected_samples += rgb_rejected_samples;
1089 result.total_samples += rgb_total_samples;
1090 result.stacked_rgb_path = output_dir / "stacked_rgb.fits";
1091 io::write_fits_rgb(result.stacked_rgb_path, R, G, B, header);
1092 }
1093 }
1094
1095 result.diagnostics_path = artifact_dir / "stacking_diagnostics.json";
1096 json diag = {
1097 {"method", result.method},
1098 {"weighting", result.weighting},
1099 {"frames_used", result.frames_used},
1100 {"frames_rejected_before_stack", result.rejected_input_frames},
1101 {"sample_rejections_or_winsor_clamps", result.rejected_samples},
1102 {"total_pixel_samples", result.total_samples},
1103 {"normalization", cfg.stacking.normalization},
1104 {"sigma_clip", {
1105 {"sigma_low", cfg.rejection.low},
1106 {"sigma_high", cfg.rejection.high},
1107 {"max_iters", cfg.rejection.max_iters},
1108 {"min_fraction", cfg.rejection.min_fraction},
1109 }},
1110 {"cosmetic_correction", cfg.stacking.cosmetic_correction},
1111 {"cosmetic_correction_sigma", cfg.stacking.cosmetic_correction_sigma},
1112 {"per_frame_cosmetic_correction", cfg.stacking.per_frame_cosmetic_correction},
1113 {"per_frame_cosmetic_correction_sigma", cfg.stacking.per_frame_cosmetic_correction_sigma},
1114 {"output_linear", result.stacked_linear_path.string()},
1115 {"accepted_indices", accepted},
1116 {"weights", weights},
1117 };
1118 if (!result.stacked_rgb_path.empty()) diag["output_rgb"] = result.stacked_rgb_path.string();
1119 if (pp.color_mode == tile_compile::ColorMode::OSC) {
1120 diag["rgb_stack_mode"] = "tile_compile_style_sub_batch_debayer_before_stack";
1121 diag["rgb_channels_stacked_separately"] = !result.stacked_rgb_path.empty();
1122 diag["linear_sub_batch_size"] = linear_batch_plan.frame_sub_batch_size;
1123 diag["rgb_sub_batch_size"] =
1124 raw_stack_sub_batch_plan(rows, cols, 3, accepted.size(), cfg.runtime_limits)
1125 .frame_sub_batch_size;
1126 diag["runtime_limits"] = {
1127 {"parallel_workers", cfg.runtime_limits.parallel_workers},
1128 {"memory_budget", cfg.runtime_limits.memory_budget},
1129 {"acceleration_backend", cfg.runtime_limits.acceleration_backend},
1130 };
1131 }
1132 core::write_text(result.diagnostics_path, diag.dump(2));
1133
1134 emitter.phase_end(run_id, prep::phase_to_string(prep::Phase::STACKING), "ok",
1135 {{"frames_used", result.frames_used},
1136 {"frames_rejected_before_stack", result.rejected_input_frames},
1137 {"sample_rejections_or_winsor_clamps", result.rejected_samples},
1138 {"output_linear", result.stacked_linear_path.string()},
1139 {"output_rgb", result.stacked_rgb_path.empty() ? json(nullptr) : json(result.stacked_rgb_path.string())},
1140 {"diagnostics", result.diagnostics_path.string()}},
1141 event_out);
1142 return result;
1143}

1173TileGrid build_preprocess_bge_grid(int rows,
1174 int cols,
1175 const prep::TileConfig& tile_cfg,
1176 float seeing_fwhm) {
1177 if (!(seeing_fwhm > 0.0f) || !std::isfinite(seeing_fwhm)) {
1178 seeing_fwhm = 3.0f;
1179 }
1180 const int tmin = std::max(16, tile_cfg.min_size);
1181 const int divisor = std::max(1, tile_cfg.max_divisor);
1182 int tmax = std::max(1, std::min(rows, cols) / divisor);
1183 if (tmax < tmin) tmax = tmin;
1184
1185 const float requested = static_cast<float>(std::max(1, tile_cfg.size_factor)) *
1186 seeing_fwhm;
1187 int tile_size = static_cast<int>(
1188 std::floor(std::min(std::max(requested, static_cast<float>(tmin)),
1189 static_cast<float>(tmax))));
1190 if (tile_size < tmin) tile_size = tmin;
1191
1192 const float overlap =
1193 std::min(0.5f, std::max(0.0f, tile_cfg.overlap_fraction));
1194 TileGrid grid;
1195 grid.tile_size = tile_size;
1196 grid.overlap_fraction = overlap;
1197 grid.rows = 0;
1198 grid.cols = 0;
1199 grid.tiles = tile_compile::pipeline::build_initial_tile_grid(cols, rows,
1200 tile_size,
1201 overlap);
1202 for (const auto& t : grid.tiles) {
1203 grid.rows = std::max(grid.rows, t.row + 1);
1204 grid.cols = std::max(grid.cols, t.col + 1);
1205 }
1206 return grid;
1207}

1209std::vector<TileMetrics> measure_bge_tile_metrics(const Matrix2Df& R,
1210 const Matrix2Df& G,
1211 const Matrix2Df& B,
1212 const TileGrid& grid) {
1213 const size_t n_tiles = grid.tiles.size();
1214 std::vector<TileMetrics> out(n_tiles);
1215 const int workers = default_parallel_workers(n_tiles, 0);
1216 parallel_for_indices(n_tiles, workers, [&](size_t ti) {
1217 const Tile& t = grid.tiles[ti];
1218 Matrix2Df tile(t.height, t.width);
1219 for (int y = 0; y < t.height; ++y) {
1220 for (int x = 0; x < t.width; ++x) {
1221 tile(y, x) = (R(t.y + y, t.x + x) + G(t.y + y, t.x + x) +
1222 B(t.y + y, t.x + x)) / 3.0f;
1223 }
1224 }
1225 TileMetrics tm = metrics::calculate_tile_metrics(tile);
1226 if (!std::isfinite(tm.background)) tm.background = 0.0f;
1227 if (!std::isfinite(tm.noise)) tm.noise = 0.0f;
1228 if (!std::isfinite(tm.gradient_energy)) tm.gradient_energy = 0.0f;
1229 if (!std::isfinite(tm.quality_score)) tm.quality_score = 1.0f;
1230 if (tm.star_count <= 0) tm.type = TileType::STRUCTURE;
1231 out[ti] = tm;
1232 });
1233 return out;
1234}

1251PreprocessPostprocessResult run_preprocess_postprocess(
1252 const std::string& run_id,
1253 const prep::Config& cfg,
1254 const PreprocessStackResult& stack,
1255 const runner::PreprocessPipelineContext& pp,
1256 const runner::QualityAnalysisContext& qa,
1257 const fs::path& run_dir,
1258 core::EventEmitter& emitter,
1259 std::ostream& event_out) {
1260 PreprocessPostprocessResult result;
1261 const fs::path artifact_dir = run_dir / "artifacts" / "preprocess";
1262 const fs::path output_dir = run_dir / "outputs";
1263 fs::create_directories(artifact_dir);
1264 fs::create_directories(output_dir);
1265
1266 astro::WCS wcs;
1267 bool have_wcs = false;
1268
1269 emitter.phase_start(run_id, prep::phase_to_string(prep::Phase::ASTROMETRY), event_out);
1270 if (!cfg.postprocess.astrometry) {
1271 add_phase_result(result.phases, "ASTROMETRY", "skipped", {{"reason", "disabled"}});
1272 emitter.phase_end(run_id, prep::phase_to_string(prep::Phase::ASTROMETRY), "skipped",
1273 {{"reason", "disabled"}}, event_out);
1274 } else if (stack.stacked_rgb_path.empty()) {
1275 add_phase_result(result.phases, "ASTROMETRY", "skipped", {{"reason", "no_rgb_stack"}});
1276 emitter.phase_end(run_id, prep::phase_to_string(prep::Phase::ASTROMETRY), "skipped",
1277 {{"reason", "no_rgb_stack"}}, event_out);
1278 } else {
1279 const std::string astap_bin_cfg = cfg.has_astrometry_config ? cfg.astrometry.astap_bin : "";
1280 const auto astap_paths = runner::resolve_astap_paths(
1281 astap_bin_cfg, cfg.has_astrometry_config ? cfg.astrometry.astap_data_dir : std::string());
1282 std::string astap_data = astap_paths.data_dir;
1283 fs::path astap_bin_path = astap_paths.binary;
1284 const int search_radius = cfg.has_astrometry_config ? cfg.astrometry.search_radius : 180;
1285 if (astap_bin_path.empty()) {
1286 const std::string reported_bin = astap_bin_cfg.empty() ? astap_data + "/astap_cli" : astap_bin_cfg;
1287 add_phase_result(result.phases, "ASTROMETRY", "skipped",
1288 {{"reason", "astap_not_found"}, {"astap_bin", reported_bin}});
1289 emitter.phase_end(run_id, prep::phase_to_string(prep::Phase::ASTROMETRY), "skipped",
1290 {{"reason", "astap_not_found"}, {"astap_bin", reported_bin}},
1291 event_out);
1292 } else {
1293 std::string astrometry_solver = "astap";
1294 int local_gaia_image_stars = 0;
1295 int local_gaia_catalog_stars = 0;
1296 int local_gaia_inlier_stars = 0;
1297 bool local_gaia_reflected = false;
1298 std::string local_gaia_error = "not_attempted";
1299 const auto astap_stamp =
1300 std::chrono::steady_clock::now().time_since_epoch().count();
1301 fs::path astap_output_prefix =
1302 fs::temp_directory_path() /
1303 ("tile_compile_astap_" + run_id + "_" +
1304 std::to_string(astap_stamp));
1305 const std::string cmd = runner::shell_quote(astap_bin_path.string()) + " -f " +
1306 runner::shell_quote(stack.stacked_rgb_path.string()) + " -d " +
1307 runner::shell_quote(astap_data) + " -r " +
1308 std::to_string(search_radius) + " -wcs -o " +
1309 runner::shell_quote(astap_output_prefix.string());
1310 const int ret = std::system(runner::system_cmd(cmd).c_str());
1311 fs::path wcs_path = astap_output_prefix;
1312 wcs_path.replace_extension(".wcs");
1313 if (ret == 0 && fs::exists(wcs_path)) {
1314 try {
1315 wcs = astro::parse_wcs_file(wcs_path.string());
1316 have_wcs = wcs.valid();
1317 } catch (...) {
1318 have_wcs = false;
1319 }
1320 }
1321 if (!have_wcs) {
1322 try {
1323 const io::RGBImage solve_rgb = io::read_fits_rgb(stack.stacked_rgb_path);
1324 const auto local_gaia = runner::solve_with_local_gaia_catalog(solve_rgb);
1325 local_gaia_image_stars = local_gaia.image_stars;
1326 local_gaia_catalog_stars = local_gaia.catalog_stars;
1327 local_gaia_inlier_stars = local_gaia.inlier_stars;
1328 local_gaia_reflected = local_gaia.reflected_solution;
1329 local_gaia_error = local_gaia.error_message;
1330 if (local_gaia.success && runner::write_wcs_sidecar(local_gaia.wcs, wcs_path)) {
1331 wcs = local_gaia.wcs;
1332 have_wcs = true;
1333 astrometry_solver = "local_gaia_dr3";
1334 }
1335 } catch (const std::exception &e) {
1336 local_gaia_error = e.what();
1337 // Keep the existing ASTAP failure contract if the local fallback
1338 // cannot read the RGB stack or catalogue.
1339 }
1340 }
1341 if (have_wcs) {
1342 result.astrometry_wcs_path = artifact_dir / "stacked_rgb.wcs";
1343 fs::copy_file(wcs_path, result.astrometry_wcs_path,
1344 fs::copy_options::overwrite_existing);
1345 io::FitsHeader hdr = io::read_fits_header(stack.stacked_rgb_path);
1346 hdr.numeric_values["CRVAL1"] = wcs.crval1;
1347 hdr.numeric_values["CRVAL2"] = wcs.crval2;
1348 hdr.numeric_values["CRPIX1"] = wcs.crpix1;
1349 hdr.numeric_values["CRPIX2"] = wcs.crpix2;
1350 hdr.numeric_values["CD1_1"] = wcs.cd1_1;
1351 hdr.numeric_values["CD1_2"] = wcs.cd1_2;
1352 hdr.numeric_values["CD2_1"] = wcs.cd2_1;
1353 hdr.numeric_values["CD2_2"] = wcs.cd2_2;
1354 hdr.numeric_values["EQUINOX"] = 2000.0;
1355 hdr.string_values["CTYPE1"] = "RA---TAN";
1356 hdr.string_values["CTYPE2"] = "DEC--TAN";
1357 hdr.string_values["CUNIT1"] = "deg";
1358 hdr.string_values["CUNIT2"] = "deg";
1359 hdr.bool_values["PLTSOLVD"] = true;
1360 io::update_fits_header_in_place(stack.stacked_rgb_path, hdr);
