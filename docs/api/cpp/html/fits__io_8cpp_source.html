146FitsHeader read_current_header(fitsfile* fptr, int& status) {
147 FitsHeader header;
148
149 char card[FLEN_CARD];
150 int nkeys = 0;
151 fits_get_hdrspace(fptr, &nkeys, nullptr, &status);
152
153 for (int i = 1; i <= nkeys; ++i) {
154 fits_read_record(fptr, i, card, &status);
155 if (status) {
156 status = 0;
157 continue;
158 }
159
160 char keyname[FLEN_KEYWORD];
161 char value[FLEN_VALUE];
162 char comment[FLEN_COMMENT];
163 int keylen = 0;
164
165 fits_get_keyname(card, keyname, &keylen, &status);
166 if (status) {
167 status = 0;
168 continue;
169 }
170
171 const std::string key(keyname);
172 if (key.empty() || key == "COMMENT" || key == "HISTORY" || key == "END") {
173 continue;
174 }
175
176 fits_parse_value(card, value, comment, &status);
177 if (status) {
178 status = 0;
179 continue;
180 }
181
182 // Determine the value type from the raw value string rather than
183 // relying on fits_get_keytype, which returns incorrect types for
184 // some non-standard FITS producers (e.g. DWARF II: FOCALLEN=100.
185 // classified as 'L', FILTER='Astro' as 'L', DATE-OBS='2026-...'
186 // as 'F'). The raw value from fits_parse_value preserves the
187 // leading quote for string keywords, which we use as the primary
188 // type discriminator.
189 std::string raw_value(value);
190 // Strip leading spaces only — keep the quote for type detection
191 size_t first_nonspace = raw_value.find_first_not_of(' ');
192 if (first_nonspace == std::string::npos) first_nonspace = 0;
193
194 const bool is_quoted_string =
195 (first_nonspace < raw_value.size() && raw_value[first_nonspace] == '\'');
196
197 // Clean value: strip spaces and single quotes for storage
198 std::string val_str(raw_value);
199 val_str.erase(0, val_str.find_first_not_of(" '"));
200 val_str.erase(val_str.find_last_not_of(" '") + 1);
201
202 if (is_quoted_string) {
203 // Definitely a string — fits_parse_value confirmed the quote
204 header.set(key, val_str);
205 } else if (val_str == "T" || val_str == "F") {
206 // FITS logical value (unquoted T or F)
207 header.set(key, val_str == "T");
208 } else {
209 // Try integer first (no decimal point), then float, then string
210 bool parsed = false;
211 if (val_str.find('.') == std::string::npos &&
212 val_str.find('e') == std::string::npos &&
213 val_str.find('E') == std::string::npos) {
214 try {
215 size_t pos = 0;
216 int int_val = std::stoi(val_str, &pos);
217 if (pos == val_str.size()) {
218 header.set(key, int_val);
219 parsed = true;
220 }
221 } catch (...) {}
222 }
223 if (!parsed) {
224 try {
225 size_t pos = 0;
226 double dbl_val = std::stod(val_str, &pos);
227 // Accept partial parse only if it consumed the entire
228 // value (e.g. "100." → 100.0, but not "2026-08-13..."
229 // which would parse as 2026 and leave a suffix).
230 if (pos == val_str.size()) {
231 header.set(key, dbl_val);
232 parsed = true;
233 }
234 } catch (...) {}
235 }
236 if (!parsed) {
237 header.set(key, val_str);
238 }
239 }
240 }
241
242 status = 0;
243 return header;
244}

266void write_header_keywords(fitsfile* fptr, const FitsHeader& header,
267 bool is_rgb_image, int& status) {
268 for (const auto& [key, value] : header.string_values) {
269 if (key.size() <= 8 && !should_skip_header_key(key, is_rgb_image)) {
270 fits_update_key(fptr, TSTRING, key.c_str(),
271 const_cast<char*>(value.c_str()), nullptr, &status);
272 if (status) status = 0;
273 }
274 }
275
276 for (const auto& [key, value] : header.numeric_values) {
277 if (key.size() <= 8 && !should_skip_header_key(key, is_rgb_image)) {
278 double val = value;
279 fits_update_key(fptr, TDOUBLE, key.c_str(), &val, nullptr, &status);
280 if (status) status = 0;
281 }
282 }
283
284 for (const auto& [key, value] : header.int_values) {
285 if (key.size() <= 8 && !should_skip_header_key(key, is_rgb_image)) {
286 int val = value;
287 fits_update_key(fptr, TINT, key.c_str(), &val, nullptr, &status);
288 if (status) status = 0;

290 }
291
292 for (const auto& [key, value] : header.bool_values) {
293 if (key.size() <= 8 && !should_skip_header_key(key, is_rgb_image)) {

621void write_fits_float(const fs::path& path, const Matrix2Df& data, const FitsHeader& header) {
622 // Atomic (plan section A6/audit 2026-09-05): stage under a private
623 // directory, fsync, then rename into place --- matches
624 // write_fits_mask_rows() above, so every reader of `path` observes either
625 // the complete previous file or the complete new one, never a partial
626 // write from cfitsio's own "!"-prefix clobber-in-place. NOTE: the two
627 // other FITS writers in this file (write_fits_rgb, write_fits_rgb_u32)
628 // still use the non-atomic "!" convention; only this function and
629 // write_fits_mask_rows are fixed so far (tracked, not silently assumed
630 // fixed elsewhere).
631 core::AtomicOutput output(path);
632 fitsfile* fptr = nullptr;
633 int status = 0;
634
635 if (fits_create_file(&fptr, output.path().string().c_str(), &status)) {
636 throw FitsError(fits_write_error_message("create FITS file", path, status));
637 }
638
639 try {
640 long naxes[2] = {static_cast<long>(data.cols()), static_cast<long>(data.rows())};
641
642 fits_create_img(fptr, FLOAT_IMG, 2, naxes, &status);
643 if (status) {
644 const int write_status = status;
645 int close_status = 0;
646 fits_close_file(fptr, &close_status);
647 fptr = nullptr;
648 throw FitsError(fits_write_error_message("create FITS image", path, write_status));
649 }
650
651 write_header_keywords(fptr, header, false, status);
652
653 // Declare row order: the internal Eigen RowMajor buffer has row 0 at the top
654 // (screen convention), which is the opposite of the default FITS bottom-up
655 // convention. ROWORDER=TOP-DOWN tells viewers (Siril, DS9, etc.) to display
656 // the image without flipping, matching the actual data layout on disk.
657 {
658 char roworder[] = "TOP-DOWN";
659 fits_update_key(fptr, TSTRING, "ROWORDER", roworder,
660 "Row order: row 0 is top of image", &status);
661 if (status) status = 0; // non-fatal: proceed even if key cannot be written
662 }
663
664 // Write pixel data row by row to avoid cfitsio internal buffer issues
665 // with very large single-write calls (e.g. 50M+ elements for expanded canvas).
666 const long nrows = static_cast<long>(data.rows());
667 const long ncols = static_cast<long>(data.cols());
668 for (long row = 0; row < nrows; ++row) {
669 long fpixel[2] = {1, row + 1};
670 const float* row_ptr = data.data() + row * ncols;
671 fits_write_pix(fptr, TFLOAT, fpixel, ncols,
672 const_cast<float*>(row_ptr), &status);
673 if (status) {
674 const int write_status = status;
675 int close_status = 0;
676 fits_close_file(fptr, &close_status);
677 fptr = nullptr;
678 throw FitsError(fits_write_error_message("write FITS pixel data", path, write_status));
679 }
680 }
681
682 int close_status = 0;
683 fits_close_file(fptr, &close_status);
684 fptr = nullptr;
685 if (close_status) {
686 throw FitsError(fits_write_error_message("close FITS file", path, close_status));
687 }
688 output.commit();
689 } catch (...) {
690 if (fptr) { int close_status = 0; fits_close_file(fptr, &close_status); }
691 throw;
692 }
693}

699void write_fits_rgb(const fs::path& path, const Matrix2Df& R, const Matrix2Df& G, const Matrix2Df& B, const FitsHeader& header) {
700 if (R.rows() != G.rows() || R.rows() != B.rows() || R.cols() != G.cols() || R.cols() != B.cols()) {
701 throw FitsError("RGB channel dimensions must match");
702 }
703
704 fitsfile* fptr = nullptr;
705 int status = 0;
706
707 std::string filepath = "!" + path.string();
708
709 if (fits_create_file(&fptr, filepath.c_str(), &status)) {
710 throw FitsError(fits_write_error_message("create FITS file", path, status));
711 }
712
713 // Create 3D image cube: NAXIS1=width, NAXIS2=height, NAXIS3=3 (RGB planes)
714 long naxes[3] = {static_cast<long>(R.cols()), static_cast<long>(R.rows()), 3};
715
716 fits_create_img(fptr, FLOAT_IMG, 3, naxes, &status);
717 if (status) {
718 const int write_status = status;
719 int close_status = 0;
720 fits_close_file(fptr, &close_status);
721 throw FitsError(fits_write_error_message("create FITS RGB image", path, write_status));
722 }
723
724 write_header_keywords(fptr, header, true, status);
725
726 // Declare row order: the internal Eigen RowMajor buffer has row 0 at the top
727 // (screen convention), which is the opposite of the default FITS bottom-up
728 // convention. ROWORDER=TOP-DOWN tells viewers (Siril, DS9, etc.) to display
729 // the image without flipping, matching the actual data layout on disk.
730 {
731 char roworder[] = "TOP-DOWN";
732 fits_update_key(fptr, TSTRING, "ROWORDER", roworder,
733 "Row order: row 0 is top of image", &status);
734 if (status) status = 0; // non-fatal: proceed even if key cannot be written
735 }
736
737 // Write each RGB plane row by row to avoid cfitsio internal buffer issues
738 // with very large single-write calls.
739 const long nrows_rgb = static_cast<long>(R.rows());
740 const long ncols_rgb = static_cast<long>(R.cols());
741 const Matrix2Df* planes[3] = {&R, &G, &B};
742 for (int p = 0; p < 3; ++p) {
743 for (long row = 0; row < nrows_rgb; ++row) {
744 long fpixel[3] = {1, row + 1, p + 1};
745 const float* row_ptr = planes[p]->data() + row * ncols_rgb;
746 fits_write_pix(fptr, TFLOAT, fpixel, ncols_rgb,
747 const_cast<float*>(row_ptr), &status);
748 if (status) break;
749 }
750 if (status) break;
751 }
752
753 if (status) {
754 const int write_status = status;
755 int close_status = 0;
756 fits_close_file(fptr, &close_status);
757 throw FitsError(fits_write_error_message("write FITS RGB pixel data", path, write_status));
758 }
759
760 fits_close_file(fptr, &status);
761 if (status) {
762 throw FitsError(fits_write_error_message("close FITS file", path, status));
763 }
764}

766void write_fits_rgb_u32(const fs::path& path, const Matrix2Df& R, const Matrix2Df& G,
767 const Matrix2Df& B, const FitsHeader& header) {
768 if (R.rows() != G.rows() || R.rows() != B.rows() || R.cols() != G.cols() ||
769 R.cols() != B.cols()) {
770 throw FitsError("RGB channel dimensions must match");
771 }
772
773 fitsfile* fptr = nullptr;
774 int status = 0;
775 const std::string filepath = "!" + path.string();
776 if (fits_create_file(&fptr, filepath.c_str(), &status)) {
777 throw FitsError(fits_write_error_message("create FITS file", path, status));
778 }
779
780 long naxes[3] = {static_cast<long>(R.cols()), static_cast<long>(R.rows()), 3};
781 fits_create_img(fptr, LONG_IMG, 3, naxes, &status);
782 if (status) {
783 const int write_status = status;
784 int close_status = 0;
785 fits_close_file(fptr, &close_status);
786 throw FitsError(fits_write_error_message("create FITS RGB uint32 image", path, write_status));
787 }
788
789 double bscale = 1.0;
790 double bzero = 2147483648.0;
791 fits_update_key(fptr, TDOUBLE, "BSCALE", &bscale, nullptr, &status);
792 fits_update_key(fptr, TDOUBLE, "BZERO", &bzero, nullptr, &status);
793 write_header_keywords(fptr, header, true, status);
794
795 char roworder[] = "TOP-DOWN";
796 fits_update_key(fptr, TSTRING, "ROWORDER", roworder,
797 "Row order: row 0 is top of image", &status);
798 if (status) status = 0;
799
800 const long nrows = static_cast<long>(R.rows());
801 const long ncols = static_cast<long>(R.cols());
802 const Matrix2Df* planes[3] = {&R, &G, &B};
803 for (int p = 0; p < 3 && !status; ++p) {
804 for (long row = 0; row < nrows && !status; ++row) {
805 long fpixel[3] = {1, row + 1, p + 1};
806 const float* row_ptr = planes[p]->data() + row * ncols;
807 std::vector<unsigned int> values(static_cast<size_t>(ncols));
808 for (long col = 0; col < ncols; ++col) {
809 const float value = row_ptr[col];
810 const double rounded = std::isfinite(value)
811 ? std::round(static_cast<double>(value)) : 0.0;
812 values[static_cast<size_t>(col)] = static_cast<unsigned int>(
813 std::clamp(rounded, 0.0, 4294967295.0));
814 }
815 fits_write_pix(fptr, TUINT, fpixel, ncols, values.data(), &status);
816 }
817 }
818
819 if (status) {
820 const int write_status = status;
821 int close_status = 0;
822 fits_close_file(fptr, &close_status);
823 throw FitsError(fits_write_error_message("write FITS RGB uint32 pixel data", path, write_status));
824 }
825
826 fits_close_file(fptr, &status);
827 if (status) {
828 throw FitsError(fits_write_error_message("close FITS file", path, status));
829 }
830}

836void update_fits_header_in_place(const fs::path& path, const FitsHeader& header) {
837 fitsfile* fptr = nullptr;
838 int status = 0;
839
840 if (fits_open_file(&fptr, path.string().c_str(), READWRITE, &status)) {
841 throw FitsError("Cannot open FITS file for header update: " +
842 path.string());
843 }
844
845 int naxis = 0;
846 long naxes[3] = {0, 0, 0};
847 int bitpix = 0;
848 if (!move_to_first_image_hdu(fptr, bitpix, naxis, naxes, status)) {
849 fits_close_file(fptr, &status);
850 throw FitsError("Cannot read FITS image parameters: " + path.string());
851 }
852
853 const bool is_rgb_image = (naxis >= 3 && naxes[2] >= 3);
854 write_header_keywords(fptr, header, is_rgb_image, status);
855 if (status) {
856 const int update_status = status;
857 int close_status = 0;
858 fits_close_file(fptr, &close_status);
859 throw FitsError(fits_write_error_message("update FITS header", path,
860 update_status));
861 }
862
863 // Ensure ROWORDER is always present after a header update.
864 {
865 char roworder[] = "TOP-DOWN";
866 int ro_status = 0;
867 fits_update_key(fptr, TSTRING, "ROWORDER", roworder,
868 "Row order: row 0 is top of image", &ro_status);
869 // non-fatal: ignore if key cannot be written
870 }
871
872 fits_close_file(fptr, &status);
873 if (status) {
874 throw FitsError(fits_write_error_message("close FITS file", path, status));
875 }

882BayerPattern detect_bayer_pattern(const FitsHeader& header) {
883 auto bayerpat = header.get_string("BAYERPAT");
884 if (bayerpat) {
885 return string_to_bayer_pattern(*bayerpat);
886 }
887
888 auto colortyp = header.get_string("COLORTYP");
889 if (colortyp) {
890 std::string ct = core::to_lower(*colortyp);
891 if (ct.find("rggb") != std::string::npos) return BayerPattern::RGGB;
892 if (ct.find("bggr") != std::string::npos) return BayerPattern::BGGR;
893 if (ct.find("grbg") != std::string::npos) return BayerPattern::GRBG;
894 if (ct.find("gbrg") != std::string::npos) return BayerPattern::GBRG;
895 }
896
897 auto xbayroff = header.get_int("XBAYROFF");
898 auto ybayroff = header.get_int("YBAYROFF");
899 if (xbayroff && ybayroff) {
900 int x = *xbayroff % 2;
901 int y = *ybayroff % 2;
902 if (x == 0 && y == 0) return BayerPattern::RGGB;
903 if (x == 1 && y == 0) return BayerPattern::GRBG;
904 if (x == 0 && y == 1) return BayerPattern::GBRG;
905 if (x == 1 && y == 1) return BayerPattern::BGGR;
906 }
907
908 return BayerPattern::UNKNOWN;
909}

915ColorMode detect_color_mode(const FitsHeader& header, int naxis) {
916 if (naxis >= 3) {
917 return ColorMode::RGB;
918 }
919
920 auto bayerpat = detect_bayer_pattern(header);
921 if (bayerpat != BayerPattern::UNKNOWN) {
922 return ColorMode::OSC;
923 }
924
925 auto colortyp = header.get_string("COLORTYP");
926 if (colortyp) {
927 std::string ct = core::to_lower(*colortyp);
928 if (ct.find("mono") != std::string::npos) return ColorMode::MONO;
929 if (ct.find("osc") != std::string::npos || ct.find("color") != std::string::npos) {
930 return ColorMode::OSC;
931 }
932 }
933
934 return ColorMode::MONO;
935}
