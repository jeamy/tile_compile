77VerifiedNormalizedSourceCache::VerifiedNormalizedSourceCache(
78 const fs::path &root,const registration::RegistrationSamplingPlan &expected,
79 size_t memory_budget_mb) : root_(root),width_(expected.source_width),
80 height_(expected.source_height) {
81 const auto ctx = context(expected);
82 const size_t bytes = frame_bytes(width_,height_);
83 if (memory_budget_mb > std::numeric_limits<size_t>::max()/(1024*1024) ||
84 memory_budget_mb < 2 || bytes > memory_budget_mb*1024*1024-1024*1024)
85 throw std::runtime_error("NORMALIZED_CACHE_MEMORY_BUDGET");
86 const auto path = root/"normalized_source_manifest.json";
87 if (!fs::is_regular_file(fs::symlink_status(path)) ||
88 fs::file_size(path)>std::min<size_t>(16*1024*1024,memory_budget_mb*1024*1024/16))
89 throw std::runtime_error("NORMALIZED_CACHE_INVALID_MANIFEST_FILE");
90 std::ifstream file(path);
91 auto manifest = json::parse(file);
92 // T1: accept schema v1 (legacy) and v2 (trusted run). v2 uses
93 // manifest_identity; v1 uses manifest_hash. Both are metadata-only hashes.
94 const int sv = manifest.at("schema_version").get<int>();
95 if (sv != 1 && sv != 2)
96 throw std::runtime_error("NORMALIZED_CACHE_UNSUPPORTED_SCHEMA");
97 const std::string id_field = sv >= 2 ? "manifest_identity" : "manifest_hash";
98 manifest_hash_ = manifest.at(id_field).get<std::string>();
99 manifest.erase(id_field);
100 if (manifest.at("context") != ctx || digest(manifest) != manifest_hash_ ||
101 !manifest.at("files").is_array() ||
102 manifest.at("files").size() != expected.frames.size())
103 throw std::runtime_error("NORMALIZED_CACHE_CONTEXT_OR_HASH_MISMATCH");
104 size_t i=0;
105 for (const auto &entry : manifest.at("files")) {
106 if (!entry.at("source_index").is_number_unsigned() || !entry.at("bytes").is_number_unsigned())
107 throw std::runtime_error("NORMALIZED_CACHE_INVALID_ENTRY_TYPE");
108 const size_t index=entry.at("source_index").get<size_t>();
109 if (index!=ctx.at("frames").at(i++).at("source_index").get<size_t>() ||
110 entry.at("bytes").get<size_t>()!=bytes)
111 throw std::runtime_error("NORMALIZED_CACHE_INVALID_ENTRY");
112 require_file(root/(std::to_string(index)+".raw"),bytes);
113 // v1 entries carry a sha256 field (ignored in trusted run); v2 does not.
114 hashes_.emplace(index, std::string{});
115 }
116 context_hash_=digest(ctx);
117 // LRU capacity (plan §30.72 O2): how many whole frames fit in the budget,
118 // leaving 1 MiB slack. At least one; never more than the manifest holds.
119 const size_t usable = memory_budget_mb*1024*1024 - 1024*1024;
120 capacity_ = std::max<size_t>(1, std::min<size_t>(hashes_.size(),
121 bytes ? usable/bytes : 1));
122}

139const Matrix2Df &VerifiedNormalizedSourceCache::load(size_t source_index) {
140 ++load_calls_;
141 const auto hit=resident_.find(source_index);
142 if (hit!=resident_.end()) {
143 const auto path=root_/(std::to_string(source_index)+".raw");
144 std::error_code ec;
145 const auto sz=fs::file_size(path,ec);
146 const auto mt=fs::last_write_time(path,ec);
147 if (!ec && sz==hit->second->file_size && mt==hit->second->mtime &&
148 mt<hit->second->verified_at) {
149 // Unchanged AND last written strictly before we verified it: promote to
150 // front, no read, no SHA-256. The `mt < verified_at` guard closes the
151 // same-mtime-tick rewrite window that a size+mtime match alone leaves.
152 ++lru_hits_;
153 lru_.splice(lru_.begin(),lru_,hit->second);
154 return hit->second->image;
155 }
156 // Size or mtime moved, a same-tick rewrite is possible, or stat failed:
157 // drop the stale entry and fall through to a full verifying reload.
158 lru_.erase(hit->second);
159 resident_.erase(hit);
160 }
161 return verify_and_insert(source_index);
162}

164const Matrix2Df &VerifiedNormalizedSourceCache::verify_and_insert(
165 size_t source_index) {
166 const auto found=hashes_.find(source_index);
167 if (found==hashes_.end()) throw std::invalid_argument("NORMALIZED_CACHE_UNKNOWN_FRAME");
168 const size_t bytes=frame_bytes(width_,height_);
169 const auto path=root_/(std::to_string(source_index)+".raw");
170 require_file(path,bytes);
171 Matrix2Df image;
172 image.resize(height_,width_);
173 std::ifstream file(path,std::ios::binary);
174 file.read(reinterpret_cast<char *>(image.data()),static_cast<std::streamsize>(bytes));
175 if (!file || file.peek()!=std::char_traits<char>::eof())
176 throw std::runtime_error("NORMALIZED_CACHE_READ_FAILED");
177 bytes_read_+=bytes;
178 std::error_code ec;
179 Entry e;
180 e.index=source_index;
181 e.image=std::move(image);
182 e.file_size=fs::file_size(path,ec);
183 e.mtime=fs::last_write_time(path,ec);
184 // Captured AFTER the read: a hit is trusted only if the file's mtime is
185 // strictly older than this instant, so a write concurrent with (or in the
186 // same fs tick as) our read is never mistaken for "unchanged".
187 e.verified_at=fs::file_time_type::clock::now();
188 lru_.push_front(std::move(e));
189 resident_[source_index]=lru_.begin();
190 // Evict least-recently-used; never the entry just inserted (front). A held
191 // reference to some OTHER frame is invalidated here, matching the previous
192 // single-buffer contract where any load() invalidated the prior reference.
193 while (lru_.size()>capacity_ && lru_.size()>1) {
194 resident_.erase(lru_.back().index);
195 lru_.pop_back();
196 ++evictions_;
197 }
198 return lru_.front().image;
199}

201void VerifiedNormalizedSourceCache::read_rect_into(
202 size_t source_index,int y0,int y1,int x0,int x1,
203 std::vector<float> &out) {
204 y0=std::clamp(y0,0,height_); y1=std::clamp(y1,0,height_);
205 x0=std::clamp(x0,0,width_); x1=std::clamp(x1,0,width_);
206 const int rows=y1-y0, cols=x1-x0;
207 if (rows<=0 || cols<=0) { out.clear(); return; }
208 if (hashes_.find(source_index)==hashes_.end())
209 throw std::invalid_argument("NORMALIZED_CACHE_UNKNOWN_FRAME");
210 out.resize(static_cast<size_t>(rows)*static_cast<size_t>(cols));
211 const size_t row_bytes=static_cast<size_t>(width_)*sizeof(float);
212 const size_t x_bytes=static_cast<size_t>(cols)*sizeof(float);
213 const auto path=root_/(std::to_string(source_index)+".raw");
214 std::ifstream file(path,std::ios::binary);
215 // Seek/read only the [x0,x1) span of each requested row directly into the
216 // output span; no covering source row is read.
217 for (int r=0;r<rows;++r) {
218 file.seekg(static_cast<std::streamoff>(
219 static_cast<size_t>(y0+r)*row_bytes+
220 static_cast<size_t>(x0)*sizeof(float)));
221 file.read(reinterpret_cast<char *>(
222 out.data()+static_cast<size_t>(r)*cols),
223 static_cast<std::streamsize>(x_bytes));
224 if (!file) throw std::runtime_error("NORMALIZED_CACHE_READ_FAILED");
225 }
226 ++rect_read_calls_;
227 bytes_read_+=static_cast<std::uint64_t>(rows)*x_bytes;
228 expanded_floats_+=static_cast<std::uint64_t>(rows)*static_cast<std::uint64_t>(cols);
229}

231void VerifiedNormalizedSourceCache::read_row_intervals_into(
232 size_t source_index, const std::vector<DrizzleAffineSourceSpan> &intervals,
233 std::vector<float> &out) {
234 out.clear();
235 if (hashes_.find(source_index)==hashes_.end())
236 throw std::invalid_argument("NORMALIZED_CACHE_UNKNOWN_FRAME");
237 size_t total=0;
238 int prev_row=-1;
239 for (const auto &s : intervals) {
240 if (s.source_y < 0 || s.source_y >= height_ || s.source_y <= prev_row ||
241 s.x_begin < 0 || s.x_end > width_ || s.x_end <= s.x_begin)
242 throw std::invalid_argument("NORMALIZED_CACHE_INVALID_INTERVAL");
243 prev_row=s.source_y;
244 total += static_cast<size_t>(s.x_end - s.x_begin);
245 }
246 out.resize(total); // resize keeps capacity for the reusable buffer
247 if (intervals.empty()) return;
248 const size_t row_bytes=static_cast<size_t>(width_)*sizeof(float);
249 const auto path=root_/(std::to_string(source_index)+".raw");
250 std::ifstream file(path,std::ios::binary);
251 if (!file) throw std::runtime_error("NORMALIZED_CACHE_READ_FAILED");
252 size_t off=0;
253 for (const auto &s : intervals) {
254 const size_t n=static_cast<size_t>(s.x_end - s.x_begin);
255 file.seekg(static_cast<std::streamoff>(
256 static_cast<size_t>(s.source_y)*row_bytes+
257 static_cast<size_t>(s.x_begin)*sizeof(float)));
258 file.read(reinterpret_cast<char *>(out.data()+off),
259 static_cast<std::streamsize>(n*sizeof(float)));
260 if (!file) throw std::runtime_error("NORMALIZED_CACHE_READ_FAILED");
261 off += n;
262 }
263 ++rect_read_calls_;
264 bytes_read_+=static_cast<std::uint64_t>(total)*sizeof(float);
265 expanded_floats_+=static_cast<std::uint64_t>(total);
266}
