123// One backend attempt over the committed-prefix tail of the store.
124// Returns false iff the device backend reported a kernel-level failure
125// (the caller then restarts the whole phase on CPU). `device_failure`
126// additionally captures why. Any other failure throws.
127bool attempt_backend(const fs::path &store_root,
128 const ForwardDrizzleV2RunPlan &plan, int source_w,
129 int source_h, const ForwardDrizzleV2FrameProvider &provider,
130 bool cuda, ForwardDrizzleV2DriverResult &result,
131 const ForwardDrizzleV2DriverOptions &options,
132 std::string *device_failure) {
133 // Re-inspect: a prior failed attempt may have deleted the generation.
134 const auto insp = inspect_forward_drizzle_v2_store(store_root, plan);
135 if (insp.status == ForwardDrizzleV2StoreStatus::corrupt)
136 throw std::runtime_error("FDV2_DRIVER_STORE_CORRUPT: " + insp.error);
137 if (insp.status == ForwardDrizzleV2StoreStatus::complete) {
138 result.generation_dir = insp.generation;
139 result.committed = true;
140 result.bands_reused = plan.band_count;
141 result.commit_hash = insp.commit_hash;
142 return true;
143 }
144
145 // The writer owns the generation until publish; on a device failure it is
146 // destroyed here (scope exit), which deletes the unpublished generation
147 // --- including an adopted prefix --- per the spec's discard contract.
148 // Heap-allocated so the simulated-kill path can leak it deliberately
149 // (a real SIGKILL never runs destructors either).
150 auto writer =
151 std::make_unique<ForwardDrizzleV2StoreWriter>(store_root, plan);
152 int first_band = 0;
153 if (insp.status == ForwardDrizzleV2StoreStatus::resumable) {
154 writer->adopt(insp.generation, insp.next_band, insp.committed);
155 first_band = insp.next_band;
156 result.bands_reused += insp.next_band;
157 } else {
158 writer->begin();
159 }
160
161 const int cols = plan.native_width; // full-width bands only
162 // One persistent kernel workspace per backend attempt: reserve for the
163 // maximum band height once, rebind per band via begin_band. On a device
164 // failure the kernel destructs (frees once) and the CPU attempt creates
165 // its own single workspace.
166 auto kernel = make_kernel(cuda);
167 // Appends the backend's last_device_error() detail (CUDA error string)
168 // when present so the fallback reason identifies the actual failure.
169 auto fail_device = [&](const char *where) {
170 *device_failure = std::string(where) + " reported a device failure";
171 const std::string detail = kernel->last_device_error();
172 if (!detail.empty()) *device_failure += ": " + detail;
173 };
174 if (!kernel->reserve(
175 cols, plan.band_rows, source_w, source_h,
176 kernel_config_for_band(plan, 0, plan.band_rows,
177 options.cached_leaf_capacity,
178 options.cpu_workers,
179 options.device_float_plane_inputs))) {
180 if (!cuda) throw std::runtime_error("FDV2_DRIVER_RESERVE_FAILED");
181 fail_device("reserve()");
182 return false;
183 }
184 std::vector<ForwardDrizzleV2PixelResult> records;
185 std::vector<ForwardDrizzleV2ProfileResult> profiles;
186 const std::size_t max_nrec =
187 static_cast<std::size_t>(plan.band_rows) * cols *
188 static_cast<std::size_t>(plan.channels);
189 records.reserve(max_nrec);
190 if (plan.emit_profiles) profiles.reserve(max_nrec);
191 int committed_this_attempt = 0;
192 std::uint64_t nonfinite_in_support = 0;
193
194 // Frame stream permutation. Historical plans stream 0..N-1. The
195 // pilot/full-frame contract streams the hash-selected pilot frames first
196 // (ascending original order), then every other frame (ascending).
197 const bool full_frame = plan.estimator == kFdV2EstimatorPilotFullFrame;
198 std::vector<std::uint64_t> frame_seq;
199 frame_seq.reserve(static_cast<std::size_t>(plan.frame_count));
200 std::size_t pilot_count = 0;
201 if (full_frame) {
202 const auto pilot = forward_drizzle_v2_selected_frame_orders(
203 plan.frame_count, plan.reservoir_size, plan.reservoir_seed);
204 std::vector<std::uint8_t> is_pilot(
205 static_cast<std::size_t>(plan.frame_count), 0);
206 for (const auto o : pilot) {
207 frame_seq.push_back(o);
208 is_pilot[static_cast<std::size_t>(o)] = 1;
209 }
210 pilot_count = frame_seq.size();
211 for (std::uint64_t o = 0; o < plan.frame_count; ++o)
212 if (!is_pilot[static_cast<std::size_t>(o)]) frame_seq.push_back(o);
213 } else {
214 for (std::uint64_t o = 0; o < plan.frame_count; ++o)
215 frame_seq.push_back(o);
216 }
217
218 for (int band = first_band; band < plan.band_count; ++band) {
219 // Test hook: pretend the device dies before this band's kernel work.
220 if (cuda && g_fd_v2_cuda_fault_after_bands.load() >= 0 &&
221 committed_this_attempt >= g_fd_v2_cuda_fault_after_bands.load()) {
222 *device_failure = "injected fault after " +
223 std::to_string(committed_this_attempt) + " bands";
224 return false;
225 }
226 const int y_begin = band * plan.band_rows;
227 const int rows = std::min(plan.band_rows, plan.native_height - y_begin);
228 if (options.progress) options.progress(band, plan.band_count);
229
230 const std::size_t rec_cap0 =
231 records.capacity() + profiles.capacity();
232 if (!kernel->begin_band(
233 rows, kernel_config_for_band(plan, y_begin, rows,
234 options.cached_leaf_capacity,
235 options.cpu_workers,
236 options.device_float_plane_inputs))) {
237 if (!cuda)
238 throw std::runtime_error("FDV2_DRIVER_BEGIN_BAND_FAILED");
239 fail_device("begin_band()");
240 return false;
241 }
242 for (std::size_t frame_pos = 0; frame_pos < frame_seq.size();
243 ++frame_pos) {
244 const std::uint64_t fr = frame_seq[frame_pos];
245 const auto frame_t0 = std::chrono::steady_clock::now();
246 // Piece-stream contract (tranche 7): the provider invokes the sink
247 // exactly once with skip=true, or with one-or-more non-skip pieces.
248 // Affine pieces carry ordered, non-overlapping native target ranges
249 // (individual empty tiles may simply be omitted); local-warp and
250 // cached-geometry frames are always exactly one full-width piece.
251 // Each piece is dispatched inside the sink call because the provider
252 // may reuse its buffers once the call returns.
253 bool any_piece = false;
254 bool saw_skip = false;
255 bool affine_open = false;
256 int affine_tx_end = 0;
257 bool emit_failed = false;
258 bool kernel_ok = true;
259 ForwardDrizzleV2FrameInput first_piece{};
260 const ForwardDrizzleV2FramePieceSink sink =
261 [&](const ForwardDrizzleV2FrameInput &piece) -> bool {
262 if (emit_failed) return false;
263 auto fail = [&]() {
264 emit_failed = true;
265 return false;
266 };
267 if (piece.skip) {
268 if (any_piece) return fail(); // mixed skip/non-skip
269 any_piece = true;
270 saw_skip = true;
271 kernel_ok = kernel->skip_frame(fr, &piece.meta);
272 return kernel_ok;
273 }
274 if (saw_skip) return fail(); // non-skip after skip
275 // A zero-sized window means "full source" (the FrameInput default);
276 // anything else must be a positive buffer contained in the source
277 // with a positive active rect contained in the buffer. The
278 // tranche-8 sample path carries no window/source pointer at all.
279 ForwardDrizzleV2SourceWindow w = piece.source_window;
280 if (w.width == 0 && w.height == 0 && w.x_begin == 0 &&
281 w.y_begin == 0)
282 w = {0, 0, source_w, source_h, 0, 0, 0, 0};
283 int aw = w.active_width, ah = w.active_height;
284 if (aw == 0 && ah == 0 && w.active_x == 0 && w.active_y == 0) {
285 aw = w.width;
286 ah = w.height;
287 }
288 if (piece.has_affine_samples) {
289 // One-shot full-target piece: canonical sample list, no source
290 // buffer/window, no target-tile fields.
291 if (piece.affine_samples == nullptr ||
292 piece.affine_sample_count == 0 ||
293 piece.affine_sample_count >
294 static_cast<std::size_t>(source_w) * source_h ||
295 piece.target_x_begin_native != 0 ||
296 piece.target_cols_native != 0)
297 return fail();
298 } else if (piece.source == nullptr || w.x_begin < 0 ||
299 w.y_begin < 0 || w.width <= 0 || w.height <= 0 ||
300 w.x_begin + w.width > source_w ||
301 w.y_begin + w.height > source_h || w.active_x < 0 ||
302 w.active_y < 0 || aw <= 0 || ah <= 0 ||
303 w.active_x + aw > w.width || w.active_y + ah > w.height)
304 return fail();
305 const bool single_piece_mode =
306 piece.has_cached_geometry || piece.has_local_model ||
307 piece.has_affine_samples;
308 if (single_piece_mode) {
309 // Local/cached frames are one full-width piece; a second piece
310 // or an affine piece first is a malformed stream.
311 if (any_piece || affine_open) return fail();
312 // target_cols_native == 0 is the full-width compatibility form.
313 if (piece.target_x_begin_native != 0 ||
314 (piece.target_cols_native != 0 &&
315 piece.target_cols_native != plan.native_width))
316 return fail();
317 } else {
318 // Affine piece: ordered, non-overlapping native target range.
319 // 0 cols resolves to the full-width compatibility piece.
320 const int tx0 = piece.target_x_begin_native;
321 const int tw = piece.target_cols_native == 0
322 ? plan.native_width
323 : piece.target_cols_native;
324 if (tx0 < 0 || tw <= 0 || tx0 + tw > plan.native_width ||
325 tx0 < affine_tx_end)
326 return fail();
327 }
328 if (any_piece) {
329 // All pieces of one frame must share transform/meta/mode.
330 if (first_piece.has_local_model != piece.has_local_model ||
331 first_piece.has_cached_geometry != piece.has_cached_geometry ||
332 first_piece.has_affine_samples != piece.has_affine_samples ||
333 std::memcmp(first_piece.affine6, piece.affine6,
334 sizeof(piece.affine6)) != 0 ||
335 std::memcmp(&first_piece.meta, &piece.meta,
336 sizeof(piece.meta)) != 0)
337 return fail();
338 } else {
339 first_piece = piece;
340 }
341 bool ok = false;
342 if (piece.has_cached_geometry) {
343 // A cached-geometry frame must carry a non-empty leaf list;
344 // empty frames are the provider's `skip` contract.
345 if (piece.cached_leaves == nullptr || piece.cached_leaf_count == 0)
346 return fail();
347 ok = kernel->accumulate_frame_cached_leaves(
348 piece.affine6, w, piece.source, piece.sigma2,
349 &piece.sigma2_model, piece.cached_leaves,
350 piece.cached_leaf_count, piece.cached_unique_source_samples,
351 fr, &piece.quality, &piece.meta);
352 } else if (piece.has_affine_samples) {
353 ok = kernel->accumulate_frame_affine_samples(
354 piece.affine6, piece.affine_samples, piece.affine_sample_count,
355 piece.sigma2_present, &piece.aligned_quality, fr, &piece.meta);
356 } else if (piece.has_local_model) {
357 ok = kernel->accumulate_frame_local_window(
358 piece.affine6, piece.warp, w, piece.source, piece.sigma2,
359 &piece.sigma2_model, fr, &piece.quality, &piece.meta);
360 } else {
361 if (!affine_open) {
362 if (!kernel->begin_affine_frame(fr, &piece.meta))
363 return fail();
364 affine_open = true;
365 }
366 const int tw = piece.target_cols_native == 0
367 ? plan.native_width
368 : piece.target_cols_native;
369 ok = kernel->accumulate_affine_piece(
370 piece.affine6, piece.target_x_begin_native, tw, w,
371 piece.source, piece.sigma2, &piece.sigma2_model,
372 &piece.quality);
373 affine_tx_end = piece.target_x_begin_native + tw;
374 }
375 if (!ok) {
376 kernel_ok = false;
377 return fail();
378 }
379 any_piece = true;
380 return true;
381 };
382 if (!provider(y_begin, rows, fr, sink))
383 throw std::runtime_error("FDV2_DRIVER_FRAME_PROVIDER_FAILED");
384 if (!any_piece || emit_failed) {
385 if (!kernel_ok && cuda) {
386 fail_device("accumulate()");
387 return false;
388 }
389 throw std::runtime_error("FDV2_DRIVER_FRAME_PROVIDER_FAILED");
390 }
391 if (affine_open && !kernel->finish_affine_frame(fr)) {
392 if (!cuda)
393 throw std::runtime_error("FDV2_DRIVER_ACCUMULATE_FAILED band=" +
394 std::to_string(band) +
395 " frame=" + std::to_string(fr));
396 fail_device("accumulate()");
397 return false;
398 }
399 if (!kernel_ok) {
400 if (!cuda)
401 throw std::runtime_error("FDV2_DRIVER_ACCUMULATE_FAILED band=" +
402 std::to_string(band) +
403 " frame=" + std::to_string(fr));
404 fail_device("accumulate()");
405 return false;
406 }
407 result.max_provider_enqueue_seconds = std::max(
408 result.max_provider_enqueue_seconds,
409 std::chrono::duration<double>(std::chrono::steady_clock::now() -
410 frame_t0)
411 .count());
412 // Pilot barrier: the frame stream position is the permuted order; the
413 // frame_order handed to the kernel/provider stays the original index.
414 if (full_frame && frame_pos + 1 == pilot_count && !kernel->end_pilot()) {
415 if (!cuda) throw std::runtime_error("FDV2_DRIVER_END_PILOT_FAILED");
416 fail_device("end_pilot()");
417 return false;
418 }
419 }
420 const std::size_t nrec = static_cast<std::size_t>(rows) * cols *
421 static_cast<std::size_t>(plan.channels);
422 records.assign(nrec, ForwardDrizzleV2PixelResult{});
423 if (plan.emit_profiles)
424 profiles.assign(nrec, ForwardDrizzleV2ProfileResult{});
425 std::uint64_t dense = 0;
426 const auto finalize_t0 = std::chrono::steady_clock::now();
427 const bool finalize_ok = kernel->finalize(
428 records.data(), plan.emit_profiles ? profiles.data() : nullptr,
429 &dense);
430 result.finalize_seconds += std::chrono::duration<double>(
431 std::chrono::steady_clock::now() - finalize_t0).count();
432 if (!finalize_ok) {
433 if (!cuda) throw std::runtime_error("FDV2_DRIVER_FINALIZE_FAILED");
434 fail_device("finalize()");
435 return false;
436 }
437 // Commit gate input (spec section 19): a nonfinite centre inside source
438 // support is a hard defect.
439 for (const auto &r : records)
440 if (r.source_fraction > 0.0f && !std::isfinite(r.value))
441 ++nonfinite_in_support;
442 const auto commit_t0 = std::chrono::steady_clock::now();
443 writer->commit_band(band, y_begin, rows, records,
444 plan.emit_profiles
445 ? std::span<const ForwardDrizzleV2ProfileResult>(
446 profiles)
447 : std::span<const ForwardDrizzleV2ProfileResult>{},
448 dense);
449 result.commit_seconds += std::chrono::duration<double>(
450 std::chrono::steady_clock::now() - commit_t0)
451 .count();
452 ++committed_this_attempt;
453 ++result.bands_committed;
454 if (records.capacity() + profiles.capacity() > rec_cap0)
455 ++result.driver_hotpath_allocations;
456 add_stats(result.totals, kernel->stats());
457 result.local_samples_discarded += kernel->stats().local_samples_discarded;
458 // Device-side worst frame: event deltas reported by the kernel.
459 result.max_frame_seconds =
460 std::max(result.max_frame_seconds,
461 kernel->stats().max_frame_seconds);
462
463 // Test-only kill simulation: leak the writer so the unpublished
464 // generation survives, then report the "kill".
465 if (options.simulate_kill_after_bands >= 0 &&
466 committed_this_attempt >= options.simulate_kill_after_bands) {
467 writer.release();
468 throw ForwardDrizzleV2SimulatedKill("FDV2_DRIVER_SIMULATED_KILL");
469 }
470 }
471
472 ForwardDrizzleV2CommitGate gate;
473 gate.nonfinite_pixels_inside_source_support = nonfinite_in_support;
474 gate.bands_processed = static_cast<std::uint64_t>(plan.band_count);
475 gate.telemetry_json =
476 "{\"backend\":\"" + std::string(cuda ? "cuda_v2" : "cpu_v2") +
477 "\",\"frames_processed\":" +
478 std::to_string(result.totals.frames_processed) +
479 ",\"candidates_streamed\":" +
480 std::to_string(result.totals.candidates_streamed) +
481 ",\"reservoir_kept_total\":" +
482 std::to_string(result.totals.reservoir_kept_total) +
483 ",\"local_samples_discarded\":" +
484 std::to_string(result.local_samples_discarded) + "}";
485 const auto finish_t0 = std::chrono::steady_clock::now();
486 result.generation_dir = writer->finish(gate);
487 result.commit_seconds += std::chrono::duration<double>(
488 std::chrono::steady_clock::now() - finish_t0)
489 .count();
490 // Record the published commit hash for run provenance/checkpointing
491 // (the complete-resume path reads it back via inspect()).
492 {
493 std::ifstream commit_in(result.generation_dir / "commit.json");
494 result.commit_hash = nlohmann::json::parse(commit_in)
495 .value("commit_hash", std::string{});
496 }
497 result.committed = true;
498 result.backend_used = cuda ? "cuda_v2" : "cpu_v2";
499 return true;
500}

508ForwardDrizzleV2DriverResult run_forward_drizzle_v2(
509 const fs::path &store_root, const ForwardDrizzleV2RunPlan &plan,
510 int source_width, int source_height,
511 const ForwardDrizzleV2FrameProvider &provider,
512 const ForwardDrizzleV2DriverOptions &options) {
513 if (plan.plan_hash.empty())
514 throw std::invalid_argument("FDV2_DRIVER_PLAN_NOT_FINALIZED");
515 if (plan.x_tiled)
516 throw std::invalid_argument("FDV2_DRIVER_X_TILED_UNSUPPORTED");
517 if (source_width <= 0 || source_height <= 0 || !provider)
518 throw std::invalid_argument("FDV2_DRIVER_INVALID_ARGS");
519 read_fault_env_once();
520
521 ForwardDrizzleV2DriverResult result;
522 result.bands_total = plan.band_count;
523 const auto phase_t0 = std::chrono::steady_clock::now();
524
525 const bool try_cuda =
526 options.prefer_cuda && forward_drizzle_cuda_runtime_available();
527 if (try_cuda) {
528 std::string device_failure;
529 try {
530 if (attempt_backend(store_root, plan, source_width, source_height,
531 provider, true, result, options, &device_failure)) {
532 result.phase_wall_seconds = std::chrono::duration<double>(

536 }
537 } catch (const ForwardDrizzleCudaError &e) {
538 device_failure = e.what();
539 }
540 // Spec: discard the uncommitted generation, restart the WHOLE phase on
541 // CPU. The failed attempt's writer already destructed (removing the
542 // generation); reset the attempt-scoped accounting.
543 result.cuda_fallback_reason = std::move(device_failure);
544 result.bands_committed = 0;
545 result.bands_reused = 0;
546 result.totals = ForwardDrizzleV2PrototypeStats{};
547 result.local_samples_discarded = 0;
548 result.max_frame_seconds = 0.0;
549 result.max_provider_enqueue_seconds = 0.0;
550 result.commit_seconds = 0.0;
551 result.finalize_seconds = 0.0;
552 result.driver_hotpath_allocations = 0;
553 }
554
555 if (!attempt_backend(store_root, plan, source_width, source_height,
556 provider, false, result, options, nullptr))
557 throw std::runtime_error("FDV2_DRIVER_CPU_BACKEND_FAILED");
558 result.phase_wall_seconds = std::chrono::duration<double>(
559 std::chrono::steady_clock::now() - phase_t0)
560 .count();
561 return result;
562}
