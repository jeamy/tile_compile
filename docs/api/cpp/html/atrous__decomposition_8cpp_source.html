15// One separable pass with the dilated B3 kernel over TWO channels that share
16// the same geometry (redundant-reload analysis N8): bounds checks and index
17// math run once for both accumulators instead of twice. `src` is a full
18// field; taps that fall outside [0, len) simply do not contribute.
19// `horizontal` selects the axis; `dilation` is the a-trous hole spacing
20// (1 << (level-1)).
21void conv_axis_pair(const std::vector<double> &src_a,
22 const std::vector<double> &src_b, int width, int height,
23 int dilation, bool horizontal,
24 std::vector<double> &out_a, std::vector<double> &out_b) {
25 out_a.resize(src_a.size());
26 out_b.resize(src_b.size());
27 // Rows are independent (pure gather from src into out); the tap order
28 // t = -2..2 and the skip rule per tap are exactly the generic form's, so
29 // the per-pixel accumulation is unchanged. The axis branch is hoisted out
30 // of the inner loop.
31 if (horizontal) {
32#pragma omp parallel for schedule(static)
33 for (int y = 0; y < height; ++y) {
34 const std::size_t row = static_cast<std::size_t>(y) * width;
35 for (int x = 0; x < width; ++x) {
36 double acc_a = 0.0, acc_b = 0.0;
37 for (int t = -2; t <= 2; ++t) {
38 const int xx = x + t * dilation;
39 if (xx < 0 || xx >= width) continue;
40 const std::size_t j = row + static_cast<std::size_t>(xx);
41 acc_a += kW[t + 2] * src_a[j];
42 acc_b += kW[t + 2] * src_b[j];
43 }
44 out_a[row + x] = acc_a;
45 out_b[row + x] = acc_b;
46 }
47 }
48 } else {
49#pragma omp parallel for schedule(static)
50 for (int y = 0; y < height; ++y) {
51 const std::size_t row = static_cast<std::size_t>(y) * width;
52 for (int x = 0; x < width; ++x) {
53 double acc_a = 0.0, acc_b = 0.0;
54 for (int t = -2; t <= 2; ++t) {
55 const int yy = y + t * dilation;
56 if (yy < 0 || yy >= height) continue;
57 const std::size_t j = static_cast<std::size_t>(yy) * width + x;
58 acc_a += kW[t + 2] * src_a[j];
59 acc_b += kW[t + 2] * src_b[j];
60 }
61 out_a[row + x] = acc_a;
62 out_b[row + x] = acc_b;
63 }
64 }
65 }
66}

85AtrousDecomposition atrous_decompose(const std::vector<float> &value,
86 const std::vector<uint8_t> &mask,
87 int width, int height, int levels) {
88 if (width <= 0 || height <= 0)
89 throw std::invalid_argument("ATROUS_BAD_DIMENSIONS");
90 const std::size_t n = static_cast<std::size_t>(width) * height;
91 if (value.size() != n)
92 throw std::invalid_argument("ATROUS_VALUE_SIZE_MISMATCH");
93 if (!mask.empty() && mask.size() != n)
94 throw std::invalid_argument("ATROUS_MASK_SIZE_MISMATCH");
95 if (levels < 1 || levels > 4)
96 throw std::invalid_argument("ATROUS_LEVELS_OUT_OF_RANGE");
97
98 // C_0, M_0.
99 std::vector<double> c_prev(n, 0.0);
100 std::vector<uint8_t> m_prev(n, 0u);
101 for (std::size_t i = 0; i < n; ++i) {
102 const bool valid = (mask.empty() || mask[i] != 0u) &&
103 std::isfinite(value[i]);
104 m_prev[i] = valid ? 1u : 0u;
105 c_prev[i] = valid ? static_cast<double>(value[i]) : 0.0;
106 }
107
108 AtrousDecomposition d;
109 d.width = width;
110 d.height = height;
111 d.levels = levels;
112 d.bands.resize(static_cast<std::size_t>(levels));
113
114 // N8: per-level scratch is hoisted; every level rewrites each buffer fully
115 // (masked_convolve assigns; c_cur is only read at entries m_cur marked this
116 // level).
117 std::vector<double> vm(n), md(n), hx_num, hx_den, num, den, c_cur(n);
118 std::vector<uint8_t> m_cur(n);
119 for (int j = 1; j <= levels; ++j) {
120 const int dilation = 1 << (j - 1);
121 for (std::size_t i = 0; i < n; ++i) {
122 md[i] = m_prev[i] ? 1.0 : 0.0;
123 vm[i] = m_prev[i] ? c_prev[i] : 0.0;
124 }
125 masked_convolve(vm, md, width, height, dilation, hx_num, hx_den, num,
126 den);
127 auto &band = d.bands[static_cast<std::size_t>(j - 1)];
128 band.level = j;
129 band.detail.assign(n, nanf_());
130 band.support.assign(n, 0u);
131
132 for (std::size_t i = 0; i < n; ++i) {
133 const bool den_ok = den[i] >= kAtrousDenMinFraction;
134 m_cur[i] = (m_prev[i] != 0u && den_ok) ? 1u : 0u;
135 if (den_ok) c_cur[i] = num[i] / den[i];
136 // D_j valid only on M_(j-1) && M_j.
137 if (m_prev[i] != 0u && m_cur[i] != 0u) {
138 band.support[i] = 1u;
139 band.detail[i] = static_cast<float>(c_prev[i] - c_cur[i]);
140 }
141 }
142 c_prev.swap(c_cur);
143 m_prev.swap(m_cur);
144 }
145
146 d.coarse.assign(n, nanf_());
147 d.coarse_support.assign(n, 0u);
148 for (std::size_t i = 0; i < n; ++i)
149 if (m_prev[i] != 0u) {
150 d.coarse_support[i] = 1u;
151 d.coarse[i] = static_cast<float>(c_prev[i]);
152 }
153 return d;
154}

156double atrous_reconstruction_max_error(const AtrousDecomposition &d,
157 const std::vector<float> &original) {
158 const std::size_t n =
159 static_cast<std::size_t>(d.width) * static_cast<std::size_t>(d.height);
160 if (original.size() != n) throw std::invalid_argument("ATROUS_ORIG_SIZE");
161 double max_err = 0.0;
162 for (std::size_t i = 0; i < n; ++i) {
163 if (!d.coarse_support[i]) continue; // tightest common valid support
164 double sum = d.coarse[i];
165 bool ok = std::isfinite(sum);
166 for (const auto &b : d.bands) {
167 if (!b.support[i] || !std::isfinite(b.detail[i])) { ok = false; break; }
168 sum += b.detail[i];
169 }
170 if (!ok || !std::isfinite(original[i])) continue;
171 max_err = std::max(max_err, std::abs(static_cast<double>(original[i]) - sum));
172 }
173 return max_err;
174}
