263 auto robust_location_scale = [](const std::vector<float> &values) {
264 std::pair<float, float> out{0.0f, 1.0f};
265 if (values.empty()) {
266 return out;
267 }
268 std::vector<float> tmp = values;
269 const float center = core::median_of(tmp);
270 for (float &v : tmp) {
271 v = std::fabs(v - center);
272 }
273 const float mad = core::median_of(tmp);
274 out.first = center;
275 out.second = std::max(1.4826f * mad, 1.0e-6f);
276 return out;
277 };
278 auto blended_zscores = [&](const std::vector<float> &local_values,
279 const std::vector<float> &pooled_values,
280 std::vector<float> *out_local_z,
281 std::vector<float> *out_blended_z) {
282 out_local_z->assign(local_values.size(), 0.0f);
283 out_blended_z->assign(local_values.size(), 0.0f);
284 if (local_values.empty()) {
285 return;
286 }
287 core::robust_zscore(local_values, *out_local_z);
288 if (!neighborhood_enabled || pooled_values.empty()) {
289 *out_blended_z = *out_local_z;
290 return;

292 const auto [center, scale] = robust_location_scale(pooled_values);
293 for (size_t i = 0; i < local_values.size(); ++i) {
294 const float z_pooled = (local_values[i] - center) / scale;
295 (*out_blended_z)[i] =
296 (1.0f - neighborhood_blend) * (*out_local_z)[i] +
297 neighborhood_blend * z_pooled;
298 }
