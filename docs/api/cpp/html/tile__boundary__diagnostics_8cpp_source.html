35 const int overlap_x0 = std::max(lhs_tile.x, rhs_tile.x);
36 const int overlap_y0 = std::max(lhs_tile.y, rhs_tile.y);

38 std::min(lhs_tile.x + lhs_tile.width, rhs_tile.x + rhs_tile.width);
39 const int overlap_y1 =
40 std::min(lhs_tile.y + lhs_tile.height, rhs_tile.y + rhs_tile.height);
41 const int overlap_w = overlap_x1 - overlap_x0;
42 const int overlap_h = overlap_y1 - overlap_y0;

44 return out;
45 }

49 std::vector<float> lhs_vals;
50 std::vector<float> rhs_vals;
51 abs_diffs.reserve(static_cast<size_t>(overlap_w * overlap_h));
52 signed_diffs.reserve(static_cast<size_t>(overlap_w * overlap_h));
53 lhs_vals.reserve(static_cast<size_t>(overlap_w * overlap_h));
54 rhs_vals.reserve(static_cast<size_t>(overlap_w * overlap_h));

58 const int rhs_y = overlap_y0 + oy - rhs_tile.y;
59 for (int ox = 0; ox < overlap_w; ++ox) {
60 const int lhs_x = overlap_x0 + ox - lhs_tile.x;
61 const int rhs_x = overlap_x0 + ox - rhs_tile.x;
62 const int global_x = overlap_x0 + ox;
63 const int global_y = overlap_y0 + oy;
64 if (!common_valid_mask.empty()) {
65 if (global_x < 0 || global_x >= common_mask_width || global_y < 0 ||
66 global_y >= common_mask_height) {
67 continue;
68 }
69 const size_t mask_idx =
70 static_cast<size_t>(global_y) *
71 static_cast<size_t>(common_mask_width) +
72 static_cast<size_t>(global_x);
73 if (mask_idx >= common_valid_mask.size() ||
74 common_valid_mask[mask_idx] == 0u) {
75 continue;
76 }
77 }
78 const float lhs_v = lhs(lhs_y, lhs_x);
79 const float rhs_v = rhs(rhs_y, rhs_x);
80 if (!std::isfinite(lhs_v) || !std::isfinite(rhs_v)) {
81 continue;
82 }
83 const float diff = rhs_v - lhs_v;
84 signed_sum += static_cast<double>(diff);
85 signed_diffs.push_back(diff);
86 abs_diffs.push_back(std::fabs(diff));
87 lhs_vals.push_back(lhs_v);
88 rhs_vals.push_back(rhs_v);
89 }
90 }
91
92 if (abs_diffs.empty()) {

94 }
95
96 out.sample_count = abs_diffs.size();
97 const double abs_sum = std::accumulate(abs_diffs.begin(), abs_diffs.end(), 0.0);
98 out.mean_abs_diff =
99 static_cast<float>(abs_sum / static_cast<double>(out.sample_count));
100 out.mean_signed_diff =
101 static_cast<float>(signed_sum / static_cast<double>(out.sample_count));
102 std::sort(abs_diffs.begin(), abs_diffs.end());
103 out.p95_abs_diff = core::percentile_from_sorted(abs_diffs, 95.0f);
104
105 std::vector<float> abs_residuals;
106 abs_residuals.reserve(signed_diffs.size());
107 for (float diff : signed_diffs) {
108 abs_residuals.push_back(std::fabs(diff - out.mean_signed_diff));
109 }
110 const double residual_sum =
111 std::accumulate(abs_residuals.begin(), abs_residuals.end(), 0.0);
112 out.mean_abs_residual = static_cast<float>(
113 residual_sum / static_cast<double>(out.sample_count));
114 std::sort(abs_residuals.begin(), abs_residuals.end());
115 out.p95_abs_residual = core::percentile_from_sorted(abs_residuals, 95.0f);
116
117 auto overlap_mad = [](std::vector<float> values) {
118 if (values.empty()) {
119 return 0.0f;
120 }
121 const float center = core::median_of(values);
122 for (float &v : values) {
123 v = std::fabs(v - center);
124 }
125 return core::median_of(values);
126 };
127 const float lhs_mad = overlap_mad(lhs_vals);
128 const float rhs_mad = overlap_mad(rhs_vals);
129 constexpr float kScaleEps = 1.0e-6f;
130 out.scale_ratio =
131 (lhs_mad > kScaleEps) ? (rhs_mad / lhs_mad)
132 : ((rhs_mad > kScaleEps) ? std::numeric_limits<float>::infinity()
133 : 1.0f);
134 out.valid = true;
135 return out;
136}

145 const std::vector<Tile> &tiles, const std::vector<Matrix2Df> &images,
146 const std::vector<uint8_t> &tile_valid,

149 TileBoundaryDiagnostics out;
150 if (tiles.empty() || images.size() != tiles.size() ||
151 tile_valid.size() != tiles.size()) {
152 return out;
153 }
154
155 auto pair_key = [](int row, int col) -> long long {
156 return (static_cast<long long>(row) << 32) ^
157 static_cast<unsigned int>(col);
158 };
159
160 std::unordered_map<long long, size_t> tile_index_by_grid;
161 tile_index_by_grid.reserve(tiles.size());
162 for (size_t ti = 0; ti < tiles.size(); ++ti) {
163 tile_index_by_grid[pair_key(tiles[ti].row, tiles[ti].col)] = ti;
164 }
165
166 std::vector<std::pair<size_t, size_t>> boundary_pairs;
167 boundary_pairs.reserve(tiles.size() * 2);
168 auto push_pair = [&](size_t lhs, size_t rhs) {
169 if (lhs >= tiles.size() || rhs >= tiles.size() || lhs == rhs ||
170 tile_valid[lhs] == 0u || tile_valid[rhs] == 0u) {
171 return;
172 }
173 boundary_pairs.push_back({lhs, rhs});
174 };
175 for (size_t ti = 0; ti < tiles.size(); ++ti) {
176 const Tile &tile = tiles[ti];
177 auto right_it = tile_index_by_grid.find(pair_key(tile.row, tile.col + 1));
178 if (right_it != tile_index_by_grid.end()) {
179 push_pair(ti, right_it->second);
180 }
181 auto down_it = tile_index_by_grid.find(pair_key(tile.row + 1, tile.col));
182 if (down_it != tile_index_by_grid.end()) {
183 push_pair(ti, down_it->second);
184 }
185 }
186 if (boundary_pairs.empty()) {
187 for (size_t lhs = 0; lhs < tiles.size(); ++lhs) {
188 if (tile_valid[lhs] == 0u) {
189 continue;
190 }
191 for (size_t rhs = lhs + 1; rhs < tiles.size(); ++rhs) {
192 if (tile_valid[rhs] == 0u) {
193 continue;
194 }
195 const Tile &a = tiles[lhs];
196 const Tile &b = tiles[rhs];
197 const bool overlap = a.x < (b.x + b.width) && b.x < (a.x + a.width) &&
198 a.y < (b.y + b.height) && b.y < (a.y + a.height);
199 if (overlap) {
200 boundary_pairs.push_back({lhs, rhs});
201 }
202 }
203 }
204 }
205 out.pair_count = boundary_pairs.size();

211 std::vector<float> pair_p95_abs_residual;

264 out.pair_scale_ratio_deviation_mean =

286
287 std::sort(out.pair_diagnostics.begin(), out.pair_diagnostics.end(),
288 [](const TileBoundaryPairDiagnostic &a,
289 const TileBoundaryPairDiagnostic &b) {
290 if (a.mean_abs_diff != b.mean_abs_diff) {
291 return a.mean_abs_diff > b.mean_abs_diff;
292 }
293 if (a.p95_abs_diff != b.p95_abs_diff) {
294 return a.p95_abs_diff > b.p95_abs_diff;
295 }
296 return a.sample_count > b.sample_count;
297 });
298 return out;
