43// central difference with one-sided fallback at borders / non-finite
44// neighbours; 0 when no finite neighbour exists.
45float v2_central_diff(const Matrix2Df &s, int y, int x, bool x_axis) {
46 const int extent = x_axis ? static_cast<int>(s.cols())
47 : static_cast<int>(s.rows());
48 const int pm = (x_axis ? x : y) - 1, pp = (x_axis ? x : y) + 1;
49 const float vm = (pm >= 0) ? (x_axis ? s(y, pm) : s(pm, x))
50 : std::numeric_limits<float>::quiet_NaN();
51 const float vp = (pp < extent) ? (x_axis ? s(y, pp) : s(pp, x))
52 : std::numeric_limits<float>::quiet_NaN();
53 const bool fm = std::isfinite(vm), fp = std::isfinite(vp);
54 if (fm && fp) return (vp - vm) * 0.5f;
55 const float vc = s(y, x);
56 if (fp && std::isfinite(vc)) return vp - vc;
57 if (fm && std::isfinite(vc)) return vc - vm;
58 return 0.0f;
59}

69RegResiduals load_reg_residuals(const fs::path &path, std::size_t count) {
70 RegResiduals out;
71 std::error_code ec;
72 if (path.empty() || !fs::is_regular_file(fs::symlink_status(path, ec)))
73 return out;
74 nlohmann::json j;
75 try {
76 std::ifstream in(path);
77 j = nlohmann::json::parse(in);
78 } catch (...) {
79 return out;
80 }
81 if (!j.contains("star_residuals") || !j["star_residuals"].is_array() ||
82 j["star_residuals"].size() < count)
83 return out;
84 out.rms_px.assign(count, std::numeric_limits<float>::quiet_NaN());
85 out.applicable.assign(count, 0u);
86 for (std::size_t i = 0; i < count; ++i) {
87 const auto &e = j["star_residuals"][i];
88 if (!e.is_object()) continue;
89 const double rms = e.value("rms_px",
90 std::numeric_limits<double>::quiet_NaN());
91 const bool appl = e.value("applicable", false);
92 out.rms_px[i] = static_cast<float>(rms);
93 out.applicable[i] = appl ? 1u : 0u;
94 }
95 return out;
96}

129std::vector<float> forward_drizzle_v2_sigma2_plane(
130 const Matrix2Df &source, double sigma_noise, double sigma_reg_px,
131 double droplet_half) {
132 const int h = static_cast<int>(source.rows()),
133 w = static_cast<int>(source.cols());
134 std::vector<float> out(static_cast<std::size_t>(h) * w);
135 for (int y = 0; y < h; ++y)
136 for (int x = 0; x < w; ++x) {
137 const double gx = v2_central_diff(source, y, x, true);
138 const double gy = v2_central_diff(source, y, x, false);
139 out[static_cast<std::size_t>(y) * w + x] = static_cast<float>(
140 forward_drizzle_v2_sigma2_model(sigma_noise, gx, gy, sigma_reg_px,
141 droplet_half));
142 }
143 return out;
144}

146std::vector<float> forward_drizzle_v2_sigma2_rect(
147 const Matrix2Df &window, int win_x0, int win_y0, int x0, int y0, int w,
148 int h, int source_w, int source_h, double sigma_noise,
149 double sigma_reg_px, double droplet_half) {
150 const int wh = static_cast<int>(window.rows());
151 const int ww = static_cast<int>(window.cols());
152 // Absolute-coordinate accessor: outside the read window (or outside the
153 // true source extent) the neighbour counts as non-finite, which reproduces
154 // the whole-plane oracle's border/NaN fallback chain.
155 auto at = [&](int ax, int ay) -> float {
156 if (ax < 0 || ay < 0 || ax >= source_w || ay >= source_h ||
157 ax < win_x0 || ay < win_y0 || ax >= win_x0 + ww ||
158 ay >= win_y0 + wh)
159 return std::numeric_limits<float>::quiet_NaN();
160 return window(ay - win_y0, ax - win_x0);
161 };
162 auto diff = [&](int sx, int sy, bool x_axis) -> float {
163 const float vm = at(sx - (x_axis ? 1 : 0), sy - (x_axis ? 0 : 1));
164 const float vp = at(sx + (x_axis ? 1 : 0), sy + (x_axis ? 0 : 1));
165 const bool fm = std::isfinite(vm), fp = std::isfinite(vp);
166 if (fm && fp) return (vp - vm) * 0.5f;
167 const float vc = at(sx, sy);
168 if (fp && std::isfinite(vc)) return vp - vc;
169 if (fm && std::isfinite(vc)) return vc - vm;
170 return 0.0f;
171 };
172 std::vector<float> out(static_cast<std::size_t>(w) * h);
173 for (int ly = 0; ly < h; ++ly)
174 for (int lx = 0; lx < w; ++lx) {
175 const int sx = x0 + lx, sy = y0 + ly;
176 out[static_cast<std::size_t>(ly) * w + lx] = static_cast<float>(
177 forward_drizzle_v2_sigma2_model(sigma_noise, diff(sx, sy, true),
178 diff(sx, sy, false), sigma_reg_px,
179 droplet_half));
180 }
181 return out;
182}

184void forward_drizzle_v2_affine_storage_spans(
185 const std::vector<DrizzleAffineSourceSpan> &active_spans, int source_w,
186 int source_h, bool need_sigma_neighbours,
187 std::vector<int> &row_active_scratch,
188 std::vector<DrizzleAffineSourceSpan> &storage_spans) {
189 storage_spans.clear();
190 if (active_spans.empty()) return;
191 if (!need_sigma_neighbours) {
192 storage_spans.assign(active_spans.begin(), active_spans.end());
193 return;
194 }
195 // Sigma2 needs the in-bounds one-pixel ring: storage row r covers its own
196 // active span expanded x+-1 plus the active spans of rows r-1/r+1 (the
197 // vertical neighbours of those rows' samples).
198 row_active_scratch.assign(static_cast<std::size_t>(source_h), -1);
199 for (std::size_t i = 0; i < active_spans.size(); ++i)
200 row_active_scratch[static_cast<std::size_t>(
201 active_spans[i].source_y)] = static_cast<int>(i);
202 for (int r = 0; r < source_h; ++r) {
203 const int self =
204 row_active_scratch[static_cast<std::size_t>(r)];
205 const int up = r > 0 ? row_active_scratch[static_cast<std::size_t>(r - 1)]
206 : -1;
207 const int dn = r + 1 < source_h
208 ? row_active_scratch[static_cast<std::size_t>(r + 1)]
209 : -1;
210 if (self < 0 && up < 0 && dn < 0) continue;
211 int x0 = source_w, x1 = 0;
212 auto cover = [&](int idx, int pad) {
213 if (idx < 0) return;
214 const auto &s = active_spans[static_cast<std::size_t>(idx)];
215 x0 = std::min(x0, s.x_begin - pad);
216 x1 = std::max(x1, s.x_end + pad);
217 };
218 cover(self, 1);
219 cover(up, 0);
220 cover(dn, 0);
221 x0 = std::max(0, x0);
222 x1 = std::min(source_w, x1);
223 if (x0 < x1) storage_spans.push_back({r, x0, x1});
224 }

227void forward_drizzle_v2_build_affine_samples(
228 const std::vector<DrizzleAffineSourceSpan> &active_spans,
229 const std::vector<DrizzleAffineSourceSpan> &storage_spans,
230 const std::vector<float> &storage_values, bool sigma2_present,
231 double sigma_noise, double sigma_reg_px, double droplet_half,
232 int source_w, int source_h, std::vector<int> &row_storage_scratch,
233 std::vector<std::size_t> &storage_offset_scratch,
234 std::vector<ForwardDrizzleV2SourceSample> &out_samples) {
235 out_samples.clear();
236 row_storage_scratch.assign(static_cast<std::size_t>(source_h), -1);
237 storage_offset_scratch.clear();
238 std::size_t off = 0;
239 for (std::size_t i = 0; i < storage_spans.size(); ++i) {
240 const auto &s = storage_spans[i];
241 row_storage_scratch[static_cast<std::size_t>(s.source_y)] =
242 static_cast<int>(i);
243 storage_offset_scratch.push_back(off);
244 off += static_cast<std::size_t>(s.x_end - s.x_begin);
245 }
246 // Absolute-coordinate accessor over the packed storage: outside the true
247 // source extent or a row's stored interval the neighbour is non-finite,
248 // reproducing the whole-plane oracle's border/NaN fallback chain.
249 auto at = [&](int ax, int ay) -> float {
250 if (ax < 0 || ay < 0 || ax >= source_w || ay >= source_h)
251 return std::numeric_limits<float>::quiet_NaN();
252 const int idx = row_storage_scratch[static_cast<std::size_t>(ay)];
253 if (idx < 0) return std::numeric_limits<float>::quiet_NaN();
254 const auto &s = storage_spans[static_cast<std::size_t>(idx)];
255 if (ax < s.x_begin || ax >= s.x_end)
256 return std::numeric_limits<float>::quiet_NaN();
257 return storage_values[storage_offset_scratch[static_cast<std::size_t>(idx)] +
258 static_cast<std::size_t>(ax - s.x_begin)];
259 };
260 auto diff = [&](int sx, int sy, bool x_axis) -> float {
261 const float vm = at(sx - (x_axis ? 1 : 0), sy - (x_axis ? 0 : 1));
262 const float vp = at(sx + (x_axis ? 1 : 0), sy + (x_axis ? 0 : 1));
263 const bool fm = std::isfinite(vm), fp = std::isfinite(vp);
264 if (fm && fp) return (vp - vm) * 0.5f;
265 const float vc = at(sx, sy);
266 if (fp && std::isfinite(vc)) return vp - vc;
267 if (fm && std::isfinite(vc)) return vc - vm;
268 return 0.0f;
269 };
270 for (const auto &span : active_spans) {
271 const int sy = span.source_y;
272 for (int sx = span.x_begin; sx < span.x_end; ++sx) {
273 ForwardDrizzleV2SourceSample smp;
274 smp.source_x = static_cast<std::uint32_t>(sx);
275 smp.source_y = static_cast<std::uint32_t>(sy);
276 smp.value = at(sx, sy);
277 smp.sigma2 =
278 sigma2_present
279 ? static_cast<float>(forward_drizzle_v2_sigma2_model(
280 sigma_noise, diff(sx, sy, true), diff(sx, sy, false),
281 sigma_reg_px, droplet_half))
282 : 0.0f;
283 out_samples.push_back(smp);
284 }
285 }

288std::size_t forward_drizzle_v2_cpu_bytes_per_native_pixel(
289 int channels, int internal_scale, int reservoir_slots, bool sigma2_plane,
290 bool emit_profiles, bool full_frame_estimator) {
291 const std::size_t res_slots =
292 static_cast<std::size_t>(reservoir_slots);
293 std::size_t per_ch =
294 sizeof(double) * 7 + sizeof(unsigned int) * 2 +
295 sizeof(unsigned short) + sizeof(std::uint64_t) +
296 res_slots * kV2ReservoirRecordBytes + kV2PixelResultBytes;
297 if (emit_profiles)
298 per_ch += res_slots * kV2ReservoirQualityBytes + kV2ProfileResultBytes;
299 if (full_frame_estimator) per_ch += kV2CpuFullAccBytes;
300 const std::size_t frame_planes =
301 (sigma2_plane ? 4 : 3) + (emit_profiles ? 5 : 0);
302 const std::size_t internal =
303 frame_planes * sizeof(double) * static_cast<std::size_t>(channels) *
304 static_cast<std::size_t>(internal_scale) *
305 static_cast<std::size_t>(internal_scale);
306 return per_ch * static_cast<std::size_t>(channels) +
307 sizeof(unsigned int) /* footprint */ + internal;
308}

310ForwardDrizzleV2RunPlan make_forward_drizzle_v2_run_plan(
311 const registration::RegistrationSamplingPlan &sampling,
312 const config::ReconstructionDrizzleConfig &drizzle_cfg,
313 const config::ReconstructionClippingConfig &clipping_cfg,
314 const config::ReconstructionMultibandConfig &multiband_cfg,
315 bool emit_profiles, std::uint64_t frame_count,
316 const std::string &normalized_cache_hash,
317 const std::string &quality_plan_hash,
318 const std::string &source_quality_cache_hash,
319 const std::string &config_snapshot_hash, std::size_t host_budget_bytes,
320 std::size_t device_budget_bytes) {
321 const OutputScaleMode osm{drizzle_cfg.internal_scale,
322 drizzle_cfg.output_scale};
323 if (!osm.valid())
324 throw std::invalid_argument("FDV2_PROD_INVALID_OUTPUT_SCALE");
325 // v2 stores on the native output grid; only output_scale == 1 is
326 // supported by the cutover contract (2/2 rejected at plan time).
327 if (drizzle_cfg.output_scale != 1)
328 throw std::invalid_argument("FDV2_PROD_OUTPUT_SCALE_UNSUPPORTED");
329
330 const int channels = sampling.color_mode == ColorMode::MONO ? 1 : 3;
331 // Reservoir slots are bound by the exact global keep set of the frame
332 // stream (default plan reservoir size 64 and the plan default seed),
333 // capped at the historical 2*64 overflow bound.
334 const auto selected_frame_orders =
335 forward_drizzle_v2_selected_frame_orders(
336 frame_count, 64, 11400714819323198485ULL);
337 const std::size_t bpp = forward_drizzle_v2_cpu_bytes_per_native_pixel(
338 channels, drizzle_cfg.internal_scale,
339 static_cast<int>(std::max<std::size_t>(
340 1, std::min(selected_frame_orders.size(), std::size_t{2 * 64}))),
341 true /*sigma2*/, emit_profiles, drizzle_cfg.full_frame_estimator);
342 // The band workspace shares the host budget with the source/Q caches and
343 // the fusion working set; give it half. When a device path is planned the
344 // nominal device bound additionally caps it (deterministic across resume;
345 // reserve() still verifies the real device per band).
346 const std::size_t meta_bytes = v2_checked_mul(
347 static_cast<std::size_t>(frame_count), kV2FrameMetaBytes);
348 std::size_t band_budget = host_budget_bytes / 2;
349 if (device_budget_bytes > 0)
350 band_budget = std::min(band_budget, device_budget_bytes);
351 band_budget = band_budget > meta_bytes ? band_budget - meta_bytes : 0;
352 const std::size_t row_bytes = v2_checked_mul(
353 bpp, static_cast<std::size_t>(sampling.canvas_width_native));
354 if (row_bytes == 0 || band_budget / row_bytes < 1)
355 throw std::invalid_argument("FDV2_PROD_BAND_BUDGET");
356 const int band_rows = static_cast<int>(std::clamp<std::size_t>(
357 band_budget / row_bytes, 1,
358 static_cast<std::size_t>(sampling.canvas_height_native)));
359
360 if (drizzle_cfg.full_frame_estimator && !emit_profiles)
361 throw std::invalid_argument("FDV2_PROD_FULL_FRAME_REQUIRES_PROFILES");
362 ForwardDrizzleV2RunPlan plan;
363 if (drizzle_cfg.full_frame_estimator)
364 plan.estimator = kFdV2EstimatorPilotFullFrame;
365 plan.source_identity_hash = sampling.source_identity_hash;
366 plan.normalized_cache_hash = normalized_cache_hash;
367 plan.quality_plan_hash = quality_plan_hash;
368 plan.sampling_plan_hash = sampling.plan_hash;
369 plan.config_snapshot_hash = config_snapshot_hash;
370 plan.source_quality_cache_hash = source_quality_cache_hash;
371 plan.min_clip_contributors = drizzle_cfg.min_clip_contributors;
372 plan.min_candidates = drizzle_cfg.min_clip_contributors;
373 plan.robust_passes = drizzle_cfg.robust_passes;
374 plan.sigma_low = clipping_cfg.clip_sigma_low;
375 plan.sigma_high = clipping_cfg.clip_sigma_high;
376 plan.shared_frame_rejection = clipping_cfg.shared_frame_rejection;
377 plan.shared_frame_rejection_consensus =
378 clipping_cfg.shared_frame_rejection_consensus;
379 plan.bimodal_veto = clipping_cfg.bimodal_veto;
380 plan.bimodal_veto_gap_sigma = clipping_cfg.bimodal_veto_gap_sigma;
381 plan.sigma2_enabled = true;
382 bool any_local = false;
383 for (const auto &f : sampling.frames)
384 if (f.valid && f.has_smooth_local_model) any_local = true;
385 plan.local_warp_representation =
386 any_local ? "smooth_local_coefficients" : "affine_only";
387 plan.emit_profiles = emit_profiles;
388 plan.multiband_levels = emit_profiles ? multiband_cfg.levels : 0;
389 plan.fine_quality_exponent = multiband_cfg.fine_quality_exponent;
390 plan.medium_quality_exponent = multiband_cfg.medium_quality_exponent;
391 plan.native_width = sampling.canvas_width_native;
392 plan.native_height = sampling.canvas_height_native;
393 plan.channels = channels;
394 plan.internal_scale = drizzle_cfg.internal_scale;
395 plan.color_mode = channels == 1 ? "MONO" : "OSC";
396 plan.pixfrac = drizzle_cfg.pixfrac;
397 plan.bayer_pattern = static_cast<int>(sampling.bayer_pattern);
398 plan.cfa_origin_x = sampling.cfa_origin_x;
399 plan.cfa_origin_y = sampling.cfa_origin_y;
400 plan.frame_count = frame_count;
401 plan.band_rows = band_rows;
402 plan.band_count = (plan.native_height + band_rows - 1) / band_rows;
403 plan.tile_cols = 0;
404 plan.halo_rows =
405 emit_profiles ? multiband_fusion_halo_rows(multiband_cfg.levels) : 0;
406 plan.x_tiled = false;
407 finalize_forward_drizzle_v2_run_plan(plan);
408 return plan;
409}

411ForwardDrizzleV2ProductionResult persist_forward_drizzle_v2_from_predecessors(
412 const fs::path &store_root, const fs::path &quality_artifact,
413 const registration::RegistrationSamplingPlan &sampling,
414 VerifiedNormalizedSourceCache &cache, const GlobalQualityConfig &quality_cfg,
415 const config::ReconstructionDrizzleConfig &drizzle_cfg,
416 const config::ReconstructionClippingConfig &clipping_cfg,
417 const config::ReconstructionMultibandConfig &multiband_cfg,
418 const fs::path &source_quality_cache_root,
419 const fs::path &source_quality_metrics_json,
420 const fs::path &global_registration_json,
421 const DrizzleGeometryCacheReader *geometry_cache,
422 const ForwardDrizzleSubdivisionParams &subdivision,
423 const std::string &config_snapshot_hash,
424 const std::string &acceleration_backend,
425 const std::function<void(int, int)> &progress, int cpu_workers) {
426 // The subdivision contract lives in the committed geometry cache; v2
427 // never re-runs it (kept in the signature for caller compatibility).
428 (void)subdivision;
429 if (source_quality_cache_root.empty())
430 throw std::invalid_argument("FDV2_PROD_REQUIRES_SOURCE_QUALITY_CACHE");
431 const std::size_t mb =
432 drizzle_cfg.memory_budget_mb ? drizzle_cfg.memory_budget_mb : 512;
433 const auto quality = load_source_quality_artifact(quality_artifact, sampling,
434 cache, quality_cfg, mb);
435 const auto weights =
436 resolve_quality_frame_weights(quality, sampling, quality_cfg, mb);
437
438 SourceQualityMapCacheReader qreader(
439 source_quality_cache_root,
440 compute_source_quality_identity_hash(sampling, cache.manifest_hash()),
441 /*expected_config_hash=*/"");
442 if (!qreader.usable())
443 throw std::runtime_error("FDV2_PROD_SOURCE_QUALITY_CACHE_UNUSABLE: " +
444 qreader.error());
445
446 const auto noise =
447 load_frame_noise(source_quality_metrics_json, sampling,
448 cache.manifest_hash(),
449 compute_source_quality_config_hash(quality_cfg));
450 const auto residuals =
451 load_reg_residuals(global_registration_json, sampling.frames.size());

453 // Participating stream: valid frames minus the local-model exclusion set,
454 // in plan order (source_index-ascending is the plan's canonical order).
455 struct Entry {
456 const registration::FrameSamplingTransform *f;
457 bool has_qc, has_q0, has_q1, has_qa, has_sigma2;
458 };
459 std::vector<Entry> stream;
460 ForwardDrizzleV2ProductionResult out;
461 const double half = 0.5 * static_cast<double>(drizzle_cfg.pixfrac);
462 for (const auto &f : sampling.frames) {
463 if (!f.valid) {
464 ++out.frames_skipped_invalid;
465 continue;
466 }
467 if (!f.source_to_canvas_affine_valid || !f.source_to_canvas.allFinite())
468 throw std::invalid_argument("DRIZZLE_INVALID_TRANSFORM");
469 if (f.has_smooth_local_model) {
470 // v2 fails closed: a local frame without a committed geometry-cache
471 // entry can never be served (no inversion/subdivision is re-run in
472 // FORWARD_DRIZZLE). The cache's finalised stats drive exclusion.
473 if (geometry_cache == nullptr)
474 throw std::runtime_error("FDV2_PROD_LOCAL_FRAME_NO_GEOMETRY_CACHE");
475 const auto st =
476 geometry_cache->frame_stats(drizzle_cfg.pixfrac, f.source_index);
477 if (!st.present)
478 throw std::runtime_error("FDV2_PROD_LOCAL_FRAME_NO_GEOMETRY_CACHE");
479 const double rate = st.subdivision_error_rate;
480 const std::uint64_t total = st.samples_total;
481 const std::uint64_t discarded = st.samples_discarded;
482 out.local_model_samples_total += total;
483 out.local_model_samples_discarded += discarded;
484 if (st.excluded) {
485 out.frames_excluded_subdivision_error_rate.emplace_back(f.frame_id,
486 rate);
487 continue;
488 }
489 }
490 Entry e;
