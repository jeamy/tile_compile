57std::vector<std::string> plane_names(const DrizzleStoreIdentity &i) {
58 const bool mb = is_multiband_mode(i.mode);
59 if (i.width <= 0 || i.height <= 0 || i.source_identity_hash.empty() ||
60 i.sampling_plan_hash.empty() || i.reconstruction_hash.empty() ||
61 (i.mode != "uniform_unclipped" && i.mode != "uniform_raw_clipped" && !mb) ||
62 (i.color_mode != ColorMode::MONO && i.color_mode != ColorMode::OSC) ||
63 (mb ? (i.multiband_levels < 1 || i.multiband_levels > 4)
64 : i.multiband_levels != 0))

66 std::vector<std::string> names;
67 std::vector<std::string> profiles = {"uniform"};
68 if (i.mode != "uniform_unclipped") profiles.push_back("raw");
69 if (mb) {
70 profiles.push_back("fine");
71 if (i.multiband_levels >= 2) profiles.push_back("medium");
72 }
73 for (const auto &profile : profiles) {
74 for (const auto &channel : {"L", "R", "G", "B"}) {
75 if ((std::string(channel) == "L") != (i.color_mode == ColorMode::MONO)) continue;
76 for (const auto &field : {"value", "weight_sum", "n_eff", "support"})
77 names.push_back(profile + "_" + channel + "_" + field);
78 }
79 }
80 if (mb)

82 names.push_back(std::string(m) + "_X_value");

185 void start() {
186 fs::create_directories(root_);

189 const uint64_t per_pixel = names.size() * sizeof(float);

199 }

201 for (const auto &name : names)
202 files_.emplace(name, std::make_unique<PlaneFile>(generation_ / (name + ".fits"),

205 void write_profile(const std::string &profile, int y, const ForwardDrizzleUniformResult &stripe) {
206 const int rows = stripe.internal_height;
207 if (stripe.internal_width != identity_.width || stripe.color_mode != identity_.color_mode ||
208 rows <= 0 || rows > identity_.height - y)
209 throw std::runtime_error("DRIZZLE_STORE_INVALID_STRIPE");
210 for (const auto &[channel, p] : std::array<std::pair<const char *, const ProfilePlane *>, 4>{
211 {{"R", &stripe.R}, {"G", &stripe.G}, {"B", &stripe.B}, {"L", &stripe.L}}}) {
212 if ((std::string(channel) == "L") != (identity_.color_mode == ColorMode::MONO)) continue;
213 const size_t count = static_cast<size_t>(rows) * identity_.width;
214 if (p->width != identity_.width || p->height != rows || p->value.size() != count ||
215 p->weight_sum.size() != count || p->n_eff.size() != count || p->support.size() != count)
216 throw std::runtime_error("DRIZZLE_STORE_INVALID_PLANE");
217 const std::string base = profile + "_" + channel + "_";
218 files_.at(base + "value")->write(y, identity_.width, p->value);
219 files_.at(base + "weight_sum")->write(y, identity_.width, p->weight_sum);
220 files_.at(base + "n_eff")->write(y, identity_.width, p->n_eff);
221 std::vector<float> row(identity_.width);
222 for (int dy = 0; dy < rows; ++dy) {
223 for (int x = 0; x < identity_.width; ++x)
224 row[x] = p->support[static_cast<size_t>(dy) * identity_.width + x] ? 1.0f : 0.0f;
225 files_.at(base + "support")->write(y + dy, identity_.width, row);
226 }
227 }

230 if (y != next_y_) throw std::runtime_error("DRIZZLE_STORE_STRIPE_ORDER");

233 next_y_ += uniform.internal_height;
234 }

236 if (next_y_ != identity_.height) throw std::runtime_error("DRIZZLE_STORE_INCOMPLETE_STRIPES");
237 for (auto &[name, file] : files_) file->finish();
238 files_.clear();

259DrizzleStoreIdentity make_drizzle_store_identity(
260 const registration::RegistrationSamplingPlan &plan,
261 const config::ReconstructionDrizzleConfig &cfg,
262 const ForwardDrizzleSubdivisionParams &subdivision,
263 const config::ReconstructionClippingConfig *clipping, const std::vector<float> &g_eff,
264 const DrizzleStorePredecessors &predecessors,
265 const MultibandStoreContract &multiband) {
266 config::ReconstructionDrizzleConfig budget_cfg = cfg;
267 const auto dims = plan_drizzle_memory_autogrow(
268 plan, budget_cfg, 1, 0, true, [](const std::string &msg) {
269 std::cout << "[DRIZZLE_STORE][warn] " << msg << std::endl;
270 });
271 const OutputScaleMode osm{cfg.internal_scale, cfg.output_scale};
272 if (!osm.valid())
273 throw std::invalid_argument("DRIZZLE_STORE_INVALID_OUTPUT_SCALE");
274 const bool mb = multiband.enabled;
275 if (mb) {
276 if (!clipping)
277 throw std::invalid_argument("DRIZZLE_STORE_MULTIBAND_REQUIRES_CLIPPING");
278 if (multiband.levels < 1 || multiband.levels > 4)
279 throw std::invalid_argument("DRIZZLE_STORE_MULTIBAND_LEVELS_RANGE");
280 }
281 DrizzleStoreIdentity i;
282 i.source_identity_hash = plan.source_identity_hash;
283 i.normalized_cache_hash = predecessors.normalized_cache_hash;
284 i.quality_plan_hash = predecessors.quality_plan_hash;
285 i.sampling_plan_hash = registration::compute_plan_hash(plan);
286 if (!plan.plan_hash.empty() && plan.plan_hash != i.sampling_plan_hash)
287 throw std::invalid_argument("DRIZZLE_STORE_STALE_SAMPLING_HASH");
288 i.mode = !clipping ? "uniform_unclipped"
289 : mb ? "uniform_raw_multiband_clipped"
290 : "uniform_raw_clipped";
291 i.multiband_levels = mb ? multiband.levels : 0;
292 // Stored geometry: internal, or halved once for the 2/1 area-average
293 // (plan 12.1). 1/1 and 2/2 store at their internal resolution.
294 i.width = osm.needs_2x2_downsample() ? dims.width / 2 : dims.width;
295 i.height = osm.needs_2x2_downsample() ? dims.height / 2 : dims.height;
296 i.color_mode = plan.color_mode;
297 json algorithm = {{"version", 1}, {"mode", i.mode}, {"kernel", cfg.kernel},
298 {"scale", cfg.internal_scale}, {"output_scale", cfg.output_scale},
299 {"pixfrac", cfg.pixfrac},
300 {"position_epsilon", subdivision.position_epsilon_internal_px},
301 {"area_epsilon", subdivision.area_relative_epsilon},
302 {"max_depth", subdivision.max_subdivision_depth},
303 {"frame_error_limit", subdivision.per_frame_inversion_error_rate_max}};
304 if (clipping) {
305 algorithm["clipping"] = {{"min_contributors", cfg.min_clip_contributors},
306 {"passes", cfg.robust_passes}, {"sigma_low", clipping->clip_sigma_low},
307 {"sigma_high", clipping->clip_sigma_high}, {"g_eff", g_eff}};
308 // M5: only present when Raw actually consumed Q_composite, so a store
309 // built without Q-maps keeps its previous reconstruction hash.
310 if (!predecessors.source_quality_cache_hash.empty())
311 algorithm["clipping"]["source_quality_cache_hash"] =
312 predecessors.source_quality_cache_hash;
313 } else if (!g_eff.empty()) throw std::invalid_argument("DRIZZLE_STORE_UNIFORM_HAS_QUALITY_WEIGHTS");
314 // M6: multiband_config_hash content (plan 16.4) --- additive, only present
315 // for a multiband store so uniform/uniform_raw stores keep their hash.
316 if (mb) {
317 const auto &a = multiband.alpha;
318 const auto &g = multiband.guard;
319 const auto &c = multiband.confidence;
320 algorithm["multiband"] = {
321 {"levels", multiband.levels},
322 {"fine_quality_exponent", multiband.fine_quality_exponent},
323 {"medium_quality_exponent", multiband.medium_quality_exponent},
324 {"atrous_den_min", kAtrousDenMinFraction},
325 {"atrous_version", kAtrousDecompositionVersion},
326 {"alpha", {{"alpha_cap", a.alpha_cap},
327 {"min_effective_samples", a.min_effective_samples},
328 {"full_effective_samples", a.full_effective_samples}}},
329 {"guard", {{"energy_limit", g.energy_limit},
330 {"bisection_iters", g.bisection_iters},
331 {"min_window_pixels", g.min_window_pixels}}},
332 {"confidence",
333 {{"min_quality_separation", c.min_quality_separation},
334 {"full_quality_separation", c.full_quality_separation},
335 {"min_artifact_contributors", c.min_artifact_contributors},
336 {"direct_fraction_lo", c.direct_fraction_lo},
337 {"direct_fraction_hi", c.direct_fraction_hi},
338 {"residual_p20_lo", c.residual_p20_lo},
339 {"residual_p20_hi", c.residual_p20_hi},
340 {"artifact_lo", c.artifact_lo},
341 {"artifact_hi", c.artifact_hi}}}};
342 }
343 i.reconstruction_hash = digest(algorithm);
344 plane_names(i);
345 return i;
346}

366DrizzleStoreValidation verify_drizzle_profile_store(
367 const fs::path &root, const DrizzleStoreIdentity &expected) {
368 DrizzleStoreValidation result;
369 try {
370 auto commit = read_small_json(root / "current.json");
371 const auto hash = commit.at("commit_hash").get<std::string>();
372 commit.erase("commit_hash");
373 if (hash != digest(commit)) throw std::runtime_error("DRIZZLE_STORE_COMMIT_HASH_MISMATCH");
374 const auto generation = commit.at("generation").get<std::string>();
375 if (!generation.starts_with("generation-") || generation.size() > 128 ||
376 generation.find_first_not_of("generation-0123456789") != std::string::npos)
377 throw std::runtime_error("DRIZZLE_STORE_INVALID_GENERATION");
378 const fs::path dir = root / generation;
379 if (!fs::is_directory(fs::symlink_status(dir)))
380 throw std::runtime_error("DRIZZLE_STORE_MISSING_GENERATION");
381 auto saved_commit = read_small_json(dir / "commit.json");
382 if (saved_commit.at("commit_hash") != hash)
383 throw std::runtime_error("DRIZZLE_STORE_GENERATION_HASH_MISMATCH");
384 saved_commit.erase("commit_hash");
385 if (saved_commit != commit) throw std::runtime_error("DRIZZLE_STORE_GENERATION_COMMIT_MISMATCH");
386 validate_generation(dir, commit, expected);
387 result.generation_dir = dir;
388 result.usable = true;
389 } catch (const std::exception &e) { result.error = e.what(); }
390 return result;

392ProfilePlane read_drizzle_profile_region_preverified(
393 const fs::path &generation_dir, const DrizzleStoreIdentity &expected,
394 const std::string &profile, const std::string &channel,
395 int x, int y, int width, int height, size_t memory_budget_mb) {
396 if (x < 0 || y < 0 || width <= 0 || height <= 0 ||
397 x > expected.width || y > expected.height ||
398 width > expected.width - x || height > expected.height - y)
399 throw std::invalid_argument("DRIZZLE_STORE_INVALID_REGION");
400 const uint64_t pixels = static_cast<uint64_t>(width) * height;
401 const uint64_t overhead = io_reserve + static_cast<uint64_t>(width) * sizeof(float);
402 if (memory_budget_mb > std::numeric_limits<size_t>::max() / (1024 * 1024) ||
403 pixels > (std::numeric_limits<uint64_t>::max() - overhead) / 13 ||
404 pixels * 13 + overhead > static_cast<uint64_t>(memory_budget_mb) * 1024 * 1024)
405 throw std::runtime_error("DRIZZLE_STORE_REGION_MEMORY_BUDGET");
406 const auto names = plane_names(expected);
407 const std::string base = profile + "_" + channel + "_";
408 if (!std::binary_search(names.begin(), names.end(), base + "value"))
409 throw std::invalid_argument("DRIZZLE_STORE_UNKNOWN_PROFILE_CHANNEL");
410 ProfilePlane result;
411 result.allocate(width, height);
412 auto read = [&](const std::string &field, std::vector<float> *values) {
413 fitsfile *file = nullptr;
414 int status = 0;
415 fits_open_diskfile(&file, (generation_dir / (base + field + ".fits")).string().c_str(),
416 READONLY, &status);
417 check(status);
418 try {
419 std::vector<float> row;
420 if (!values) row.resize(width);
421 for (int dy = 0; dy < height; ++dy) {
422 long first[2] = {static_cast<long>(x) + 1, static_cast<long>(y) + dy + 1};
423 float *dst = values ? values->data() + static_cast<size_t>(dy) * width : row.data();
424 float null_value = std::numeric_limits<float>::quiet_NaN();
425 int any_null = 0;
426 fits_read_pix(file, TFLOAT, first, width, &null_value, dst, &any_null, &status);
427 check(status);
428 if (!values) for (int dx = 0; dx < width; ++dx) {
429 if (dst[dx] != 0.0f && dst[dx] != 1.0f)
430 throw std::runtime_error("DRIZZLE_STORE_INVALID_SUPPORT");
431 result.support[static_cast<size_t>(dy) * width + dx] = dst[dx] == 1.0f;
432 }
433 }
434 int close_status = 0;
435 fits_close_file(file, &close_status);
436 file = nullptr;
437 check(close_status);
438 } catch (...) {
439 if (file) { int ignored = 0; fits_close_file(file, &ignored); }
440 throw;
441 }
442 };
443 read("value", &result.value);
444 // Alpha-confidence pseudo-planes ("alpha_*_X") carry only a value field.
445 const bool alpha_map =
446 channel == "X" &&
447 std::find(kAlphaMapNames.begin(), kAlphaMapNames.end(), profile) !=
448 kAlphaMapNames.end();
449 if (!alpha_map) {
450 read("weight_sum", &result.weight_sum);
451 read("n_eff", &result.n_eff);
452 read("support", nullptr);
453 } else {
454 for (int k = 0; k < width * height; ++k)
455 result.support[k] = std::isfinite(result.value[k]) ? 1u : 0u;
456 }
457 return result;
458}
