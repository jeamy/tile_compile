21bool valid_manifest_shape(const ProfileStoreManifest &m) {
22 if (m.profile.empty() || m.internal_width <= 0 || m.internal_height <= 0 || m.planes.empty())
23 return false;
24 std::set<std::string> names;
25 for (const auto &p : m.planes) {
26 // Logical stems only: no absolute paths, traversal or alternate streams.
27 if (p.name.empty() || p.name.find_first_not_of(
28 "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_-") != std::string::npos ||
29 !names.insert(p.name).second || p.width != m.internal_width ||
30 p.height != m.internal_height)
31 return false;
32 // T1: v1 requires a 64-hex sha256; v2 requires bytes > 0.
33 if (m.schema_version >= 2) {
34 if (p.bytes == 0) return false;
35 } else {
36 if (p.sha256.size() != 64 ||
37 p.sha256.find_first_not_of("0123456789abcdef") != std::string::npos)
38 return false;
39 }
40 }
41 return true;
42}

46std::string compute_profile_store_manifest_hash(const ProfileStoreManifest& m) {
47 ByteSink s;
48 s.str("profile_store_manifest:v1");
49 s.i32(m.schema_version);
50 s.str(m.profile);
51 s.i32(m.internal_width);
52 s.i32(m.internal_height);
53 s.u64(m.planes.size());
54 for (const auto& p : m.planes) {
55 s.str(p.name);
56 // T1: v1 manifests hash the sha256 field; v2 manifests hash the bytes field.
57 if (m.schema_version >= 2)
58 s.u64(p.bytes);
59 else
60 s.str(p.sha256);
61 s.i32(p.width);
62 s.i32(p.height);
63 }
64 return core::sha256_bytes(s.bytes);
65}

79 for (const auto& name : names) {

82 throw std::invalid_argument("PROFILE_STORE_INVALID_FILE");
83 ProfileStoreEntry e;
84 e.name = name;
85 e.width = internal_width;
86 e.height = internal_height;

90 if (!valid_manifest_shape(probe))

92 m.planes.push_back(std::move(e));
93 }

95 throw std::invalid_argument("PROFILE_STORE_INVALID_MANIFEST");
96 m.manifest_hash = compute_profile_store_manifest_hash(m);

124bool parse_profile_store_manifest(const std::string& text, ProfileStoreManifest& out,
125 std::string& error) {
126 try {
127 const json j = json::parse(text);
128 const int sv = j.value("schema_version", -1);
129 if (sv != 1 && sv != 2) {
130 error = "unsupported profile_store_manifest schema_version";
131 return false;
132 }
133 ProfileStoreManifest m;
134 m.schema_version = sv;
135 m.profile = j.at("profile").get<std::string>();
136 m.internal_width = j.at("internal_width").get<int>();
137 m.internal_height = j.at("internal_height").get<int>();
138 m.manifest_hash = j.at("manifest_hash").get<std::string>();
139 if (!j.at("planes").is_array())
140 throw std::invalid_argument("PROFILE_STORE_PLANES_NOT_ARRAY");
141 for (const auto& jp : j.at("planes")) {
142 ProfileStoreEntry e;
143 e.name = jp.at("name").get<std::string>();
144 e.width = jp.at("width").get<int>();
145 e.height = jp.at("height").get<int>();
146 // T1: v1 has sha256, v2 has bytes. Parse whichever is present.
147 if (jp.contains("sha256"))
148 e.sha256 = jp.at("sha256").get<std::string>();
149 if (jp.contains("bytes"))
150 e.bytes = jp.at("bytes").get<std::uintmax_t>();
151 m.planes.push_back(std::move(e));
152 }
153 if (m.manifest_hash != compute_profile_store_manifest_hash(m)) {
154 error = "profile_store_manifest hash mismatch";
155 return false;
156 }
157 if (!valid_manifest_shape(m))
158 throw std::invalid_argument("PROFILE_STORE_INVALID_MANIFEST");
159 error.clear();
160 out = std::move(m);
161 return true;
162 } catch (const std::exception& e) {
163 error = std::string("profile_store_manifest parse error: ") + e.what();
164 return false;
165 }
166}

168ProfileStoreVerifyResult verify_profile_store(const fs::path& dir,
169 const ProfileStoreManifest& manifest) {
170 ProfileStoreVerifyResult r;
171 r.manifest_hash_ok =
172 manifest.manifest_hash == compute_profile_store_manifest_hash(manifest);
173 if (!r.manifest_hash_ok || !valid_manifest_shape(manifest)) return r;
174 for (const auto& p : manifest.planes) {
175 const fs::path f = dir / (p.name + ".fits");
176 std::error_code ec;
177 if (!fs::exists(f, ec)) {
178 r.missing.push_back(p.name);
179 continue;
180 }
181 if (!fs::is_regular_file(fs::symlink_status(f, ec)) || ec) {
182 r.corrupt.push_back(p.name);
183 continue;
184 }
185 // T1: size check only (trusted run). v1 manifests without a bytes field
186 // fall back to existence check only.
187 if (manifest.schema_version >= 2 && p.bytes > 0) {
188 if (fs::file_size(f) != p.bytes) r.corrupt.push_back(p.name);
189 }
190 }
191 r.usable = r.manifest_hash_ok && r.missing.empty() && r.corrupt.empty();
192 return r;
193}
