59float compute_eccentricity(float fwhm_x, float fwhm_y) {
60 if (fwhm_x <= 0.0f || fwhm_y <= 0.0f) return 0.0f;
61 const float mn = std::min(fwhm_x, fwhm_y);
62 const float mx = std::max(fwhm_x, fwhm_y);
63 return 1.0f - mn / mx;
64}

66float measure_clip_fraction_ptr(const float* data, int rows, int cols) {
67 if (data == nullptr || rows <= 0 || cols <= 0) return 0.0f;
68 float min_v = std::numeric_limits<float>::infinity();
69 float max_v = -std::numeric_limits<float>::infinity();
70 size_t finite = 0;
71 const size_t total = static_cast<size_t>(rows) * static_cast<size_t>(cols);
72 for (size_t i = 0; i < total; ++i) {
73 const float v = data[i];
74 if (!std::isfinite(v)) continue;
75 min_v = std::min(min_v, v);
76 max_v = std::max(max_v, v);
77 ++finite;
78 }
79 if (finite == 0) return 0.0f;
80
81 const float hi_threshold = max_v <= 1.25f ? 0.999f : max_v - std::max(1e-6f, std::fabs(max_v) * 1e-5f);
82 const float lo_threshold = min_v >= -0.01f ? 0.0f : min_v + std::max(1e-6f, std::fabs(min_v) * 1e-5f);
83 size_t clipped = 0;
84 for (size_t i = 0; i < total; ++i) {
85 const float v = data[i];
86 if (!std::isfinite(v)) continue;
87 if (v <= lo_threshold || v >= hi_threshold) ++clipped;
88 }
89 return static_cast<float>(clipped) / static_cast<float>(finite);
90}

93float estimate_snr(float gradient_energy, float noise) {
94 if (noise <= 0.0f || !std::isfinite(noise)) return 0.0f;
95 return std::sqrt(std::max(0.0f, gradient_energy)) / noise;

100std::pair<float, float> robust_stats(std::vector<float> values) {
101 if (values.empty()) return {0.0f, 0.0f};
102 std::sort(values.begin(), values.end());
103 const size_t n = values.size();
104 const float med = values[n / 2];
105 std::vector<float> abs_dev;
106 abs_dev.reserve(n);
107 for (float v : values) abs_dev.push_back(std::fabs(v - med));
108 std::sort(abs_dev.begin(), abs_dev.end());
109 const float mad = abs_dev[n / 2];
110 const float sigma = mad * 1.4826f; // consistent estimator
111 return {med, sigma};
112}

118void apply_sigma_rejection_high(const std::vector<float>& values,
119 float k,
120 std::vector<bool>& excluded) {
121 std::vector<float> active;
122 active.reserve(values.size());
123 for (size_t i = 0; i < values.size(); ++i) {
124 if (!excluded[i] && std::isfinite(values[i])) active.push_back(values[i]);
125 }
126 auto [med, sig] = robust_stats(active);
127 if (sig <= 0.0f) return;
128 const float threshold = med + k * sig;
129 for (size_t i = 0; i < values.size(); ++i) {
130 if (!excluded[i] && values[i] > threshold) excluded[i] = true;
131 }
132}

207 const std::string& run_id,
208 const preprocessing::Config& cfg,
209 const PreprocessPipelineContext& pipeline_ctx,
210 const std::vector<metrics::FrameStarMetrics>& star_metrics,
211 const fs::path& run_dir,
212 core::EventEmitter& emitter,
213 std::ostream& log_file,
214 QualityAnalysisContext& out)

217 // Phase: QUALITY_ANALYSIS
218 // -----------------------------------------------------------------------
219 emitter.phase_start(run_id, pname(preprocessing::Phase::QUALITY_ANALYSIS), log_file);
220
221 const size_t n = pipeline_ctx.effective_frames.size();
222 out.records.resize(n);
223
224 std::atomic<size_t> measured{0};
225 std::mutex progress_mutex;
226 parallel_for_indices(n, default_parallel_workers(n), [&](size_t i) {
227 FrameQualityRecord& r = out.records[i];
228 r.index = static_cast<int>(i);
229 r.filename = pipeline_ctx.effective_frames[i].filename().string();
230
231 // --- Star metrics ---
232 if (i < star_metrics.size()) {
233 r.star_count = star_metrics[i].star_count;
234 r.fwhm = star_metrics[i].fwhm;
235 r.fwhm_x = star_metrics[i].fwhm_x;
236 r.fwhm_y = star_metrics[i].fwhm_y;
237 r.eccentricity = compute_eccentricity(r.fwhm_x, r.fwhm_y);
238 }
239
240 // --- Background / noise from FrameMetrics ---
241 if (i < pipeline_ctx.frame_metrics.size()) {
242 const auto& fm = pipeline_ctx.frame_metrics[i];
243 r.background_median = fm.background;
244 r.background_rms = fm.noise;
245 r.snr_estimate = estimate_snr(fm.gradient_energy, fm.noise);
246 r.quality_score = fm.quality_score;
247 if (pipeline_ctx.prewarped_frames.has_data(i)) {
248 r.clip_fraction = measure_clip_fraction_ptr(
249 pipeline_ctx.prewarped_frames.frame_data(i),
250 pipeline_ctx.prewarped_frames.rows(),
251 pipeline_ctx.prewarped_frames.cols());
252 }
253 }
254
255 // --- Registration correlation ---
256 if (i < pipeline_ctx.frame_cc.size()) {
257 r.registration_cc = pipeline_ctx.frame_cc[i];
258 }
259
260 r.included = pipeline_ctx.frame_has_data[i] != 0;
261 if (!r.included) {
262 r.exclusion_reason = FrameExclusionReason::registration_cc_too_low;
263 r.exclusion_detail = "frame not prewarped (registration failed)";
264 }
265
266 const size_t done = measured.fetch_add(1) + 1;
267 std::lock_guard<std::mutex> lock(progress_mutex);
268 emitter.phase_progress(run_id, pname(preprocessing::Phase::QUALITY_ANALYSIS),
269 static_cast<float>(done) / static_cast<float>(n),
270 "measured frame " + std::to_string(done) + "/" +
271 std::to_string(n), log_file);
272 });
273
274 for (const auto& override_item : cfg.manual_frame_overrides) {
275 for (auto& record : out.records) {
276 const bool index_match = override_item.index >= 0 && record.index == override_item.index;
277 const bool name_match = !override_item.filename.empty() &&
278 record.filename == fs::path(override_item.filename).filename().string();
279 if (index_match || name_match) {
280 record.manual_override = true;
281 record.manual_included = override_item.include;
282 }
283 }
284 }
285
286 // Write artifacts now – even before filtering
287 const fs::path artifact_dir = run_dir / "artifacts" / "preprocess";
288 fs::create_directories(artifact_dir);
289 out.csv_path = artifact_dir / "frame_quality.csv";
290 write_frame_quality_csv(out.csv_path, out.records);
291
292 emitter.phase_end(run_id, pname(preprocessing::Phase::QUALITY_ANALYSIS), "ok",
293 {
294 {"n_frames", static_cast<int>(n)},
295 {"csv_path", out.csv_path.string()},
296 }, log_file);
297
298 // -----------------------------------------------------------------------
299 // Phase: FRAME_FILTERING
300 // -----------------------------------------------------------------------
301 emitter.phase_start(run_id, pname(preprocessing::Phase::FRAME_FILTERING), log_file);
302
303 const auto& qf = cfg.quality_filter;
304 const std::string mode = qf.mode; // "auto", "strict", "relaxed", "off"
305
306 std::vector<bool> excluded(n, false);
307 // Propagate frames that failed registration
308 for (size_t i = 0; i < n; ++i) {
309 if (!out.records[i].included) excluded[i] = true;
310 }
311
312 if (mode != "off") {
313 // --- Hard threshold: registration correlation ---
314 for (size_t i = 0; i < n; ++i) {
315 if (!excluded[i] && out.records[i].registration_cc < qf.min_correlation &&
316 out.records[i].registration_cc > 0.0f) {
317 excluded[i] = true;
318 out.records[i].exclusion_reason =
319 FrameExclusionReason::registration_cc_too_low;
320 out.records[i].exclusion_detail =
321 "cc=" + std::to_string(out.records[i].registration_cc) +
322 " < min_correlation=" + std::to_string(qf.min_correlation);
323 }
324 }
325
326 // --- Hard threshold: minimum star count ---
327 for (size_t i = 0; i < n; ++i) {
328 if (!excluded[i] && out.records[i].star_count > 0 &&
329 out.records[i].star_count < qf.min_stars) {
330 excluded[i] = true;
331 out.records[i].exclusion_reason =
332 FrameExclusionReason::star_count_too_low;
333 out.records[i].exclusion_detail =
334 "stars=" + std::to_string(out.records[i].star_count) +
335 " < min_stars=" + std::to_string(qf.min_stars);
336 }
337 }
338
339 // --- Hard threshold: max eccentricity ---
340 for (size_t i = 0; i < n; ++i) {
341 if (!excluded[i] && out.records[i].eccentricity > qf.max_eccentricity) {
342 excluded[i] = true;
343 out.records[i].exclusion_reason =
344 FrameExclusionReason::eccentricity_too_high;
345 out.records[i].exclusion_detail =
346 "eccentricity=" + std::to_string(out.records[i].eccentricity) +
347 " > max_eccentricity=" + std::to_string(qf.max_eccentricity);
348 }
349 }
350
351 // --- Sigma-based FWHM rejection ---
352 if (mode == "auto" || mode == "strict" || mode == "relaxed") {
353 const float k_fwhm = (mode == "strict") ? 1.5f :
354 (mode == "relaxed") ? 3.0f :
355 qf.max_fwhm_sigma; // "auto" uses configured value
356 std::vector<float> fwhm_vals(n);
357 for (size_t i = 0; i < n; ++i) fwhm_vals[i] = out.records[i].fwhm;
358 apply_sigma_rejection_high(fwhm_vals, k_fwhm, excluded);
359 for (size_t i = 0; i < n; ++i) {
360 if (excluded[i] &&
361 out.records[i].exclusion_reason == FrameExclusionReason::not_excluded) {
362 out.records[i].exclusion_reason =
363 FrameExclusionReason::fwhm_too_high;
364 out.records[i].exclusion_detail =
365 "fwhm=" + std::to_string(out.records[i].fwhm) +
366 " sigma-rejected (k=" + std::to_string(k_fwhm) + ")";
367 }
368 }
369 }
370 }
371
372 // Apply manual overrides (overrides auto-rejection in both directions)
373 for (size_t i = 0; i < n; ++i) {
374 if (out.records[i].manual_override) {
375 excluded[i] = !out.records[i].manual_included;
376 if (excluded[i]) {
377 out.records[i].exclusion_reason = FrameExclusionReason::manual_exclude;
378 out.records[i].exclusion_detail = "manually excluded";
379 }
380 }
381 }
382
383 // Build accepted / rejected index lists
384 for (size_t i = 0; i < n; ++i) {
385 out.records[i].included = !excluded[i];
386 if (!excluded[i]) {
387 out.accepted_indices.push_back(static_cast<int>(i));
388 } else {
389 out.rejected_indices.push_back(static_cast<int>(i));
390 }
391 }
392
393 // Rewrite CSV with final filter decision
394 write_frame_quality_csv(out.csv_path, out.records);
395 pipeline_ctx.prewarped_frames.clear_mappings();
396
397 // Write quality_analysis.json
398 const core::json qa_json = build_quality_json(out, qf, mode);
399 core::write_text(artifact_dir / "quality_analysis.json", qa_json.dump(2));
400
401 // Emit warning if many frames rejected
402 if (!out.records.empty()) {
403 const float reject_fraction =
404 static_cast<float>(out.rejected_indices.size()) /
405 static_cast<float>(out.records.size());
406 if (reject_fraction >= 0.5f) {
407 emitter.warning(run_id,
408 "FRAME_FILTERING: " + std::to_string(out.rejected_indices.size()) +
409 "/" + std::to_string(out.records.size()) +
410 " frames rejected (" +
411 std::to_string(static_cast<int>(reject_fraction * 100.0f)) + "%)",
412 log_file);
413 }
414 }
415
416 if (out.accepted_indices.empty()) {
417 emitter.phase_end(run_id, pname(preprocessing::Phase::FRAME_FILTERING), "error",
418 {{"error", "all frames were rejected by quality filter"}},
419 log_file);
420 emitter.run_end(run_id, false, "error", log_file,
421 {{"message", "all frames were rejected by quality filter"}});
422 return false;
423 }
424
425 emitter.phase_end(run_id, pname(preprocessing::Phase::FRAME_FILTERING), "ok",
426 {
427 {"rejection_mode", mode},
428 {"n_total", static_cast<int>(n)},
429 {"n_accepted", static_cast<int>(out.accepted_indices.size())},
430 {"n_rejected", static_cast<int>(out.rejected_indices.size())},
431 {"csv_path", out.csv_path.string()},
432 }, log_file);
433
434 return true;
435}
