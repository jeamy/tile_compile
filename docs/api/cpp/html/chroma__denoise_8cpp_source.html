17void rgb_to_chroma_space(const cv::Mat& R, const cv::Mat& G, const cv::Mat& B,
18 const std::string& color_space,
19 cv::Mat& Y, cv::Mat& C1, cv::Mat& C2) {
20 if (color_space == "opponent_linear") {
21 Y = (R + G + B) / 3.0f;
22 C1 = R - G;
23 C2 = B - G;
24 return;
25 }
26
27 Y = 0.25f * R + 0.5f * G + 0.25f * B;
28 C1 = B - Y;
29 C2 = R - Y;
30}

36void chroma_space_to_rgb(const cv::Mat& Y, const cv::Mat& C1, const cv::Mat& C2,
37 const std::string& color_space,
38 cv::Mat& R, cv::Mat& G, cv::Mat& B) {
39 if (color_space == "opponent_linear") {
40 G = Y - (C1 + C2) / 3.0f;
41 R = G + C1;
42 B = G + C2;
43 return;
44 }
45
46 R = Y + C2;
47 B = Y + C1;
48 G = 2.0f * Y - 0.5f * (R + B);
49}

55cv::Mat build_protection_mask(const cv::Mat& y, const cv::Mat& c1,
56 const cv::Mat& c2, const cv::Mat& valid,
57 const config::ChromaDenoiseConfig& cfg,
58 ChromaDenoiseStats* stats,
59 ChromaDenoiseDiagnostics* diagnostics) {
60 cv::Mat mask = cv::Mat::zeros(y.size(), CV_32F);
61 const double pixels = static_cast<double>(cv::countNonZero(valid));
62
63 if (cfg.star_protection.enabled) {
64 const float sigma = robust_sigma_mad_from_mat(y, valid);
65 const float sky_median = percentile_from_mat(y, 50.0f, valid);
66 const float thr = sky_median + cfg.star_protection.threshold_sigma *
67 (sigma + 1.0e-6f);
68 cv::Mat stars;
69 cv::threshold(y, stars, thr, 1.0, cv::THRESH_BINARY);
70 stars.convertTo(stars, CV_32F);
71 stars.setTo(0.0f, valid == 0);
72 if (cfg.star_protection.dilate_px > 0) {
73 const int k = std::max(1, cfg.star_protection.dilate_px * 2 + 1);
74 cv::Mat ker = cv::getStructuringElement(cv::MORPH_ELLIPSE, cv::Size(k, k));
75 cv::dilate(stars, stars, ker);
76 }
77 if (stats && pixels > 0.0)
78 stats->star_protected_fraction = cv::countNonZero(stars > 0.5f) / pixels;
79 // Feather this component's own boundary, scaled to ITS OWN dilation
80 // radius -- not the combined mask's largest radius. star_protection's
81 // disk is typically much smaller than extended_source_protection's
82 // (a handful of px vs. tens of px for a galaxy disc); feathering it
83 // by the latter's scale would smear the precise star mask far more
84 // than the ring artifact requires, and the reverse -- feathering a
85 // large extended-source disc by only the star mask's small scale --
86 // would leave the extended-source boundary just as sharp as before.
87 {
88 const float star_feather = std::max(
89 1.0f, static_cast<float>(cfg.star_protection.dilate_px) / 3.0f);
90 cv::GaussianBlur(stars, stars, cv::Size(0, 0), star_feather,
91 star_feather, cv::BORDER_REFLECT_101);
92 }
93 if (diagnostics) copy_to_matrix(stars, diagnostics->star_mask);
94 cv::max(mask, stars, mask);
95 }
96
97 if (cfg.structure_protection.enabled) {
98 cv::Mat gx, gy, mag_y, mag_c1, mag_c2, mag;
99 const cv::Mat y_grad = masked_gaussian_blur(y, valid, 0.5);
100 const cv::Mat c1_grad = masked_gaussian_blur(c1, valid, 0.5);
101 const cv::Mat c2_grad = masked_gaussian_blur(c2, valid, 0.5);
102 cv::Sobel(y_grad, gx, CV_32F, 1, 0, 3);
103 cv::Sobel(y_grad, gy, CV_32F, 0, 1, 3);
104 cv::magnitude(gx, gy, mag_y);
105 cv::Sobel(c1_grad, gx, CV_32F, 1, 0, 3);
106 cv::Sobel(c1_grad, gy, CV_32F, 0, 1, 3);
107 cv::magnitude(gx, gy, mag_c1);
108 cv::Sobel(c2_grad, gx, CV_32F, 1, 0, 3);
109 cv::Sobel(c2_grad, gy, CV_32F, 0, 1, 3);
110 cv::magnitude(gx, gy, mag_c2);
111 mag = cv::max(mag_y, cv::max(mag_c1, mag_c2));
112 const float med = percentile_from_mat(mag, 50.0f, valid);
113 cv::Mat deviation = cv::abs(mag - med);
114 const float sig = robust_sigma_mad_from_mat(deviation, valid);
115 const float p = std::max(
116 percentile_from_mat(mag, cfg.structure_protection.gradient_percentile,
117 valid),
118 med + 3.0f * sig);
119 cv::Mat structures;
120 cv::threshold(mag, structures, p, 1.0, cv::THRESH_BINARY);
121 structures.convertTo(structures, CV_32F);
122 structures.setTo(0.0f, valid == 0);
123 if (stats && pixels > 0.0)
124 stats->structure_protected_fraction =
125 cv::countNonZero(structures > 0.5f) / pixels;
126 // Own-scale feather, same reasoning as star_protection/
127 // extended_source_protection above. This component has no
128 // dilate_px to derive a radius from (a raw gradient threshold, not
129 // a disk), so a fixed sigma matched to the other two components'
130 // typical scale is used instead. Without this, structure_
131 // protection was the one piece of the combined mask left as a hard
132 // per-pixel binary threshold, relying only on the final 1px
133 // anti-aliasing pass below -- for a percentile threshold that
134 // catches scattered single/few-pixel gradient spikes (real edges
135 // and noise both), that leaves sharp, per-pixel denoise/no-denoise
136 // transitions exactly where this whole feathering fix was meant to
137 // remove them.
138 cv::GaussianBlur(structures, structures, cv::Size(0, 0), 2.0f, 2.0f,
139 cv::BORDER_REFLECT_101);
140 if (diagnostics)
141 copy_to_matrix(structures, diagnostics->structure_mask);
142 cv::max(mask, structures, mask);
143 }
144
145 if (cfg.extended_source_protection.enabled) {
146 // Heavily blur Y to suppress point sources; only extended smooth emission
147 // survives at this scale (sigma ≈ 2 % of image short axis).
148 const float blur_sigma = std::max(5.0f, static_cast<float>(
149 std::min(y.rows, y.cols)) * 0.02f);
150 cv::Mat y_smooth;
151 y_smooth = masked_gaussian_blur(y, valid, blur_sigma);
152
153 // Estimate sky stats from the lowest-brightness half of the smoothed luma.
154 const float sky_med = percentile_from_mat(y_smooth, 50.0f, valid);
155 std::vector<float> sky_vals;
156 sky_vals.reserve(static_cast<size_t>(y_smooth.total()));
157 for (int row = 0; row < y_smooth.rows; ++row) {
158 const float* ptr = y_smooth.ptr<float>(row);
159 for (int col = 0; col < y_smooth.cols; ++col)
160 if (valid.at<std::uint8_t>(row, col) != 0 &&
161 std::isfinite(ptr[col])) sky_vals.push_back(ptr[col]);
162 }
163 const float sky_sigma = tile_compile::core::robust_sigma_mad(sky_vals);
164 const float thr = sky_med + cfg.extended_source_protection.luma_sigma * sky_sigma;
165
166 cv::Mat ext_src;
167 cv::threshold(y_smooth, ext_src, static_cast<double>(thr), 1.0, cv::THRESH_BINARY);
168 ext_src.convertTo(ext_src, CV_32F);
169 ext_src.setTo(0.0f, valid == 0);
170 // A large Gaussian turns bright stars into smooth islands as well; a
171 // threshold alone therefore cannot distinguish a point source from an
172 // extended target. Keep only connected regions whose area is large at
173 // the detection scale. Compact sources remain covered by the separate
174 // star mask instead of inflating this mask into broad stellar discs.
175 cv::Mat labels, cc_stats, centroids;
176 cv::Mat ext_u8;
177 ext_src.convertTo(ext_u8, CV_8U, 255.0);
178 const int n_labels = cv::connectedComponentsWithStats(
179 ext_u8, labels, cc_stats, centroids, 8, CV_32S);
180 cv::Mat extended_only = cv::Mat::zeros(ext_src.size(), CV_32F);
181 const int min_component_area = std::max(
182 64, static_cast<int>(std::ceil(4.0 * CV_PI * blur_sigma * blur_sigma)));
183 // Keep table + one pass over the label image instead of one full-frame
184 // setTo(labels == label) per component (O(labels x N)).
185 std::vector<std::uint8_t> keep_label(static_cast<std::size_t>(n_labels), 0u);
186 for (int label = 1; label < n_labels; ++label) {
187 if (cc_stats.at<int>(label, cv::CC_STAT_AREA) >= min_component_area)
188 keep_label[static_cast<std::size_t>(label)] = 1u;
189 }
190 for (int y = 0; y < labels.rows; ++y) {
191 const int* lrow = labels.ptr<int>(y);
192 float* orow = extended_only.ptr<float>(y);
193 for (int x = 0; x < labels.cols; ++x)
194 if (keep_label[static_cast<std::size_t>(lrow[x])]) orow[x] = 1.0f;
195 }
196 ext_src = extended_only;
197 if (stats) {
198 stats->extended_source_sky_median = sky_med;
199 stats->extended_source_sky_sigma = sky_sigma;
200 stats->extended_source_threshold = thr;
201 if (pixels > 0.0)
202 stats->extended_source_raw_fraction =
203 cv::countNonZero(ext_src > 0.5f) / pixels;
204 }
205 if (cfg.extended_source_protection.dilate_px > 0) {
206 const int k = std::max(1, cfg.extended_source_protection.dilate_px * 2 + 1);
207 cv::Mat ker = cv::getStructuringElement(cv::MORPH_ELLIPSE, cv::Size(k, k));
208 cv::dilate(ext_src, ext_src, ker);
209 }
210 if (stats && pixels > 0.0)
211 stats->extended_source_protected_fraction =
212 cv::countNonZero(ext_src > 0.5f) / pixels;
213 // Own-scale feather -- see the star_protection block above for why
214 // this must not share a radius with the other components.
215 {
216 const float ext_feather = std::max(
217 1.0f,
218 static_cast<float>(cfg.extended_source_protection.dilate_px) /
219 3.0f);
220 cv::GaussianBlur(ext_src, ext_src, cv::Size(0, 0), ext_feather,
221 ext_feather, cv::BORDER_REFLECT_101);
222 }
223 if (diagnostics)
224 copy_to_matrix(ext_src, diagnostics->extended_source_mask);
225 cv::max(mask, ext_src, mask);
226 }
227
228 // Final small anti-aliasing pass on the combined mask (star_protection
229 // and extended_source_protection are already feathered at their own
230 // scale above; this just smooths the max() seams between components).
231 cv::GaussianBlur(mask, mask, cv::Size(0, 0), 1.0, 1.0, cv::BORDER_REFLECT_101);
232 cv::min(mask, 1.0, mask);
233 cv::max(mask, 0.0, mask);
234 mask.setTo(0.0f, valid == 0);
235 if (stats && pixels > 0.0) {
236 stats->combined_protected_fraction =
237 cv::countNonZero(mask > 0.5f) / pixels;
238 stats->mean_protection = cv::mean(mask, valid)[0];
239 }
240 if (diagnostics) copy_to_matrix(mask, diagnostics->combined_mask);
241 return mask;
242}

275double flatten_large_scale_chroma_bias(
276 cv::Mat& c, const cv::Mat& protect_mask, const cv::Mat& valid_mask,
277 const config::ChromaDenoiseConfig::LargeScaleBiasConfig& cfg,
278 double* holes_filled_fraction) {
279 if (holes_filled_fraction) *holes_filled_fraction = 0.0;
280 if (!cfg.enabled || c.empty()) return 0.0;
281
282 const int bs = std::max(4, cfg.block_size);
283 const int gy = std::max(1, (c.rows + bs - 1) / bs);
284 const int gx = std::max(1, (c.cols + bs - 1) / bs);
285 cv::Mat grid(gy, gx, CV_32F, cv::Scalar(0.0f));
286 cv::Mat grid_valid(gy, gx, CV_32F, cv::Scalar(0.0f));
287
288 std::vector<float> block_vals;
289 for (int by = 0; by < gy; ++by) {
290 const int r0 = by * bs, r1 = std::min(c.rows, r0 + bs);
291 for (int bx = 0; bx < gx; ++bx) {
292 const int c0 = bx * bs, c1 = std::min(c.cols, c0 + bs);
293 block_vals.clear();
294 int valid_background = 0;
295 int valid_total = 0;
296 for (int r = r0; r < r1; ++r) {
297 const float* crow = c.ptr<float>(r);
298 const float* mrow =
299 protect_mask.empty() ? nullptr : protect_mask.ptr<float>(r);
300 const std::uint8_t* vrow = valid_mask.ptr<std::uint8_t>(r);
301 for (int cc = c0; cc < c1; ++cc) {
302 if (vrow[cc] == 0 || !std::isfinite(crow[cc])) continue;
303 ++valid_total;
304 if (mrow && mrow[cc] > 0.1f) continue;
305 block_vals.push_back(crow[cc]);
306 ++valid_background;
307 }
308 }
309 if (valid_total == 0 || valid_background < 16 ||
310 valid_background * 2 < valid_total) continue;
311 std::nth_element(block_vals.begin(),
312 block_vals.begin() + block_vals.size() / 2,
313 block_vals.end());
314 grid.at<float>(by, bx) = block_vals[block_vals.size() / 2];
315 grid_valid.at<float>(by, bx) = 1.0f;
316 }
317 }
318
319 const int n_cells = gy * gx;
320 const int n_valid = cv::countNonZero(grid_valid);
321 const int required_valid = std::min(
322 n_cells, std::max(4, static_cast<int>(std::ceil(0.25 * n_cells))));
323 if (n_valid < required_valid) return 0.0;
324 if (holes_filled_fraction)
325 *holes_filled_fraction =
326 static_cast<double>(n_cells - n_valid) / static_cast<double>(n_cells);
327
328 // Push-pull fill: propagate valid block values into holes via a growing
329 // normalized-convolution box filter, so every cell ends up covered.
330 if (n_valid < n_cells) {
331 cv::Mat value = grid.mul(grid_valid);
332 cv::Mat weight = grid_valid.clone();
333 int ksize = 3;
334 const int kmax = 2 * std::max(gy, gx) + 1;
335 while (cv::countNonZero(weight) < n_cells && ksize <= kmax) {
336 cv::Mat blurred_value, blurred_weight;
337 cv::boxFilter(value, blurred_value, -1, cv::Size(ksize, ksize),
338 cv::Point(-1, -1), false, cv::BORDER_REPLICATE);
339 cv::boxFilter(weight, blurred_weight, -1, cv::Size(ksize, ksize),
340 cv::Point(-1, -1), false, cv::BORDER_REPLICATE);
341 cv::Mat still_empty;
342 cv::compare(weight, 0.5f, still_empty, cv::CMP_LT);
343 cv::Mat reached;
344 cv::compare(blurred_weight, 1.0e-6f, reached, cv::CMP_GT);
345 cv::Mat newly;
346 cv::bitwise_and(still_empty, reached, newly);
347 cv::Mat filled_value;
348 cv::divide(blurred_value, blurred_weight, filled_value);
349 filled_value.copyTo(value, newly);
350 cv::Mat ones(weight.size(), CV_32F, cv::Scalar(1.0f));
351 ones.copyTo(weight, newly);
352 ksize += 2;
353 }
354 grid = value;
355 }
356
357 // Reference level: median of the ORIGINALLY-valid block medians only --
358 // the plane's overall background chroma level, preserved unchanged.
359 std::vector<float> gv;
360 gv.reserve(static_cast<size_t>(n_valid));
361 for (int by = 0; by < gy; ++by)
362 for (int bx = 0; bx < gx; ++bx)
363 if (grid_valid.at<float>(by, bx) > 0.5f)
364 gv.push_back(grid.at<float>(by, bx));
365 std::nth_element(gv.begin(), gv.begin() + gv.size() / 2, gv.end());
366 const float ref = gv[gv.size() / 2];
367
368 cv::Mat surface;
369 cv::resize(grid, surface, c.size(), 0, 0, cv::INTER_LINEAR);
370 if (cfg.blur_sigma > 0.0f)
371 cv::GaussianBlur(surface, surface, cv::Size(0, 0), cfg.blur_sigma,
372 cfg.blur_sigma, cv::BORDER_REFLECT_101);
373
374 cv::Mat correction = (surface - ref) * cfg.strength;
375 correction.setTo(0.0f, valid_mask == 0);
376 cv::Scalar mean, stddev;
377 cv::meanStdDev(correction, mean, stddev, valid_mask);
378 c -= correction;
379 return stddev[0];
380}

386void denoise_chroma_plane_inplace(cv::Mat& c,
387 const config::ChromaDenoiseConfig& cfg,
388 const cv::Mat& valid,
389 const cv::Mat& protect) {
390 if (cfg.chroma_wavelet.enabled) {
391 cv::Mat approximation = c.clone();
392 std::vector<cv::Mat> details;
393 details.reserve(static_cast<size_t>(std::max(1, cfg.chroma_wavelet.levels)));
394 const int levels = std::max(1, cfg.chroma_wavelet.levels);
395 for (int lvl = 0; lvl < levels; ++lvl) {
396 const double sigma = std::pow(2.0, static_cast<double>(lvl)) * 0.75;
397 cv::Mat low;
398 low = masked_gaussian_blur(approximation, valid, sigma);
399 details.push_back(approximation - low);
400 approximation = low;
401 }
402 cv::Mat reconstructed = approximation;
403 cv::Mat noise_mask = valid.clone();
404 if (!protect.empty()) noise_mask.setTo(0, protect > 0.1f);
405 for (cv::Mat& detail : details) {
406 const float sigma_n = robust_sigma_mad_from_mat(detail, noise_mask);
407 const float tau = cfg.chroma_wavelet.threshold_scale *
408 sigma_n;
409 cv::Mat abs_detail = cv::abs(detail);
410 cv::Mat ratio;
411 cv::divide(tau, abs_detail + 1.0e-12f, ratio);
412 cv::pow(ratio, cfg.chroma_wavelet.soft_k, ratio);
413 cv::Mat gain = 1.0f - ratio;
414 cv::max(gain, 0.0f, gain);
415 reconstructed += detail.mul(gain);
416 }
417 c = reconstructed;
418 }
419
420 if (cfg.chroma_bilateral.enabled) {
421 cv::Mat noise_mask = valid.clone();
422 if (!protect.empty()) noise_mask.setTo(0, protect > 0.1f);
423 const float sigma_noise = robust_sigma_mad_from_mat(c, noise_mask);
424 cv::Mat bilateral_input = c.clone();
425 cv::Mat boundary_fill = masked_gaussian_blur(
426 c, valid, std::max(0.5f, cfg.chroma_bilateral.sigma_spatial));
427 boundary_fill.copyTo(bilateral_input, valid == 0);
428 cv::Mat out;
429 cv::bilateralFilter(bilateral_input, out, 0,
430 std::max(1.0e-6f,
431 cfg.chroma_bilateral.sigma_range * sigma_noise),
432 cfg.chroma_bilateral.sigma_spatial,
433 cv::BORDER_REFLECT_101);
434 c.copyTo(out, valid == 0);
435 c = out;
436 }
437}

440ChromaDenoiseStats chroma_denoise_rgb_inplace(
441 Matrix2Df& r, Matrix2Df& g, Matrix2Df& b,
442 const config::ChromaDenoiseConfig& cfg,
443 Matrix2Df* protection_mask_out,
444 const std::vector<std::uint8_t>* valid_mask,
445 ChromaDenoiseDiagnostics* diagnostics) {
446 ChromaDenoiseStats stats;
447 if (!cfg.enabled) return stats;
448 if (r.size() <= 0 || g.size() <= 0 || b.size() <= 0) return stats;
449 if (r.rows() != g.rows() || r.cols() != g.cols() ||
450 r.rows() != b.rows() || r.cols() != b.cols()) {
451 return stats;
452 }
453
454 cv::Mat R(r.rows(), r.cols(), CV_32F, r.data());
455 cv::Mat G(g.rows(), g.cols(), CV_32F, g.data());
456 cv::Mat B(b.rows(), b.cols(), CV_32F, b.data());
457 const cv::Mat valid = build_valid_mask(R, G, B, valid_mask);
458 const auto valid_pixels = static_cast<std::uint64_t>(cv::countNonZero(valid));
459 if (valid_pixels == 0) return stats;
460
461 if (cfg.blend.mode != "chroma_only") return stats;
462
463 cv::Mat Y, C1, C2;
464 rgb_to_chroma_space(R, G, B, cfg.color_space, Y, C1, C2);
465
466 // Dataset-aware adaptation: scale denoise strength from measured chroma
467 // noise, expressed RELATIVE to the image's own luma-noise sigma (not a
468 // fixed absolute constant -- the old ref_sigma=0.02f was calibrated for
469 // [0,1]-normalized data and saturated the clamp on real, unnormalized
470 // ADU-scale data, where absolute chroma sigma is routinely in the tens).
471 // This keeps fine detail on clean data and increases suppression on noisy data.
472 config::ChromaDenoiseConfig tuned = cfg;
473 const float sigma_c1 = high_frequency_sigma(C1, valid);
474 const float sigma_c2 = high_frequency_sigma(C2, valid);
475 const float chroma_sigma = 0.5f * (sigma_c1 + sigma_c2);
476 const float luma_sigma = std::max(1.0e-6f, high_frequency_sigma(Y, valid));
477 const float ref_ratio = std::max(1.0e-6f, cfg.adaptation_reference_ratio);
478 const float adapt =
479 std::clamp((chroma_sigma / luma_sigma) / ref_ratio, 0.8f, 1.4f);
480 tuned.blend.amount = std::clamp(cfg.blend.amount * adapt, 0.0f, 1.0f);
481 tuned.chroma_wavelet.threshold_scale =
482 std::max(0.1f, cfg.chroma_wavelet.threshold_scale * adapt);
483 tuned.chroma_bilateral.sigma_range =
484 std::max(1.0e-4f, cfg.chroma_bilateral.sigma_range * std::sqrt(adapt));
485 stats.applied = true;
486 stats.valid_pixels = valid_pixels;
487 stats.input_chroma_sigma = chroma_sigma;
488 stats.input_luma_sigma = luma_sigma;
489 stats.adaptation = adapt;
490 stats.effective_blend_amount = tuned.blend.amount;
491
492 // Protection mask, computed once up front: it depends only on
493 // star_protection/structure_protection/extended_source_protection/
494 // luma_guard_strength, none of which the adaptation above touches, and
495 // is reused both for the large-scale bias estimate below (background-
496 // only pixels) and for the final blend-back amount_map.
497 cv::Mat protect;
498 if (tuned.protect_luma)
499 protect = build_protection_mask(Y, C1, C2, valid, tuned, &stats,
500 diagnostics);
501 if (protection_mask_out != nullptr) {
502 protection_mask_out->resize(r.rows(), r.cols());
503 if (!protect.empty()) {
504 cv::Mat out_view(r.rows(), r.cols(), CV_32F, protection_mask_out->data());
505 protect.copyTo(out_view);
506 } else {
507 protection_mask_out->setZero();
508 }
509 }
510
511 cv::Mat C1_orig = C1.clone();
512 cv::Mat C2_orig = C2.clone();
513
514 if (tuned.large_scale_bias.enabled) {
515 double holes_c1 = 0.0, holes_c2 = 0.0;
516 stats.large_scale_bias_removed_rms_c1 = flatten_large_scale_chroma_bias(
517 C1, protect, valid, tuned.large_scale_bias, &holes_c1);
518 stats.large_scale_bias_removed_rms_c2 = flatten_large_scale_chroma_bias(
519 C2, protect, valid, tuned.large_scale_bias, &holes_c2);
520 stats.large_scale_bias_grid_holes_filled_fraction =
521 std::max(holes_c1, holes_c2);
522 }
523
524 if (tuned.large_scale_bias.enabled) {
525 cv::Mat bias_amount(Y.size(), CV_32F,
526 cv::Scalar(tuned.blend.amount));
527 if (!protect.empty()) bias_amount = bias_amount.mul(1.0f - protect);
528 C1 = C1_orig.mul(1.0f - bias_amount) + C1.mul(bias_amount);
529 C2 = C2_orig.mul(1.0f - bias_amount) + C2.mul(bias_amount);
530 }
531
532 cv::Mat C1_bias = C1.clone();
533 cv::Mat C2_bias = C2.clone();
534
535 denoise_chroma_plane_inplace(C1, tuned, valid, protect);
536 denoise_chroma_plane_inplace(C2, tuned, valid, protect);
537
538 cv::Mat amount_map(Y.size(), CV_32F, cv::Scalar(tuned.blend.amount));
539 if (tuned.protect_luma) {
540 amount_map = amount_map.mul(1.0f - tuned.luma_guard_strength * protect);
541 cv::min(amount_map, tuned.blend.amount, amount_map);
542 cv::max(amount_map, 0.0, amount_map);
543 }
544 stats.mean_denoise_fraction = cv::mean(amount_map, valid)[0];
545 if (diagnostics) copy_to_matrix(amount_map, diagnostics->effective_amount);
546
547 cv::Mat one_minus = 1.0f - amount_map;
548 cv::Mat C1_mix = C1_bias.mul(one_minus) + C1.mul(amount_map);
549 cv::Mat C2_mix = C2_bias.mul(one_minus) + C2.mul(amount_map);
550 C1_orig.copyTo(C1_mix, valid == 0);
551 C2_orig.copyTo(C2_mix, valid == 0);
552
553 cv::Mat R_new, G_new, B_new;
554 chroma_space_to_rgb(Y, C1_mix, C2_mix, cfg.color_space, R_new, G_new, B_new);
555
556 R_new.copyTo(R);
557 G_new.copyTo(G);
558 B_new.copyTo(B);
559 return stats;
560}
