3// Forward-Drizzle-v2 Gate-7: versioned artifact schema, checkpoint
4// identities and band-boundary resume with atomic commit. Implements the
5// frozen contract in
6// docs/forward_drizzle_v2_gate7_transaction_resume_spec_2026-09-12.json.
7//
8// Layout under a caller-chosen store root:
9//
10// <root>/forward_drizzle_v2_generation-<unique>/
11// plan.json binding run plan (written first, immutable)
12// band-%04d.bin self-describing band artifact (64 B header +
13// record_count * 64 B ForwardDrizzleV2PixelResult)
14// band-%04d.profiles.bin profile payload (Gate-9/10, only when
15// plan.emit_profiles): same header shape,
16// record_count * 80 B ForwardDrizzleV2ProfileResult
17// checkpoint.json committed band prefix, rewritten atomically
18// after every band commit
19// commit.json final marker, after full re-verification
20// <root>/current.json pointer; written atomically as the LAST step
21//
22// Resume granularity (frozen decision): atomically committed band
23// boundaries only. A partially written band is discarded and recomputed;
24// reservoir/accumulator state is transient and never persisted. A phase
25// event alone never establishes resumability --- only checkpoint.json
26// content does. Unpublished generations are owned and removed by their
27// writer; after a publication attempt they are never deleted.
28//
29// Not wired into the production runner.
30

177// Band writer. begin() creates the unpublished generation and writes
178// plan.json plus an empty checkpoint. commit_band() appends exactly the
179// next expected band index in a contiguous y order: payload staged,
180// fsynced, renamed, hashed, THEN checkpoint.json is updated atomically
181// (the commit mark is always the last durable step). finish() re-verifies
182// every artifact, evaluates the gate, writes commit.json and publishes
183// current.json. The destructor removes the generation iff publish was
184// never attempted.
185class ForwardDrizzleV2StoreWriter {
186 public:
187 ForwardDrizzleV2StoreWriter(fs::path root, ForwardDrizzleV2RunPlan plan);
188 ~ForwardDrizzleV2StoreWriter();
189 ForwardDrizzleV2StoreWriter(const ForwardDrizzleV2StoreWriter &) = delete;
190 ForwardDrizzleV2StoreWriter &operator=(
191 const ForwardDrizzleV2StoreWriter &) = delete;
192
193 // Creates (fresh) or adopts (resumed) the unpublished generation and
194 // persists plan.json + checkpoint.json. Must be called before
195 // commit_band().
196 void begin();
197 // Adopts the generation named by a resumable inspect state instead of
198 // creating a new one; `resume_from_band` must equal the verified
199 // committed prefix length.
200 void adopt(const fs::path &generation, int resume_from_band,
201 std::vector<ForwardDrizzleV2BandCommit> committed);
202
203 // Appends band `band_index` covering native rows [y_begin, y_begin+rows).
204 // `results` must hold rows*native_cols*channels records (channel-major).
205 // When plan_.emit_profiles is set, `profiles` must hold the same count of
206 // ForwardDrizzleV2ProfileResult records; it is committed inside the same
207 // band transaction (durable before the checkpoint mark). When the plan
208 // has no emit_profiles, `profiles` must be empty.
209 // Enforces the contiguous prefix and the plan's band geometry.
210 void commit_band(int band_index, int y_begin, int rows,
211 std::span<const ForwardDrizzleV2PixelResult> results,
212 std::span<const ForwardDrizzleV2ProfileResult> profiles,
213 std::uint64_t dense_overlap_count);
214
215 // Verifies completeness, evaluates `gate`, writes commit.json and
216 // publishes current.json. Throws on any violation; a thrown finish()
217 // leaves the generation unpublished (destructor cleans it up).
218 fs::path finish(const ForwardDrizzleV2CommitGate &gate);
219
220 const ForwardDrizzleV2RunPlan &plan() const { return plan_; }
221 const fs::path &generation() const { return generation_; }
222 int committed_bands() const { return static_cast<int>(bands_.size()); }
223 bool published() const { return publish_attempted_; }
224
225 private:
226 fs::path root_;
227 ForwardDrizzleV2RunPlan plan_;
228 fs::path generation_;
229 std::vector<ForwardDrizzleV2BandCommit> bands_;
230 int next_y_ = 0;
231 bool begun_ = false;
232 bool publish_attempted_ = false;
233};

236 fresh, // no generation: caller starts a new run

240};
