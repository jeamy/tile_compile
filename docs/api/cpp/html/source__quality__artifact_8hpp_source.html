59// Plan 11.13(2): the three plan-15 candidate images at FULL per-channel
60// resolution are streamed stripe-wise into a caller-owned scratch directory
61// instead of held whole in RAM. Peak resident candidate data is then one fusion
62// stripe (nch * chunk * W floats) during the fuse pass and one plane during
63// delivery read-back --- not 3*nch whole planes. `nch` is 1 (MONO -> "L") or 3
64// (OSC -> R,G,B). Each spooled plane is host-native `float`, row-major
65// width*height, NaN off support. The runner reads back only the delivered
66// candidates (Raw baseline + selected + optionally the `full`-diagnostics
67// controls), one plane at a time, straight to FITS.

85// Plan 11.13(1)+(4): the shared, overflow-safe host working-set estimate for the
86// MULTIBAND fuse / validation / export phase, computed BEFORE any large
87// allocation. Every term is an upper bound on data that is concurrently
88// resident. `fuse_multiband_store_to_image` fails closed
89// (throws "MULTIBAND_MEMORY_BUDGET") when `estimated_peak_bytes > budget_bytes`,
90// before it touches the final image, the candidate buffers or the spool ---
91// nothing already committed is disturbed.
92struct MultibandFusionMemoryPlan {
93 std::size_t final_image_bytes = 0; // nch * N * 4 (held whole)
94 std::size_t stripe_working_bytes = 0; // U/R/F/M + fuse + alpha + luma scratch
95 std::size_t candidate_luma_bytes = 0; // with_candidate_luma: 3 luma + support + alpha
96 std::size_t spool_stripe_bytes = 0; // with_candidate_spool: 3 * nch * chunk * W * 4
97 std::size_t delivery_readback_bytes = 0; // one plane read back for FITS export
98 std::size_t margin_bytes = 0; // max(64 MiB, 5% of the raw sum)
99 std::size_t estimated_peak_bytes = 0; // RAM sum of the above
100 std::size_t budget_bytes = 0;
101 bool fits = false; // estimated_peak_bytes <= budget_bytes
102 // Plan 11.13(2) + 11.11: bytes the candidate spool writes to the temp
103 // filesystem (3 candidates * nch * N * 4), and the free space that must be
104 // available for it: estimated_temp_peak * 1.20 + max(2 GiB, 5% of capacity).
105 // `available_temp_bytes` / `temp_space_ok` are filled by
106 // fuse_multiband_store_to_image once the spool directory is known (0 / true
107 // when no spool is requested).
108 std::size_t spool_temp_bytes = 0;
109 std::size_t required_free_temp_bytes = 0;
110 std::size_t available_temp_bytes = 0;
111 bool temp_space_ok = true;
112};
