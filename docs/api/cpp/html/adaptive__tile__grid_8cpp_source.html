51// Compute coverage fraction for a single tile against the canvas mask.
56float tile_coverage(const Tile& t,
57 const std::vector<uint8_t>& mask,
58 int canvas_width, int canvas_height) {
59 const int x0 = std::max(0, t.x);
60 const int y0 = std::max(0, t.y);
61 const int x1 = std::min(canvas_width, t.x + t.width);
62 const int y1 = std::min(canvas_height, t.y + t.height);
63 if (x1 <= x0 || y1 <= y0) return 0.0f;
64
65 const int total = (x1 - x0) * (y1 - y0);
66 int covered = 0;
67 for (int y = y0; y < y1; ++y) {
68 const uint8_t* row = mask.data() + y * canvas_width;
69 for (int x = x0; x < x1; ++x)
70 if (row[x]) ++covered;
71 }
72 return static_cast<float>(covered) / static_cast<float>(total);
73}
