14// cfa_channel_for_source_pixel's result depends only on (sx & 1, sy & 1) for
15// a fixed (bayer, cfa_origin_x, cfa_origin_y) --- the origin only shifts
16// parity, never the period. Building this 2x2 lookup once (redundant-reload
17// analysis C4) instead of re-deriving it from the Bayer-offset switch every
18// pixel is exact, not an approximation: it enumerates the same 4 inputs the
19// per-pixel calls would ever see.
20std::array<std::array<CfaChannel, 2>, 2> build_cfa_channel_2x2_lut(
21 BayerPattern bayer, int cfa_origin_x, int cfa_origin_y) {
22 std::array<std::array<CfaChannel, 2>, 2> lut{};
23 for (int py = 0; py < 2; ++py)
24 for (int px = 0; px < 2; ++px)
25 lut[py][px] = cfa_channel_for_source_pixel(px, py, bayer, cfa_origin_x,
26 cfa_origin_y);
27 return lut;
28}

30// Separable [1,4,6,4,1]/16 pass along one axis, clamp-to-edge boundary.
31void b3_pass_rows(const Matrix2Df &in, Matrix2Df &out) {
32 const int H = static_cast<int>(in.rows());
33 const int W = static_cast<int>(in.cols());
34 out.resize(H, W);
35 static const double k[5] = {1.0, 4.0, 6.0, 4.0, 1.0};
36 for (int y = 0; y < H; ++y) {
37 for (int x = 0; x < W; ++x) {
38 double sum = 0.0;
39 for (int t = -2; t <= 2; ++t) {
40 const int xx = std::clamp(x + t, 0, W - 1);
41 sum += k[t + 2] * static_cast<double>(in(y, xx));
42 }
43 out(y, x) = static_cast<float>(sum / 16.0);
44 }
45 }
46}

74double median_absolute_deviation_sigma(const Matrix2Df &values) {
75 std::vector<float> finite;
76 finite.reserve(static_cast<size_t>(values.rows()) * static_cast<size_t>(values.cols()));
77 for (int y = 0; y < values.rows(); ++y)
78 for (int x = 0; x < values.cols(); ++x) {
79 const float v = values(y, x);
80 if (std::isfinite(v)) finite.push_back(v);
81 }
82 if (finite.empty()) return 0.0;
83
84 const double med = tile_compile::core::median_of_or_nan_inplace(finite);
85 for (float &v : finite) v = static_cast<float>(std::abs(v - med));
86 const double mad = tile_compile::core::median_of_or_nan_inplace(finite);
87 return tile_compile::core::kMadToSigma * mad;
88}

92// Edge-aware full-resolution green proxy at one R/B site (plan 13.2 step 3):
93// prefers the neighbour-pair direction (horizontal or vertical) with the
94// lower local gradient, so the proxy never interpolates across an edge.
95float edge_aware_green_at(const Matrix2Df &source, int x, int y, int width, int height) {
96 const bool have_w = x > 0, have_e = x < width - 1;
97 const bool have_n = y > 0, have_s = y < height - 1;
98 const double west = have_w ? static_cast<double>(source(y, x - 1)) : 0.0;
99 const double east = have_e ? static_cast<double>(source(y, x + 1)) : 0.0;
100 const double north = have_n ? static_cast<double>(source(y - 1, x)) : 0.0;
101 const double south = have_s ? static_cast<double>(source(y + 1, x)) : 0.0;
102
103 const bool horiz_ok = have_w && have_e;
104 const bool vert_ok = have_n && have_s;
105 const bool horiz_avail = have_w || have_e;
106 const bool vert_avail = have_n || have_s;
107
108 const double horiz_val =
109 horiz_ok ? 0.5 * (west + east) : (have_w ? west : (have_e ? east : 0.0));
110 const double vert_val =
111 vert_ok ? 0.5 * (north + south) : (have_n ? north : (have_s ? south : 0.0));
112 const double horiz_grad = horiz_ok ? std::abs(west - east) : std::numeric_limits<double>::infinity();
113 const double vert_grad = vert_ok ? std::abs(north - south) : std::numeric_limits<double>::infinity();
114
115 if (!horiz_avail && !vert_avail) return source(y, x); // 1x1 degenerate case
116 if (!vert_avail) return static_cast<float>(horiz_val);
117 if (!horiz_avail) return static_cast<float>(vert_val);
118 return static_cast<float>(horiz_grad <= vert_grad ? horiz_val : vert_val);
119}

123SourceQualityProxyResult compute_source_quality_proxy_v1(const Matrix2Df &source,
124 ColorMode color_mode,
125 BayerPattern bayer_pattern,
126 int cfa_origin_x, int cfa_origin_y) {
127 SourceQualityProxyResult result;
128 const int H = static_cast<int>(source.rows());
129 const int W = static_cast<int>(source.cols());
130
131 if (color_mode == ColorMode::MONO) {
132 // Plan 13.2: "MONO verwendet direkt die normalisierte L-Ebene als Proxy
133 // ... ohne CFA-Interpolation."
134 result.proxy_full = source;
135 result.sigma_green = median_absolute_deviation_sigma(source - b3_spline_blur(source));
136 return result;
137 }
138
139 // --- Quad-Green grid (plan 13.2 step 1) ---
140 const auto cfa_lut = build_cfa_channel_2x2_lut(bayer_pattern, cfa_origin_x, cfa_origin_y);
141 const int qw = W / 2, qh = H / 2;
142 result.quad_width = qw;
143 result.quad_height = qh;
144 result.quad_green.resize(qh, qw);
145 for (int qy = 0; qy < qh; ++qy) {
146 for (int qx = 0; qx < qw; ++qx) {
147 const int sx = qx * 2, sy = qy * 2;
148 double green_sum = 0.0;
149 for (int dy = 0; dy < 2; ++dy) {
150 for (int dx = 0; dx < 2; ++dx) {
151 // sx is always even, so (sx+dx)&1 == dx (and likewise sy/dy):
152 // the lookup key reduces to (dy, dx) directly.
153 if (cfa_lut[dy][dx] == CfaChannel::G) {
154 green_sum += source(sy + dy, sx + dx);
155 }
156 }
157 }
158 result.quad_green(qy, qx) = static_cast<float>(green_sum * 0.5); // 0.5*(G1+G2)
159 }
160 }
161
162 // --- global sigma_green (plan 13.2 step 2), on the Quad-Green grid ---
163 if (qw > 0 && qh > 0) {
164 result.sigma_green =
165 median_absolute_deviation_sigma(result.quad_green - b3_spline_blur(result.quad_green));
166 }
167
168 // --- full-resolution edge-aware green proxy (plan 13.2 step 3) ---
169 result.proxy_full.resize(H, W);
170 for (int y = 0; y < H; ++y) {
171 for (int x = 0; x < W; ++x) {
172 const CfaChannel c = cfa_lut[y & 1][x & 1];
173 result.proxy_full(y, x) =
174 (c == CfaChannel::G) ? source(y, x) : edge_aware_green_at(source, x, y, W, H);
175 }
176 }
177
178 return result;
179}
