23BGEValueStats stats_from_values_local(const std::vector<float>& values) {
24 BGEValueStats stats;
25 std::vector<float> finite;
26 finite.reserve(values.size());
27 double sum = 0.0;
28 for (float v : values) {
29 if (!std::isfinite(v))
30 continue;
31 finite.push_back(v);
32 sum += static_cast<double>(v);
33 }
34 stats.n = static_cast<int>(finite.size());
35 if (finite.empty())
36 return stats;
37 // Selection instead of a full sort: the same min/max/median values as the
38 // sorted form (the even-size lower median is the max of the lower half).
39 const auto mm = std::minmax_element(finite.begin(), finite.end());
40 stats.min = *mm.first;
41 stats.max = *mm.second;
42 const size_t mid = finite.size() / 2;
43 std::nth_element(finite.begin(), finite.begin() + static_cast<std::ptrdiff_t>(mid),
44 finite.end());
45 stats.median = (finite.size() % 2 != 0)
46 ? finite[mid]
47 : 0.5f * (*std::max_element(finite.begin(),
48 finite.begin() + static_cast<std::ptrdiff_t>(mid)) +
49 finite[mid]);
50 stats.mean = static_cast<float>(sum / static_cast<double>(finite.size()));
51 double var = 0.0;
52 for (float v : finite) {
53 const double d = static_cast<double>(v) - stats.mean;
54 var += d * d;
55 }
56 stats.std = static_cast<float>(std::sqrt(var / static_cast<double>(finite.size())));
57 return stats;
58}

98float patch_estimate(const Matrix2Df& img, int cx, int cy, int patch_size,
99 const std::string& estimator) {
100 // Grow-only per-thread scratch: the gradient descent calls this thousands
101 // of times per seed; no heap traffic per call once warmed up. The in-place
102 // medians permute these buffers, which is fine because every pass below
103 // only needs the multiset of values.
104 thread_local std::vector<float> vals, dev, clipped;
105 vals.clear();
106 const int half = patch_size / 2;
107 const int r0 = std::max(0, cy - half);
108 const int r1 = std::min(static_cast<int>(img.rows()) - 1, cy + half);
109 const int c0 = std::max(0, cx - half);
110 const int c1 = std::min(static_cast<int>(img.cols()) - 1, cx + half);
111 for (int r = r0; r <= r1; ++r)
112 for (int c = c0; c <= c1; ++c) {
113 const float v = img(r, c);
114 if (std::isfinite(v) && v > 0.0f) vals.push_back(v);
115 }
116 if (vals.empty())
117 return 0.0f;
118 if (estimator != "sigma_clipped_median")
119 return tile_compile::core::median_of_or_nan_inplace(vals);
120
121 const float med = tile_compile::core::median_of_or_nan_inplace(vals);
122 dev.clear();
123 for (float v : vals)
124 dev.push_back(std::fabs(v - med));
125 const float mad = tile_compile::core::median_of_or_nan_inplace(dev);
126 const float sigma = tile_compile::core::kMadToSigma * mad;
127 if (!(sigma > kTiny))
128 return med;
129
130 clipped.clear();
131 const float lo = med - 3.0f * sigma;
132 const float hi = med + 3.0f * sigma;
133 for (float v : vals) {
134 if (v >= lo && v <= hi)
135 clipped.push_back(v);
136 }
137 return clipped.empty() ? med
138 : tile_compile::core::median_of_or_nan_inplace(clipped);
139}

141float patch_sigma(const Matrix2Df& img, int cx, int cy, int patch_size) {
142 std::vector<float> vals = collect_patch_values(img, cx, cy, patch_size);
143 if (vals.size() < 2) return 0.0f;
144 const float med = tile_compile::core::median_of(vals);
145 double sum_sq = 0.0;
146 for (float v : vals) {
147 const double d = static_cast<double>(v) - med;
148 sum_sq += d * d;
149 }
150 return static_cast<float>(std::sqrt(sum_sq / static_cast<double>(vals.size())));
151}

153// Reject sample points whose local patch variance is high relative to the
154// global distribution of patch sigmas. Points on signal (nebula, stars)
155// have significantly higher local variance than pure background points.
156std::vector<SamplePoint> filter_sample_points_by_variance(
157 const Matrix2Df& image_downsampled,
158 const std::vector<SamplePoint>& points,
159 int patch_size,
160 float sigma_clip_factor) {
161 if (points.size() < 8) return points;
162 std::vector<float> sigmas;
163 sigmas.reserve(points.size());
164 for (const auto& p : points)
165 sigmas.push_back(patch_sigma(image_downsampled, p.x, p.y, patch_size));
166 const float med_sigma = tile_compile::core::median_of(sigmas);
167 std::vector<float> dev;
168 dev.reserve(sigmas.size());
169 for (float s : sigmas) dev.push_back(std::fabs(s - med_sigma));
170 const float mad = tile_compile::core::median_of(std::move(dev));
171 const float sigma_mad = tile_compile::core::kMadToSigma * mad;
172 if (!(sigma_mad > kTiny)) return points;
173 const float threshold = med_sigma + sigma_clip_factor * sigma_mad;
174 std::vector<SamplePoint> filtered;
175 filtered.reserve(points.size());
176 for (size_t i = 0; i < points.size(); ++i) {
177 if (sigmas[i] <= threshold)
178 filtered.push_back(points[i]);
179 }
180 return filtered;
181}

220Matrix2Df transform_to_autobge_working_space(
221 const Matrix2Df& channel, const BGEConfig::AutoBGEConfig& config,
222 StretchParams* params, int channel_index,
223 const std::vector<uint8_t>* valid_mask) {
224
225 const int rows = static_cast<int>(channel.rows());
226 const int cols = static_cast<int>(channel.cols());
227
228 if (config.stretch_mode == "none") {
229 if (params) {
230 if (static_cast<int>(params->original_mins.size()) <= channel_index) {
231 params->original_mins.resize(channel_index + 1, 0.0f);
232 params->original_medians.resize(channel_index + 1, 0.0f);
233 params->linear_offsets.resize(channel_index + 1, 0.0f);
234 params->linear_scales.resize(channel_index + 1, 1.0f);
235 params->mtf_targets.resize(channel_index + 1, 0.25f);
236 params->mtf_scales.resize(channel_index + 1, 1.0f);
237 }
238 params->mode = "none";
239 }
240 return channel;
241 }
242
243 // Collect valid, positive, finite pixels for percentile computation
244 std::vector<float> valid_vals;
245 valid_vals.reserve(rows * cols);
246 for (int r = 0; r < rows; ++r)
247 for (int c = 0; c < cols; ++c) {
248 const float v = channel(r, c);
249 if (std::isfinite(v) && v > 0.0f) {
250 if (!valid_mask || (*valid_mask)[r * cols + c] != 0)
251 valid_vals.push_back(v);
252 }
253 }
254
255 if (valid_vals.size() < 16) {
256 // Fallback to none for this channel
257 if (params) {
258 if (static_cast<int>(params->original_mins.size()) <= channel_index) {
259 params->original_mins.resize(channel_index + 1, 0.0f);
260 params->original_medians.resize(channel_index + 1, 0.0f);
261 params->linear_offsets.resize(channel_index + 1, 0.0f);
262 params->linear_scales.resize(channel_index + 1, 1.0f);
263 params->mtf_targets.resize(channel_index + 1, 0.25f);
264 params->mtf_scales.resize(channel_index + 1, 1.0f);
265 }
266 params->mode = "none";
267 }
268 return channel;
269 }
270
271 // Three single quantiles by selection instead of a full sort (same element
272 // percentile_from_sorted picks: index round(p * (n - 1))). The shifted
273 // median is max(0, v - vmin) applied to the median element: that map is
274 // monotone non-decreasing, so it commutes with the rank selection exactly.
275 auto select_quantile = [&](float p) -> float {
276 // Exactly percentile_from_sorted's index expression (pct in percent).
277 const float clamped = std::min(std::max(p * 100.0f, 0.0f), 100.0f);
278 const float pos = (clamped / 100.0f) * static_cast<float>(valid_vals.size() - 1);
279 size_t idx = static_cast<size_t>(std::round(pos));
280 idx = std::min(idx, valid_vals.size() - 1);
281 std::nth_element(valid_vals.begin(),
282 valid_vals.begin() + static_cast<std::ptrdiff_t>(idx),
283 valid_vals.end());
284 return valid_vals[idx];
285 };
286 const float vmin = *std::min_element(valid_vals.begin(), valid_vals.end());
287 const float p01 = select_quantile(0.01f);
288 const float p99 = select_quantile(0.99f);
289 const float median = select_quantile(0.50f);
290 const float median_shifted = std::max(0.0f, median - vmin);
291
292 if (params) {
293 if (static_cast<int>(params->original_mins.size()) <= channel_index) {
294 params->original_mins.resize(channel_index + 1, 0.0f);
295 params->original_medians.resize(channel_index + 1, 0.0f);
296 params->linear_offsets.resize(channel_index + 1, 0.0f);
297 params->linear_scales.resize(channel_index + 1, 1.0f);
298 params->mtf_targets.resize(channel_index + 1, 0.25f);
299 params->mtf_scales.resize(channel_index + 1, 1.0f);
300 }
301 params->original_mins[channel_index] = vmin;
302 params->original_medians[channel_index] = median;
303 }
304
305 if (config.stretch_mode == "linear") {
306 const float scale = std::max(p99 - p01, kTiny);
307 if (params) {
308 params->linear_offsets[channel_index] = p01;
309 params->linear_scales[channel_index] = scale;
310 params->mode = "linear";
311 }
312 Matrix2Df out = (channel.array() - p01) / scale;
313 return out;
314 }
315
316 // mtf: AutoBGE-style unlinked non-linear stretch on channel-min-shifted data.
317 // The formula is designed for normalized inputs (values in [0,1]).
318 // Normalize by the 99th percentile range to keep the pole outside the data.
319 const float target = config.stretch_target_median;
320 const float scale = std::max(p99 - vmin, kTiny);
321 const float mtf_median = median_shifted > 0.0f ? median_shifted / scale : 1.0f;
322 if (params) {
323 params->original_mins[channel_index] = vmin;
324 params->original_medians[channel_index] = mtf_median; // normalized median
325 params->linear_offsets[channel_index] = 0.0f;
326 params->linear_scales[channel_index] = 1.0f;
327 params->mtf_targets[channel_index] = target;
328 params->mtf_scales[channel_index] = scale;
329 params->mode = "mtf";
330 }
331
332 Matrix2Df out(rows, cols);
333 const float eps = 1.0e-10f;
334 for (int r = 0; r < rows; ++r)
335 for (int c = 0; c < cols; ++c) {
336 const float v = channel(r, c);
337 if (!std::isfinite(v) || v <= 0.0f) {
338 out(r, c) = 0.0f;
339 continue;
340 }
341 const float shifted = std::max(0.0f, v - vmin) / scale;
342 const float numerator = (mtf_median - 1.0f) * target * shifted;
343 const float denominator =
344 mtf_median * (target + shifted - 1.0f) - target * shifted;
345 out(r, c) = numerator / (denominator + eps);
346 }
347 return out;
348}

350Matrix2Df transform_from_autobge_working_space(
351 const Matrix2Df& channel, const StretchParams& params, int channel_index) {
352
353 if (params.mode == "none" || channel_index >= static_cast<int>(params.linear_scales.size()))
354 return channel;
355
356 if (params.mode == "linear") {
357 const float offset = params.linear_offsets[channel_index];
358 const float scale = params.linear_scales[channel_index];
359 return channel.array() * scale + offset;
360 }
361
362 // mtf inverse for:
363 // y = ((m - 1) * t * x) / (m * (t + x - 1) - t * x)
364 // where x = shifted / scale (normalized) and m is the normalized median.
365 // Solve for x, then scale back: shifted = x * scale, original = shifted + min.
366 const int rows = static_cast<int>(channel.rows());
367 const int cols = static_cast<int>(channel.cols());
368 const float mtf_median = params.original_medians[channel_index];
369 const float orig_min = params.original_mins[channel_index];
370 const float target =
371 channel_index < static_cast<int>(params.mtf_targets.size())
372 ? params.mtf_targets[channel_index]
373 : 0.25f;
374 const float scale =
375 channel_index < static_cast<int>(params.mtf_scales.size())
376 ? params.mtf_scales[channel_index]
377 : 1.0f;
378
379 Matrix2Df out(rows, cols);
380 const float eps = 1.0e-10f;
381 for (int r = 0; r < rows; ++r)
382 for (int c = 0; c < cols; ++c) {
383 const float corrected = channel(r, c);
384 if (!std::isfinite(corrected)) {
385 out(r, c) = corrected;
386 continue;
387 }
388 const float numerator = corrected * mtf_median * (target - 1.0f);
389 const float denominator =
390 (mtf_median - 1.0f) * target -
391 corrected * (mtf_median - target);
392 const float x = numerator / (denominator + eps);
393 out(r, c) = x * scale + orig_min;
394 }
395 return out;
396}

398Matrix2Df downsample_area(const Matrix2Df& image, int scale) {
399 if (scale <= 1) return image;
400 const int src_rows = static_cast<int>(image.rows());
401 const int src_cols = static_cast<int>(image.cols());
402 const int dst_rows = src_rows / scale;
403 const int dst_cols = src_cols / scale;
404 if (dst_rows < 1 || dst_cols < 1) return image;
405
406 Matrix2Df out(dst_rows, dst_cols);
407 const float inv_area = 1.0f / static_cast<float>(scale * scale);
408 for (int dr = 0; dr < dst_rows; ++dr) {
409 for (int dc = 0; dc < dst_cols; ++dc) {
410 float sum = 0.0f;
411 int count = 0;
412 for (int sy = 0; sy < scale; ++sy) {
413 const int sr = dr * scale + sy;
414 if (sr >= src_rows) break;
415 for (int sx = 0; sx < scale; ++sx) {
416 const int sc = dc * scale + sx;
417 if (sc >= src_cols) break;
418 const float v = image(sr, sc);
419 if (std::isfinite(v)) {
420 sum += v;
421 ++count;
422 }
423 }
424 }
425 out(dr, dc) = count > 0 ? sum * inv_area : 0.0f;
426 }
427 }
428 return out;
429}

431Matrix2Df upscale_lanczos4(const Matrix2Df& background, int target_rows, int target_cols) {
432 const int src_rows = static_cast<int>(background.rows());
433 const int src_cols = static_cast<int>(background.cols());
434 if (src_rows == target_rows && src_cols == target_cols) return background;
435
436 Matrix2Df out(target_rows, target_cols);
437 const float scale_r = static_cast<float>(src_rows) / target_rows;
438 const float scale_c = static_cast<float>(src_cols) / target_cols;
439 constexpr int a = 4; // Lanczos kernel size
440
441 auto lanczos_kernel = [](float x, int n) -> float {
442 if (x == 0.0f) return 1.0f;
443 if (std::abs(x) >= n) return 0.0f;
444 const float px = static_cast<float>(M_PI) * x;
445 const float npx = px / n;
446 return std::sin(px) * std::sin(npx) / (px * npx);
447 };
448
449 // The kernel is separable: the column taps (c0, c1, wc[]) depend only on
450 // tc and the row taps only on tr. Tabulate both once (2*a taps each) and
451 // keep the per-pixel accumulation loop, order and arithmetic unchanged, so
452 // the result is identical to the direct per-tap evaluation.
453 constexpr int taps = 2 * a;
454 struct AxisTaps {
455 int i0 = 0, i1 = -1;
456 float w[taps] = {};
457 };
458 auto tabulate = [&](int target_n, int src_n, float scale,
459 std::vector<AxisTaps>& table) {
460 table.resize(static_cast<size_t>(target_n));
461 for (int t = 0; t < target_n; ++t) {
462 const float src = (t + 0.5f) * scale - 0.5f;
463 AxisTaps& at = table[static_cast<size_t>(t)];
464 at.i0 = std::max(0, static_cast<int>(std::floor(src)) - a + 1);
465 at.i1 = std::min(src_n - 1, static_cast<int>(std::floor(src)) + a);
466 for (int i = at.i0; i <= at.i1; ++i)
467 at.w[i - at.i0] = lanczos_kernel(src - i, a);
468 }
469 };
470 std::vector<AxisTaps> row_taps, col_taps;
471 tabulate(target_rows, src_rows, scale_r, row_taps);
472 tabulate(target_cols, src_cols, scale_c, col_taps);
473
474#pragma omp parallel for schedule(static)
475 for (int tr = 0; tr < target_rows; ++tr) {
476 const AxisTaps& rt = row_taps[static_cast<size_t>(tr)];
477 for (int tc = 0; tc < target_cols; ++tc) {
478 const AxisTaps& ct = col_taps[static_cast<size_t>(tc)];
479 float weight_sum = 0.0f;
480 float value_sum = 0.0f;
481 for (int sr = rt.i0; sr <= rt.i1; ++sr) {
482 const float wr = rt.w[sr - rt.i0];
483 if (wr == 0.0f) continue;
484 for (int sc = ct.i0; sc <= ct.i1; ++sc) {
485 const float wc = ct.w[sc - ct.i0];
486 if (wc == 0.0f) continue;
487 const float w = wr * wc;
488 value_sum += w * background(sr, sc);
489 weight_sum += w;
490 }
491 }
492 out(tr, tc) = weight_sum > kTiny ? value_sum / weight_sum : 0.0f;
493 }
494 }
495 return out;
496}

498// Generate AutoBGE sample points: border + grid + gradient-descent refined points
499std::vector<SamplePoint> generate_autobge_sample_points(
500 const Matrix2Df& image_downsampled,
501 const BGEConfig::AutoBGEConfig& config,
502 const std::vector<uint8_t>* valid_mask_downsampled,
503 std::mt19937* rng,
504 bool random_downselection) {
505
506 const int rows = static_cast<int>(image_downsampled.rows());
507 const int cols = static_cast<int>(image_downsampled.cols());
508 if (rows < 4 || cols < 4) return {};
509
510 std::mt19937 local_rng;
511 if (rng) {
512 local_rng = *rng;
513 } else {
514 local_rng.seed(42);
515 }
516
517 // Compute bright exclusion threshold
518 std::vector<float> valid_vals;
519 valid_vals.reserve(rows * cols);
520 for (int r = 0; r < rows; ++r)
521 for (int c = 0; c < cols; ++c) {
522 const float v = image_downsampled(r, c);
523 if (std::isfinite(v) && v > 0.0f) {
524 if (!valid_mask_downsampled || (*valid_mask_downsampled)[r * cols + c] != 0)
525 valid_vals.push_back(v);
526 }
527 }
528 if (valid_vals.empty()) return {};
529
530 // bright_exclusion_fraction is the fraction at the bright end to reject.
531 // Example: 0.30 keeps the darkest 70%, matching AutoBGE.py semantics.
532 const float keep_fraction = 1.0f - config.bright_exclusion_fraction;
533 const int bright_idx = static_cast<int>(
534 static_cast<float>(valid_vals.size()) * keep_fraction);
535 const int bright_sel = std::min(bright_idx,
536 static_cast<int>(valid_vals.size()) - 1);
537 std::nth_element(valid_vals.begin(), valid_vals.begin() + bright_sel,
538 valid_vals.end());
539 const float bright_threshold = valid_vals[bright_sel];
540
541 // Helper: check if a point is valid (not bright, not masked)
542 auto is_valid_point = [&](int x, int y) -> bool {
543 if (x < 0 || x >= cols || y < 0 || y >= rows) return false;
544 if (valid_mask_downsampled && (*valid_mask_downsampled)[y * cols + x] == 0)
545 return false;
546 const float v = image_downsampled(y, x);
547 if (!std::isfinite(v) || v <= 0.0f) return false;
548 if (v >= bright_threshold) return false;
549 return true;
550 };
551
552 // Determine number of points before refinement so the descent radius can be
553 // tied to the nominal sample spacing.
554 int target_points = config.num_sample_points;
555 if (target_points <= 0) {
556 target_points = std::clamp((rows * cols) / 800, 200, 3000);
557 }
558 const float nominal_spacing = std::sqrt(
559 static_cast<float>(rows * cols) / std::max(1, target_points));
560 const int max_descent_distance =
561 std::max(2, static_cast<int>(std::floor(0.45f * nominal_spacing)));
562
563 // Gradient descent to a nearby dim spot. Restricting displacement to less
564 // than half the nominal spacing prevents neighboring grid seeds from all
565 // collapsing into the same dark local minimum.
566 auto gradient_descent_to_dim = [&](int& x, int& y) -> void {
567 const int max_iters = config.gradient_descent_max_iters;
568 const int origin_x = x;
569 const int origin_y = y;
570 constexpr int step = 1;
571 for (int iter = 0; iter < max_iters; ++iter) {
572 float current_val = patch_estimate(image_downsampled, x, y,
573 config.patch_size,
574 config.patch_estimator);
575 bool moved = false;
576 // Check the immediate 8-neighborhood, as in the Python reference.
577 float best_val = current_val;
578 int best_x = x, best_y = y;
579 for (int dy = -step; dy <= step; dy += step) {
580 for (int dx = -step; dx <= step; dx += step) {
581 if (dx == 0 && dy == 0) continue;
582 int nx = x + dx, ny = y + dy;
583 if (std::abs(nx - origin_x) > max_descent_distance ||
584 std::abs(ny - origin_y) > max_descent_distance)
585 continue;
586 if (!is_valid_point(nx, ny)) continue;
587 float nv = patch_estimate(image_downsampled, nx, ny,
588 config.patch_size,
589 config.patch_estimator);
590 if (nv < best_val) {
591 best_val = nv;
592 best_x = nx;
593 best_y = ny;
594 }
595 }
596 }
597 if (best_x != x || best_y != y) {
598 x = best_x;
599 y = best_y;
600 moved = true;
601 }
602 if (!moved) break;
603 }
604 };
605
606 std::vector<SamplePoint> candidates;
607
608 const int margin = config.border_margin;
609 const int usable_w = cols - 2 * margin;
610 const int usable_h = rows - 2 * margin;
611 if (usable_w < 4 || usable_h < 4) return {};
612
613 auto add_candidate = [&](int x, int y) {
614 x = std::clamp(x, margin, cols - margin - 1);
615 y = std::clamp(y, margin, rows - margin - 1);
616 if (!is_valid_point(x, y))
617 return;
618 gradient_descent_to_dim(x, y);
619 if (is_valid_point(x, y))
620 candidates.push_back({x, y});
621 };
622
623 // Explicit corners and edge anchors stabilize large-scale planes.
624 add_candidate(margin, margin);
625 add_candidate(cols - margin - 1, margin);
626 add_candidate(margin, rows - margin - 1);
627 add_candidate(cols - margin - 1, rows - margin - 1);
628 const int edge_steps = std::max(2, static_cast<int>(std::sqrt(target_points)));
629 for (int i = 1; i < edge_steps; ++i) {
630 const int x = margin + i * usable_w / edge_steps;
631 const int y = margin + i * usable_h / edge_steps;
632 add_candidate(x, margin);
633 add_candidate(x, rows - margin - 1);
634 add_candidate(margin, y);
635 add_candidate(cols - margin - 1, y);
636 }
637
638 // Generate grid points
639 const int grid_nx = static_cast<int>(std::sqrt(static_cast<float>(target_points)
640 * usable_w / usable_h));
641 const int grid_ny = static_cast<int>(std::sqrt(static_cast<float>(target_points)
642 * usable_h / usable_w));
643 const int actual_nx = std::max(2, grid_nx);
644 const int actual_ny = std::max(2, grid_ny);
645
646 for (int gy = 0; gy < actual_ny; ++gy) {
647 for (int gx = 0; gx < actual_nx; ++gx) {
648 int x = margin + static_cast<int>(static_cast<float>(gx + 0.5f) * usable_w / actual_nx);
649 int y = margin + static_cast<int>(static_cast<float>(gy + 0.5f) * usable_h / actual_ny);
650 if (!is_valid_point(x, y)) {
651 // Try to find nearest valid point
652 bool found = false;
653 for (int radius = 1; radius < std::max(rows, cols) && !found; ++radius) {
654 for (int dy = -radius; dy <= radius && !found; ++dy) {
655 for (int dx = -radius; dx <= radius && !found; ++dx) {
656 if (std::abs(dx) != radius && std::abs(dy) != radius) continue;
657 int nx = x + dx, ny = y + dy;
658 if (is_valid_point(nx, ny)) {
659 x = nx; y = ny; found = true;
660 }
661 }
662 }
663 }
664 if (!found) continue;
665 }
666 // Apply gradient descent to find dim spot
667 gradient_descent_to_dim(x, y);
668 candidates.push_back({x, y});
669 }
670 }
671
672 // Convert normalized user-provided sample points to the downsampled working
673 // space used by this function. They are always kept and bypass the random
674 // downselection so users can force the model to honour specific background
675 // locations.
676 std::vector<SamplePoint> user_points;
677 user_points.reserve(config.user_sample_points.size());
678 for (const auto& up : config.user_sample_points) {
679 if (up[0] < 0.0f || up[0] > 1.0f || up[1] < 0.0f || up[1] > 1.0f) continue;
680 const int px = static_cast<int>(std::floor(up[0] * (cols - 1) + 0.5f));
681 const int py = static_cast<int>(std::floor(up[1] * (rows - 1) + 0.5f));
682 if (px < 0 || px >= cols || py < 0 || py >= rows) continue;
683 if (valid_mask_downsampled && (*valid_mask_downsampled)[py * cols + px] == 0) continue;
684 user_points.push_back({px, py});
685 }
686
687 // Remove duplicate points (gradient descent may converge multiple to same spot)
688 std::sort(candidates.begin(), candidates.end(), [](const SamplePoint& a, const SamplePoint& b) {
689 return a.y * 100000 + a.x < b.y * 100000 + b.x;
690 });
691 candidates.erase(std::unique(candidates.begin(), candidates.end(), [](const SamplePoint& a, const SamplePoint& b) {
692 return a.x == b.x && a.y == b.y;
693 }), candidates.end());
694
695 // Helper to test whether a point is already in the selected set.
696 auto already_selected = [&](const SamplePoint& p, const std::vector<SamplePoint>& selected) -> bool {
697 return std::any_of(selected.begin(), selected.end(),
698 [&](const SamplePoint& e) { return e.x == p.x && e.y == p.y; });
699 };
700
701 if (!random_downselection) {
702 for (const auto& up : user_points) {
703 if (!already_selected(up, candidates)) candidates.push_back(up);
704 }
705 return candidates;
706 }
707
708 const int user_count = static_cast<int>(user_points.size());
709 const int auto_target = std::max(0, target_points - user_count);
710
711 std::array<std::vector<SamplePoint>, 4> quartiles;
712 for (const auto& p : candidates) {
713 if (already_selected(p, user_points)) continue;
714 const int qx = p.x < cols / 2 ? 0 : 1;
715 const int qy = p.y < rows / 2 ? 0 : 1;
716 quartiles[static_cast<size_t>(qy * 2 + qx)].push_back(p);
717 }
718
719 std::vector<SamplePoint> points;
720 points.reserve(static_cast<size_t>(user_count + auto_target));
721 for (const auto& up : user_points) points.push_back(up);
722
723 // Reject near-duplicates, not only exact duplicates. A modest fraction of
724 // nominal spacing preserves coverage while still allowing dense sampling.
725 const float min_distance = std::max(1.0f, nominal_spacing * 0.35f);
726 const float min_distance_sq = min_distance * min_distance;
727 auto far_enough = [&](const SamplePoint& p) {
728 for (size_t i = static_cast<size_t>(user_count); i < points.size(); ++i) {
729 const float dx = static_cast<float>(p.x - points[i].x);
730 const float dy = static_cast<float>(p.y - points[i].y);
731 if (dx * dx + dy * dy < min_distance_sq) return false;
732 }
733 return true;
734 };
735
736 for (auto& q : quartiles) {
737 std::shuffle(q.begin(), q.end(), local_rng);
738 const int quota = std::max(1, auto_target / 4);
739 int accepted = 0;
740 for (const auto& p : q) {
741 if (!far_enough(p)) continue;
742 points.push_back(p);
743 if (++accepted >= quota) break;
744 }
745 }
746 if (static_cast<int>(points.size()) < target_points) {
747 std::shuffle(candidates.begin(), candidates.end(), local_rng);
748 for (const auto& p : candidates) {
749 if (!already_selected(p, points) && far_enough(p)) points.push_back(p);
750 if (static_cast<int>(points.size()) >= target_points) break;
751 }
752 }
753
754 return points;
755}

757Matrix2Df fit_polynomial_autobge(
758 const Matrix2Df& image_downsampled,
759 const std::vector<SamplePoint>& points,
760 const BGEConfig::AutoBGEConfig& config,
761 int target_rows, int target_cols) {
762
763 const int rows = static_cast<int>(image_downsampled.rows());
764 const int cols = static_cast<int>(image_downsampled.cols());
765 if (points.empty() || rows < 2 || cols < 2) {
766 return Matrix2Df::Zero(target_rows, target_cols);
767 }
768
769 const int degree = config.poly_degree;
770 // Count polynomial terms: for 2D polynomial of degree d:
771 // number of terms = (d+1)*(d+2)/2
772 const int n_terms = autobge_polynomial_term_count(degree);
773 const int n_points = static_cast<int>(points.size());
774
775 if (n_points < n_terms + 4) {
776 return Matrix2Df::Zero(target_rows, target_cols);
777 }
778
779 // Build design matrix and observation vector
780 Eigen::MatrixXf A(n_points, n_terms);
781 Eigen::VectorXf b(n_points);
782
783 // Normalize coordinates to [-1, 1]
784 const float norm_x = 2.0f / (cols - 1);
785 const float norm_y = 2.0f / (rows - 1);
786
787 for (int i = 0; i < n_points; ++i) {
788 const float x = static_cast<float>(points[i].x) * norm_x - 1.0f;
789 const float y = static_cast<float>(points[i].y) * norm_y - 1.0f;
790 const float val = patch_estimate(image_downsampled, points[i].x, points[i].y,
791 config.patch_size,
792 config.patch_estimator);
793 b(i) = val;
794 int col = 0;
795 for (int dy = 0; dy <= degree; ++dy) {
796 for (int dx = 0; dx <= degree - dy; ++dx) {
797 A(i, col) = std::pow(x, dx) * std::pow(y, dy);
798 ++col;
799 }
800 }
801 }
802
803 // Solve least squares
804 Eigen::VectorXf coeffs = A.colPivHouseholderQr().solve(b);
805
806 // Render at downsampled resolution, then upscale
807 Matrix2Df bg_down(rows, cols);
808 for (int r = 0; r < rows; ++r) {
809 const float y = static_cast<float>(r) * norm_y - 1.0f;
810 for (int c = 0; c < cols; ++c) {
811 const float x = static_cast<float>(c) * norm_x - 1.0f;
812 float val = 0.0f;
813 int idx = 0;
814 for (int dy = 0; dy <= degree; ++dy) {
815 for (int dx = 0; dx <= degree - dy; ++dx) {
816 val += coeffs(idx) * std::pow(x, dx) * std::pow(y, dy);
817 ++idx;
818 }
819 }
820 bg_down(r, c) = val;
821 }
822 }
823
824 return upscale_lanczos4(bg_down, target_rows, target_cols);
825}
826
827Matrix2Df fit_rbf_autobge(
828 const Matrix2Df& image_downsampled,
829 const std::vector<SamplePoint>& points,
830 const BGEConfig::AutoBGEConfig& config,
831 int target_rows, int target_cols) {
832
833 const int rows = static_cast<int>(image_downsampled.rows());
834 const int cols = static_cast<int>(image_downsampled.cols());
835 const int n = static_cast<int>(points.size());
836 if (n < kAutoBGEMinStage2Points || rows < 2 || cols < 2) {
837 return Matrix2Df::Zero(target_rows, target_cols);
838 }
839
840 // Compute sample values at each point
841 Eigen::VectorXf f(n);
842 for (int i = 0; i < n; ++i) {
843 f(i) = patch_estimate(image_downsampled, points[i].x, points[i].y,
844 config.patch_size, config.patch_estimator);
845 }
846
847 constexpr float epsilon = 1.0f;
848 const float smooth = config.rbf_smooth;
849
850 auto norm_x = [&](int x) -> float {
851 return cols > 1 ? static_cast<float>(x) / static_cast<float>(cols - 1) : 0.0f;
852 };
853 auto norm_y = [&](int y) -> float {
854 return rows > 1 ? static_cast<float>(y) / static_cast<float>(rows - 1) : 0.0f;
855 };
856 auto multiquadric_kernel = [](float r, float eps) -> float {
857 return std::sqrt((r / eps) * (r / eps) + 1.0f);
858 };
859
860 Eigen::MatrixXf W = Eigen::MatrixXf::Zero(n + 3, n + 3);
861 Eigen::VectorXf rhs = Eigen::VectorXf::Zero(n + 3);
862 for (int i = 0; i < n; ++i) {
863 rhs(i) = f(i);
864 for (int j = 0; j < n; ++j) {
865 const float dx = norm_x(points[i].x) - norm_x(points[j].x);
866 const float dy = norm_y(points[i].y) - norm_y(points[j].y);
867 const float r = std::sqrt(dx * dx + dy * dy);
868 W(i, j) = multiquadric_kernel(r, epsilon);
869 }
870 // Smoothing diagonal
871 W(i, i) += smooth;
872 // Polynomial terms (linear: 1, x, y) for TPS with affine component
873 W(i, n) = 1.0f;
874 W(i, n + 1) = norm_x(points[i].x);
875 W(i, n + 2) = norm_y(points[i].y);
876 W(n, i) = 1.0f;
877 W(n + 1, i) = norm_x(points[i].x);
878 W(n + 2, i) = norm_y(points[i].y);
879 }
880
881 Eigen::VectorXf weights = W.colPivHouseholderQr().solve(rhs);
882
883 // Render at downsampled resolution
884 Matrix2Df bg_down(rows, cols);
885 for (int r = 0; r < rows; ++r) {
886 for (int c = 0; c < cols; ++c) {
887 const float x = norm_x(c);
888 const float y = norm_y(r);
889 float val = weights(n) + weights(n + 1) * x + weights(n + 2) * y;
890 for (int i = 0; i < n; ++i) {
891 const float dx = x - norm_x(points[i].x);
892 const float dy = y - norm_y(points[i].y);
893 const float dist = std::sqrt(dx * dx + dy * dy);
894 val += weights(i) * multiquadric_kernel(dist, epsilon);
895 }
896 bg_down(r, c) = val;
897 }
898 }
899
900 return upscale_lanczos4(bg_down, target_rows, target_cols);
901}
902
903AutoBGEResult build_autobge_models(
904 const Matrix2Df& R, const Matrix2Df& G, const Matrix2Df& B,
905 const BGEConfig& config) {
906
907 AutoBGEResult result;
908 const int rows = static_cast<int>(R.rows());
909 const int cols = static_cast<int>(R.cols());
910
911 // Detect mono input: all three channels identical
912 const bool mono = (R.array() == G.array()).all() && (G.array() == B.array()).all();
913 result.mono_input = mono;
914
915 // Determine which channels to process
916 std::array<const Matrix2Df*, 3> channels = {&R, &G, &B};
917 std::array<std::string, 3> names = {"R", "G", "B"};
918 int n_channels = mono ? 1 : 3;
919
920 // Determine if we should process mono
921 if (mono && config.autobge.mono_mode == "disabled") {
922 result.success = false;
923 return result;
924 }
925
926 const std::vector<uint8_t>* valid_mask = nullptr;
927 if (config.common_valid_mask.size() == static_cast<size_t>(rows * cols) &&
928 config.common_mask_rows == rows && config.common_mask_cols == cols) {
929 valid_mask = &config.common_valid_mask;
930 }
931 if (!valid_mask) {
932 result.success = false;
933 return result;
934 }
935
936 std::vector<uint8_t> sampling_mask = *valid_mask;
937 const bool have_sampling_mask =
938 config.sampling_valid_mask.size() == static_cast<size_t>(rows * cols) &&
939 config.sampling_mask_rows == rows && config.sampling_mask_cols == cols;
940 if (have_sampling_mask) {
941 for (size_t i = 0; i < sampling_mask.size(); ++i)
942 sampling_mask[i] = (sampling_mask[i] != 0 &&
943 config.sampling_valid_mask[i] != 0) ? 1u : 0u;
944 }
945 Matrix2Df luma(rows, cols);
946 std::vector<float> valid_luma;
947 valid_luma.reserve(static_cast<size_t>(rows * cols));
948 for (int y = 0; y < rows; ++y) {
949 for (int x = 0; x < cols; ++x) {
950 const size_t idx = static_cast<size_t>(y * cols + x);
951 const float rv = R(y, x);
952 const float gv = G(y, x);
953 const float bv = B(y, x);
954 const float lv = 0.2126f * rv + 0.7152f * gv + 0.0722f * bv;
955 luma(y, x) = lv;
956 if ((*valid_mask)[idx] != 0 && std::isfinite(lv) && lv > 0.0f)
957 valid_luma.push_back(lv);
958 }
959 }
960 if (valid_luma.size() >= 16) {
961 // sorted_quantile_local interpolates between ranks lo and lo+1; select
962 // those two order statistics instead of sorting all values.
963 const float q = std::max(0.0f, std::min(1.0f, config.structure_thresh_percentile));
964 const float pos = q * static_cast<float>(valid_luma.size() - 1);
965 const size_t lo = static_cast<size_t>(std::floor(pos));
966 const size_t hi = std::min(valid_luma.size() - 1, lo + 1);
967 const float t = pos - static_cast<float>(lo);
968 std::nth_element(valid_luma.begin(),
969 valid_luma.begin() + static_cast<std::ptrdiff_t>(lo),
970 valid_luma.end());
971 const float v_lo = valid_luma[lo];
972 const float v_hi = hi == lo ? v_lo
973 : *std::min_element(valid_luma.begin() + static_cast<std::ptrdiff_t>(lo + 1),
974 valid_luma.end());
975 const float fg_thresh = v_lo * (1.0f - t) + v_hi * t;
976 for (int y = 0; y < rows; ++y) {
977 for (int x = 0; x < cols; ++x) {
978 const size_t idx = static_cast<size_t>(y * cols + x);
979 if ((*valid_mask)[idx] == 0 || !std::isfinite(luma(y, x)) ||
980 luma(y, x) >= fg_thresh) {
981 sampling_mask[idx] = 0;
982 }
983 }
984 }
985 }
986
987 const int scale = config.autobge.downsample_scale;
988 std::mt19937 rng(static_cast<uint32_t>(config.autobge.random_seed));
989
990 for (int ch = 0; ch < n_channels; ++ch) {
991 const Matrix2Df& channel = *channels[ch];
992
993 // Step 1: Transform to working space
994 StretchParams stretch_params;
995 Matrix2Df stretched = transform_to_autobge_working_space(
996 channel, config.autobge, &stretch_params, ch, valid_mask);
997
998 // Step 2: Downsample
999 Matrix2Df downsampled = downsample_area(stretched, scale);
1000
1001 // Downsample valid mask if present
1002 std::vector<uint8_t> mask_down;
1003 const std::vector<uint8_t>* mask_down_ptr = nullptr;
1004 if (valid_mask) {
1005 const int ds_rows = static_cast<int>(downsampled.rows());
1006 const int ds_cols = static_cast<int>(downsampled.cols());
1007 mask_down = downsample_mask_majority(sampling_mask, rows, cols, scale,
1008 ds_rows, ds_cols);
1009 mask_down_ptr = &mask_down;
1010 }
1011
1012 auto generate_points_with_retry = [&](const Matrix2Df& image,
1013 int min_points,
1014 const char* stage_name) {
1015 std::vector<SamplePoint> generated = generate_autobge_sample_points(
1016 image, config.autobge, mask_down_ptr, &rng);
1017 if (static_cast<int>(generated.size()) >= min_points)
1018 return generated;
1019
1020 BGEConfig::AutoBGEConfig relaxed = config.autobge;
1021 relaxed.bright_exclusion_fraction =
1022 std::min(0.98f,
1023 std::max(relaxed.bright_exclusion_fraction + 0.20f, 0.90f));
1024 std::vector<SamplePoint> retry = generate_autobge_sample_points(
1025 image, relaxed, mask_down_ptr, &rng, false);
1026 std::cerr << "[AutoBGE] Channel " << names[ch] << " " << stage_name
1027 << ": relaxed sparse sampling retry " << generated.size()
1028 << " -> " << retry.size() << " points" << std::endl;
1029 return retry;
1030 };
1031
1032 // Step 3: Generate sample points
1033 const int min_stage1_points = autobge_min_stage1_points(config.autobge);
1034 std::vector<SamplePoint> points = generate_points_with_retry(
1035 downsampled, min_stage1_points, "stage1");
1036
1037 // Post-filter: reject sample points whose local patch variance is high
1038 // relative to the global distribution. In the MTF working space, nebula
1039 // signal can appear compressed and slip through bright-exclusion. The
1040 // variance filter catches it because signal regions always have higher
1041 // local scatter than pure background.
1042 if (points.size() > 8) {
1043 const size_t before = points.size();
1044 points = filter_sample_points_by_variance(
1045 downsampled, points, config.autobge.patch_size, 2.5f);
1046 if (points.size() < before) {
1047 std::cerr << "[AutoBGE] Channel " << names[ch]
1048 << ": variance filter removed " << (before - points.size())
1049 << " of " << before << " stage1 points" << std::endl;
1050 }
1051 }
1052
1053 if (static_cast<int>(points.size()) < min_stage1_points) {
1054 std::cerr << "[AutoBGE] Channel " << names[ch]
1055 << ": insufficient stage1 sample points (" << points.size()
1056 << ", required " << min_stage1_points << ")" << std::endl;
1057 result.channel_models[ch].success = false;
1058 result.channel_models[ch].error_message = "insufficient_sample_points";
1059 continue;
1060 }
1061
1062 // Step 4: Two-stage fitting — polynomial first, then RBF on residuals
1063 Matrix2Df poly_bg = fit_polynomial_autobge(
1064 downsampled, points, config.autobge, rows, cols);
1065
1066 // Compute residual after polynomial subtraction
1067 Matrix2Df poly_down = downsample_area(poly_bg, scale);
1068 Matrix2Df residual_down = downsampled - poly_down;
1069 if (config.autobge.normalize_between_stages) {
1070 std::vector<float> source_vals;
1071 std::vector<float> residual_vals;
1072 source_vals.reserve(points.size());
1073 residual_vals.reserve(points.size());
1074 for (const auto& p : points) {
1075 source_vals.push_back(patch_estimate(downsampled, p.x, p.y,
1076 config.autobge.patch_size,
1077 config.autobge.patch_estimator));
1078 residual_vals.push_back(patch_estimate(residual_down, p.x, p.y,
1079 config.autobge.patch_size,
1080 config.autobge.patch_estimator));
1081 }
1082 const float source_med = tile_compile::core::median_of(std::move(source_vals));
1083 const float residual_med = tile_compile::core::median_of(std::move(residual_vals));
1084 residual_down.array() += (source_med - residual_med);
1085 }
1086
1087 // Generate sample points on the residual image
1088 std::vector<SamplePoint> residual_points = generate_points_with_retry(
1089 residual_down, kAutoBGEMinStage2Points, "stage2");
1090
1091 Matrix2Df rbf_bg;
1092 if (static_cast<int>(residual_points.size()) < kAutoBGEMinStage2Points) {
1093 std::cerr << "[AutoBGE] Channel " << names[ch]
1094 << ": insufficient stage2 sample points ("
1095 << residual_points.size() << ", required "
1096 << kAutoBGEMinStage2Points << ")" << std::endl;
1097 result.channel_models[ch].success = false;
1098 result.channel_models[ch].error_message =
1099 "insufficient_residual_sample_points";
1100 continue;
1101 }
1102 rbf_bg = fit_rbf_autobge(
1103 residual_down, residual_points, config.autobge, rows, cols);
1104
1105 // Combine: total background = polynomial + RBF residual
1106 Matrix2Df total_bg = poly_bg + rbf_bg;
1107
1108 // Step 5: derive original-domain model from corrected image. This avoids
1109 // treating nonlinear working-space deltas as directly additive backgrounds.
1110 Matrix2Df corrected_working = stretched - total_bg;
1111 Matrix2Df corrected_original = transform_from_autobge_working_space(
1112 corrected_working, stretch_params, ch);
1113 Matrix2Df bg_original = channel - corrected_original;
1114
1115 // Fill model
1116 result.channel_models[ch].model = bg_original;
1117 result.channel_models[ch].success = true;
1118 result.channel_models[ch].n_valid_cells = static_cast<int>(points.size());
1119 result.channel_models[ch].grid_cells.clear();
1120 result.channel_models[ch].grid_cells.reserve(points.size());
1121 std::vector<float> residual_values;
1122 residual_values.reserve(points.size());
1123 for (const auto& p : points) {
1124 GridCell gc;
1125 gc.center_x = static_cast<float>(p.x * scale);
1126 gc.center_y = static_cast<float>(p.y * scale);
1127 gc.bg_value = patch_estimate(downsampled, p.x, p.y,
1128 config.autobge.patch_size,
1129 config.autobge.patch_estimator);
1130 gc.weight = 1.0f;
1131 gc.n_samples = 1;
1132 gc.valid = true;
1133 result.channel_models[ch].grid_cells.push_back(gc);
1134 const int yy = std::clamp(p.y * scale, 0, rows - 1);
1135 const int xx = std::clamp(p.x * scale, 0, cols - 1);
1136 residual_values.push_back(gc.bg_value - total_bg(yy, xx));
1137 }
1138 result.channel_models[ch].rms_residual = 0.0f;
1139 if (!residual_values.empty()) {
1140 double sum_sq = 0.0;
1141 for (float v : residual_values)
1142 sum_sq += static_cast<double>(v) * static_cast<double>(v);
1143 result.channel_models[ch].rms_residual =
1144 static_cast<float>(std::sqrt(sum_sq / residual_values.size()));
1145 }
1146
1147 // Build channel diagnostics
1148 BGEChannelDiagnostics ch_diag;
1149 ch_diag.channel_name = names[ch];
1150 ch_diag.applied = false;
1151 ch_diag.fit_success = true;
1152 ch_diag.fit_rms_residual = result.channel_models[ch].rms_residual;
1153 ch_diag.tile_samples_total = static_cast<int>(points.size());
1154 ch_diag.tile_samples_valid = static_cast<int>(points.size());
1155 ch_diag.grid_cells = result.channel_models[ch].grid_cells;
1156 ch_diag.grid_cells_valid = static_cast<int>(ch_diag.grid_cells.size());
1157 for (const auto& gc : ch_diag.grid_cells) {
1158 ch_diag.sample_bg_values.push_back(gc.bg_value);
1159 ch_diag.sample_weight_values.push_back(gc.weight);
1160 }
1161 ch_diag.sample_bg_stats = stats_from_values_local(ch_diag.sample_bg_values);
1162 ch_diag.sample_weight_stats = stats_from_values_local(ch_diag.sample_weight_values);
1163 ch_diag.residual_values = residual_values;
1164 ch_diag.residual_stats = stats_from_values_local(ch_diag.residual_values);
1165 result.channel_diagnostics.push_back(ch_diag);
1166 }
1167
1168 // If mono with rgb_duplicate, copy the R model to G and B
1169 if (mono && config.autobge.mono_mode == "rgb_duplicate") {
1170 result.channel_models[1] = result.channel_models[0];
1171 result.channel_models[2] = result.channel_models[0];
1172 if (!result.channel_diagnostics.empty()) {
1173 BGEChannelDiagnostics g_diag = result.channel_diagnostics[0];
1174 g_diag.channel_name = "G";
1175 BGEChannelDiagnostics b_diag = result.channel_diagnostics[0];
1176 b_diag.channel_name = "B";
1177 result.channel_diagnostics.push_back(g_diag);
1178 result.channel_diagnostics.push_back(b_diag);
1179 }
1180 }
1181
1182 // Check if at least one channel succeeded
1183 bool any_success = false;
1184 for (int ch = 0; ch < 3; ++ch) {
1185 if (result.channel_models[ch].success) {
1186 any_success = true;
1187 break;
1188 }
1189 }
1190 result.success = any_success;
1191 return result;
1192}

1196 const std::array<BackgroundModel, 3>& channel_models,
1197 const std::vector<BGEChannelDiagnostics>& channel_diagnostics,
1198 const BGEConfig& config,
1199 BGEDiagnostics* diagnostics) {
1200
1201 const int rows = static_cast<int>(R.rows());
1202 const int cols = static_cast<int>(R.cols());
1203 const bool have_canvas_mask =
1204 config.common_valid_mask.size() == static_cast<size_t>(rows * cols) &&
1205 config.common_mask_rows == rows && config.common_mask_cols == cols;
1206 if (!have_canvas_mask) {
1207 if (diagnostics) {
1208 diagnostics->attempted = true;
1209 diagnostics->success = false;
1210 diagnostics->bge_method = "autobge";
1211 diagnostics->method = "autobge";
1212 diagnostics->failure_reason = "missing_canvas_mask";
1213 diagnostics->channels = channel_diagnostics;
1214 }
1215 return false;
1216 }
1217
1218 std::array<const Matrix2Df*, 3> inputs = {&R, &G, &B};
1219 std::array<std::string, 3> names = {"R", "G", "B"};
1220 std::array<Matrix2Df, 3> corrected;
1221 std::vector<BGEChannelDiagnostics> out_diags = channel_diagnostics;
1222 if (out_diags.size() < 3)
1223 out_diags.resize(3);
1224 // This function is all-or-nothing at the pixel level (R/G/B are only written back at the
1225 // very end, once every channel has cleared its guards). But out_diags[ch].applied gets set
1226 // true the moment a channel individually clears its own per-channel guard (below), which can
1227 // happen for R/G before a later channel (B) causes an early return. Without this reset, the
1228 // persisted bge.json would report e.g. R/G "applied": true with real mean_shift/output_stats
1229 // even though those corrections were discarded and never reached the delivered image -- call
1230 // this on every false-return path so the diagnostics can't claim a channel was applied when
1231 // the whole correction was rejected.
1232 auto reject_all_applied = [&out_diags]() {
1233 for (auto& d : out_diags) d.applied = false;
1234 };
1235
1236 // Pass 1: validate every channel model and collect stats. All model
1237 // medians must be known before any correction so the apply below can
1238 // anchor every channel at one shared pedestal level.
1239 float shared_pedestal = std::numeric_limits<float>::infinity();
1240 for (int ch = 0; ch < 3; ++ch) {
1241 if (out_diags[static_cast<size_t>(ch)].channel_name.empty())
1242 out_diags[static_cast<size_t>(ch)].channel_name = names[ch];
1243 if (!channel_models[ch].success) {
1244 std::cerr << "[AutoBGE] Channel " << names[ch]
1245 << ": model not available, rejecting RGB apply" << std::endl;
1246 if (diagnostics) {
1247 diagnostics->attempted = true;
1248 diagnostics->success = false;
1249 diagnostics->bge_method = "autobge";
1250 diagnostics->method = "autobge";
1251 diagnostics->failure_reason = "partial_channel_model";
1252 reject_all_applied();
1253 diagnostics->channels = std::move(out_diags);
1254 }
1255 return false;
1256 }
1257
1258 const Matrix2Df& bg = channel_models[ch].model;
1259 if (bg.rows() != rows || bg.cols() != cols) {
1260 std::cerr << "[AutoBGE] Channel " << names[ch]
1261 << ": model dimension mismatch (" << bg.rows() << "x" << bg.cols()
1262 << " vs " << rows << "x" << cols << ")" << std::endl;
1263 if (diagnostics) {
1264 diagnostics->attempted = true;
1265 diagnostics->success = false;
1266 diagnostics->bge_method = "autobge";
1267 diagnostics->method = "autobge";
1268 diagnostics->failure_reason = "model_dimension_mismatch";
1269 reject_all_applied();
1270 diagnostics->channels = std::move(out_diags);
1271 }
1272 return false;
1273 }
1274
1275 auto& ch_diag = out_diags[static_cast<size_t>(ch)];
1276 ch_diag.input_stats = stats_from_matrix_local(*inputs[ch]);
1277 ch_diag.model_stats = stats_from_matrix_local(bg);
1278 if (std::isfinite(ch_diag.model_stats.median))
1279 shared_pedestal = std::min(shared_pedestal, ch_diag.model_stats.median);
1280 }
1281 if (!std::isfinite(shared_pedestal))
1282 shared_pedestal = 0.0f;
1283 // Shared pedestal (darkest channel level): re-anchoring every channel at
1284 // the same background level equalizes the R/G/B floors. Per-channel
1285 // pedestals preserve floor offsets that downstream stages (HMS) amplify
1286 // into a color cast.
1287
1288 // Pass 2: apply the correction with the shared pedestal. flatness_worsened
1289 // stays an immediate reject; slope_worsened is deferred so the channel-
1290 // level equalization check below can still rescue an already flat field.
1291 std::array<bool, 3> slope_rejected{false, false, false};
1292 for (int ch = 0; ch < 3; ++ch) {
1293 auto& ch_diag = out_diags[static_cast<size_t>(ch)];
1294 const Matrix2Df& img = *inputs[ch];
1295 const Matrix2Df& bg = channel_models[ch].model;
1296 corrected[ch] = img;
1297 for (int r = 0; r < rows; ++r) {
1298 for (int c = 0; c < cols; ++c) {
1299 const size_t idx = static_cast<size_t>(r * cols + c);
1300 if (config.common_valid_mask[idx] == 0) {
1301 corrected[ch](r, c) = 0.0f;
1302 continue;
1303 }
1304 const float vin = img(r, c);
1305 const float mv = bg(r, c);
1306 if (!std::isfinite(vin) || !std::isfinite(mv)) {
1307 corrected[ch](r, c) = std::numeric_limits<float>::quiet_NaN();
1308 continue;
1309 }
1310 corrected[ch](r, c) = std::max(0.0f, vin - mv + shared_pedestal);
1311 }
1312 }
1313
1314 if (config.autobge.apply_guards) {
1315 const float flat_pre =
1316 spatial_background_spread(img, &config.common_valid_mask);
1317 const float flat_post =
1318 spatial_background_spread(corrected[ch], &config.common_valid_mask);
1319 const float slope_pre =
1320 coarse_background_plane_slope(img, &config.common_valid_mask);
1321 const float slope_post =
1322 coarse_background_plane_slope(corrected[ch], &config.common_valid_mask);
1323 ch_diag.guard_flat_pre = flat_pre;
1324 ch_diag.guard_flat_post = flat_post;
1325 ch_diag.guard_slope_pre = slope_pre;
1326 ch_diag.guard_slope_post = slope_post;
1327 const float max_flatness_worsen_factor =
1328 config.internal_relaxed_channel_guards ? 1.35f : 1.15f;
1329 const float max_slope_worsen_factor =
1330 config.internal_relaxed_channel_guards ? 1.15f : 1.08f;
1331 if (std::isfinite(flat_pre) && std::isfinite(flat_post) &&
1332 flat_post > flat_pre * max_flatness_worsen_factor) {
1333 ch_diag.guard_rejected = true;
1334 ch_diag.guard_reason = "flatness_worsened";
1335 std::cerr << "[AutoBGE] Channel " << names[ch]
1336 << " guard rejected RGB apply: " << ch_diag.guard_reason
1337 << std::endl;
1338 if (diagnostics) {
1339 diagnostics->attempted = true;
1340 diagnostics->success = false;
1341 diagnostics->bge_method = "autobge";
1342 diagnostics->method = "autobge";
1343 diagnostics->failure_reason = ch_diag.guard_reason;
1344 reject_all_applied();
1345 diagnostics->channels = std::move(out_diags);
1346 }
1347 return false;
1348 }
1349 if (!ch_diag.guard_rejected && std::isfinite(slope_pre) &&
1350 std::isfinite(slope_post) &&
1351 slope_post > slope_pre * max_slope_worsen_factor) {
1352 ch_diag.guard_rejected = true;
1353 ch_diag.guard_reason = "slope_worsened";
1354 slope_rejected[static_cast<size_t>(ch)] = true;
1355 }
1356 }
1357
1358 ch_diag.fit_success = true;
1359 ch_diag.fit_rms_residual = channel_models[ch].rms_residual;
1360 ch_diag.output_stats = stats_from_matrix_local(corrected[ch]);
1361 ch_diag.mean_shift = ch_diag.output_stats.mean - ch_diag.input_stats.mean;
1362 ch_diag.applied = true;
1363 }
1364
1365 // Deferred slope decision: a worsened coarse-plane slope vetoes the apply
1366 // only when the correction cannot simultaneously equalize the per-channel
1367 // background levels. On an already flat field the fitted model can add a
1368 // harmless micro-tilt while still removing real channel floor offsets --
1369 // that trade is accepted and reported instead of discarding the run's only
1370 // pedestal equalization step. The shared-pedestal variant keeps each
1371 // channel's residual offset (in_med - model_med); when the per-channel
1372 // models disagree about the absolute level (e.g. a channel whose model
1373 // under-estimates the pedestal) the residual spread can stay or even grow.
1374 // The rescue therefore re-anchors every channel at the darkest residual
1375 // level: corrected_c = in_c - model_c + (shared_pedestal + min_res -
1376 // res_med_c), which makes the corrected medians equal by construction.
1377 if (config.autobge.apply_guards &&
1378 (slope_rejected[0] || slope_rejected[1] || slope_rejected[2])) {
1379 const size_t total_px = static_cast<size_t>(rows) * static_cast<size_t>(cols);
1380 const size_t stride = std::max<size_t>(1, total_px / 200000u);
1381 auto masked_level = [&](const Matrix2Df &m) -> float {
1382 std::vector<float> vals;
1383 for (size_t i = 0; i < total_px; i += stride) {
1384 if (config.common_valid_mask[i] == 0)
1385 continue;
1386 const float v = m.data()[i];
1387 if (std::isfinite(v) && v > 0.0f)
1388 vals.push_back(v);
1389 }
1390 if (vals.empty())
1391 return std::numeric_limits<float>::quiet_NaN();
1392 std::nth_element(vals.begin(), vals.begin() + vals.size() / 2, vals.end());
1393 return vals[vals.size() / 2];
1394 };
1395 // Residual median of (input - model) per channel; residuals are legitimate
1396 // level offsets, so unlike masked_level no v>0 filter applies.
1397 auto masked_residual = [&](const Matrix2Df &img,
1398 const Matrix2Df &bg) -> float {
1399 std::vector<float> vals;
1400 for (size_t i = 0; i < total_px; i += stride) {
1401 if (config.common_valid_mask[i] == 0)
1402 continue;
1403 const float v = img.data()[i] - bg.data()[i];
1404 if (std::isfinite(v))
1405 vals.push_back(v);
1406 }
1407 if (vals.empty())
1408 return std::numeric_limits<float>::quiet_NaN();
1409 std::nth_element(vals.begin(), vals.begin() + vals.size() / 2, vals.end());
1410 return vals[vals.size() / 2];
1411 };
1412 float in_med[3], res_med[3];
1413 float in_lo = std::numeric_limits<float>::infinity(),
1414 in_hi = -in_lo;
1415 float min_res = std::numeric_limits<float>::infinity();
1416 bool have_levels = true;
1417 for (int ch = 0; ch < 3; ++ch) {
1418 in_med[ch] = masked_level(*inputs[ch]);
1419 res_med[ch] = masked_residual(*inputs[ch], channel_models[ch].model);
1420 if (!std::isfinite(in_med[ch]) || !std::isfinite(res_med[ch]))
1421 have_levels = false;
1422 in_lo = std::min(in_lo, in_med[ch]);
1423 in_hi = std::max(in_hi, in_med[ch]);
1424 min_res = std::min(min_res, res_med[ch]);
1425 }
1426 const float spread_pre = in_hi - in_lo;
1427 // A constant per-channel shift does not change the within-channel
1428 // flatness/slope metrics, so the earlier guard measurements stay valid.
1429 // The rescue is still bounded: the worsened slope may not exceed a
1430 // moderate factor over the input slope, and the channel-level spread
1431 // must strictly improve.
1432 const float slope_bound =
1433 config.internal_relaxed_channel_guards ? 2.0f : 1.6f;
1434 // The ratio bound alone is meaningless on an already flat input: when
1435 // slope_pre ~ 1e-5 DN/px (~0.05 DN edge-to-edge), even a model adding a
1436 // negligible micro-tilt trips any multiplicative guard. The absolute
1437 // fallback therefore caps the post-correction tilt amplitude
1438 // (slope * frame diagonal, in DN) at a fraction of the channel spread the
1439 // rescue is about to remove -- a tilt far below the corrected offset is
1440 // always a net uniformity win.
1441 const float diag_px = std::sqrt(static_cast<float>(rows) * rows +
1442 static_cast<float>(cols) * cols);
1443 const float tilt_bound_dn = 0.25f * spread_pre;
1444 bool slope_bounded = have_levels;
1445 for (int ch = 0; ch < 3 && slope_bounded; ++ch) {
1446 if (!slope_rejected[static_cast<size_t>(ch)])
1447 continue;
1448 const auto &d = out_diags[static_cast<size_t>(ch)];
1449 const float tilt_amp_dn =
1450 std::isfinite(d.guard_slope_post) ? d.guard_slope_post * diag_px
1451 : std::numeric_limits<float>::infinity();
1452 slope_bounded = d.guard_slope_post <= d.guard_slope_pre * slope_bound ||
1453 tilt_amp_dn <= tilt_bound_dn;
1454 }
1455 if (have_levels && slope_bounded) {
1456 for (int ch = 0; ch < 3; ++ch) {
1457 const Matrix2Df &img = *inputs[ch];
1458 const Matrix2Df &bg = channel_models[ch].model;
1459 const float anchor =
1460 shared_pedestal + (min_res - res_med[ch]);
1461 for (int r = 0; r < rows; ++r) {
1462 for (int c = 0; c < cols; ++c) {
1463 const size_t idx = static_cast<size_t>(r * cols + c);
1464 if (config.common_valid_mask[idx] == 0) {
1465 corrected[ch](r, c) = 0.0f;
1466 continue;
1467 }
1468 const float vin = img(r, c);
1469 const float mv = bg(r, c);
1470 if (!std::isfinite(vin) || !std::isfinite(mv)) {
1471 corrected[ch](r, c) = std::numeric_limits<float>::quiet_NaN();
1472 continue;
1473 }
1474 corrected[ch](r, c) = std::max(0.0f, vin - mv + anchor);
1475 }
1476 }
1477 auto &d = out_diags[static_cast<size_t>(ch)];
1478 d.output_stats = stats_from_matrix_local(corrected[ch]);
1479 d.mean_shift = d.output_stats.mean - d.input_stats.mean;
1480 }
1481 const float spread_post = [&] {
1482 float lo = std::numeric_limits<float>::infinity(),
1483 hi = -lo;
1484 for (int ch = 0; ch < 3; ++ch) {
1485 const float lv = masked_level(corrected[ch]);
1486 if (!std::isfinite(lv)) return std::numeric_limits<float>::quiet_NaN();
1487 lo = std::min(lo, lv);
1488 hi = std::max(hi, lv);
1489 }
1490 return hi - lo;
1491 }();
1492 if (std::isfinite(spread_post) && spread_post < spread_pre) {
1493 std::cerr << "[AutoBGE] slope guard overridden: channel background "
1494 "level equalization improved (spread "
1495 << spread_pre << " -> " << spread_post << ")" << std::endl;
1496 for (int ch = 0; ch < 3; ++ch) {
1497 if (slope_rejected[static_cast<size_t>(ch)]) {
1498 auto &d = out_diags[static_cast<size_t>(ch)];
1499 d.guard_rejected = false;
1500 d.guard_reason = "slope_worsened_but_level_equalized";
1501 slope_rejected[static_cast<size_t>(ch)] = false;
1502 }
1503 }
1504 if (diagnostics)
1505 diagnostics->guard_override = "level_equalization";
1506 } else {
1507 // Anchored variant did not improve the spread --- restore the shared
1508 // pedestal form so the rejection below reports consistent stats.
1509 for (int ch = 0; ch < 3; ++ch) {
1510 const Matrix2Df &img = *inputs[ch];
1511 const Matrix2Df &bg = channel_models[ch].model;
1512 for (int r = 0; r < rows; ++r)
1513 for (int c = 0; c < cols; ++c) {
1514 const size_t idx = static_cast<size_t>(r * cols + c);
1515 if (config.common_valid_mask[idx] == 0) {
1516 corrected[ch](r, c) = 0.0f;
1517 continue;
1518 }
1519 const float vin = img(r, c);
1520 const float mv = bg(r, c);
1521 corrected[ch](r, c) = (!std::isfinite(vin) || !std::isfinite(mv))
1522 ? std::numeric_limits<float>::quiet_NaN()
1523 : std::max(0.0f, vin - mv + shared_pedestal);
1524 }
1525 auto &d = out_diags[static_cast<size_t>(ch)];
1526 d.output_stats = stats_from_matrix_local(corrected[ch]);
1527 d.mean_shift = d.output_stats.mean - d.input_stats.mean;
1528 }
1529 }
1530 }
1531 // Any channel still slope-rejected after the rescue rejects the apply.
1532 const bool still_rejected = slope_rejected[0] || slope_rejected[1] ||
1533 slope_rejected[2];
1534 if (still_rejected) {
1535 for (int ch = 0; ch < 3; ++ch) {
1536 if (!slope_rejected[static_cast<size_t>(ch)])
1537 continue;
1538 std::cerr << "[AutoBGE] Channel " << names[ch]
1539 << " guard rejected RGB apply: "
1540 << out_diags[static_cast<size_t>(ch)].guard_reason
1541 << std::endl;
1542 if (diagnostics) {
1543 diagnostics->attempted = true;
1544 diagnostics->success = false;
1545 diagnostics->bge_method = "autobge";
1546 diagnostics->method = "autobge";
1547 diagnostics->failure_reason =
1548 out_diags[static_cast<size_t>(ch)].guard_reason;
1549 reject_all_applied();
1550 diagnostics->channels = std::move(out_diags);
1551 }
1552 return false;
1553 }
1554 }
1555 }
1556
1557 if (config.autobge.apply_guards) {
1558 const std::vector<uint8_t> bg_mask =
1559 build_chroma_background_mask_from_rgb(R, G, B, &config.common_valid_mask);
1560 // A per-channel pedestal damps the log-ratio scatter of the input (the
1561 // offset compresses both noise and true spatial variation of the
1562 // ratio), so comparing raw input std against a level-equalized
1563 // correction falsely rejects exactly the equalization BGE is for.
1564 // Measure the pre-std on level-equalized input channels so both sides
1565 // sit at a common level and the comparison is apples-to-apples.
1566 const auto masked_median = [&](const Matrix2Df &ch) {
1567 std::vector<float> vals;
1568 vals.reserve(bg_mask.size() / 4);
1569 for (size_t i = 0; i < bg_mask.size(); ++i) {
1570 if (bg_mask[i] == 0)
1571 continue;
1572 const float v = ch.data()[i];
1573 if (std::isfinite(v) && v > 0.0f)
1574 vals.push_back(v);
1575 }
1576 return vals.empty() ? 0.0f
1577 : tile_compile::core::median_of(std::move(vals));
1578 };
1579 const std::array<float, 3> in_med = {masked_median(R), masked_median(G),
1580 masked_median(B)};
1581 const float shared_level =
1582 std::min(in_med[0], std::min(in_med[1], in_med[2]));
1583 const Matrix2Df R_eq =
1584 (R.array() + (shared_level - in_med[0])).matrix();
1585 const Matrix2Df G_eq =
1586 (G.array() + (shared_level - in_med[1])).matrix();
1587 const Matrix2Df B_eq =
1588 (B.array() + (shared_level - in_med[2])).matrix();
1589 const float pre_rg_std = log_chroma_std_background(R_eq, G_eq, bg_mask);
1590 const float pre_bg_std = log_chroma_std_background(B_eq, G_eq, bg_mask);
1591 const float post_rg_std =
1592 log_chroma_std_background(corrected[0], corrected[1], bg_mask);
1593 const float post_bg_std =
1594 log_chroma_std_background(corrected[2], corrected[1], bg_mask);
1595 constexpr float kMaxChromaStdWorsenFactor = 1.08f;
1596 if ((std::isfinite(pre_rg_std) && std::isfinite(post_rg_std) &&
