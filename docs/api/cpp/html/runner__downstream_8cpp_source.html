51std::vector<TileMetrics> build_bge_tile_metrics_from_rgb(
52 const TileGrid &grid, const tile_compile::Matrix2Df &R,
53 const tile_compile::Matrix2Df &G, const tile_compile::Matrix2Df &B,
54 const std::vector<uint8_t> &valid_mask, int mask_rows, int mask_cols) {
55 std::vector<TileMetrics> out;
56 out.reserve(grid.tiles.size());
57 const bool mask_ok =
58 mask_rows == R.rows() && mask_cols == R.cols() &&
59 G.rows() == R.rows() && B.rows() == R.rows() &&
60 G.cols() == R.cols() && B.cols() == R.cols() &&
61 valid_mask.size() == static_cast<size_t>(R.rows() * R.cols());
62
63 for (const auto &tile : grid.tiles) {
64 TileMetrics tm{};
65 tm.fwhm = 0.0f;
66 tm.roundness = 0.0f;
67 tm.contrast = 0.0f;
68 tm.sharpness = 0.0f;
69 tm.background = 0.0f;
70 tm.noise = 0.0f;
71 tm.gradient_energy = 0.0f;
72 tm.star_count = 0;
73 tm.type = TileType::STRUCTURE;
74 tm.quality_score = 0.0f;
75
76 const int x0 = std::max(0, tile.x);
77 const int y0 = std::max(0, tile.y);
78 const int x1 = std::min(tile.x + tile.width, static_cast<int>(R.cols()));
79 const int y1 = std::min(tile.y + tile.height, static_cast<int>(R.rows()));
80 if (x1 <= x0 || y1 <= y0 || !mask_ok) {
81 out.push_back(tm);
82 continue;
83 }
84
85 std::vector<float> values;
86 values.reserve(static_cast<size_t>((x1 - x0) * (y1 - y0)));
87 double gradient_sum = 0.0;
88 size_t gradient_count = 0;
89 for (int y = y0; y < y1; ++y) {
90 for (int x = x0; x < x1; ++x) {
91 const size_t idx =
92 static_cast<size_t>(y) * static_cast<size_t>(mask_cols) +
93 static_cast<size_t>(x);
94 if (valid_mask[idx] == 0) {
95 continue;
96 }
97 const float rv = R(y, x);
98 const float gv = G(y, x);
99 const float bv = B(y, x);
100 if (!(std::isfinite(rv) && std::isfinite(gv) && std::isfinite(bv))) {
101 continue;
102 }
103 const float luma = 0.2126f * rv + 0.7152f * gv + 0.0722f * bv;
104 if (!std::isfinite(luma)) {
105 continue;
106 }
107 values.push_back(luma);
108
109 const int xm = std::max(x0, x - 1);
110 const int xp = std::min(x1 - 1, x + 1);
111 const int ym = std::max(y0, y - 1);
112 const int yp = std::min(y1 - 1, y + 1);
113 const float l_xm =
114 0.2126f * R(y, xm) + 0.7152f * G(y, xm) + 0.0722f * B(y, xm);
115 const float l_xp =
116 0.2126f * R(y, xp) + 0.7152f * G(y, xp) + 0.0722f * B(y, xp);
117 const float l_ym =
118 0.2126f * R(ym, x) + 0.7152f * G(ym, x) + 0.0722f * B(ym, x);
119 const float l_yp =
120 0.2126f * R(yp, x) + 0.7152f * G(yp, x) + 0.0722f * B(yp, x);
121 if (std::isfinite(l_xm) && std::isfinite(l_xp) &&
122 std::isfinite(l_ym) && std::isfinite(l_yp)) {
123 gradient_sum += std::fabs(l_xp - l_xm) + std::fabs(l_yp - l_ym);
124 ++gradient_count;
125 }
126 }
127 }
128 if (!values.empty()) {
129 std::vector<float> median_values = values;
130 tm.background = tile_compile::core::median_of(median_values);
131 std::vector<float> noise_values = values;
132 tm.noise = tile_compile::core::robust_sigma_mad(noise_values);
133 tm.gradient_energy =
134 gradient_count > 0
135 ? static_cast<float>(gradient_sum /
136 static_cast<double>(gradient_count))
137 : 0.0f;
138 tm.contrast = tm.gradient_energy;
139 tm.sharpness = tm.gradient_energy;
140 }
141 out.push_back(tm);
142 }
143
144 return out;
145}
