70void migrate_legacy_config_node(YAML::Node& node,
71 ConfigMigrationReport& report) {
72 if (!node || !node.IsMap()) {
73 return; // nothing structured to migrate
74 }
75
76 // --- 1. semantic legacy keys: fail-closed, never stripped ----------------
77 if (node["method"] && node["method"].IsDefined() &&
78 !node["method"].IsNull()) {
79 throw ConfigError(
80 "UNKNOWN_LEGACY_KEY: top-level 'method' is not accepted by the "
81 "single-method pipeline. There is exactly one reconstruction "
82 "method now (plan section 6.5). Remove the 'method' key, or run "
83 "'tile_compile_cli migrate-config <in> <out>'.");
84 }
85 // A reconstruction-selection 'engine' key, if one ever existed.
86 if (node["reconstruction"] && node["reconstruction"].IsMap() &&
87 node["reconstruction"]["engine"]) {
88 throw ConfigError(
89 "UNKNOWN_LEGACY_KEY: 'reconstruction.engine' selects a reconstruction "
90 "engine; the single-method pipeline has none (plan section 6.5).");
91 }
92 // --- 2. removed structural blocks: strip with warning -------------------
93 for (const char* block : kRemovedTopLevelBlocks) {
94 if (node[block] && node[block].IsDefined()) {
95 node.remove(block);
96 warn_stripped(block, report);
97 }
98 }
99 for (const auto& [block, key] : kRemovedSubKeys) {
100 if (node[block] && node[block].IsMap() && node[block][key] &&
101 node[block][key].IsDefined()) {
102 node[block].remove(key);
103 warn_stripped(std::string(block) + "." + key, report);
104 }
105 }
106 for (const auto& [block, sub, key] : kRemovedNestedKeys) {
107 if (node[block] && node[block].IsMap() && node[block][sub] &&
108 node[block][sub].IsMap() && node[block][sub][key] &&
109 node[block][sub][key].IsDefined()) {
110 node[block][sub].remove(key);
111 warn_stripped(std::string(block) + "." + sub + "." + key, report);
112 }
113 }
114
115 // --- 3. renames ---------------------------------------------------------
116 // stacking.common_overlap_required_fraction -> reconstruction.* (same knob,
117 // moved when STACKING became a pass-through).
118 auto move_key = [&](YAML::Node src_parent, const char* src_key,
119 YAML::Node dst_parent, const char* dst_key,
120 const std::string& from, const std::string& to) {
121 if (!src_parent || !src_parent.IsMap() || !src_parent[src_key] ||
122 !src_parent[src_key].IsDefined()) {
123 return;
124 }
125 // An explicitly set new-style key wins; the legacy value is still dropped.
126 if (!dst_parent[dst_key] || !dst_parent[dst_key].IsDefined()) {
127 dst_parent[dst_key] = src_parent[src_key];
128 }
129 src_parent.remove(src_key);
130 std::cerr << "[CONFIG-MIGRATION] WARN renamed legacy config key '" << from
131 << "' -> '" << to << "'." << std::endl;
132 report.renamed_keys.emplace_back(from, to);
133 report.applied = true;
134 };
135
136 if (node["stacking"] && node["stacking"].IsMap() &&
137 node["stacking"]["common_overlap_required_fraction"] &&
138 node["stacking"]["common_overlap_required_fraction"].IsDefined()) {
139 if (!node["reconstruction"] || !node["reconstruction"].IsMap()) {
140 node["reconstruction"] = YAML::Node(YAML::NodeType::Map);
141 }
142 move_key(node["stacking"], "common_overlap_required_fraction",
143 node["reconstruction"], "common_overlap_required_fraction",
144 "stacking.common_overlap_required_fraction",
145 "reconstruction.common_overlap_required_fraction");
146 }
