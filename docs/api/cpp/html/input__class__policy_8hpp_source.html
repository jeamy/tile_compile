3// Input-class policy --- milestone M0 of the CFA-forward-drizzle plan
4// (docs/AQMH/aqmh_cfa_forward_drizzle_multiband_implementierungsplan_de.md
5// sections 3.1.1, 23/M0).
6//
7// The single-method pipeline supports exactly two input classes: OSC raw with a
8// known Bayer pattern, and MONO raw. Already-debayered RGB cubes are rejected
9// fail-closed at SCAN_INPUT with a stable "UNSUPPORTED_INPUT" error; unknown /
10// contradictory colour metadata is also rejected fail-closed.

18enum class InputClassDecision {
19 accept_osc, // OSC with a known Bayer pattern
20 accept_mono, // MONO single plane
21 reject_rgb, // already-debayered RGB cube --- out of scope for the cutover
22 reject_osc_no_bayer, // OSC without a resolvable Bayer pattern
23 reject_unknown, // no usable colour classification
24};
25
26inline InputClassDecision classify_input_for_single_method(
27 ColorMode mode, BayerPattern bayer) {
28 switch (mode) {
29 case ColorMode::MONO:
30 return InputClassDecision::accept_mono;
31 case ColorMode::OSC:
32 return bayer == BayerPattern::UNKNOWN
33 ? InputClassDecision::reject_osc_no_bayer
34 : InputClassDecision::accept_osc;
35 case ColorMode::RGB:
36 return InputClassDecision::reject_rgb;
37 default:
38 return InputClassDecision::reject_unknown;
39 }
40}
41
42inline bool input_class_accepted(InputClassDecision d) {
43 return d == InputClassDecision::accept_osc ||
44 d == InputClassDecision::accept_mono;
45}

47inline std::string input_class_rejection_message(InputClassDecision d) {
48 switch (d) {
49 case InputClassDecision::reject_rgb:
50 return "UNSUPPORTED_INPUT: already-debayered RGB frames are not accepted "
51 "by the CFA-forward-drizzle single-method pipeline (plan section "
52 "3.1.1). Supported input classes: OSC raw (with a known Bayer "
53 "pattern) and MONO raw.";
54 case InputClassDecision::reject_osc_no_bayer:
55 return "UNSUPPORTED_INPUT: OSC input without a resolvable Bayer pattern "
56 "(BAYERPAT / COLORTYP / data.bayer_pattern). The single-method "
57 "pipeline does not guess a CFA pattern (plan section 3.1.1).";
58 case InputClassDecision::reject_unknown:
59 default:
60 return "UNSUPPORTED_INPUT: the colour mode of the input frames could not "
61 "be classified as OSC or MONO (plan section 3.1.1).";
62 }
63}
