47double shoelace_area(const double *x, const double *y, int n) {
48 if (n < 3)
49 return 0.0;
50 double s = 0.0;
51 for (int i = 0; i < n; ++i) {
52 const int j = (i + 1) % n;
53 s += x[i] * y[j] - x[j] * y[i];
54 }
55 return std::abs(s) * 0.5;
56}

60// Sutherland-Hodgman clip of a convex polygon against one axis-aligned
61// half-plane, generic over the plane test/intersection so all four sides of
62// a rectangle reuse it.
63template <typename InsideFn, typename IntersectFn>
64int clip_one_plane(const double *in_x, const double *in_y, int in_n,
65 InsideFn inside, IntersectFn intersect, double *out_x,
66 double *out_y) {
67 if (in_n == 0)
68 return 0;
69 int out_n = 0;
70 for (int i = 0; i < in_n; ++i) {
71 const int j = i + 1 == in_n ? 0 : i + 1;
72 const bool in_i = inside(in_x[i], in_y[i]);
73 const bool in_j = inside(in_x[j], in_y[j]);
74 if (in_i) {
75 out_x[out_n] = in_x[i];
76 out_y[out_n] = in_y[i];
77 ++out_n;
78 }
79 if (in_i != in_j) {
80 double ix = 0.0, iy = 0.0;
81 intersect(in_x[i], in_y[i], in_x[j], in_y[j], ix, iy);
82 out_x[out_n] = ix;
83 out_y[out_n] = iy;
84 ++out_n;
85 }
86 }
87 return out_n;
88}

92// The rectangle clip is split into its column half (x >= rx0, x <= rx1) and
93// its row half (y >= ry0, y <= ry1) so a caller that scatters one polygon over
94// a block of cells clips each column once. The two halves run the four
95// Sutherland-Hodgman planes in the historical order with the historical
96// arithmetic, so composing them is bit-identical to the one-shot clip. The
97// shortcuts below are exact by construction (each replaces a plane pass whose
98// result is known without evaluating it):
99// - every vertex passes a plane: the plane returns its input unchanged, in
100// order;
101// - every vertex fails a plane's test (and the plane is the first one able to
102// reject, or the failing vertices all exceed the far bound so that earlier
103// planes pass them untouched): no output vertices, area exactly 0.0.
104// The rejects use the strict "beyond the far bound" form for the second plane
105// of each pair and require lo <= hi, so an edge crossing the first plane can
106// never produce a vertex that survives the second.
107int polygon_clip_columns(const double poly_x[4], const double poly_y[4],
108 double rx0, double rx1, double out_x[8],
109 double out_y[8]) {
110 bool all_in = true;
111 bool all_below = true; // every vertex fails x >= rx0
112 bool all_above = rx0 <= rx1; // every vertex has x > rx1
113 for (int i = 0; i < 4; ++i) {
114 const double x = poly_x[i];
115 const bool ge = x >= rx0;
116 const bool le = x <= rx1;
117 all_in = all_in && ge && le;
118 all_below = all_below && !ge;
119 all_above = all_above && x > rx1;
120 }
121 if (all_below || all_above)
122 return 0;
123 if (all_in) {
124 for (int i = 0; i < 4; ++i) {
125 out_x[i] = poly_x[i];
126 out_y[i] = poly_y[i];
127 }
128 return 4;
129 }
130
131 double bx[8], by[8];
132 int n = clip_one_plane(
133 poly_x, poly_y, 4, [&](double x, double /*y*/) { return x >= rx0; },
134 [&](double x0, double y0, double x1, double y1, double &ix, double &iy) {
135 const double t = (rx0 - x0) / (x1 - x0);
136 ix = rx0;
137 iy = y0 + t * (y1 - y0);
138 },
139 bx, by);
140 return clip_one_plane(
141 bx, by, n, [&](double x, double /*y*/) { return x <= rx1; },
142 [&](double x0, double y0, double x1, double y1, double &ix, double &iy) {
143 const double t = (rx1 - x0) / (x1 - x0);
144 ix = rx1;
145 iy = y0 + t * (y1 - y0);
146 },
147 out_x, out_y);
148}

150double polygon_clipped_rows_area(const double *x, const double *y, int n,
151 double ry0, double ry1) {
152 if (n == 0)
153 return 0.0;
154 bool all_in = true;
155 bool all_below = true; // every vertex fails y >= ry0
156 bool all_above = ry0 <= ry1; // every vertex has y > ry1
157 for (int i = 0; i < n; ++i) {
158 const bool ge = y[i] >= ry0;
159 const bool le = y[i] <= ry1;
160 all_in = all_in && ge && le;
161 all_below = all_below && !ge;
162 all_above = all_above && y[i] > ry1;
163 }
164 if (all_below || all_above)
165 return 0.0;
166 if (all_in)
167 return shoelace_area(x, y, n);
168
169 double bx[8], by[8];
170 double cx[8], cy[8];
171 int m = clip_one_plane(
172 x, y, n, [&](double /*x*/, double yy) { return yy >= ry0; },
173 [&](double x0, double y0, double x1, double y1, double &ix, double &iy) {
174 const double t = (ry0 - y0) / (y1 - y0);
175 ix = x0 + t * (x1 - x0);
176 iy = ry0;
177 },
178 bx, by);
179 m = clip_one_plane(
180 bx, by, m, [&](double /*x*/, double yy) { return yy <= ry1; },
181 [&](double x0, double y0, double x1, double y1, double &ix, double &iy) {
182 const double t = (ry1 - y0) / (y1 - y0);
183 ix = x0 + t * (x1 - x0);
184 iy = ry1;
185 },
186 cx, cy);
187 return shoelace_area(cx, cy, m);
188}

212// Affine forward map (native canvas), using the frame's checked
213// source_to_canvas (plan 7.2). Returns false if the affine inverse was never
214// established for this frame.
215bool affine_forward(const FrameSamplingTransform &f, double sx, double sy,
216 double &qx, double &qy) {
217 if (!f.source_to_canvas_affine_valid)
218 return false;
219 const auto &s2c = f.source_to_canvas;
220 qx = static_cast<double>(s2c(0, 0)) * sx +
221 static_cast<double>(s2c(0, 1)) * sy + static_cast<double>(s2c(0, 2));
222 qy = static_cast<double>(s2c(1, 0)) * sx +
223 static_cast<double>(s2c(1, 1)) * sy + static_cast<double>(s2c(1, 2));
224 return std::isfinite(qx) && std::isfinite(qy);
225}

244// Builds the exact affine droplet leaf for one source sample: the mapped
245// parallelogram of the square [sx-h,sx+h] x [sy-h,sy+h] (native source
246// space), in internal-canvas coordinates.
247bool build_affine_leaf(const FrameSamplingTransform &f, double sx, double sy,
248 double half, int internal_scale, Leaf &out) {
249 const double corner_sx[4] = {sx - half, sx + half, sx + half, sx - half};
250 const double corner_sy[4] = {sy - half, sy - half, sy + half, sy + half};
251 for (int i = 0; i < 4; ++i) {
252 double qx = 0.0, qy = 0.0;
253 if (!affine_forward(f, corner_sx[i], corner_sy[i], qx, qy))
254 return false;
255 to_internal(qx, qy, internal_scale, out.x[i], out.y[i]);
256 }
257 return true;
258}

260// Every accepted leaf passes both tests, including at maximum depth. The
261// extra midpoint grid is a convergence probe, not another accepted level.
262bool subdivide_local(const FrameSamplingTransform &f, double x0, double y0,
263 double x1, double y1, int depth, int cw, int ch, int scale,
264 const ForwardDrizzleSubdivisionParams &p,
265 const LocalInversionParams &inv,
266 std::vector<Leaf> &leaves) {
267 if (geomstats::registry().enabled)
268 ++geomstats::registry().cur().subdivide_local_calls;
269 double x[3][3], y[3][3];
270 for (int j = 0; j < 3; ++j)
271 for (int i = 0; i < 3; ++i) {
272 double qx, qy;
273 if (!local_forward(f, x0 + (x1 - x0) * i / 2, y0 + (y1 - y0) * j / 2, cw,
274 ch, inv, qx, qy))
275 return false;
276 to_internal(qx, qy, scale, x[j][i], y[j][i]);
277 }
278 Leaf parent{{x[0][0], x[0][2], x[2][2], x[2][0]},
279 {y[0][0], y[0][2], y[2][2], y[2][0]}};
280 double error = 0, child_area = 0;
281 for (int j = 0; j < 3; ++j)
282 for (int i = 0; i < 3; ++i) {
283 const double u = i / 2.0, v = j / 2.0;
284 const double bx = (1 - u) * (1 - v) * parent.x[0] +
285 u * (1 - v) * parent.x[1] + u * v * parent.x[2] +
286 (1 - u) * v * parent.x[3];
287 const double by = (1 - u) * (1 - v) * parent.y[0] +
288 u * (1 - v) * parent.y[1] + u * v * parent.y[2] +
289 (1 - u) * v * parent.y[3];
290 error = std::max(error, std::hypot(x[j][i] - bx, y[j][i] - by));
291 }
292 for (int j = 0; j < 2; ++j)
293 for (int i = 0; i < 2; ++i) {
294 double cx[] = {x[j][i], x[j][i + 1], x[j + 1][i + 1], x[j + 1][i]};
295 double cy[] = {y[j][i], y[j][i + 1], y[j + 1][i + 1], y[j + 1][i]};
296 child_area += shoelace_area(cx, cy, 4);
297 }
298 const double area = shoelace_area(parent.x, parent.y, 4);
299 if (area > 0 && error <= p.position_epsilon_internal_px &&
300 std::abs(child_area - area) / area <= p.area_relative_epsilon) {
301 leaves.push_back(parent);
302 return true;
303 }
304 if (depth >= p.max_subdivision_depth)
305 return false;
306 const size_t before = leaves.size();
307 for (int j = 0; j < 2; ++j)
308 for (int i = 0; i < 2; ++i) {
309 if (!subdivide_local(f, x0 + (x1 - x0) * i / 2, y0 + (y1 - y0) * j / 2,
310 x0 + (x1 - x0) * (i + 1) / 2,
311 y0 + (y1 - y0) * (j + 1) / 2, depth + 1, cw, ch,
312 scale, p, inv, leaves)) {
313 leaves.resize(before);
314 return false; // reject the entire source sample
315 }
316 }
317 return true;
318}

320size_t available_memory_headroom() {
321 size_t available = std::numeric_limits<size_t>::max();
322#ifdef __linux__
323 std::ifstream info("/proc/meminfo");
324 std::string line;
325 while (std::getline(info, line)) {
326 if (line.rfind("MemAvailable:", 0) == 0) {
327 std::istringstream stream(line.substr(13));
328 size_t kb = 0;
329 if (stream >> kb)
330 available = kb * 1024;
331 break;
332 }
333 }
334 // cgroup v2 limit is often tighter than host MemAvailable. Unknown/unlimited
335 // values leave the configured allocation budget in charge.
336 std::ifstream max_file("/sys/fs/cgroup/memory.max"),
337 current_file("/sys/fs/cgroup/memory.current");
338 size_t maximum = 0, current = 0;
339 if ((max_file >> maximum) && (current_file >> current))
340 available = std::min(available, maximum > current ? maximum - current : 0);
341#endif
342 return available;
343}

383DrizzleMemoryPlan
384plan_drizzle_memory(const RegistrationSamplingPlan &plan,
385 const config::ReconstructionDrizzleConfig &cfg,
386 size_t bytes_per_pixel, size_t retained_bytes,
387 bool loads_source) {
388 if (plan.source_width <= 0 || plan.source_height <= 0 ||
389 plan.canvas_width_native <= 0 || plan.canvas_height_native <= 0 ||
390 cfg.internal_scale < 1 || cfg.internal_scale > 2 ||
391 !std::isfinite(cfg.pixfrac) || cfg.pixfrac <= 0 || cfg.pixfrac > 1 ||
392 cfg.kernel != "square" || cfg.chunk_rows < 0 ||
393 cfg.chunk_halo_rows < -1 ||
394 plan.canvas_width_native >
395 std::numeric_limits<int>::max() / cfg.internal_scale ||
396 plan.canvas_height_native >
397 std::numeric_limits<int>::max() / cfg.internal_scale)
398 throw std::invalid_argument("DRIZZLE_INVALID_GEOMETRY");
399 if (plan.color_mode != ColorMode::MONO && plan.color_mode != ColorMode::OSC)
400 throw std::invalid_argument("DRIZZLE_UNSUPPORTED_COLOR_MODE");
401 if (plan.color_mode == ColorMode::OSC &&
402 plan.bayer_pattern == BayerPattern::UNKNOWN)
403 throw std::invalid_argument("DRIZZLE_UNKNOWN_CFA");
404 DrizzleMemoryPlan m;
405 m.width = plan.canvas_width_native * cfg.internal_scale;
406 m.height = plan.canvas_height_native * cfg.internal_scale;
407 m.budget_bytes = checked_product(
408 cfg.memory_budget_mb ? cfg.memory_budget_mb : 512, 1024 * 1024);
409 const size_t available = available_memory_headroom();
410 if (available != std::numeric_limits<size_t>::max())
411 m.budget_bytes = std::min(m.budget_bytes, available - available / 5);
412 size_t fixed =
413 checked_product(plan.frames.size(), sizeof(FrameSamplingTransform) + 128);
414 const size_t source =
415 loads_source ? checked_product(
416 checked_product(plan.source_width, plan.source_height),
417 sizeof(float) * 2)
418 : 0;
419 // Include one source plus a transient load/normalization copy and 1 MiB for
420 // bounded geometry scratch, callbacks and allocator overhead.
421 if (retained_bytes > m.budget_bytes ||
422 source > m.budget_bytes - retained_bytes ||
423 fixed > m.budget_bytes - retained_bytes - source ||
424 m.budget_bytes - retained_bytes - source - fixed < 1024 * 1024) {
425 std::ostringstream msg;
426 msg << "DRIZZLE_MEMORY_BUDGET: retained/source buffers exceed budget "
427 << "(retained=" << retained_bytes / (1024 * 1024) << " MiB, source="
428 << source / (1024 * 1024) << " MiB, fixed=" << fixed / (1024 * 1024)
429 << " MiB, budget=" << m.budget_bytes / (1024 * 1024) << " MiB)";
430 throw std::runtime_error(msg.str());
431 }
432 fixed += retained_bytes + source + 1024 * 1024;
433 const size_t per_row = checked_product(m.width, bytes_per_pixel);
434 const size_t max_rows = per_row ? (m.budget_bytes - fixed) / per_row : 0;
435 if (!max_rows)
436 throw std::runtime_error(
437 "DRIZZLE_MEMORY_BUDGET: one stripe row does not fit");
438 m.rows = static_cast<int>(std::min<size_t>(m.height, max_rows));
439 if (cfg.chunk_rows) {
440 const int requested = std::min(cfg.chunk_rows, m.height);
441 if (static_cast<size_t>(requested) > max_rows)
442 throw std::runtime_error(
443 "DRIZZLE_MEMORY_BUDGET: explicit chunk_rows exceeds budget");
444 m.rows = requested;
445 } else
446 m.rows = std::min(m.rows, 256);
447 m.estimated_peak_bytes = fixed + per_row * m.rows;
448 return m;
449}

451DrizzleMemoryPlan plan_drizzle_memory_autogrow(
452 const registration::RegistrationSamplingPlan &plan,
453 config::ReconstructionDrizzleConfig &cfg, size_t bytes_per_pixel,
454 size_t retained_bytes, bool loads_source,
455 const std::function<void(const std::string &)> &warn) {
456 size_t mb = cfg.memory_budget_mb ? cfg.memory_budget_mb : 512;
457 for (;;) {
458 config::ReconstructionDrizzleConfig attempt = cfg;
459 attempt.memory_budget_mb = mb;
460 try {
461 const auto result = plan_drizzle_memory(plan, attempt, bytes_per_pixel,
462 retained_bytes, loads_source);
463 if (mb != cfg.memory_budget_mb) cfg.memory_budget_mb = mb;
464 return result;
465 } catch (const std::runtime_error &e) {
466 if (std::string(e.what()).rfind("DRIZZLE_MEMORY_BUDGET", 0) != 0)
467 throw;
468 // The effective budget is min(configured, ~80% of the live memory
469 // headroom); once the headroom cap binds, further +1 GiB steps cannot
470 // grow it, so the working set genuinely does not fit.

476 throw;
477 const size_t next_mb = mb + 1024;
478 if (std::min(next_mb * 1024 * 1024, cap) <=
479 std::min(mb * 1024 * 1024, cap))
480 throw;
481 if (warn)
482 warn(std::string(e.what()) +
483 " — raising drizzle memory budget to " +
484 std::to_string(next_mb) + " MiB");
485 mb = next_mb;
486 }
487 }

491prepare_drizzle_frames(const RegistrationSamplingPlan &plan,
492 const config::ReconstructionDrizzleConfig &cfg,
493 const ForwardDrizzleSubdivisionParams &p) {
494 if (p.max_subdivision_depth < 0 || p.max_subdivision_depth > 2 ||
495 !(p.position_epsilon_internal_px > 0) ||
496 !(p.area_relative_epsilon >= 0) ||
497 !(p.per_frame_inversion_error_rate_max >= 0 &&

499 throw std::invalid_argument("DRIZZLE_INVALID_SUBDIVISION");

507 throw std::invalid_argument("DRIZZLE_DUPLICATE_FRAME_ID");
508 if (!f.valid)
509 continue;
510 if (!f.source_to_canvas_affine_valid || !f.source_to_canvas.allFinite())
511 throw std::invalid_argument("DRIZZLE_INVALID_TRANSFORM");

514 // per-sample sweep for this frame and finalised its exclusion rate.
515 // Consume that instead of re-running sample_leaves (§11.14.3: exclusion
516 // counting folds into the build; chunk height does not change the set).
517 if (const auto *cache = active_geometry_cache()) {
518 const auto st = cache->frame_stats(cfg.pixfrac, f.source_index);
519 if (st.present) {
520 result.diagnostics.local_model_samples_total +=
521 static_cast<long long>(st.samples_total);
522 result.diagnostics.local_model_samples_discarded +=
523 static_cast<long long>(st.samples_discarded);
524 if (st.excluded) {
525 result.diagnostics.frames_excluded_subdivision_error_rate

527 continue;
528 }
529 result.frames.push_back(&f);
530 continue;
531 }
532 }

534 cfg.pixfrac);

536 if (geomstats::registry().enabled) {
537 ++geomstats::registry().cur().enumerate_calls;
538 geomstats::registry().cur().source_rows_scanned += plan.source_height;
539 geomstats::registry().cur().source_samples_visited +=
540 static_cast<std::uint64_t>(plan.source_width) * plan.source_height;
541 }

543 plan.source_height,
544 discarded = 0;
545 for (int y = 0; y < plan.source_height; ++y)
546 for (int x = 0; x < plan.source_width; ++x)
547 if (!sample_leaves(plan, f, x, y, cfg.internal_scale, cfg.pixfrac, p,
548 leaves))
549 ++discarded;
550 result.diagnostics.local_model_samples_total += total;
551 result.diagnostics.local_model_samples_discarded += discarded;
552 const double rate = static_cast<double>(discarded) / total;
553 if (rate > p.per_frame_inversion_error_rate_max) {
554 result.diagnostics.frames_excluded_subdivision_error_rate.emplace_back(
555 f.frame_id, rate);
556 continue;
557 }
558 }
559 result.frames.push_back(&f);
560 }
561 std::sort(result.frames.begin(), result.frames.end(),
562 [](auto a, auto b) { return a->source_index < b->source_index; });
563 return result;
564}

566DrizzleSourceScanBox drizzle_source_scan_box(
567 const RegistrationSamplingPlan &plan, const FrameSamplingTransform &f,
568 int scale, int y_begin, int rows, int x_begin, int cols) {
569 const int W_full = plan.canvas_width_native * scale;
570 const int xb = std::clamp(x_begin, 0, W_full);
571 const int xe = cols < 0 ? W_full : std::clamp(x_begin + cols, xb, W_full);
572 DrizzleSourceScanBox box{0, plan.source_height, 0, plan.source_width};
573 if (f.has_smooth_local_model) return box;
574 // Inverse-map the destination-stripe rectangle back into the source, in
575 // BOTH axes (plus a one-source-pixel droplet margin). A source pixel
576 // outside this box maps entirely outside the stripe, so sample_leaves
577 // there would only ever produce leaves whose y/x bbox clamps empty and the
578 // cell sink is never called --- skipping it is bit-identical. This is the
579 // same argument the Y bound already relied on, now extended to X so a frame
580 // that covers only part of the canvas width (rotation, shear, a smaller
581 // frame) does not rescan full source rows.
582 WarpMatrix inverse;
583 if (!registration::invert_affine_2x3(f.source_to_canvas, 1e-12f, 1e12f,
584 inverse))
585 throw std::invalid_argument("DRIZZLE_SINGULAR_TRANSFORM");
586 double lo = std::numeric_limits<double>::infinity(), hi = -lo;
587 double xlo = lo, xhi = hi;
588 // §30.81: inverse-map the TARGET-COLUMN-WINDOW rectangle (not the full
589 // canvas width) so a narrow tile only scans the source pixels that can
590 // reach it. xb/xe collapse to 0/W for the default window.
591 for (double x : {static_cast<double>(xb) / scale,
592 static_cast<double>(xe) / scale})
593 for (double y : {static_cast<double>(y_begin) / scale,
594 static_cast<double>(y_begin + rows) / scale}) {
595 const double sx = inverse(0, 0) * x + inverse(0, 1) * y + inverse(0, 2);
596 const double sy = inverse(1, 0) * x + inverse(1, 1) * y + inverse(1, 2);
597 lo = std::min(lo, sy);
598 hi = std::max(hi, sy);
599 xlo = std::min(xlo, sx);
600 xhi = std::max(xhi, sx);
601 }
602 box.y0 = static_cast<int>(std::clamp(
603 std::floor(lo - 1), 0.0, static_cast<double>(plan.source_height)));
604 box.y1 = static_cast<int>(std::clamp(
605 std::ceil(hi + 1), 0.0, static_cast<double>(plan.source_height)));
606 box.x0 = static_cast<int>(std::clamp(
607 std::floor(xlo - 1), 0.0, static_cast<double>(plan.source_width)));
608 box.x1 = static_cast<int>(std::clamp(
609 std::ceil(xhi + 1), 0.0, static_cast<double>(plan.source_width)));
610 return box;
611}

613void drizzle_affine_source_spans_into(
614 const RegistrationSamplingPlan &plan, const FrameSamplingTransform &f,
615 float pixfrac, int band_y_begin_native, int band_rows_native,
616 std::vector<DrizzleAffineSourceSpan> &out) {
617 out.clear();
618 if (!f.valid || f.has_smooth_local_model ||
619 !f.source_to_canvas_affine_valid || !std::isfinite(pixfrac) ||
620 pixfrac <= 0.0f || band_rows_native <= 0 || band_y_begin_native < 0 ||
621 band_y_begin_native + band_rows_native > plan.canvas_height_native)
622 return;
623 const int sw = plan.source_width, sh = plan.source_height;
624 const double W = static_cast<double>(plan.canvas_width_native);
625 const double y0 = static_cast<double>(band_y_begin_native);
626 const double y1 = y0 + static_cast<double>(band_rows_native);
627 const double a0 = f.source_to_canvas(0, 0), a1 = f.source_to_canvas(0, 1);
628 const double a2 = f.source_to_canvas(0, 2), a3 = f.source_to_canvas(1, 0);
629 const double a4 = f.source_to_canvas(1, 1), a5 = f.source_to_canvas(1, 2);
630 const double half = 0.5 * static_cast<double>(pixfrac);
631 const double rx = half * (std::abs(a0) + std::abs(a1));
632 const double ry = half * (std::abs(a3) + std::abs(a4));
633 const double inf = std::numeric_limits<double>::infinity();
634 for (int sy = 0; sy < sh; ++sy) {
635 const double syc = static_cast<double>(sy) + 0.5;
636 // covers() is the exact per-pixel membership predicate: the four strict
637 // droplet-bbox inequalities evaluated on integer source coordinates.
638 auto covers = [&](int sx) {
639 const double sxc = static_cast<double>(sx) + 0.5;
640 const double qx = a0 * sxc + a1 * syc + a2;
641 const double qy = a3 * sxc + a4 * syc + a5;
642 return qx + rx > 0.0 && qx - rx < W && qy + ry > y0 && qy - ry < y1;
643 };
644 // Analytic interval: each inequality is linear in sx' = sx + 0.5 and is
645 // rewritten as coef * sx' > rhs.
646 double lo = -inf, hi = inf;
647 bool impossible = false;
648 auto add = [&](double coef, double rhs) {
649 if (coef > 0.0)
650 lo = std::max(lo, rhs / coef);
651 else if (coef < 0.0)
652 hi = std::min(hi, rhs / coef);
653 else if (!(0.0 > rhs))
654 impossible = true;
655 };
656 add(a0, -(a1 * syc + a2 + rx)); // qx_max > 0
657 add(-a0, a1 * syc + a2 - rx - W); // qx_min < W
658 add(a3, y0 - ry - a4 * syc - a5); // qy_max > y0
659 add(-a3, a4 * syc + a5 - ry - y1); // qy_min < y1
660 if (impossible) continue;
661 // sx+0.5 in (lo, hi) -> sx in (lo-.5, hi-.5); pad 3 px against rounding,
662 // then covers() decides the true edges (shrink failing boundary pixels,
663 // extend while membership holds).
664 int x0 = std::isinf(lo) ? 0
665 : static_cast<int>(std::floor(lo - 0.5)) - 3;
666 int x1 = std::isinf(hi) ? sw
667 : static_cast<int>(std::ceil(hi - 0.5)) + 3;
668 x0 = std::clamp(x0, 0, sw);
669 x1 = std::clamp(x1, 0, sw);
670 if (x0 >= x1) {
671 // Degenerate (near-)empty analytic interval: probe the bound positions
672 // and midpoint before declaring the row inactive.
673 for (double px : {lo, hi, 0.5 * (lo + hi)}) {
674 if (!std::isfinite(px)) continue;
675 const int p = std::clamp(
676 static_cast<int>(std::lround(px - 0.5)), 0, sw - 1);
677 if (covers(p)) {
678 x0 = p;
679 x1 = p + 1;
680 break;
681 }
682 }
683 if (x0 >= x1) continue;
684 }
685 while (x0 < x1 && !covers(x0)) ++x0;
686 while (x1 > x0 && !covers(x1 - 1)) --x1;
687 while (x0 > 0 && covers(x0 - 1)) --x0;
688 while (x1 < sw && covers(x1)) ++x1;
689 if (x0 < x1) out.push_back({sy, x0, x1});
690 }
691}

702void enumerate_drizzle_stripe_leaf_cells(
703 const RegistrationSamplingPlan &plan, const FrameSamplingTransform &f,
704 int scale, float pixfrac, int y_begin, int rows,
705 const DrizzleLeafCellSink &sink, const ForwardDrizzleSubdivisionParams &p,
706 int x_begin, int cols) {
707 namespace gs = geomstats;
708 const bool instrument = gs::registry().enabled;
709 // §30.81 target-column window. `xb == 0 && xe == W` reproduces the historical
710 // full-width scan exactly (same source box, same bbox clamp, same order).
711 const int W_full = plan.canvas_width_native * scale;
712 const int xb = std::clamp(x_begin, 0, W_full);
713 const int xe = cols < 0 ? W_full : std::clamp(x_begin + cols, xb, W_full);
714
715 // Plan 11.14 P1/P2: for a LOCAL-WARP frame that a published geometry cache
716 // holds, replay the pre-built leaves for this stripe instead of re-running
717 // sample_leaves over the whole source. Bit-identical to the scan below
718 // (same corners, same bbox clamp, same canonical order); adds zero local
719 // basis evaluations. Affine frames and cache misses fall through unchanged.
720 if (f.has_smooth_local_model) {
721 const auto *cache = active_geometry_cache();
722 if (cache && cache->has_frame(pixfrac, f.source_index)) {
723 if (instrument) ++gs::registry().cur().enumerate_calls;
724 if (xb == 0 && xe == W_full) {
725 cache->enumerate_stripe(pixfrac, f.source_index, scale, y_begin, rows,
726 sink);
727 } else {
728 // §30.81: the cache replay contract is full-width; a narrower window
729 // drops the out-of-window cells here. Order and per-cell corners are
730 // untouched, so the retained subset is byte-identical to the windowed
731 // affine scan. (Local-warp frames do not reach the tiled CUDA store
732 // path; this keeps the CPU streaming build correct under a window.)
733 cache->enumerate_stripe(
734 pixfrac, f.source_index, scale, y_begin, rows,
735 [&](int sx, int sy, int c, int lo, int cx, int cy, const double *lx,
736 const double *ly) {
737 if (cx >= xb && cx < xe)
738 sink(sx, sy, c, lo, cx, cy, lx, ly);
739 });
740 }
741 return;
742 }
743 }
744
745 // The scanned source box is shared with every banded source/quality read
746 // (A1): drizzle_source_scan_box reproduces the exact inverse-mapped box the
747 // scan below uses (full extent for local-warp frames).
748 const auto scan_box =
749 drizzle_source_scan_box(plan, f, scale, y_begin, rows, x_begin, cols);
750 const int source_y0 = scan_box.y0, source_y1 = scan_box.y1;
751 const int source_x0 = scan_box.x0, source_x1 = scan_box.x1;
752 if (instrument) {
753 auto &c = gs::registry().cur();
754 ++c.enumerate_calls;
755 c.source_rows_scanned +=
756 static_cast<std::uint64_t>(std::max(0, source_y1 - source_y0));
757 c.source_samples_visited +=
758 static_cast<std::uint64_t>(std::max(0, source_y1 - source_y0)) *
759 static_cast<std::uint64_t>(std::max(0, source_x1 - source_x0));
760 }
761 std::vector<Leaf> leaves;
762 leaves.reserve(16);
763 // Hoisted out of the per-source-pixel loop (redundant-reload analysis C5):
764 // invariant for the whole call. `sink` is an opaque std::function the
765 // compiler cannot assume leaves `plan` (a const-ref alias) unmodified
766 // across the call, so without this it must re-read every iteration.
767 const bool is_osc = plan.color_mode == ColorMode::OSC;
768 const BayerPattern bayer_pattern = plan.bayer_pattern;
769 const int cfa_origin_x = plan.cfa_origin_x;
770 const int cfa_origin_y = plan.cfa_origin_y;
771 for (int sy = source_y0; sy < source_y1; ++sy)
772 for (int sx = source_x0; sx < source_x1; ++sx) {
773 if (!sample_leaves(plan, f, sx, sy, scale, pixfrac, p, leaves))
774 continue;
775 int c = 0;
776 if (is_osc) {
777 const auto channel = cfa_channel_for_source_pixel(
778 sx, sy, bayer_pattern, cfa_origin_x, cfa_origin_y);
779 c = channel == CfaChannel::R ? 0 : channel == CfaChannel::G ? 1 : 2;
780 }
781 for (size_t li = 0; li < leaves.size(); ++li) {
782 const auto &leaf = leaves[li];
783 double xmin = *std::min_element(leaf.x, leaf.x + 4),
784 xmax = *std::max_element(leaf.x, leaf.x + 4);
785 double ymin = *std::min_element(leaf.y, leaf.y + 4),
786 ymax = *std::max_element(leaf.y, leaf.y + 4);
787 int x0 = static_cast<int>(std::clamp(
788 std::floor(xmin), static_cast<double>(xb), static_cast<double>(xe)));
789 int x1 = static_cast<int>(std::clamp(
790 std::ceil(xmax), static_cast<double>(xb), static_cast<double>(xe)));
791 int y0 = static_cast<int>(
792 std::clamp(std::floor(ymin), static_cast<double>(y_begin),
793 static_cast<double>(y_begin + rows)));
794 int y1 = static_cast<int>(
795 std::clamp(std::ceil(ymax), static_cast<double>(y_begin),
796 static_cast<double>(y_begin + rows)));
797 for (int y = y0; y < y1; ++y)
798 for (int x = x0; x < x1; ++x) {
799 if (instrument)
800 ++gs::registry().cur().leaf_cells_emitted;
801 sink(sx, sy, c, static_cast<int>(li), x, y, leaf.x, leaf.y);
802 }
803 }
804 }

832// x-bbox of a convex quad intersected with the horizontal slab [ylo, yhi].
833// Returns false when the intersection is empty. Used only to bound which
834// internal cells of a stripe row need explicit classification against the
835// dense footprint parallelogram --- the classification itself is exact.
836bool convex_quad_slab_x_range(const double *px, const double *py, double ylo,
837 double yhi, double &xlo, double &xhi) {
838 xlo = std::numeric_limits<double>::infinity();
839 xhi = -xlo;
840 for (int i = 0; i < 4; ++i) {
841 const int j = (i + 1) & 3;
842 const double ax = px[i], ay = py[i], bx = px[j], by = py[j];
843 if (ay >= ylo && ay <= yhi) {
844 xlo = std::min(xlo, ax);
845 xhi = std::max(xhi, ax);
846 }
847 for (double yc : {ylo, yhi}) {
848 if ((ay < yc) != (by < yc)) {
849 const double t = (yc - ay) / (by - ay);
850 const double xc = ax + t * (bx - ax);
851 xlo = std::min(xlo, xc);
852 xhi = std::max(xhi, xc);
853 }
854 }
855 }
856 return xlo <= xhi;
857}

861void dense_footprint_touched_stripe(const RegistrationSamplingPlan &plan,
862 const FrameSamplingTransform &f, int scale,
863 int y_begin, int rows,
864 std::vector<std::uint8_t> &touched,
865 bool force_exact) {
866 const int W = plan.canvas_width_native * scale;
867 const size_t n = static_cast<size_t>(W) * static_cast<size_t>(std::max(0, rows));
868 touched.assign(n, 0);
869 if (rows <= 0 || W <= 0)
870 return;
871
872 const ForwardDrizzleSubdivisionParams p{};
873
874 // The exact reference path, kept verbatim: it is the definition this routine
875 // must reproduce byte-for-byte.
876 auto exact_reference = [&] {
877 rasterize_drizzle_stripe(
878 plan, f, scale, 1.0f, y_begin, rows,
879 [&](int, int, int, int, size_t i, double) { touched[i] = 1; }, p);
880 };
881
882 if (force_exact || f.has_smooth_local_model ||
883 !f.source_to_canvas_affine_valid) {
884 exact_reference();
885 return;
886 }
887
888 // P = affine_f([0, W_src] x [0, H_src]) in internal-canvas coordinates: the
889 // union of every unshrunk source-pixel square's image.
890 const double src_x[4] = {0.0, static_cast<double>(plan.source_width),
891 static_cast<double>(plan.source_width), 0.0};
892 const double src_y[4] = {0.0, 0.0, static_cast<double>(plan.source_height),
893 static_cast<double>(plan.source_height)};
894 double Px[4], Py[4];
895 for (int k = 0; k < 4; ++k) {
896 double qx = 0.0, qy = 0.0;
897 if (!affine_forward(f, src_x[k], src_y[k], qx, qy)) {
898 exact_reference();
899 return;
900 }
901 to_internal(qx, qy, scale, Px[k], Py[k]);
902 }
903
904 WarpMatrix inverse;
905 if (!registration::invert_affine_2x3(f.source_to_canvas, 1e-12f, 1e12f,
906 inverse)) {
907 exact_reference(); // matches enumerate_drizzle_stripe_leaf_cells' throw
908 return; // path being taken by the reference rasterize
909 }
910
911 // Inward unit normals + a support vertex for each of P's four edges. Signed
912 // distance dot(n_e, q - v_e) is >= 0 for points inside that edge's half-plane.
913 const double cx = 0.25 * (Px[0] + Px[1] + Px[2] + Px[3]);
914 const double cy = 0.25 * (Py[0] + Py[1] + Py[2] + Py[3]);
915 double nx[4], ny[4];
916 for (int e = 0; e < 4; ++e) {
917 const int j = (e + 1) & 3;
918 double ex = Px[j] - Px[e], ey = Py[j] - Py[e];
919 double cand_x = -ey, cand_y = ex;
920 const double len = std::hypot(cand_x, cand_y);
921 if (!(len > 0.0)) { // degenerate edge -> fall back wholesale
922 exact_reference();
923 return;
924 }
925 cand_x /= len;
926 cand_y /= len;
927 if (cand_x * (cx - Px[e]) + cand_y * (cy - Py[e]) < 0.0) {
928 cand_x = -cand_x;
929 cand_y = -cand_y;
930 }
931 nx[e] = cand_x;
932 ny[e] = cand_y;
933 }
934 auto signed_dist = [&](int e, double x, double y) {
935 return nx[e] * (x - Px[e]) + ny[e] * (y - Py[e]);
936 };
937
938 // Interior margin + soundness of the interior fast-fill.
939 //
940 // `ext` bounds one mapped source-pixel square's internal-canvas extent along
941 // either axis: ext = scale * max(|a|+|b|, |c|+|d|). A cell whose four corners
942 // are all >= `margin = 2*ext + 3` inside every edge of P is:
943 // (1) fully tiled by mapped source-pixel squares --- every point of the
944 // cell is farther from dP than one square's diameter, and the squares
945 // tile P (affine image of a partition), so the cell lies in the union;
946 // (2) touched by the reference: the unit cell's area (= 1) is partitioned
947 // among the tiles that meet it; at most (1+ext)^2 tiles can meet a unit
948 // cell (their bounding boxes are ext x ext), so some tile contributes
949 // >= 1/(1+ext)^2 of the cell. With ext <= 64 that share is >= ~2.4e-4,
950 // far above any double-precision rounding in
951 // polygon_rectangle_intersection_area --- it returns > 0, and the
952 // reference sets touched = 1.
953 // This holds for any non-singular linear part (invert_affine_2x3 already
954 // rejected singular ones above); it does not need a near-unit determinant.
955 // ext > 64 (a large scale-up / extreme shear) is bounced to the exact path
956 // so the 1/(1+ext)^2 argument always applies.
957 const auto &s2c = f.source_to_canvas;
958 const double lin_x = std::fabs(static_cast<double>(s2c(0, 0))) +
959 std::fabs(static_cast<double>(s2c(0, 1)));
960 const double lin_y = std::fabs(static_cast<double>(s2c(1, 0))) +
961 std::fabs(static_cast<double>(s2c(1, 1)));
962 const double ext = static_cast<double>(scale) * std::max(lin_x, lin_y);
963 if (!(ext > 0.0) || ext > 64.0) {
964 exact_reference();
965 return;
966 }
967 const double margin = 2.0 * ext + 3.0;
968
969 std::vector<Leaf> leaves;
970 leaves.reserve(4);
971
972 // Exact per-cell fallback for a boundary cell: replay the reference test
973 // (sample_leaves + polygon_rectangle_intersection_area > 0) over exactly the
974 // source pixels whose unshrunk square can reach this cell. The inverse-mapped
975 // cell rectangle plus a one-source-pixel margin is the same superset rule
976 // enumerate_drizzle_stripe_leaf_cells already relies on for its stripe bound.
977 auto boundary_cell_touched = [&](int gx, int gy) -> bool {
978 double slo_x = std::numeric_limits<double>::infinity(), shi_x = -slo_x;
979 double slo_y = slo_x, shi_y = shi_x;
980 for (double dx : {static_cast<double>(gx) - 1.0,
981 static_cast<double>(gx) + 2.0})
982 for (double dy : {static_cast<double>(gy) - 1.0,
983 static_cast<double>(gy) + 2.0}) {
984 const double xn = dx / scale, yn = dy / scale;
985 const double sx = inverse(0, 0) * xn + inverse(0, 1) * yn + inverse(0, 2);
986 const double sy = inverse(1, 0) * xn + inverse(1, 1) * yn + inverse(1, 2);
987 slo_x = std::min(slo_x, sx);
988 shi_x = std::max(shi_x, sx);
989 slo_y = std::min(slo_y, sy);
990 shi_y = std::max(shi_y, sy);
991 }
992 const int sx0 = static_cast<int>(std::clamp(
993 std::floor(slo_x - 1.0), 0.0, static_cast<double>(plan.source_width)));
994 const int sx1 = static_cast<int>(std::clamp(
995 std::ceil(shi_x + 1.0), 0.0, static_cast<double>(plan.source_width)));
996 const int sy0 = static_cast<int>(std::clamp(
997 std::floor(slo_y - 1.0), 0.0, static_cast<double>(plan.source_height)));
998 const int sy1 = static_cast<int>(std::clamp(
999 std::ceil(shi_y + 1.0), 0.0, static_cast<double>(plan.source_height)));
1000 for (int sy = sy0; sy < sy1; ++sy)
1001 for (int sx = sx0; sx < sx1; ++sx) {
1002 if (!sample_leaves(plan, f, sx, sy, scale, 1.0f, p, leaves))
1003 continue;
1004 for (const auto &L : leaves)
1005 if (polygon_rectangle_intersection_area(L.x, L.y, gx, gy, gx + 1.0,
1006 gy + 1.0) > 0)
1007 return true;
1008 }
1009 return false;
1010 };
1011
1012 for (int gy = y_begin; gy < y_begin + rows; ++gy) {
1013 double xr_lo, xr_hi;
1014 if (!convex_quad_slab_x_range(Px, Py, static_cast<double>(gy),
1015 static_cast<double>(gy) + 1.0, xr_lo, xr_hi))
1016 continue; // P does not reach this stripe row at all
1017 int gx0 = static_cast<int>(
1018 std::clamp(std::floor(xr_lo), 0.0, static_cast<double>(W)));
1019 int gx1 = static_cast<int>(
1020 std::clamp(std::ceil(xr_hi), 0.0, static_cast<double>(W)));
1021 const size_t row_off = static_cast<size_t>(gy - y_begin) * W;

1055 const RegistrationSamplingPlan &plan, const SourceImageProvider &source_of,
1056 const config::ReconstructionDrizzleConfig &cfg,
1057 const UniformStripeSink &sink,
1058 const ForwardDrizzleSubdivisionParams &subdivision, size_t retained_bytes,
1059 const SourceImageRectProvider &source_rect_of) {
1060 const int channels = plan.color_mode == ColorMode::MONO ? 1 : 3;
1061 const auto memory = plan_drizzle_memory(
1062 plan, cfg, channels * (5 * sizeof(double) + 13), retained_bytes);
1063 auto prepared = prepare_drizzle_frames(plan, cfg, subdivision);
1064 auto diag = prepared.diagnostics;
1065 diag.estimated_peak_bytes = memory.estimated_peak_bytes;
1066 diag.resolved_chunk_rows = memory.rows;
1067 {
1068 int local_n = 0;
1069 for (const auto *pf : prepared.frames)
1070 if (pf->has_smooth_local_model)
1071 ++local_n;
1072 geomstats::stamp_context(plan.source_width, plan.source_height,
1073 plan.canvas_width_native,
1074 plan.canvas_height_native, cfg.internal_scale,
1075 memory.rows, memory.height,
1076 static_cast<int>(prepared.frames.size()), local_n);
1077 }
1078 // Hoisted out of the stripe loop (redundant-reload analysis D2): these were
1079 // previously declared fresh every stripe iteration, forcing a malloc/free
1080 // per stripe per array. `n` shrinks only for the final (possibly partial)
1081 // stripe, so hoisting the std::array<vector> declarations lets .assign(n,0)
1082 // below reuse each vector's already-grown capacity across stripes instead
1083 // of reallocating. Bit-identical: same values assigned, same n per stripe.
1084 std::array<std::vector<double>, 3> wx, w, w2, A, B;
1085 // D3: the stripe result object is reused across stripes (sinks consume it
1086 // synchronously and may not retain it); ProfilePlane::allocate assigns, so
1087 // the plane vectors keep their grown capacity instead of reallocating per
1088 // stripe.
1089 ForwardDrizzleUniformResult stripe;
1090 for (int y = 0; y < memory.height; y += memory.rows) {
1091 const int rows = std::min(memory.rows, memory.height - y);
1092 const size_t n = static_cast<size_t>(memory.width) * rows;
1093 for (int c = 0; c < channels; ++c) {
1094 wx[c].assign(n, 0);
1095 w[c].assign(n, 0);
1096 w2[c].assign(n, 0);
1097 A[c].assign(n, 0);
1098 B[c].assign(n, 0);
1099 }
1100 for (const auto *f : prepared.frames) {
1101 // A3: with a banded source provider only the stripe's inverse-mapped
1102 // source box is read; without it the historical full-frame load runs.
1103 Matrix2Df src_rect;
1104 const Matrix2Df *src_p = nullptr;
1105 int src_yo = 0, src_xo = 0;
1106 if (source_rect_of) {
1107 const auto box = drizzle_source_scan_box(
1108 plan, *f, cfg.internal_scale, y, rows, 0, -1);
1109 if (box.y1 <= box.y0 || box.x1 <= box.x0)
1110 continue; // no scanned source pixels -> no contributions
1111 src_rect = source_rect_of(f->source_index, box.y0, box.y1, box.x0,
1112 box.x1);
1113 if (src_rect.rows() != box.y1 - box.y0 ||
1114 src_rect.cols() != box.x1 - box.x0)
1115 throw std::invalid_argument("DRIZZLE_SOURCE_SHAPE_MISMATCH");
1116 src_yo = box.y0;
1117 src_xo = box.x0;
1118 src_p = &src_rect;
1119 } else {
1120 src_p = &source_of(f->source_index);
1121 if (src_p->rows() != plan.source_height ||
1122 src_p->cols() != plan.source_width)
1123 throw std::invalid_argument("DRIZZLE_SOURCE_SHAPE_MISMATCH");
1124 }
1125 const Matrix2Df &source = *src_p;
1126 for (int c = 0; c < channels; ++c) {
1127 std::fill(A[c].begin(), A[c].end(), 0);
1128 std::fill(B[c].begin(), B[c].end(), 0);
1129 }
1130 {
1131 geomstats::ScopedVariant _v(
1132 geomstats::Variant::kUniformDiagnostic, cfg.pixfrac);
1133 geomstats::ScopedGeometryTimer _t;
1134 rasterize_drizzle_stripe(
1135 plan, *f, cfg.internal_scale, cfg.pixfrac, y, rows,
1136 [&](int sx, int sy, int c, int /*leaf*/, size_t i, double k) {
1137 const double v = source(sy - src_yo, sx - src_xo);
1138 if (std::isfinite(v)) {
1139 A[c][i] += k * v;
1140 B[c][i] += k;
1141 }
1142 },
1143 subdivision);
1144 }
1145 for (int c = 0; c < channels; ++c)
1146 for (size_t i = 0; i < n; ++i)
1147 if (B[c][i] > 0) {
1148 wx[c][i] += A[c][i];
1149 w[c][i] += B[c][i];
1150 w2[c][i] += B[c][i] * B[c][i];
1151 }
1152 }
1153 stripe.color_mode = plan.color_mode;
1154 stripe.internal_width = memory.width;
1155 stripe.internal_height = rows;
1156 std::array<ProfilePlane *, 3> planes =
1157 channels == 1
1158 ? std::array<ProfilePlane *, 3>{&stripe.L, nullptr, nullptr}
1159 : std::array<ProfilePlane *, 3>{&stripe.R, &stripe.G, &stripe.B};
1160 for (int c = 0; c < channels; ++c) {
1161 auto &plane = *planes[c];
1162 plane.allocate(memory.width, rows);
1163 for (size_t i = 0; i < n; ++i)
1164 if (w[c][i] > 0) {
1165 plane.value[i] = wx[c][i] / w[c][i];
1166 plane.weight_sum[i] = w[c][i];
1167 plane.n_eff[i] = w[c][i] * w[c][i] / w2[c][i];
1168 plane.support[i] = 1;
1169 }
1170 }
1171 sink(y, stripe);
1172 }
1173 return diag;
1174}
