20// Fit an elliptical 2D Gaussian proxy using weighted second central moments.
21// The principal-axis sigmas come from the covariance eigenvalues.
26static PsfFit2D fit_elliptical_psf_2d(const cv::Mat& patch, float bg) {
27 PsfFit2D out;
28 if (patch.empty()) return out;
29
30 constexpr double kFwhmScale = 2.3548200450309493; // 2*sqrt(2*ln(2))
31
32 double sum_w = 0.0;
33 double mx = 0.0;
34 double my = 0.0;
35 for (int y = 0; y < patch.rows; ++y) {
36 const float* row = patch.ptr<float>(y);
37 for (int x = 0; x < patch.cols; ++x) {
38 const double w = std::max(0.0, static_cast<double>(row[x]) - static_cast<double>(bg));
39 sum_w += w;
40 mx += w * static_cast<double>(x);
41 my += w * static_cast<double>(y);
42 }
43 }
44 if (!(sum_w > 0.0)) return out;
45
46 mx /= sum_w;
47 my /= sum_w;
48
49 double cxx = 0.0;
50 double cyy = 0.0;
51 double cxy = 0.0;
52 for (int y = 0; y < patch.rows; ++y) {
53 const float* row = patch.ptr<float>(y);
54 for (int x = 0; x < patch.cols; ++x) {
55 const double w = std::max(0.0, static_cast<double>(row[x]) - static_cast<double>(bg));
56 if (!(w > 0.0)) continue;
57 const double dx = static_cast<double>(x) - mx;
58 const double dy = static_cast<double>(y) - my;
59 cxx += w * dx * dx;
60 cyy += w * dy * dy;
61 cxy += w * dx * dy;
62 }
63 }
64
65 cxx /= sum_w;
66 cyy /= sum_w;
67 cxy /= sum_w;
68
69 const double tr = cxx + cyy;
70 const double det = cxx * cyy - cxy * cxy;
71 const double disc = std::max(0.0, tr * tr - 4.0 * det);
72 const double root = std::sqrt(disc);
73 const double lambda_major = 0.5 * (tr + root);
74 const double lambda_minor = 0.5 * (tr - root);
75 if (!(lambda_major > 0.0) || !(lambda_minor > 0.0)) return out;
76
77 const double fwhm_major = kFwhmScale * std::sqrt(lambda_major);
78 const double fwhm_minor = kFwhmScale * std::sqrt(lambda_minor);
79 if (!(fwhm_major > 0.2 && fwhm_major < 50.0 &&
80 fwhm_minor > 0.2 && fwhm_minor < 50.0)) {
81 return out;
82 }
83
84 out.fwhm_major = static_cast<float>(fwhm_major);
85 out.fwhm_minor = static_cast<float>(fwhm_minor);
86 return out;
87}

110// Measure FWHM, roundness, and contrast from a small patch around a star.
111// Returns valid=true only if both X and Y fits succeed.
116bool measure_star_patch(const cv::Mat& tile_cv, const cv::Point2f& pt,
117 int patch_radius, float tile_bg, float tile_sigma,
118 StarMeasurement& out) {
119 int cx = static_cast<int>(std::round(pt.x));
120 int cy = static_cast<int>(std::round(pt.y));
121 int psz = 2 * patch_radius + 1;
122 int x0 = cx - patch_radius;
123 int y0 = cy - patch_radius;
124
125 if (x0 < 0 || y0 < 0 || (x0 + psz) > tile_cv.cols ||
126 (y0 + psz) > tile_cv.rows)
127 return false;
128
129 cv::Mat patch = tile_cv(cv::Rect(x0, y0, psz, psz));
130
131 // Find peak in patch
132 double maxv = 0.0;
133 cv::Point peak_loc;
134 cv::minMaxLoc(patch, nullptr, &maxv, nullptr, &peak_loc);
135
136 // Reject if peak is not significantly above background
137 if (maxv <= static_cast<double>(tile_bg) + 3.0 * static_cast<double>(tile_sigma))
138 return false;
139
140 const PsfFit2D fit = fit_elliptical_psf_2d(patch, tile_bg);
141 float fx = fit.fwhm_major;
142 float fy = fit.fwhm_minor;
143
144 if (!(fx > 0.0f) || !(fy > 0.0f))
145 return false;
146
147 out.fwhm_x = fx;
148 out.fwhm_y = fy;
149 out.fwhm = std::sqrt(fx * fy);
150 out.roundness = std::min(fx, fy) / std::max(fx, fy);
151 float peak_flux = static_cast<float>(maxv);
152 out.contrast = (tile_bg > 1e-8f) ? (peak_flux / tile_bg) : 0.0f;
153 return true;
154}

288 m.roundness = core::median_of(scratch.tmp);
289 scratch.tmp = scratch.contrasts;

291 }

294}
