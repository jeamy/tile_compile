15PositiveMedianEstimate positive_median(const Matrix2Df &src, bool abs_values,
16 float center) {
17 PositiveMedianEstimate out;
18 if (src.size() <= 0) {
19 return out;
20 }
21 out.total_count = static_cast<size_t>(src.size());
22 std::vector<float> values;
23 values.reserve(out.total_count);
24 for (Eigen::Index i = 0; i < src.size(); ++i) {
25 const float v = src.data()[i];
26 if (!(std::isfinite(v) && v > 0.0f)) {
27 continue;

30 }

37 values.end());
38 out.value = values[mid];
39 return out;
40}
