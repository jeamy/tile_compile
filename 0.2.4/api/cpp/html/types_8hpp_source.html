42inline BayerPattern string_to_bayer_pattern(const std::string& s) {
43 std::string norm = s;
44 auto not_space = [](unsigned char c) { return !std::isspace(c); };
45 norm.erase(norm.begin(),
46 std::find_if(norm.begin(), norm.end(), not_space));
47 norm.erase(std::find_if(norm.rbegin(), norm.rend(), not_space).base(),
48 norm.end());
49 std::transform(norm.begin(), norm.end(), norm.begin(),
50 [](unsigned char c) { return static_cast<char>(std::toupper(c)); });
51
52 if (norm == "RGGB") return BayerPattern::RGGB;
53 if (norm == "BGGR") return BayerPattern::BGGR;
54 if (norm == "GRBG") return BayerPattern::GRBG;
55 if (norm == "GBRG") return BayerPattern::GBRG;
56 return BayerPattern::UNKNOWN;
57}
