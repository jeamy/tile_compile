77std::string trim_trailing_zeros(std::string text) {
78 const auto dot = text.find('.');
79 if (dot == std::string::npos) {
80 return text;
81 }
82 while (!text.empty() && text.back() == '0') {
83 text.pop_back();
84 }
85 if (!text.empty() && text.back() == '.') {
86 text.pop_back();
87 }
88 if (text == "-0") {
89 return "0";
90 }
91 return text.empty() ? "0" : text;

98std::string format_config_float_scalar(double value) {
99 if (!std::isfinite(value)) {
100 return "0";
101 }
102
103 const double rounded = std::round(value * 100.0) / 100.0;
104 if (rounded == 0.0 && value != 0.0 && std::fabs(value) < 0.01) {
105 std::ostringstream oss;
106 oss << std::scientific << std::setprecision(2) << value;
107 return oss.str();
108 }
109
110 std::ostringstream oss;
111 oss << std::fixed << std::setprecision(2) << rounded;
112 return trim_trailing_zeros(oss.str());

119void round_yaml_numeric_scalars_inplace(YAML::Node node) {
120 if (!node || node.IsNull()) {
121 return;
122 }
123 if (node.IsMap()) {
124 for (auto it = node.begin(); it != node.end(); ++it) {
125 round_yaml_numeric_scalars_inplace(it->second);
126 }
127 return;
128 }
129 if (node.IsSequence()) {
130 for (auto it = node.begin(); it != node.end(); ++it) {
131 round_yaml_numeric_scalars_inplace(*it);
132 }
133 return;
134 }
135 if (!node.IsScalar()) {
136 return;
137 }
138
139 const std::string raw = node.Scalar();
140 if (!scalar_looks_like_float(raw)) {
141 return;
142 }
143
144 char *end = nullptr;
145 errno = 0;
146 const double value = std::strtod(raw.c_str(), &end);
147 if (errno != 0 || end == raw.c_str() || (end && *end != '\0') ||
148 !std::isfinite(value)) {
149 return;
150 }
151 node = format_config_float_scalar(value);
152}
