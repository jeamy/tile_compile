32std::optional<double> extract_exposure_seconds(const io::FitsHeader &header) {
33 auto read_positive = [&](const char *key) -> std::optional<double> {
34 if (auto value = header.get_double(key);
35 value && std::isfinite(*value) && *value > 0.0) {
36 return value;
37 }
38 if (auto value = header.get_int(key); value && *value > 0) {
39 return static_cast<double>(*value);
40 }
41 if (auto value = header.get_string(key)) {
42 char *end = nullptr;
43 errno = 0;
44 const double d = std::strtod(value->c_str(), &end);
45 if (errno == 0 && end != value->c_str() && *end == '\0' &&
46 std::isfinite(d) && d > 0.0) {
47 return d;
48 }
49 }
50 return std::nullopt;
51 };
52
53 for (const char *key : {"EXPTIME", "EXPOSURE", "EXPOSURETIME",
54 "EXPOSURE_TIME", "EXP_TIME", "DURATION", "EXPOS"}) {
55 if (auto value = read_positive(key)) {
56 return value;
57 }
58 }
59 return std::nullopt;
60}

638 global_metrics_worker();
639 }
640
641 if (gm_failed.load(std::memory_order_relaxed)) {
642 emitter.phase_end(run_id, Phase::GLOBAL_METRICS, "error",
643 {{"error", gm_error.empty() ? "unknown_error"
644 : gm_error}},
645 log_file);
646 emitter.run_end(run_id, false, "error", log_file);
647 std::cerr << "Error during GLOBAL_METRICS: "
648 << (gm_error.empty() ? "unknown_error" : gm_error) << std::endl;
649 return false;
650 }
651
652 // Determine reference star count (max) and recompute wFWHM
653 for (const auto &sm : frame_star_metrics) {
654 if (sm.star_count > ref_star_count)
655 ref_star_count = sm.star_count;
656 }
657 if (ref_star_count > 0) {
658 for (auto &sm : frame_star_metrics) {
659 if (sm.star_count > 0 && sm.fwhm > 0) {
660 sm.wfwhm = sm.fwhm * static_cast<float>(ref_star_count) /
661 static_cast<float>(sm.star_count);
662 }
663 }
664 }
665
666 out.global_weights = metrics::calculate_global_weights(
667 frame_metrics, cfg.global_metrics.weights.background,
668 cfg.global_metrics.weights.noise, cfg.global_metrics.weights.gradient,
669 cfg.global_metrics.clamp[0], cfg.global_metrics.clamp[1],
670 cfg.global_metrics.adaptive_weights,
671 cfg.global_metrics.weight_exponent_scale);
672 auto &global_weights = out.global_weights;
673
674 {
675 core::json artifact;
676 artifact["metrics"] = core::json::array();
677 for (size_t i = 0; i < frame_metrics.size(); ++i) {
678 core::json m;
679 m["background"] = frame_metrics[i].background;
680 m["noise"] = frame_metrics[i].noise;
681 m["gradient_energy"] = frame_metrics[i].gradient_energy;
682 m["quality_score"] = frame_metrics[i].quality_score;
683 m["global_weight"] = (i < static_cast<size_t>(global_weights.size()))
684 ? global_weights[static_cast<int>(i)]
685 : 0.0f;
686 m["fwhm"] = frame_star_metrics[i].fwhm;
687 m["fwhm_x"] = frame_star_metrics[i].fwhm_x;
688 m["fwhm_y"] = frame_star_metrics[i].fwhm_y;
689 m["roundness"] = frame_star_metrics[i].roundness;
690 m["wfwhm"] = frame_star_metrics[i].wfwhm;
691 m["star_count"] = frame_star_metrics[i].star_count;
692 artifact["metrics"].push_back(m);
693 }
694
695 artifact["weights"] = {
696 {"background", cfg.global_metrics.weights.background},
697 {"noise", cfg.global_metrics.weights.noise},
698 {"gradient", cfg.global_metrics.weights.gradient}};
699 artifact["clamp"] = {cfg.global_metrics.clamp[0],
700 cfg.global_metrics.clamp[1]};
701 artifact["adaptive_weights"] = cfg.global_metrics.adaptive_weights;
702 artifact["adaptive_weighting_method"] =
703 "leave_one_out_positive_correlation_squared";
704 artifact["adaptive_weighting_target"] =
705 "For each metric, predict the leave-one-out consensus of the other two "
706 "higher-is-better normalized signals using static weights renormalized "
707 "over the remaining metrics.";
708 artifact["adaptive_weighting_tie_break"] =
709 "If utilities are degenerate or nearly tied, keep the static weights. "
710 "Otherwise clip adaptive weights to [0.1,0.7] and renormalize.";
711 core::write_text(run_dir / "artifacts" / "global_metrics.json",
712 artifact.dump(2));
713 }
714
715 emitter.phase_end(run_id, Phase::GLOBAL_METRICS, "ok",
716 {
717 {"num_frames", static_cast<int>(frame_metrics.size())},
718 },
719 log_file);
720
721 return true;
722}
