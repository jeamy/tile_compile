25Matrix2Df downsample2x2_mean(const Matrix2Df &in) {
26 const int h = in.rows();
27 const int w = in.cols();
28 const int h2 = h - (h % 2);

36 const int sx = x * 2;

41 out(y, x) = 0.25f * (a + b + c + d);

58// Invert a 2×3 affine warp matrix: given M→R, return R→M (or vice versa).
63static WarpMatrix invert_warp_2x3(const WarpMatrix &w) {
64 const float a00 = w(0, 0), a01 = w(0, 1), tx = w(0, 2);
65 const float a10 = w(1, 0), a11 = w(1, 1), ty = w(1, 2);
66 const float det = a00 * a11 - a01 * a10;
67 if (std::fabs(det) < 1e-12f)
68 return w; // degenerate — return as-is
69 const float inv_det = 1.0f / det;
70 WarpMatrix inv;
71 inv(0, 0) = a11 * inv_det;
72 inv(0, 1) = -a01 * inv_det;
73 inv(1, 0) = -a10 * inv_det;
74 inv(1, 1) = a00 * inv_det;
75 inv(0, 2) = -(inv(0, 0) * tx + inv(0, 1) * ty);
76 inv(1, 2) = -(inv(1, 0) * tx + inv(1, 1) * ty);
77 return inv;
78}

88float estimate_rotation_logpolar(const cv::Mat &ref, const cv::Mat &mov) {

97 cv::magnitude(planes_ref[0], planes_ref[1], mag_ref);
98 cv::magnitude(planes_mov[0], planes_mov[1], mag_mov);
99

101 mag_mov += 1.0e-9f;

119cv::Mat to_uint8_stretch(const Matrix2Df &src) {
120 cv::Mat f(src.rows(), src.cols(), CV_32F, const_cast<float *>(src.data()));
121 std::vector<float> vals;
122 vals.reserve(static_cast<size_t>(src.size()));
123 for (int r = 0; r < f.rows; ++r) {
124 const float *p = f.ptr<float>(r);
125 vals.insert(vals.end(), p, p + f.cols);
126 }
127 if (vals.empty())
128 return cv::Mat();
129 const size_t n = vals.size();
130 auto nth = [&](size_t k) {
131 std::nth_element(vals.begin(), vals.begin() + k, vals.end());
132 return vals[k];
133 };
134 float lo = nth(static_cast<size_t>(0.01 * n));
135 float hi = nth(static_cast<size_t>(0.99 * n));
136 if (hi <= lo)
137 hi = lo + 1.0f;
138 cv::Mat out;
139 f.convertTo(out, CV_8U, 255.0 / (hi - lo), -255.0 * lo / (hi - lo));

165static bool invert_forward_affine_to_warp(const cv::Mat &A,
166 bool allow_rotation, WarpMatrix &warp,
167 std::string *error_message = nullptr) {
168 if (A.empty()) {
169 if (error_message) {
170 *error_message = "transform_fail";
171 }
172 return false;
173 }
174
175 float a00_fw = static_cast<float>(A.at<double>(0, 0));
176 float a01_fw = static_cast<float>(A.at<double>(0, 1));
177 float a10_fw = static_cast<float>(A.at<double>(1, 0));
178 float a11_fw = static_cast<float>(A.at<double>(1, 1));
179 float tx_fw = static_cast<float>(A.at<double>(0, 2));
180 float ty_fw = static_cast<float>(A.at<double>(1, 2));
181
182 if (!allow_rotation) {
183 a00_fw = 1.0f;
184 a01_fw = 0.0f;
185 a10_fw = 0.0f;
186 a11_fw = 1.0f;
187 }
188
189 const float det = a00_fw * a11_fw - a01_fw * a10_fw;
190 if (std::fabs(det) < 1e-8f) {
191 if (error_message) {
192 *error_message = "singular_matrix";
193 }
194 return false;
195 }
196 const float inv_det = 1.0f / det;
197 const float a00_inv = a11_fw * inv_det;
198 const float a01_inv = -a01_fw * inv_det;
199 const float a10_inv = -a10_fw * inv_det;
200 const float a11_inv = a00_fw * inv_det;
201 const float tx_inv = -(a00_inv * tx_fw + a01_inv * ty_fw);
202 const float ty_inv = -(a10_inv * tx_fw + a11_inv * ty_fw);
203 warp << a00_inv, a01_inv, tx_inv, a10_inv, a11_inv, ty_inv;
204 return true;
205}

286static std::vector<StarPoint> detect_stars_with_threshold(
287 const Matrix2Df &img, int topk, float sigma_multiplier,
288 float med, float sigma) {
289 const int h = img.rows();
290 const int w = img.cols();
291 const float thresh = med + sigma_multiplier * sigma;
292
293 std::vector<StarPoint> stars;
294 stars.reserve(static_cast<size_t>(topk) * 2);
295 for (int y = 1; y < h - 1; ++y) {
296 const float *row = img.data() + static_cast<size_t>(y) * w;
297 for (int x = 1; x < w - 1; ++x) {
298 const float v = row[x];
299 if (v < thresh)
300 continue;
301 bool is_max = true;
302 for (int dy = -1; dy <= 1 && is_max; ++dy) {
303 const float *r2 = img.data() + static_cast<size_t>(y + dy) * w;
304 for (int dx = -1; dx <= 1; ++dx) {
305 if (dx == 0 && dy == 0)
306 continue;
307 if (r2[x + dx] >= v) {
308 is_max = false;
309 break;
310 }
311 }
312 }
313 if (!is_max)
314 continue;
315
316 // Compute flux-weighted centroid and hot pixel / elongation metrics
317 // in a 5×5 neighborhood for better discrimination
318 const int hw = 2; // half-width for analysis
319 float wsum = 0.0f;
320 float xs = 0.0f;
321 float ys = 0.0f;
322 float central_flux = row[x] - med;
323 if (central_flux < 0.0f)
324 central_flux = 0.0f;
325 float Ixx = 0.0f, Iyy = 0.0f, Ixy = 0.0f;
326 for (int dy = -hw; dy <= hw; ++dy) {
327 if (y + dy < 0 || y + dy >= h)
328 continue;
329 const float *r2 = img.data() + static_cast<size_t>(y + dy) * w;
330 for (int dx = -hw; dx <= hw; ++dx) {
331 if (x + dx < 0 || x + dx >= w)
332 continue;
333 const float val = r2[x + dx] - med;
334 if (val <= 0.0f)
335 continue;
336 wsum += val;
337 xs += (static_cast<float>(x + dx) * val);
338 ys += (static_cast<float>(y + dy) * val);
339 Ixx += static_cast<float>(dx * dx) * val;
340 Iyy += static_cast<float>(dy * dy) * val;
341 Ixy += static_cast<float>(dx * dy) * val;
342 }
343 }
344 if (wsum <= 0.0f)
345 continue;
346
347 // Hot pixel rejection: real stars spread flux over multiple pixels.
348 // Hot pixels concentrate >80% in the central pixel.
349 const float concentration = central_flux / wsum;
350 if (concentration > 0.8f)
351 continue;
352
353 // Roundness filter via second moments: reject elongated sources
354 if (Ixx + Iyy > 1e-6f) {
355 float trace = Ixx + Iyy;
356 float det2 = Ixx * Iyy - Ixy * Ixy;
357 float disc = trace * trace - 4.0f * det2;
358 if (disc < 0.0f) disc = 0.0f;
359 float sqrt_disc = std::sqrt(disc);
360 float lam1 = (trace + sqrt_disc) * 0.5f;
361 float lam2 = (trace - sqrt_disc) * 0.5f;
362 float roundness = (lam1 > 1e-6f) ? (lam2 / lam1) : 0.0f;
363 if (roundness < 0.15f)
364 continue; // too elongated (trail or artifact)
365 }
366
367 StarPoint s;
368 s.x = xs / wsum;
369 s.y = ys / wsum;
370 s.flux = wsum;
371 stars.push_back(s);
372 }
373 }
374 std::sort(
375 stars.begin(), stars.end(),
376 [](const StarPoint &a, const StarPoint &b) { return a.flux > b.flux; });
377 if (static_cast<int>(stars.size()) > topk) {
378 stars.resize(static_cast<size_t>(topk));
379 }
380 return stars;
381}

387std::vector<StarPoint> detect_stars_simple(const Matrix2Df &img, int topk,
388 bool enable_local_background_subtraction) {
389 const int h = img.rows();
390 const int w = img.cols();
391 if (h < 5 || w < 5)
392 return {};
393
394 // §4.4, §8.D — Lokale Hintergrundsubtraktion bei Gradienten/Mondlicht
395 Matrix2Df processed_img;
396 const Matrix2Df *img_ptr = &img;
397 if (enable_local_background_subtraction) {
398 // Lokale Hintergrundschätzung mit Box-Blur (31x31 Pixel)
399 processed_img = img; // Kopie
400 const int kernel_size = 31;
401 const int half_k = kernel_size / 2;
402 // Einfacher Box-Blur für Hintergrundschätzung
403 for (int y = 0; y < h; ++y) {
404 for (int x = 0; x < w; ++x) {
405 float sum = 0.0f;
406 int count = 0;
407 for (int dy = -half_k; dy <= half_k; ++dy) {
408 int py = std::clamp(y + dy, 0, h - 1);
409 for (int dx = -half_k; dx <= half_k; ++dx) {
410 int px = std::clamp(x + dx, 0, w - 1);
411 sum += img(py, px);
412 ++count;
413 }
414 }
415 float background = sum / count;
416 processed_img(y, x) = std::max(0.0f, img(y, x) - background);
417 }
418 }
419 img_ptr = &processed_img;
420 }
421
422 std::vector<float> pixels;
423 pixels.reserve(static_cast<size_t>(img_ptr->size()));
424 for (int y = 0; y < h; ++y) {
425 const float *row = img_ptr->data() + static_cast<size_t>(y) * w;
426 pixels.insert(pixels.end(), row, row + w);
427 }
428 float med = core::median_of(pixels);
429 float sigma = core::robust_sigma_mad(pixels);
430 if (sigma < 1.0e-6f)
431 sigma = 1.0f;
432
433 // Try standard threshold (3.5σ) — mit optionaler Hintergrundsubtraktion
434 std::vector<StarPoint> stars = detect_stars_with_threshold(*img_ptr, topk, 3.5f, med, sigma);
435
436 // Adaptive fallback: if we found very few stars, retry with lower threshold
437 // This helps with clouds/nebula where stars appear fainter
438 const int min_expected = std::max(4, topk / 2);
439 if (static_cast<int>(stars.size()) < min_expected) {
440 std::vector<StarPoint> stars_relaxed = detect_stars_with_threshold(*img_ptr, topk, 2.5f, med, sigma);
441 if (stars_relaxed.size() > stars.size()) {
442 stars = std::move(stars_relaxed);
443 }
444 }
445
446 return stars;
447}

466SimilarityResult score_similarity(const std::vector<StarPoint> &mov,
467 const std::vector<StarPoint> &ref,
468 float scale, float theta,
469 const Eigen::Vector2f &t,
470 float inlier_tol_px) {
471 if (mov.empty() || ref.empty())
472 return {};
473 const float ct = std::cos(theta);
474 const float st = std::sin(theta);
475 int inl = 0;
476 float err_sum = 0.0f;
477 for (const auto &m : mov) {
478 const float xr = scale * (ct * m.x - st * m.y) + t.x();
479 const float yr = scale * (st * m.x + ct * m.y) + t.y();
480 float best = std::numeric_limits<float>::max();
481 for (const auto &r : ref) {
482 const float dx = xr - r.x;
483 const float dy = yr - r.y;
484 const float d = std::sqrt(dx * dx + dy * dy);
485 if (d < best)
486 best = d;
487 }
488 if (best < inlier_tol_px) {
489 ++inl;
490 err_sum += best;
491 }
492 }
493 SimilarityResult res;
494 res.ok = inl > 0;
495 res.inliers = inl;
496 res.mean_err = (inl > 0) ? (err_sum / static_cast<float>(inl)) : res.mean_err;
497 res.scale = scale;
498 res.theta = theta;
499 res.t = t;
500 return res;
501}

503static std::vector<std::pair<cv::Point2f, cv::Point2f>>
508build_similarity_consensus_pairs(const std::vector<StarPoint> &mov,
509 const std::vector<StarPoint> &ref,
510 const SimilarityResult &best,
511 float inlier_tol_px) {
512 struct Candidate {
513 int mov_idx = -1;
514 int ref_idx = -1;
515 float dist = 0.0f;
516 };
517
518 std::vector<Candidate> candidates;
519 candidates.reserve(mov.size());
520 const float ct = std::cos(best.theta);
521 const float st = std::sin(best.theta);
522 for (size_t mi = 0; mi < mov.size(); ++mi) {
523 const auto &m = mov[mi];
524 const float xr = best.scale * (ct * m.x - st * m.y) + best.t.x();
525 const float yr = best.scale * (st * m.x + ct * m.y) + best.t.y();
526 int best_ref_idx = -1;
527 float best_dist = std::numeric_limits<float>::max();
528 for (size_t ri = 0; ri < ref.size(); ++ri) {
529 const float dx = xr - ref[ri].x;
530 const float dy = yr - ref[ri].y;
531 const float d = std::sqrt(dx * dx + dy * dy);
532 if (d < best_dist) {
533 best_dist = d;
534 best_ref_idx = static_cast<int>(ri);
535 }
536 }
537 if (best_ref_idx >= 0 && best_dist < inlier_tol_px) {
538 candidates.push_back(
539 {static_cast<int>(mi), best_ref_idx, best_dist});
540 }
541 }
542
543 std::sort(candidates.begin(), candidates.end(),
544 [](const Candidate &a, const Candidate &b) {
545 return a.dist < b.dist;
546 });
547 std::vector<char> ref_used(ref.size(), 0);
548 std::vector<std::pair<cv::Point2f, cv::Point2f>> pairs;
549 pairs.reserve(candidates.size());
550 for (const Candidate &c : candidates) {
551 if (c.ref_idx < 0 || c.ref_idx >= static_cast<int>(ref.size()) ||
552 ref_used[static_cast<size_t>(c.ref_idx)] != 0) {
553 continue;
554 }
555 ref_used[static_cast<size_t>(c.ref_idx)] = 1;
556 pairs.emplace_back(cv::Point2f(mov[static_cast<size_t>(c.mov_idx)].x,
557 mov[static_cast<size_t>(c.mov_idx)].y),
558 cv::Point2f(ref[static_cast<size_t>(c.ref_idx)].x,
559 ref[static_cast<size_t>(c.ref_idx)].y));
560 }
561 return pairs;
562}

620bool similarity_from_pairs(const Eigen::Vector2f &m1, const Eigen::Vector2f &m2,
621 const Eigen::Vector2f &r1, const Eigen::Vector2f &r2,
622 bool allow_rotation, float &scale, float &theta,
623 Eigen::Vector2f &t) {
624 Eigen::Vector2f v_m = m2 - m1;
625 Eigen::Vector2f v_r = r2 - r1;
626 const float len_m = v_m.norm();
627 const float len_r = v_r.norm();
628 if (len_m < 1.0e-3f || len_r < 1.0e-3f)
629 return false;
630 scale = len_r / len_m;
631 if (!std::isfinite(scale) || scale <= 0.0f)
632 return false;
633 theta = allow_rotation
634 ? (std::atan2(v_r.y(), v_r.x()) - std::atan2(v_m.y(), v_m.x()))
635 : 0.0f;
636 const float ct = std::cos(theta);
637 const float st = std::sin(theta);
638 Eigen::Matrix2f R;
639 R << ct, -st, st, ct;
640 t = r1 - scale * (R * m1);
641 return true;
642}

654 cv::Mat f(img.rows(), img.cols(), CV_32F,
655 const_cast<float *>(img.data()));
656 // Laplacian of Gaussian: removes low-frequency gradients (nebula, clouds)
657 // while preserving high-frequency structure (stars, edges)
658 cv::Mat blurred;
659 cv::GaussianBlur(f, blurred, cv::Size(0, 0), 2.0);
660 cv::Mat laplacian;
661 cv::Laplacian(blurred, laplacian, CV_32F, 3);
662
663 // Normalize to [0,1]
664 double minV, maxV;
665 cv::minMaxLoc(laplacian, &minV, &maxV);
666 if (maxV - minV < 1e-10)
667 maxV = minV + 1.0;
668 cv::Mat norm = (laplacian - minV) / (maxV - minV);
669
670 Matrix2Df result(img.rows(), img.cols());
671 std::memcpy(result.data(), norm.data,
672 static_cast<size_t>(img.size()) * sizeof(float));
673 return result;
674}

680RegistrationResult robust_phase_ecc(const Matrix2Df &mov,
681 const Matrix2Df &ref,
682 bool allow_rotation) {
683 RegistrationResult res;
684 res.warp = identity_warp();
685 res.success = false;
686 res.correlation = 0.0f;
687
688 // Gradient pre-processing: removes nebula/cloud gradients
689 Matrix2Df mov_grad = gradient_preprocess(mov);
690 Matrix2Df ref_grad = gradient_preprocess(ref);
691
692 // Multi-scale pyramid: coarse-to-fine for large displacements/rotations
693 // Level 2 (4x downsampled) → Level 1 (2x) → Level 0 (original)
694 std::vector<Matrix2Df> mov_pyr = {mov_grad};
695 std::vector<Matrix2Df> ref_pyr = {ref_grad};
696
697 // Build 2 pyramid levels
698 for (int level = 0; level < 2; ++level) {
699 mov_pyr.push_back(downsample2x2_mean(mov_pyr.back()));
700 ref_pyr.push_back(downsample2x2_mean(ref_pyr.back()));
701 }
702
703 // Start from coarsest level
704 WarpMatrix current_warp = identity_warp();
705
706 for (int level = static_cast<int>(mov_pyr.size()) - 1; level >= 0; --level) {
707 const Matrix2Df &m = mov_pyr[static_cast<size_t>(level)];
708 const Matrix2Df &r = ref_pyr[static_cast<size_t>(level)];
709
710 Matrix2Df m_ecc = prepare_ecc_image(m);
711 Matrix2Df r_ecc = prepare_ecc_image(r);
712
713 if (level == static_cast<int>(mov_pyr.size()) - 1) {
714 // Coarsest level: estimate initial translation + rotation
715 auto [dx, dy] = phasecorr_translation(m_ecc, r_ecc);
716 current_warp(0, 2) = dx;
717 current_warp(1, 2) = dy;
718
719 if (allow_rotation) {
720 cv::Mat r_cv(r_ecc.rows(), r_ecc.cols(), CV_32F,
721 const_cast<float *>(r_ecc.data()));
722 cv::Mat m_cv(m_ecc.rows(), m_ecc.cols(), CV_32F,
723 const_cast<float *>(m_ecc.data()));
724 float rot = estimate_rotation_logpolar(r_cv, m_cv);
725 float th = rot * 3.14159265f / 180.0f;
726 float ct = std::cos(th);
727 float st = std::sin(th);
728 float cx = static_cast<float>(m_ecc.cols()) * 0.5f;
729 float cy = static_cast<float>(m_ecc.rows()) * 0.5f;
730 float tx_rot = cx * (1.0f - ct) + cy * st;
731 float ty_rot = cy * (1.0f - ct) - cx * st;
732 current_warp << ct, -st, dx + tx_rot, st, ct, dy + ty_rot;
733 }
734 } else {
735 // Scale warp from previous (coarser) level to current level
736 current_warp(0, 2) *= 2.0f;
737 current_warp(1, 2) *= 2.0f;
738 }
739
740 // ECC refinement at this level
741 int iters = (level == 0) ? 200 : 100;
742 float eps = (level == 0) ? 1e-6f : 1e-4f;
743 RegistrationResult level_res =
744 ecc_warp(m_ecc, r_ecc, allow_rotation, current_warp, iters, eps);
745 if (level_res.success) {
746 current_warp = level_res.warp;
747 res = level_res;
748 }
749 // If ECC fails at coarse level, still try finer levels with current seed
750 }
751
752 // NOTE: OpenCV findTransformECC returns a warp that is intended to be used
753 // directly with warpAffine(..., WARP_INVERSE_MAP) to align moving→reference.
754 // Our apply_warp() also uses WARP_INVERSE_MAP, so we must NOT invert here.
755 return res;
756}

812RegistrationResult
817star_registration_similarity(const Matrix2Df &mov, const Matrix2Df &ref,
818 bool allow_rotation,
819 int topk_stars, int min_inliers,
820 float inlier_tol_px, float dist_bin_px,
821 const std::string &transform_model,
822 bool enable_local_background_subtraction) {
823 RegistrationResult res;
824 res.warp = identity_warp();
825 res.success = false;
826 res.correlation = 0.0f;
827
828 auto mov_stars = detect_stars_simple(mov, topk_stars, enable_local_background_subtraction);
829 auto ref_stars = detect_stars_simple(ref, topk_stars, enable_local_background_subtraction);
830 if (mov_stars.size() < 3 || ref_stars.size() < 3) {
831 res.error_message = "too_few_stars";
832 return res;
833 }
834
835 struct Pair {
836 int i;
837 int j;
838 float dist;
839 };
840 std::vector<Pair> ref_pairs;
841 std::vector<Pair> mov_pairs;
842
843 auto build_pairs = [&](const std::vector<StarPoint> &stars,
844 std::vector<Pair> &out) {
845 const int n = static_cast<int>(stars.size());
846 out.reserve(static_cast<size_t>(n) * static_cast<size_t>(n) / 2);
847 for (int i = 0; i < n; ++i) {
848 for (int j = i + 1; j < n; ++j) {
849 const float dx = stars[j].x - stars[i].x;
850 const float dy = stars[j].y - stars[i].y;
851 const float d = std::sqrt(dx * dx + dy * dy);
852 if (d > 1.0f)
853 out.push_back({i, j, d});
854 }
855 }
856 std::sort(out.begin(), out.end(),
857 [](const Pair &a, const Pair &b) { return a.dist > b.dist; });
858 const size_t limit = std::min<size_t>(out.size(), 800);
859 out.resize(limit);
860 };
861
862 build_pairs(ref_stars, ref_pairs);
863 build_pairs(mov_stars, mov_pairs);
864 if (ref_pairs.empty() || mov_pairs.empty()) {
865 res.error_message = "no_pairs";
866 return res;
867 }
868
869 std::unordered_map<int, std::vector<Pair>> ref_bucket;
870 ref_bucket.reserve(ref_pairs.size() * 2);
871 for (const auto &p : ref_pairs) {
872 int key = static_cast<int>(std::round(p.dist / dist_bin_px));
873 ref_bucket[key].push_back(p);
874 }
875
876 SimilarityResult best;
877 int attempts = 0;
878 for (const auto &pm : mov_pairs) {
879 int key = static_cast<int>(std::round(pm.dist / dist_bin_px));
880 for (int dk = -1; dk <= 1; ++dk) {
881 auto it = ref_bucket.find(key + dk);
882 if (it == ref_bucket.end())
883 continue;
884 for (const auto &pr : it->second) {
885 ++attempts;
886 float scale = 1.0f;
887 float theta = 0.0f;
888 Eigen::Vector2f t;
889 const Eigen::Vector2f m1(mov_stars[pm.i].x, mov_stars[pm.i].y);
890 const Eigen::Vector2f m2(mov_stars[pm.j].x, mov_stars[pm.j].y);
891 const Eigen::Vector2f r1(ref_stars[pr.i].x, ref_stars[pr.i].y);
892 const Eigen::Vector2f r2(ref_stars[pr.j].x, ref_stars[pr.j].y);
893 if (!similarity_from_pairs(m1, m2, r1, r2, allow_rotation, scale, theta,
894 t))
895 continue;
896 // No hard rotation limit — RANSAC consensus handles outliers.
897 if (!std::isfinite(scale) || scale < 0.5f || scale > 2.0f)
898 continue;
899
900 SimilarityResult sr = score_similarity(mov_stars, ref_stars, scale,
901 theta, t, inlier_tol_px);
902 if (!sr.ok)
903 continue;
904 if (sr.inliers > best.inliers ||
905 (sr.inliers == best.inliers && sr.mean_err < best.mean_err)) {
906 best = sr;
907 }
908 }
909 }
910 if (attempts > 4000 && best.inliers >= min_inliers)
911 break;
912 }
913
914 if (best.inliers >= min_inliers && best.mean_err < inlier_tol_px * 1.2f) {
915 res.success = true;
916 res.correlation = static_cast<float>(best.inliers) /
917 static_cast<float>(std::max<size_t>(1, mov_stars.size()));
918 if (maybe_refine_similarity_to_affine(
919 mov_stars, ref_stars, best, allow_rotation, inlier_tol_px,
920 min_inliers, transform_model, res.warp, res.correlation,
921 res.error_message)) {
922 return res;
923 }
924 // Construct Forward Matrix (M -> R)
925 float s_fw = best.scale;
926 float th_fw = best.theta;
927 float tx_fw = best.t.x();
928 float ty_fw = best.t.y();
929
930 float c_fw = std::cos(th_fw);
931 float sn_fw = std::sin(th_fw);
932
933 // We need the Inverse Matrix (R -> M) for apply_warp with WARP_INVERSE_MAP
934 // M = (1/s) * R^T * (R_coord - t)
935 // Scale_inv = 1/s
936 // Rot_inv = -theta
937 // T_inv = - (1/s) * R^T * t
938
939 float s_inv = 1.0f / s_fw;
940 float c_inv = c_fw; // cos(-th) = cos(th)
941 float sn_inv = -sn_fw; // sin(-th) = -sin(th)
942
943 float a00 = s_inv * c_inv;
944 float a01 = s_inv * -sn_inv; // -sin inside rot matrix
945 float a10 = s_inv * sn_inv;
946 float a11 = s_inv * c_inv;
947
948 // t_inv = - S_inv * t_fw = - [a00 a01; a10 a11] * [tx; ty]
949 float tx_inv = -(a00 * tx_fw + a01 * ty_fw);
950 float ty_inv = -(a10 * tx_fw + a11 * ty_fw);
951
952 res.warp << a00, a01, tx_inv, a10, a11, ty_inv;
953 } else {
954 res.error_message = "no_consensus";
955 }
956 return res;
957}

959// =====================================================================
960// Triangle-based asterism matching (astroalign-style, rotation-invariant)
961// =====================================================================
962
963struct Triangle {
964 int i, j, k; // star indices
965 float sides[3]; // sorted side lengths (ascending)
966 float perimeter;
967 float ratios[2]; // sides[0]/sides[2], sides[1]/sides[2] — invariants
968 float signed_area2; // orientation-preserving triangle discriminator
969};
970
971static std::vector<Triangle>
976build_triangles(const std::vector<StarPoint> &stars, int max_triangles) {
977 const int n = static_cast<int>(stars.size());
978 std::vector<Triangle> tris;
979 if (n < 3)
980 return tris;
981
982 // Limit combinatorial explosion: use a wider star subset. The previous
983 // hard cap of 30 caused dense fields to collapse into many ambiguous
984 // asterisms, even though plenty of stars were available.
985 const int limit = std::min(n, 60);
986 tris.reserve(static_cast<size_t>(limit * (limit - 1) * (limit - 2) / 6));
987
988 for (int i = 0; i < limit; ++i) {
989 for (int j = i + 1; j < limit; ++j) {
990 for (int k = j + 1; k < limit; ++k) {
991 float dx_ij = stars[j].x - stars[i].x;
992 float dy_ij = stars[j].y - stars[i].y;
993 float dx_ik = stars[k].x - stars[i].x;
994 float dy_ik = stars[k].y - stars[i].y;
995 float dx_jk = stars[k].x - stars[j].x;
996 float dy_jk = stars[k].y - stars[j].y;
997
998 float d_ij = std::sqrt(dx_ij * dx_ij + dy_ij * dy_ij);
999 float d_ik = std::sqrt(dx_ik * dx_ik + dy_ik * dy_ik);
1000 float d_jk = std::sqrt(dx_jk * dx_jk + dy_jk * dy_jk);
1001 const float signed_area2 = dx_ij * dy_ik - dy_ij * dx_ik;
1002
1003 // Skip degenerate triangles
1004 if (d_ij < 3.0f || d_ik < 3.0f || d_jk < 3.0f)
1005 continue;
1006 if (std::fabs(signed_area2) < 9.0f)
1007 continue;
1008
1009 Triangle t;
1010 t.i = i;
1011 t.j = j;
1012 t.k = k;
1013
1014 // Sort sides ascending
1015 float s[3] = {d_ij, d_ik, d_jk};
1016 if (s[0] > s[1]) std::swap(s[0], s[1]);
1017 if (s[1] > s[2]) std::swap(s[1], s[2]);
1018 if (s[0] > s[1]) std::swap(s[0], s[1]);
1019
1020 t.sides[0] = s[0];
1021 t.sides[1] = s[1];
1022 t.sides[2] = s[2];
1023 t.perimeter = s[0] + s[1] + s[2];
1024
1025 // Invariant ratios (scale + rotation invariant)
1026 t.ratios[0] = s[0] / s[2];
1027 t.ratios[1] = s[1] / s[2];
1028 t.signed_area2 = signed_area2;
1029
1030 // Near-equilateral and near-isosceles triangles are highly ambiguous
1031 // under ratio-only matching and poison the downstream affine fit.
1032 if (t.ratios[1] > 0.97f || (t.ratios[1] - t.ratios[0]) < 0.02f) {
1033 continue;
1034 }
1035
1036 tris.push_back(t);
1037 }
1038 }
1039 }
1040
1041 // Keep largest triangles (more robust)
1042 if (static_cast<int>(tris.size()) > max_triangles) {
1043 std::sort(tris.begin(), tris.end(),
1044 [](const Triangle &a, const Triangle &b) {
1045 return a.perimeter > b.perimeter;
1046 });
1047 tris.resize(static_cast<size_t>(max_triangles));
1048 }
1049
1050 return tris;
1051}

1053RegistrationResult
1058triangle_star_matching(const Matrix2Df &mov, const Matrix2Df &ref,
1059 bool allow_rotation,
1060 int topk_stars, int min_inliers,
1061 float inlier_tol_px,
1062 const std::string &transform_model,
1063 bool enable_local_background_subtraction,
1064 float shift_radius_px) {
1065 RegistrationResult res;
1066 res.warp = identity_warp();
1067 res.success = false;
1068 res.correlation = 0.0f;
1069
1070 auto mov_stars = detect_stars_simple(mov, topk_stars, enable_local_background_subtraction);
1071 auto ref_stars = detect_stars_simple(ref, topk_stars, enable_local_background_subtraction);
1072
1073 if (mov_stars.size() < 3 || ref_stars.size() < 3) {
1074 res.error_message = "too_few_stars";
1075 return res;
1076 }
1077
1078 const int max_tris = 600;
1079 auto mov_tris = build_triangles(mov_stars, max_tris);
1080 auto ref_tris = build_triangles(ref_stars, max_tris);
1081
1082 if (mov_tris.empty() || ref_tris.empty()) {
1083 res.error_message = "no_triangles";
1084 return res;
1085 }
1086
1087 // Match triangles by invariant ratios
1088 // Calibrated via Python simulation on M104 star data:
1089 // ratio_tol=0.03 -> too many false matches (crowded fields)
1090 // ratio_tol=0.008, ambiguity_margin=0.007 -> correct balance
1091 const float ratio_tol = 0.008f;
1092 const float ambiguity_margin = 0.007f;
1093
1094 struct PairVote {
1095 int mov_idx = -1;
1096 int ref_idx = -1;
1097 int votes = 0;
1098 float score_sum = 0.0f;
1099 };
1100 std::unordered_map<std::uint64_t, PairVote> pair_votes;
1101 pair_votes.reserve(static_cast<size_t>(max_tris));
1102
1103 int matches_found = 0;
1104 for (const auto &mt : mov_tris) {
1105 float best_dist = ratio_tol * 2.0f;
1106 float second_best_dist = ratio_tol * 2.0f;
1107 const Triangle *best_rt = nullptr;
1108
1109 for (const auto &rt : ref_tris) {
1110 if ((mt.signed_area2 > 0.0f) != (rt.signed_area2 > 0.0f)) {
1111 continue;
1112 }
1113 float dr0 = mt.ratios[0] - rt.ratios[0];
1114 float dr1 = mt.ratios[1] - rt.ratios[1];
1115 float d = std::sqrt(dr0 * dr0 + dr1 * dr1);
1116 if (d < best_dist) {
1117 second_best_dist = best_dist;
1118 best_dist = d;
1119 best_rt = &rt;
1120 } else if (d < second_best_dist) {
1121 second_best_dist = d;
1122 }
1123 }
1124
1125 if (!best_rt || best_dist >= ratio_tol)
1126 continue;
1127 if (second_best_dist - best_dist < ambiguity_margin)
1128 continue;
1129
1130 matches_found++;
1131
1132 // Determine vertex correspondence by matching sorted side lengths
1133 // The vertices opposite to the shortest/medium/longest sides correspond
1134 // mov triangle vertices: i, j, k
1135 // We need to figure out which vertex is opposite which side
1136 // Side ij is between i,j — opposite vertex is k
1137 // Side ik is between i,k — opposite vertex is j
1138 // Side jk is between j,k — opposite vertex is i
1139 auto vertex_order = [](const std::vector<StarPoint> &stars,
1140 const Triangle &t) -> std::array<int, 3> {
1141 float d_ij = 0, d_ik = 0, d_jk = 0;
1142 {
1143 float dx = stars[t.j].x - stars[t.i].x;
1144 float dy = stars[t.j].y - stars[t.i].y;
1145 d_ij = std::sqrt(dx * dx + dy * dy);
1146 }
1147 {
1148 float dx = stars[t.k].x - stars[t.i].x;
1149 float dy = stars[t.k].y - stars[t.i].y;
1150 d_ik = std::sqrt(dx * dx + dy * dy);
1151 }
1152 {
1153 float dx = stars[t.k].x - stars[t.j].x;
1154 float dy = stars[t.k].y - stars[t.j].y;
1155 d_jk = std::sqrt(dx * dx + dy * dy);
1156 }
1157 // vertex opposite shortest side, medium side, longest side
1158 struct SideVtx {
1159 float len;
1160 int opposite;
1161 };
1162 SideVtx sv[3] = {{d_ij, t.k}, {d_ik, t.j}, {d_jk, t.i}};
1163 if (sv[0].len > sv[1].len) std::swap(sv[0], sv[1]);
1164 if (sv[1].len > sv[2].len) std::swap(sv[1], sv[2]);
1165 if (sv[0].len > sv[1].len) std::swap(sv[0], sv[1]);
1166 return {sv[0].opposite, sv[1].opposite, sv[2].opposite};
1167 };
1168
1169 auto mov_order = vertex_order(mov_stars, mt);
1170 auto ref_order = vertex_order(ref_stars, *best_rt);
1171
1172 for (int v = 0; v < 3; ++v) {
1173 const int mov_idx = mov_order[v];
1174 const int ref_idx = ref_order[v];
1175 const std::uint64_t key =
1176 (static_cast<std::uint64_t>(static_cast<std::uint32_t>(mov_idx))
1177 << 32) |
1178 static_cast<std::uint32_t>(ref_idx);
1179 auto &vote = pair_votes[key];
1180 vote.mov_idx = mov_idx;
1181 vote.ref_idx = ref_idx;
1182 vote.votes += 1;
1183 vote.score_sum += best_dist;
1184 }
1185
1186 if (matches_found > 200)
1187 break;
1188 }
1189
1190 std::vector<PairVote> ranked_pairs;
1191 ranked_pairs.reserve(pair_votes.size());
1192 for (const auto &entry : pair_votes) {
1193 ranked_pairs.push_back(entry.second);
1194 }
1195 std::sort(ranked_pairs.begin(), ranked_pairs.end(),
1196 [](const PairVote &a, const PairVote &b) {
1197 if (a.votes != b.votes) {
1198 return a.votes > b.votes;
1199 }
1200 return a.score_sum < b.score_sum;
1201 });
1202
1203 // Shift-consistency filter: each candidate pair implies a translation
1204 // (dx,dy). Compute per-pair support = sum of votes of other pairs whose
1205 // implied shift is within shift_radius px. The pair with highest support
1206 // is the shift-cluster anchor; keep only pairs within shift_radius of it.
1207 // This eliminates false matches that dominate vote counts in crowded fields.
1208 const float shift_radius = shift_radius_px;
1209 struct PairShift {
1210 int mov_idx, ref_idx, votes;
1211 float dx, dy;
1212 int support;
1213 };
1214 std::vector<PairShift> cand_pairs;
1215 cand_pairs.reserve(ranked_pairs.size());
1216 for (const PairVote &vote : ranked_pairs) {
1217 if (vote.mov_idx < 0 || vote.ref_idx < 0) continue;
1218 if (vote.mov_idx >= static_cast<int>(mov_stars.size()) ||
1219 vote.ref_idx >= static_cast<int>(ref_stars.size())) continue;
1220 float dx = ref_stars[static_cast<size_t>(vote.ref_idx)].x -
1221 mov_stars[static_cast<size_t>(vote.mov_idx)].x;
1222 float dy = ref_stars[static_cast<size_t>(vote.ref_idx)].y -
1223 mov_stars[static_cast<size_t>(vote.mov_idx)].y;
1224 cand_pairs.push_back({vote.mov_idx, vote.ref_idx, vote.votes, dx, dy, 0});
1225 }
1226 // Compute shift support for each pair (O(n^2), n <= ~150 pairs — fast enough)
1227 for (size_t i = 0; i < cand_pairs.size(); ++i) {
1228 int sup = 0;
1229 for (size_t j = 0; j < cand_pairs.size(); ++j) {
1230 if (i == j) continue;
1231 float ddx = cand_pairs[i].dx - cand_pairs[j].dx;
1232 float ddy = cand_pairs[i].dy - cand_pairs[j].dy;
1233 if (std::sqrt(ddx * ddx + ddy * ddy) <= shift_radius)
1234 sup += cand_pairs[j].votes;
1235 }
1236 cand_pairs[i].support = sup;
1237 }
1238 // Find anchor = pair with maximum weighted shift support
1239 size_t best_anchor = 0;
1240 int best_anchor_sup = -1;
1241 for (size_t i = 0; i < cand_pairs.size(); ++i) {
1242 if (cand_pairs[i].support > best_anchor_sup) {
1243 best_anchor_sup = cand_pairs[i].support;
1244 best_anchor = i;
1245 }
1246 }
1247 // Keep only pairs consistent with the anchor shift
1248 if (!cand_pairs.empty()) {
1249 float anchor_dx = cand_pairs[best_anchor].dx;
1250 float anchor_dy = cand_pairs[best_anchor].dy;
1251 cand_pairs.erase(
1252 std::remove_if(cand_pairs.begin(), cand_pairs.end(),
1253 [anchor_dx, anchor_dy, shift_radius](const PairShift &p) {
1254 float ddx = p.dx - anchor_dx;
1255 float ddy = p.dy - anchor_dy;
1256 return std::sqrt(ddx * ddx + ddy * ddy) > shift_radius;
1257 }),
1258 cand_pairs.end());
1259 }
1260 // Re-sort by vote count
1261 std::sort(cand_pairs.begin(), cand_pairs.end(),
1262 [](const PairShift &a, const PairShift &b) {
1263 return a.votes > b.votes;
1264 });
1265
1266 std::vector<cv::Point2f> pts_mov, pts_ref;
1267 pts_mov.reserve(cand_pairs.size());
1268 pts_ref.reserve(cand_pairs.size());
1269 std::vector<uint8_t> mov_used(mov_stars.size(), 0);
1270 std::vector<uint8_t> ref_used(ref_stars.size(), 0);
1271 for (const PairShift &p : cand_pairs) {
1272 if (mov_used[static_cast<size_t>(p.mov_idx)] != 0 ||
1273 ref_used[static_cast<size_t>(p.ref_idx)] != 0) {
1274 continue;
1275 }
1276 mov_used[static_cast<size_t>(p.mov_idx)] = 1;
1277 ref_used[static_cast<size_t>(p.ref_idx)] = 1;
1278 pts_mov.push_back(cv::Point2f(mov_stars[static_cast<size_t>(p.mov_idx)].x,
1279 mov_stars[static_cast<size_t>(p.mov_idx)].y));
1280 pts_ref.push_back(cv::Point2f(ref_stars[static_cast<size_t>(p.ref_idx)].x,
1281 ref_stars[static_cast<size_t>(p.ref_idx)].y));
1282 }
1283
1284 if (pts_mov.size() < static_cast<size_t>(std::max(3, min_inliers))) {
1285 res.error_message = "few_triangle_matches";
1286 return res;
1287 }
1288
1289 // Use RANSAC to find the best similarity transform from correspondences
1290 cv::Mat inliers;
1291 cv::Mat A = estimate_affine_family_transform(pts_mov, pts_ref,
1292 transform_model, inliers);
1293 if (A.empty()) {
1294 res.error_message = "transform_fail";
1295 return res;
1296 }
1297 if (!invert_forward_affine_to_warp(A, allow_rotation, res.warp,
1298 &res.error_message)) {
1299 return res;
1300 }
1301 int inl = inliers.empty() ? 0 : cv::countNonZero(inliers);
1302 res.correlation =
1303 pts_mov.empty()
1304 ? 0.0f
1305 : static_cast<float>(inl) / static_cast<float>(pts_mov.size());
1306 res.success = inl >= min_inliers;
1307 return res;
1308}

1314RegistrationResult hybrid_phase_ecc(const Matrix2Df &mov, const Matrix2Df &ref,
1315 bool allow_rotation) {
1316 RegistrationResult res;
1317 res.warp = identity_warp();
1318 res.success = false;
1319 res.correlation = 0.0f;
1320
1321 Matrix2Df mov_ecc = prepare_ecc_image(mov);
1322 Matrix2Df ref_ecc = prepare_ecc_image(ref);
1323
1324 auto [dx, dy] = phasecorr_translation(mov_ecc, ref_ecc);
1325
1326 WarpMatrix init = identity_warp();
1327 init(0, 2) = dx;
1328 init(1, 2) = dy;
1329
1330 if (allow_rotation) {
1331 cv::Mat ref_cv(ref_ecc.rows(), ref_ecc.cols(), CV_32F,
1332 const_cast<float *>(ref_ecc.data()));
1333 cv::Mat mov_cv(mov_ecc.rows(), mov_ecc.cols(), CV_32F,
1334 const_cast<float *>(mov_ecc.data()));
1335 float rot = estimate_rotation_logpolar(ref_cv, mov_cv);
1336 // Use actual detected rotation as ECC seed — clamping causes
1337 // convergence failures when real rotation exceeds the limit.
1338 const float th = rot * 3.14159265f / 180.0f;
1339 const float ct = std::cos(th);
1340 const float st = std::sin(th);
1341
1342 // Correct for center of rotation
1343 const float cx = static_cast<float>(mov_ecc.cols()) * 0.5f;
1344 const float cy = static_cast<float>(mov_ecc.rows()) * 0.5f;
1345 const float tx_rot = cx * (1.0f - ct) + cy * st;
1346 const float ty_rot = cy * (1.0f - ct) - cx * st;
1347
1348 init << ct, -st, dx + tx_rot, st, ct, dy + ty_rot;
1349 }
1350
1351 res = ecc_warp(mov_ecc, ref_ecc, allow_rotation, init, 200, 1e-6f);
1352
1353 // NOTE: OpenCV findTransformECC returns a warp that is intended to be used
1354 // directly with warpAffine(..., WARP_INVERSE_MAP) to align moving→reference.
1355 // Our apply_warp() also uses WARP_INVERSE_MAP, so we must NOT invert here.
1356 return res;
1357}

1367static float compute_ncc(const Matrix2Df &a, const Matrix2Df &b) {
1368 const int n = a.size();
1369 if (n <= 0 || n != b.size())
1370 return 0.0f;
1371 const float *da = a.data();
1372 const float *db = b.data();
1373 double ma = 0, mb = 0;
1374 for (int i = 0; i < n; ++i) {
1375 ma += da[i];
1376 mb += db[i];
1377 }
1378 ma /= n;
1379 mb /= n;
1380 double sab = 0, saa = 0, sbb = 0;
1381 for (int i = 0; i < n; ++i) {
1382 double va = da[i] - ma;
1383 double vb = db[i] - mb;
1384 sab += va * vb;
1385 saa += va * va;
1386 sbb += vb * vb;
1387 }
1388 double den = std::sqrt(saa * sbb);
1389 return (den > 1e-10) ? static_cast<float>(sab / den) : 0.0f;
1390}

1396cv::Mat warp_valid_mask(const Matrix2Df &img, const WarpMatrix &warp) {
1397 cv::Mat ones(img.rows(), img.cols(), CV_32F, cv::Scalar(1.0f));
1398 cv::Mat warp_matrix(2, 3, CV_32F);
1399 for (int i = 0; i < 2; ++i) {
1400 for (int j = 0; j < 3; ++j) {
1401 warp_matrix.at<float>(i, j) = warp(i, j);
1402 }
1403 }
1404 cv::Mat warped_mask;
1405 cv::warpAffine(ones, warped_mask, warp_matrix, ones.size(),
1406 cv::INTER_NEAREST | cv::WARP_INVERSE_MAP,
1407 cv::BORDER_CONSTANT, cv::Scalar(0.0f));
1408 return warped_mask;
1409}

1415float compute_ncc_masked(const Matrix2Df &a, const Matrix2Df &b,
1416 const cv::Mat &mask, int *used_pixels) {
1417 if (a.size() <= 0 || a.size() != b.size() || mask.empty() ||
1418 mask.rows != a.rows() || mask.cols != a.cols()) {
1419 if (used_pixels) {
1420 *used_pixels = 0;
1421 }
1422 return 0.0f;
1423 }
1424
1425 const float *da = a.data();
1426 const float *db = b.data();
1427 double ma = 0.0;
1428 double mb = 0.0;
1429 int n = 0;
1430 for (int y = 0; y < mask.rows; ++y) {
1431 const float *pm = mask.ptr<float>(y);
1432 const int row_off = y * mask.cols;
1433 for (int x = 0; x < mask.cols; ++x) {
1434 if (pm[x] <= 0.5f) {
1435 continue;
1436 }
1437 const int idx = row_off + x;
1438 ma += da[idx];
1439 mb += db[idx];
1440 ++n;
1441 }
1442 }
1443 if (used_pixels) {
1444 *used_pixels = n;
1445 }
1446 if (n <= 1) {
1447 return 0.0f;
1448 }
1449
1450 ma /= static_cast<double>(n);
1451 mb /= static_cast<double>(n);
1452 double sab = 0.0;
1453 double saa = 0.0;
1454 double sbb = 0.0;
1455 for (int y = 0; y < mask.rows; ++y) {
1456 const float *pm = mask.ptr<float>(y);
1457 const int row_off = y * mask.cols;
1458 for (int x = 0; x < mask.cols; ++x) {
1459 if (pm[x] <= 0.5f) {
1460 continue;
1461 }
1462 const int idx = row_off + x;
1463 const double va = static_cast<double>(da[idx]) - ma;
1464 const double vb = static_cast<double>(db[idx]) - mb;
1465 sab += va * vb;
1466 saa += va * va;
1467 sbb += vb * vb;
1468 }
1469 }
1470 const double den = std::sqrt(saa * sbb);
1471 return (den > 1e-10) ? static_cast<float>(sab / den) : 0.0f;
1472}

1478SingleFrameRegResult register_single_frame(const Matrix2Df &mov,
1479 const Matrix2Df &ref,
1480 const config::RegistrationConfig &rcfg,
1481 float min_ncc_improvement) {
1482 // Thread-safe diagnostic counter (only log first few calls)
1483 static std::atomic<int> diag_counter{0};
1484 const int diag_id = diag_counter.fetch_add(1);
1485 const bool diag = (diag_id < 10);
1486
1487 SingleFrameRegResult out;
1488 out.reg.warp = identity_warp();
1489 out.reg.success = false;
1490 out.reg.correlation = 0.0f;
1491 out.reg.error_message.clear();
1492 out.method_used = "identity";
1493 out.ncc_identity = compute_ncc(mov, ref);
1494 out.ncc_warped = out.ncc_identity;
1495
1496 const float identity_gate_margin = std::max(min_ncc_improvement, 0.0f);
1497 if (out.ncc_identity >= 1.0f - identity_gate_margin) {
1498 // Spec §4.2 binding edge case: if identity is already near-perfect,
1499 // no warp can satisfy the strict NCC improvement gate. Accept identity
1500 // directly and do not count it as a cascade failure.
1501 out.reg.success = true;
1502 out.reg.correlation = out.ncc_identity;
1503 if (diag) {
1504 std::cout << "[REG-DIAG#" << diag_id
1505 << "] identity accepted directly: ncc_identity="
1506 << out.ncc_identity
1507 << " threshold=" << (1.0f - identity_gate_margin)
1508 << std::endl;
1509 }
1510 return out;
1511 }
1512
1513 if (diag) {
1514 auto stars_ref = detect_stars_simple(ref, rcfg.star_topk, rcfg.enable_local_background_subtraction);
1515 auto stars_mov = detect_stars_simple(mov, rcfg.star_topk, rcfg.enable_local_background_subtraction);
1516 std::cout << "[REG-DIAG#" << diag_id << "] ncc_identity=" << out.ncc_identity
1517 << " stars_ref=" << stars_ref.size()
1518 << " stars_mov=" << stars_mov.size()
1519 << " img=" << mov.rows() << "x" << mov.cols()
1520 << std::endl;
1521 }
1522
1523 bool accepted = false;
1524
1525 // Try a cascade method and validate with NCC
1526 auto try_method = [&](RegistrationResult rr,
1527 const std::string &method) -> bool {
1528 if (!rr.success) {
1529 if (diag) {
1530 std::cout << "[REG-DIAG#" << diag_id << "] " << method
1531 << " FAIL: " << rr.error_message << std::endl;
1532 }
1533 return false;
1534 }
1535 Matrix2Df warped = apply_warp(mov, rr.warp);
1536 const cv::Mat valid_mask = warp_valid_mask(mov, rr.warp);
1537 int overlap_pixels = 0;
1538 // Blur before NCC: makes the quality metric robust against hot pixels and
1539 // sharp point sources that cause extreme NCC sensitivity to sub-pixel shifts.
1540 // sigma=1.5 matches the ECC pre-processing blur (prepare_ecc_image).
1541 // Important: clamp to zero first so that negative background-subtracted
1542 // values do not bleed into star peaks via Gaussian blur.
1543 auto blur_for_ncc = [](const Matrix2Df &img) -> Matrix2Df {
1544 cv::Mat m(img.rows(), img.cols(), CV_32F,
1545 const_cast<float *>(img.data()));
1546 cv::Mat clamped;
1547 cv::max(m, 0.0f, clamped); // remove negative background artefacts
1548 cv::GaussianBlur(clamped, clamped, cv::Size(0, 0), 1.5);
1549 Matrix2Df result(img.rows(), img.cols());
1550 std::memcpy(result.data(), clamped.data,
1551 static_cast<size_t>(img.size()) * sizeof(float));
1552 return result;
1553 };
1554 const Matrix2Df mov_b = blur_for_ncc(mov);
1555 const Matrix2Df warped_b = blur_for_ncc(warped);
1556 const Matrix2Df ref_b = blur_for_ncc(ref);
1557 const float ncc_identity_overlap =
1558 compute_ncc_masked(mov_b, ref_b, valid_mask, &overlap_pixels);
1559 float ncc = compute_ncc_masked(warped_b, ref_b, valid_mask);
1560 if (diag) {
1561 const float angle_deg = std::atan2(-rr.warp(0,1), rr.warp(0,0)) * (180.0f / 3.14159265f);
1562 const float scale = std::sqrt(rr.warp(0,0)*rr.warp(0,0) + rr.warp(0,1)*rr.warp(0,1));
1563 std::cout << "[REG-DIAG#" << diag_id << "] " << method
1564 << " success=" << rr.success << " ncc_warped=" << ncc
1565 << " ncc_identity_overlap=" << ncc_identity_overlap
1566 << " overlap_px=" << overlap_pixels
1567 << " threshold=" << (ncc_identity_overlap + min_ncc_improvement)
1568 << " tx=" << rr.warp(0,2) << " ty=" << rr.warp(1,2)
1569 << " rot_deg=" << angle_deg << " scale=" << scale
1570 << " a00=" << rr.warp(0,0) << " a01=" << rr.warp(0,1)
1571 << " a10=" << rr.warp(1,0) << " a11=" << rr.warp(1,1)
1572 << std::endl;
1573 }
1574 // Near-identity bypass: if the warp is geometrically trivial (sub-pixel
1575 // shift + negligible rotation) AND does not degrade NCC significantly,
1576 // bilinear interpolation may slightly lower NCC vs. identity. Accept the
1577 // warp and report ncc_identity_overlap as the quality score so downstream
1578 // metrics are not penalised by interpolation blur.
1579 const float shift_total = std::sqrt(rr.warp(0,2)*rr.warp(0,2) +
1580 rr.warp(1,2)*rr.warp(1,2));
1581 const float angle_rad = std::atan2(-rr.warp(0,1), rr.warp(0,0));
1582 const float angle_abs = std::fabs(angle_rad) * (180.0f / 3.14159265f);
1583 // Only bypass the NCC gate when the warp is small AND non-degrading AND
1584 // the frame is already well-aligned with the reference (ncc_identity > 0.7).
1585 // For frames far from the reference a near-zero warp from star_pair just
1586 // means the method found no valid shift, not that the frame is already
1587 // aligned. A drop > 0.02 below blurred-identity indicates a false warp
1588 // even at small shifts, so we keep the NCC gate in that case.
1589 const bool near_identity = (shift_total < rcfg.star_inlier_tol_px) &&
1590 (angle_abs < 0.1f) &&
1591 (ncc >= ncc_identity_overlap - 0.02f) &&
1592 (out.ncc_identity > 0.7f);
1593 if (overlap_pixels <= 16 ||
1594 (!near_identity && ncc < ncc_identity_overlap + min_ncc_improvement))
1595 return false; // warp doesn't improve alignment — reject
1596 out.reg = rr;
1597 // For near-identity warps use the uninterpolated NCC so downstream
1598 // quality metrics are not degraded by interpolation blur.
1599 out.reg.correlation = near_identity ? ncc_identity_overlap : ncc;
1600 out.method_used = method;
1601 out.ncc_warped = out.reg.correlation;
1602 return true;
1603 };
1604
1605 auto try_star_methods = [&]() -> bool {
1606 bool ok = try_method(
1607 triangle_star_matching(mov, ref, rcfg.allow_rotation,
1608 rcfg.star_topk, rcfg.star_min_inliers,
1609 rcfg.star_inlier_tol_px,
1610 rcfg.transform_model,
1611 rcfg.enable_local_background_subtraction,
1612 rcfg.star_shift_radius_px),
1613 "triangle");
1614 if (!ok && rcfg.enable_star_pair_fallback) {
1615 ok = try_method(
1616 star_registration_similarity(
1617 mov, ref, rcfg.allow_rotation, rcfg.star_topk,
1618 rcfg.star_min_inliers, rcfg.star_inlier_tol_px,
1619 rcfg.star_dist_bin_px, rcfg.transform_model,
1620 rcfg.enable_local_background_subtraction),
1621 "star_pair");
1622 }
1623 return ok;
1624 };
1625
1626 // 1) Primary engine
1627 if (!accepted) {
1628 if (rcfg.engine == "triangle_star_matching" ||
1629 rcfg.engine == "star_similarity" || rcfg.engine.empty()) {
1630 accepted = try_star_methods();
1631 } else if (rcfg.engine == "opencv_feature") {
1632 accepted = try_method(
1633 feature_registration_similarity(mov, ref, rcfg.allow_rotation,
1634 rcfg.transform_model),
1635 "akaze");
1636 } else if (rcfg.engine == "robust_phase_ecc") {
1637 accepted = try_method(
1638 robust_phase_ecc(mov, ref, rcfg.allow_rotation),
1639 "robust_phase_ecc");
1640 } else {
1641 // Default: triangle
1642 accepted = try_method(
1643 triangle_star_matching(mov, ref, rcfg.allow_rotation,
1644 rcfg.star_topk, rcfg.star_min_inliers,
1645 rcfg.star_inlier_tol_px,
1646 rcfg.transform_model,
1647 rcfg.enable_local_background_subtraction,
1648 rcfg.star_shift_radius_px),
1649 "triangle");
1650 }
1651 }
1652
1653 // 2) Fallback cascade
1654 if (!accepted) {
1655 accepted = try_method(
1656 feature_registration_similarity(mov, ref, rcfg.allow_rotation,
1657 rcfg.transform_model),
1658 "akaze");
1659 }
1660 if (!accepted) {
1661 accepted = try_method(
1662 robust_phase_ecc(mov, ref, rcfg.allow_rotation),
1663 "robust_phase_ecc");
1664 }
1665
1666 // 3) Final fallback: identity
1667 if (!accepted) {
1668 out.reg.warp = identity_warp();
1669 out.reg.correlation = 0.0f;
1670 out.reg.success = false;
1671 out.reg.error_message = "identity_fallback";
1672 out.method_used = "identity";
1673 out.ncc_warped = out.ncc_identity;
1674 }
1675
1676 return out;
1677}
