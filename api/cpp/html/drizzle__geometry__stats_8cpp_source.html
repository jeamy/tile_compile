53void stamp_context(int source_width, int source_height, int canvas_width_native,
54 int canvas_height_native, int internal_scale, int chunk_rows,
55 int internal_height, int prepared_frames,
56 int prepared_local_frames) {
57 auto &r = registry();
58 if (!r.enabled)
59 return;
60 r.source_width = source_width;
61 r.source_height = source_height;
62 r.canvas_width_native = canvas_width_native;
63 r.canvas_height_native = canvas_height_native;
64 r.internal_scale = internal_scale;
65 r.resolved_chunk_rows = chunk_rows;
66 r.stripe_count =
67 chunk_rows > 0 ? (internal_height + chunk_rows - 1) / chunk_rows : 0;
68 r.prepared_frames = prepared_frames;
69 r.prepared_local_frames = prepared_local_frames;
70}

72std::string to_json() {

84 << ",\"prepared_frames\":" << r.prepared_frames
85 << ",\"prepared_local_frames\":" << r.prepared_local_frames << "}";
86 o << ",\"variants\":{";
87 bool first = true;
88 for (int i = 0; i < static_cast<int>(Variant::kVariantCount); ++i) {
89 const auto &c = r.v[static_cast<std::size_t>(i)];
90 // Skip variants that were never touched to keep the artifact compact.
91 const bool touched = c.enumerate_calls || c.source_samples_visited ||
92 c.top_level_sample_leaves_calls ||
93 c.leaf_cells_emitted;
94 if (!touched)
95 continue;
96 if (!first)
97 o << ",";
98 first = false;
99 o << "\"" << variant_name(static_cast<Variant>(i)) << "\":{"
100 << "\"pixfrac\":" << c.pixfrac
101 << ",\"enumerate_calls\":" << c.enumerate_calls
102 << ",\"source_rows_scanned\":" << c.source_rows_scanned
103 << ",\"source_samples_visited\":" << c.source_samples_visited
104 << ",\"top_level_sample_leaves_calls\":" << c.top_level_sample_leaves_calls
105 << ",\"sample_leaves_discarded\":" << c.sample_leaves_discarded
106 << ",\"leaves_generated\":" << c.leaves_generated
107 << ",\"subdivide_local_calls\":" << c.subdivide_local_calls
108 << ",\"local_forward_calls\":" << c.local_forward_calls
109 << ",\"invert_calls\":" << c.invert_calls
110 << ",\"invert_iterations\":" << c.invert_iterations
111 << ",\"basis_evaluations\":" << c.invert_iterations
112 << ",\"exp_calls\":" << (c.invert_iterations * 16u)
113 << ",\"leaf_cells_emitted\":" << c.leaf_cells_emitted
114 << ",\"geometry_wall_s\":" << c.geometry_wall_s
115 << ",\"geometry_cpu_s\":" << c.geometry_cpu_s << "}";
116 }
117 o << "}}";

119}
