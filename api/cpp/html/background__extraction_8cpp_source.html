69float robust_median_inplace(std::vector<float> &values) {
70 if (values.empty())
71 return 0.0f;

198std::vector<float> sigma_clipped_values(std::vector<float> values,
199 float sigma = 3.0f,
200 int max_iters = 5) {
201 values.erase(std::remove_if(values.begin(), values.end(),
202 [](float v) { return !std::isfinite(v); }),
203 values.end());
204 for (int iter = 0; iter < max_iters && values.size() >= 8; ++iter) {
205 std::vector<float> work = values;
206 const float center = robust_median_inplace(work);
207 const float scale = 1.4826f * robust_mad(values, center);
208 if (!(std::isfinite(scale) && scale > kTiny))
209 break;
210 const float limit = sigma * scale;
211 std::vector<float> kept;
212 kept.reserve(values.size());
213 for (float v : values) {
214 if (std::abs(v - center) <= limit)
215 kept.push_back(v);
216 }
217 if (kept.size() == values.size() || kept.size() < 8)
218 break;
219 values.swap(kept);
220 }
221 return values;
222}

224float biweight_location(std::vector<float> values) {
225 values.erase(std::remove_if(values.begin(), values.end(),
226 [](float v) { return !std::isfinite(v); }),
227 values.end());
228 if (values.empty())
229 return std::numeric_limits<float>::quiet_NaN();
230 std::vector<float> work = values;
231 const float m = robust_median_inplace(work);
232 const float mad = robust_mad(values, m);
233 const float scale = 9.0f * std::max(mad, kTiny);
234 double num = 0.0;
235 double den = 0.0;
236 for (float v : values) {
237 const float u = (v - m) / scale;
238 if (std::abs(u) >= 1.0f)
239 continue;
240 const float one_minus_u2 = 1.0f - u * u;
241 const float w = one_minus_u2 * one_minus_u2;
242 num += static_cast<double>(v) * static_cast<double>(w);
243 den += static_cast<double>(w);
244 }
245 if (den <= 0.0)
246 return m;
247 return static_cast<float>(num / den);
248}

250float estimate_tile_background_value(std::vector<float> pixels,
251 const BGEConfig &config) {
252 pixels.erase(std::remove_if(pixels.begin(), pixels.end(),
253 [](float v) { return !std::isfinite(v); }),
254 pixels.end());
255 if (pixels.empty())
256 return std::numeric_limits<float>::quiet_NaN();
257
258 if (config.sample_estimator == "sigma_clipped_median") {
259 auto clipped = sigma_clipped_values(std::move(pixels));
260 if (clipped.empty())
261 return std::numeric_limits<float>::quiet_NaN();
262 return robust_median_inplace(clipped);
263 }
264
265 if (config.sample_estimator == "sextractor_mode") {
266 auto clipped = sigma_clipped_values(std::move(pixels));
267 if (clipped.empty())
268 return std::numeric_limits<float>::quiet_NaN();
269 std::vector<float> med_work = clipped;
270 const float median = robust_median_inplace(med_work);
271 const float mean = robust_mean(clipped);
272 const float mode = 2.5f * median - 1.5f * mean;
273 const float denom = std::max(1.0f, std::abs(median));
274 if (!std::isfinite(mode) || std::abs(mode - median) > 0.30f * denom)
275 return median;
276 return mode;
277 }
278
279 if (config.sample_estimator == "biweight") {
280 return biweight_location(std::move(pixels));
281 }
282
283 return robust_quantile_inplace(pixels, config.sample_quantile);
284}

299void box_blur_subregion(const Matrix2Df &image, int x0, int y0, int w, int h,
300 int radius, std::vector<double> *integral,
301 std::vector<float> *out) {
302 out->assign(static_cast<size_t>(std::max(0, h * w)), 0.0f);
303 if (h <= 0 || w <= 0)
304 return;
305 radius = std::max(0, radius);
306
307 integral->assign(static_cast<size_t>((h + 1) * (w + 1)), 0.0);
308 auto idx = [w](int y, int x) { return static_cast<size_t>(y * (w + 1) + x); };
309 const int stride = static_cast<int>(image.cols());
310 const float *base = image.data();
311 for (int y = 0; y < h; ++y) {
312 double row_sum = 0.0;
313 const float *src =
314 base + static_cast<size_t>(y0 + y) * static_cast<size_t>(stride) + x0;
315 for (int x = 0; x < w; ++x) {
316 float v = src[x];
317 if (!std::isfinite(v))
318 v = 0.0f;
319 row_sum += static_cast<double>(v);
320 (*integral)[idx(y + 1, x + 1)] = (*integral)[idx(y, x + 1)] + row_sum;
321 }
322 }
323
324 for (int y = 0; y < h; ++y) {
325 const int y0 = std::max(0, y - radius);
326 const int y1 = std::min(h - 1, y + radius);
327 for (int x = 0; x < w; ++x) {
328 const int x0 = std::max(0, x - radius);
329 const int x1 = std::min(w - 1, x + radius);
330 const double sum = (*integral)[idx(y1 + 1, x1 + 1)] -
331 (*integral)[idx(y0, x1 + 1)] -
332 (*integral)[idx(y1 + 1, x0)] +
333 (*integral)[idx(y0, x0)];
334 const int area = (y1 - y0 + 1) * (x1 - x0 + 1);
335 (*out)[static_cast<size_t>(y * w + x)] =
336 static_cast<float>(sum / std::max(1, area));
337 }
338 }

345void dilate_mask_in_place(std::vector<uint8_t> *mask, int w, int h, int radius,
346 std::vector<uint8_t> *scratch) {
347 if (radius <= 0)
348 return;
349 scratch->assign(mask->size(), 0);
350 const int r2 = radius * radius;
351 for (int y = 0; y < h; ++y) {
352 for (int x = 0; x < w; ++x) {
353 uint8_t hit = 0;
354 for (int dy = -radius; dy <= radius && !hit; ++dy) {
355 const int yy = y + dy;
356 if (yy < 0 || yy >= h)
357 continue;
358 for (int dx = -radius; dx <= radius; ++dx) {
359 const int xx = x + dx;
360 if (xx < 0 || xx >= w)
361 continue;
362 if (dx * dx + dy * dy > r2)
363 continue;
364 if ((*mask)[static_cast<size_t>(yy * w + xx)] != 0) {
365 hit = 1;
366 break;
367 }
368 }
369 }
370 (*scratch)[static_cast<size_t>(y * w + x)] = hit;
371 }
372 }
373 mask->swap(*scratch);
374}

441float compute_structure_score_from_background_mask(
442 const Matrix2Df &channel, int x0, int y0, int tw, int th, float grad_thresh,
443 float noise_scale, const TileSampleScratch &scratch,
444 TileSampleScratch *scratch_mut) {

446 const float noise2 =
447 std::max(kTiny * kTiny, noise_scale * noise_scale);
448 box_blur_subregion(channel, x0, y0, tw, th, kStructureBlurRadiusPx,
449 &scratch_mut->blur_integral, &scratch_mut->blur_large);
450 scratch_mut->structure_values.clear();

464 scratch_mut->structure_values.push_back((hp * hp) / noise2);

468 return std::numeric_limits<float>::infinity();

568 sum += static_cast<double>(v);

643float spatial_background_spread(const Matrix2Df &img,
644 const std::vector<uint8_t> *valid_mask) {
645 const int H = static_cast<int>(img.rows());
646 const int W = static_cast<int>(img.cols());
647 if (H <= 0 || W <= 0)
648 return std::numeric_limits<float>::infinity();
649
650 const size_t total_px = static_cast<size_t>(H * W);
651 const bool use_mask =
652 (valid_mask != nullptr && valid_mask->size() == total_px);
653
654 std::vector<float> valid_values;
655 valid_values.reserve(total_px);
656 for (int y = 0; y < H; ++y) {
657 for (int x = 0; x < W; ++x) {
658 const size_t i = static_cast<size_t>(y * W + x);
659 if (use_mask && (*valid_mask)[i] == 0)
660 continue;
661 const float v = img.data()[i];
662 if (std::isfinite(v) && v > 0.0f)
663 valid_values.push_back(v);
664 }
665 }
666 if (valid_values.size() < 4096)
667 return std::numeric_limits<float>::infinity();
668
669 const float value_thresh = robust_quantile_inplace(valid_values, 0.65f);
670
671 std::vector<float> grad_map(static_cast<size_t>(H * W), 0.0f);
672 std::vector<float> grad_values;
673 grad_values.reserve(valid_values.size());
674 for (int y = 0; y < H; ++y) {
675 const int ym = std::max(0, y - 1);
676 const int yp = std::min(H - 1, y + 1);
677 for (int x = 0; x < W; ++x) {
678 const size_t idx = static_cast<size_t>(y * W + x);
679 if (use_mask && (*valid_mask)[idx] == 0)
680 continue;
681 const float v = img(y, x);
682 if (!(std::isfinite(v) && v > 0.0f))
683 continue;
684
685 const int xm = std::max(0, x - 1);
686 const int xp = std::min(W - 1, x + 1);
687
688 float vxm = img(y, xm);
689 float vxp = img(y, xp);
690 float vym = img(ym, x);
691 float vyp = img(yp, x);
692
693 if (!std::isfinite(vxm))
694 vxm = v;
695 if (!std::isfinite(vxp))
696 vxp = v;
697 if (!std::isfinite(vym))
698 vym = v;
699 if (!std::isfinite(vyp))
700 vyp = v;
701
702 const float g = std::abs(vxp - vxm) + std::abs(vyp - vym);
703 grad_map[idx] = g;
704 grad_values.push_back(g);
705 }
706 }
707 if (grad_values.size() < 4096)
708 return std::numeric_limits<float>::infinity();
709
710 const float grad_thresh = robust_quantile_inplace(grad_values, 0.70f);
711 constexpr int kBlockSize = 128;
712 constexpr int kMinPixelsPerBlock = 256;
713
714 std::vector<float> block_medians;
715 std::vector<float> block_values;
716 for (int y0 = 0; y0 < H; y0 += kBlockSize) {
717 for (int x0 = 0; x0 < W; x0 += kBlockSize) {
718 const int y1 = std::min(H, y0 + kBlockSize);
719 const int x1 = std::min(W, x0 + kBlockSize);
720 block_values.clear();
721 block_values.reserve(static_cast<size_t>((y1 - y0) * (x1 - x0)));
722
723 for (int y = y0; y < y1; ++y) {
724 for (int x = x0; x < x1; ++x) {
725 const size_t idx = static_cast<size_t>(y * W + x);
726 if (use_mask && (*valid_mask)[idx] == 0)
727 continue;
728 const float v = img(y, x);
729 if (!(std::isfinite(v) && v > 0.0f))
730 continue;
731 if (v > value_thresh)
732 continue;
733 const float g = grad_map[idx];
734 if (!(std::isfinite(g) && g <= grad_thresh))
735 continue;
736 block_values.push_back(v);
737 }
738 }
739
740 if (static_cast<int>(block_values.size()) >= kMinPixelsPerBlock) {
741 block_medians.push_back(robust_median_inplace(block_values));
742 }
743 }
744 }
745
746 if (block_medians.size() < 8)
747 return std::numeric_limits<float>::infinity();
748
749 std::sort(block_medians.begin(), block_medians.end());
750 const float p10 = sorted_quantile(block_medians, 0.10f);
751 const float p90 = sorted_quantile(block_medians, 0.90f);
752 return p90 - p10;
753}

759float coarse_background_plane_slope(const Matrix2Df &img,
760 const std::vector<uint8_t> *valid_mask) {
761 const int H = static_cast<int>(img.rows());
762 const int W = static_cast<int>(img.cols());
763 if (H <= 0 || W <= 0)
764 return std::numeric_limits<float>::infinity();
765
766 const size_t total_px = static_cast<size_t>(H * W);
767 const bool use_mask =
768 (valid_mask != nullptr && valid_mask->size() == total_px);
769
770 std::vector<float> valid_values;
771 valid_values.reserve(total_px);
772 for (int y = 0; y < H; ++y) {
773 for (int x = 0; x < W; ++x) {
774 const size_t i = static_cast<size_t>(y * W + x);
775 if (use_mask && (*valid_mask)[i] == 0)
776 continue;
777 const float v = img.data()[i];
778 if (std::isfinite(v) && v > 0.0f)
779 valid_values.push_back(v);
780 }
781 }
782 if (valid_values.size() < 4096)
783 return std::numeric_limits<float>::infinity();
784
785 const float value_thresh = robust_quantile_inplace(valid_values, 0.65f);
786
787 std::vector<float> grad_map(static_cast<size_t>(H * W), 0.0f);
788 std::vector<float> grad_values;
789 grad_values.reserve(valid_values.size());
790 for (int y = 0; y < H; ++y) {
791 const int ym = std::max(0, y - 1);
792 const int yp = std::min(H - 1, y + 1);
793 for (int x = 0; x < W; ++x) {
794 const size_t idx = static_cast<size_t>(y * W + x);
795 if (use_mask && (*valid_mask)[idx] == 0)
796 continue;
797 const float v = img(y, x);
798 if (!(std::isfinite(v) && v > 0.0f))
799 continue;
800
801 const int xm = std::max(0, x - 1);
802 const int xp = std::min(W - 1, x + 1);
803
804 float vxm = img(y, xm);
805 float vxp = img(y, xp);
806 float vym = img(ym, x);
807 float vyp = img(yp, x);
808 if (!std::isfinite(vxm))
809 vxm = v;
810 if (!std::isfinite(vxp))
811 vxp = v;
812 if (!std::isfinite(vym))
813 vym = v;
814 if (!std::isfinite(vyp))
815 vyp = v;
816
817 const float g = std::abs(vxp - vxm) + std::abs(vyp - vym);
818 grad_map[idx] = g;
819 grad_values.push_back(g);
820 }
821 }
822 if (grad_values.size() < 4096)
823 return std::numeric_limits<float>::infinity();
824
825 const float grad_thresh = robust_quantile_inplace(grad_values, 0.70f);
826 constexpr int kBlockSize = 128;
827 constexpr int kMinPixelsPerBlock = 256;
828
829 std::vector<std::array<float, 3>> samples;
830 std::vector<float> block_values;
831 for (int y0 = 0; y0 < H; y0 += kBlockSize) {
832 for (int x0 = 0; x0 < W; x0 += kBlockSize) {
833 const int y1 = std::min(H, y0 + kBlockSize);
834 const int x1 = std::min(W, x0 + kBlockSize);
835 block_values.clear();
836 block_values.reserve(static_cast<size_t>((y1 - y0) * (x1 - x0)));
837 for (int y = y0; y < y1; ++y) {
838 for (int x = x0; x < x1; ++x) {
839 const size_t idx = static_cast<size_t>(y * W + x);
840 if (use_mask && (*valid_mask)[idx] == 0)
841 continue;
842 const float v = img(y, x);
843 if (!(std::isfinite(v) && v > 0.0f))
844 continue;
845 if (v > value_thresh)
846 continue;
847 const float g = grad_map[idx];
848 if (!(std::isfinite(g) && g <= grad_thresh))
849 continue;
850 block_values.push_back(v);
851 }
852 }
853 if (static_cast<int>(block_values.size()) < kMinPixelsPerBlock)
854 continue;
855 const float z = robust_median_inplace(block_values);
856 const float cx = 0.5f * static_cast<float>(x0 + x1 - 1);
857 const float cy = 0.5f * static_cast<float>(y0 + y1 - 1);
858 samples.push_back({cx, cy, z});
859 }
860 }
861 if (samples.size() < 8)
862 return std::numeric_limits<float>::infinity();
863
864 double mx = 0.0, my = 0.0, mz = 0.0;
865 for (const auto &s : samples) {
866 mx += s[0];
867 my += s[1];
868 mz += s[2];
869 }
870 const double n = static_cast<double>(samples.size());
871 mx /= n;
872 my /= n;
873 mz /= n;
874
875 double sxx = 0.0, syy = 0.0, sxy = 0.0, sxz = 0.0, syz = 0.0;
876 for (const auto &s : samples) {
877 const double dx = static_cast<double>(s[0]) - mx;
878 const double dy = static_cast<double>(s[1]) - my;
879 const double dz = static_cast<double>(s[2]) - mz;
880 sxx += dx * dx;
881 syy += dy * dy;
882 sxy += dx * dy;
883 sxz += dx * dz;
884 syz += dy * dz;
885 }
886
887 const double det = sxx * syy - sxy * sxy;
888 if (!(std::isfinite(det)) || std::abs(det) < 1.0e-12) {
889 return std::numeric_limits<float>::infinity();
890 }
891
892 const double ax = (sxz * syy - syz * sxy) / det;
893 const double ay = (syz * sxx - sxz * sxy) / det;
894 if (!(std::isfinite(ax) && std::isfinite(ay))) {
895 return std::numeric_limits<float>::infinity();
896 }
897 return static_cast<float>(std::sqrt(ax * ax + ay * ay));
898}

906std::vector<uint8_t> build_chroma_background_mask_from_rgb_impl(
907 const Matrix2Df &R, const Matrix2Df &G, const Matrix2Df &B,
908 const std::vector<uint8_t> *valid_mask) {
909 const int H = static_cast<int>(R.rows());
910 const int W = static_cast<int>(R.cols());
911 std::vector<uint8_t> mask(static_cast<size_t>(std::max(0, H * W)), 0);
912 if (H <= 0 || W <= 0)
913 return mask;
914
915 const size_t total_px = static_cast<size_t>(H * W);
916 const bool use_external_mask =
917 (valid_mask != nullptr && valid_mask->size() == total_px);
918
919 std::vector<float> luma(total_px, 0.0f);
920 std::vector<float> lum_samples;
921 lum_samples.reserve(static_cast<size_t>((H / 2 + 1) * (W / 2 + 1)));
922 for (int y = 0; y < H; ++y) {
923 for (int x = 0; x < W; ++x) {
924 const size_t idx = static_cast<size_t>(y * W + x);
925 if (use_external_mask && (*valid_mask)[idx] == 0)
926 continue;
927 const float rv = R(y, x);
928 const float gv = G(y, x);
929 const float bv = B(y, x);
930 if (!(std::isfinite(rv) && rv > 0.0f && std::isfinite(gv) && gv > 0.0f &&
931 std::isfinite(bv) && bv > 0.0f)) {
932 continue;
933 }
934 const float lv = 0.2126f * rv + 0.7152f * gv + 0.0722f * bv;
935 luma[idx] = lv;
936 if ((y % 2) == 0 && (x % 2) == 0)
937 lum_samples.push_back(lv);
938 }
939 }
940 if (lum_samples.size() < 4096) {
941 // Low-data fallback: treat all finite positive pixels as safe background.
942 for (int y = 0; y < H; ++y) {
943 for (int x = 0; x < W; ++x) {
944 const size_t idx = static_cast<size_t>(y * W + x);
945 mask[idx] = (luma[idx] > 0.0f) ? 1 : 0;
946 }
947 }
948 return mask;
949 }
950
951 const float lum_thresh = robust_quantile(lum_samples, 0.60f);
952 std::vector<float> grad(static_cast<size_t>(H * W), 0.0f);
953 std::vector<float> grad_samples;
954 grad_samples.reserve(lum_samples.size());
955 for (int y = 0; y < H; ++y) {
956 const int ym = std::max(0, y - 1);
957 const int yp = std::min(H - 1, y + 1);
958 for (int x = 0; x < W; ++x) {
959 const int xm = std::max(0, x - 1);
960 const int xp = std::min(W - 1, x + 1);
961 const size_t idx = static_cast<size_t>(y * W + x);
962 const float lv = luma[idx];
963 if (!(std::isfinite(lv) && lv > 0.0f))
964 continue;
965 const float gx = std::abs(luma[static_cast<size_t>(y * W + xp)] -
966 luma[static_cast<size_t>(y * W + xm)]);
967 const float gy = std::abs(luma[static_cast<size_t>(yp * W + x)] -
968 luma[static_cast<size_t>(ym * W + x)]);
969 const float gv = gx + gy;
970 grad[idx] = gv;
971 if ((y % 2) == 0 && (x % 2) == 0)
972 grad_samples.push_back(gv);
973 }
974 }
975 if (grad_samples.size() < 4096) {
976 for (int y = 0; y < H; ++y) {
977 for (int x = 0; x < W; ++x) {
978 const size_t idx = static_cast<size_t>(y * W + x);
979 mask[idx] = (luma[idx] > 0.0f && luma[idx] <= lum_thresh) ? 1 : 0;
980 }
981 }
982 return mask;
983 }
984
985 const float grad_thresh = robust_quantile(grad_samples, 0.70f);
986 for (int y = 0; y < H; ++y) {
987 for (int x = 0; x < W; ++x) {
988 const size_t idx = static_cast<size_t>(y * W + x);
989 const float lv = luma[idx];
990 const float gv = grad[idx];
991 mask[idx] = (std::isfinite(lv) && lv > 0.0f && lv <= lum_thresh &&
992 std::isfinite(gv) && gv <= grad_thresh)
993 ? 1
994 : 0;
995 }
996 }
997 return mask;
998}

1010

1014 for (int x = 0; x < W; ++x) {

1033 }

1086void extract_connected_components_above_threshold(
1087 const Matrix2Df &luma, float threshold,
1088 std::vector<ForegroundComponent> *components, std::vector<int> *labels) {
1089 components->clear();
1090 const int H = static_cast<int>(luma.rows());
1091 const int W = static_cast<int>(luma.cols());
1092 labels->assign(static_cast<size_t>(H * W), 0);
1093 if (H <= 0 || W <= 0)
1094 return;
1095
1096 auto is_fg = [&](int y, int x) {
1097 const float v = luma(y, x);
1098 return std::isfinite(v) && v > threshold;
1099 };
1100
1101 int next_label = 1;
1102 std::queue<int> q;
1103 const int dx8[8] = {-1, 0, 1, -1, 1, -1, 0, 1};
1104 const int dy8[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
1105
1106 for (int y = 0; y < H; ++y) {
1107 for (int x = 0; x < W; ++x) {
1108 const int root = y * W + x;
1109 if ((*labels)[static_cast<size_t>(root)] != 0)
1110 continue;
1111 if (!is_fg(y, x))
1112 continue;
1113
1114 ForegroundComponent comp;
1115 comp.label = next_label;
1116 comp.area = 0;
1117 comp.peak = luma(y, x);
1118 comp.peak_x = x;
1119 comp.peak_y = y;
1120 comp.min_x = x;
1121 comp.max_x = x;
1122 comp.min_y = y;
1123 comp.max_y = y;
1124
1125 (*labels)[static_cast<size_t>(root)] = next_label;
1126 q.push(root);
1127
1128 while (!q.empty()) {
1129 const int idx = q.front();
1130 q.pop();
1131 const int cy = idx / W;
1132 const int cx = idx - cy * W;
1133 const float cv = luma(cy, cx);
1134
1135 comp.pixels.push_back(idx);
1136 ++comp.area;
1137 if (cv > comp.peak) {
1138 comp.peak = cv;
1139 comp.peak_x = cx;
1140 comp.peak_y = cy;
1141 }
1142 comp.min_x = std::min(comp.min_x, cx);
1143 comp.max_x = std::max(comp.max_x, cx);
1144 comp.min_y = std::min(comp.min_y, cy);
1145 comp.max_y = std::max(comp.max_y, cy);
1146
1147 for (int k = 0; k < 8; ++k) {
1148 const int nx = cx + dx8[k];
1149 const int ny = cy + dy8[k];
1150 if (nx < 0 || ny < 0 || nx >= W || ny >= H)
1151 continue;
1152 const int nidx = ny * W + nx;
1153 int &lab = (*labels)[static_cast<size_t>(nidx)];
1154 if (lab != 0)
1155 continue;
1156 if (!is_fg(ny, nx))
1157 continue;
1158 lab = next_label;
1159 q.push(nidx);
1160 }
1161 }
1162
1163 if (comp.area >= 4) {
1164 components->push_back(std::move(comp));
1165 ++next_label;
1166 } else {
1167 for (int idx : comp.pixels) {
1168 (*labels)[static_cast<size_t>(idx)] = 0;
1169 }
1170 }
1171 }
1172 }
1173}

1180build_disk_offsets_table(int max_radius) {

1190 offsets.push_back({dx, dy});
1191 }
1192 }
1193 }
1194 return table;

1197std::vector<uint8_t>
1202build_modeled_foreground_mask(const Matrix2Df &luma, const BGEConfig &config,
1203 std::vector<ForegroundComponent> *components,
1204 float *out_threshold, float *out_sigma) {
1205 const int H = static_cast<int>(luma.rows());
1206 const int W = static_cast<int>(luma.cols());
1207 std::vector<uint8_t> out_mask(static_cast<size_t>(std::max(0, H * W)), 0);
1208 if (H <= 0 || W <= 0)
1209 return out_mask;
1210
1211 std::vector<float> sample_vals;
1212 sample_vals.reserve(static_cast<size_t>((H / 4 + 1) * (W / 4 + 1)));
1213 for (int y = 0; y < H; y += 4) {
1214 for (int x = 0; x < W; x += 4) {
1215 const float v = luma(y, x);
1216 if (std::isfinite(v) && v > 0.0f)
1217 sample_vals.push_back(v);
1218 }
1219 }
1220 if (sample_vals.size() < 256)
1221 return out_mask;
1222
1223 const float med = robust_median_inplace(sample_vals);
1224 const float sigma = std::max(1.0e-6f, 1.4826f * robust_mad(sample_vals, med));
1225 const float thresh = med + 0.8f * sigma;
1226 if (out_threshold)
1227 *out_threshold = thresh;
1228 if (out_sigma)
1229 *out_sigma = sigma;
1230
1231 std::vector<int> labels;
1232 extract_connected_components_above_threshold(luma, thresh, components,
1233 &labels);
1234 if (components->empty())
1235 return out_mask;
1236
1237 const int base_radius = std::max(1, config.mask.star_dilate_px);
1238 const int max_radius = base_radius + 20;
1239 const auto offsets_table = build_disk_offsets_table(max_radius);
1240 constexpr float kPi = 3.14159265359f;
1241
1242 for (auto &comp : *components) {
1243 const float peak_sigma =
1244 std::max(0.0f, (comp.peak - thresh) / std::max(1.0e-6f, sigma));
1245 const float area_scale =
1246 std::sqrt(static_cast<float>(std::max(1, comp.area)) / kPi);
1247 int rad =
1248 base_radius + static_cast<int>(std::lround(
1249 1.6f * std::log1p(peak_sigma) + 0.15f * area_scale));
1250 rad = std::clamp(rad, base_radius, max_radius);
1251 comp.dilation_radius = rad;
1252
1253 const auto &offsets = offsets_table[static_cast<size_t>(rad)];
1254 for (int idx : comp.pixels) {
1255 const int y = idx / W;
1256 const int x = idx - y * W;
1257 for (const auto &[dx, dy] : offsets) {
1258 const int xx = x + dx;
1259 const int yy = y + dy;
1260 if (xx < 0 || yy < 0 || xx >= W || yy >= H)
1261 continue;
1262 out_mask[static_cast<size_t>(yy * W + xx)] = 1;
1263 }
1264 }
1265 }
1266 return out_mask;
1267}

1273float robust_mesh_background_estimate(std::vector<float> values) {
1274 if (values.size() < 16)
1275 return std::numeric_limits<float>::quiet_NaN();
1276 for (int iter = 0; iter < 4; ++iter) {
1277 const float med = robust_median_inplace(values);
1278 const float sigma = 1.4826f * robust_mad(values, med);
1279 if (!(std::isfinite(sigma) && sigma > 1.0e-6f))
1280 break;
1281 const float clip = 2.5f * sigma;
1282 std::vector<float> kept;
1283 kept.reserve(values.size());
1284 for (float v : values) {
1285 if (std::isfinite(v) && std::abs(v - med) <= clip)
1286 kept.push_back(v);
1287 }
1288 if (kept.size() < 16 || kept.size() == values.size())
1289 break;
1290 values.swap(kept);
1291 }
1292 if (values.size() < 8)
1293 return std::numeric_limits<float>::quiet_NaN();
1294 return robust_median_inplace(values);

1355 const float q995 = sampled_positive_quantile(luma, 0.995f, 4);
1356 const float bright_thresh =
1357 std::max(q995, low_threshold + 6.0f * std::max(1.0e-6f, sigma));
1358
1359 std::vector<int> order(components.size());
1360 for (size_t i = 0; i < components.size(); ++i)
1361 order[i] = static_cast<int>(i);
1362 std::sort(order.begin(), order.end(), [&](int a, int b) {
1363 return components[static_cast<size_t>(a)].peak >
1364 components[static_cast<size_t>(b)].peak;
1365 });
1366
1367 constexpr float kPi = 3.14159265359f;
1368 int modeled = 0;
1369 const int max_models = 1200;
1370 for (int oi : order) {
1371 if (modeled >= max_models)
1372 break;
1373 const auto &comp = components[static_cast<size_t>(oi)];
1374 if (comp.area <= 2 || comp.area > 256)
1375 continue;
1376 if (!(std::isfinite(comp.peak) && comp.peak >= bright_thresh))
1377 continue;
1378
1379 const int cx = comp.peak_x;
1380 const int cy = comp.peak_y;
1381 if (cx < 0 || cy < 0 || cx >= W || cy >= H)
1382 continue;
1383
1384 const int ann_in = std::max(3, comp.dilation_radius + 2);
1385 const int ann_out = ann_in + 10;
1386 const float bg_local =
1387 annulus_background_median(sky_work, cx, cy, ann_in, ann_out, fg_mask);
1388 if (!(std::isfinite(bg_local) && bg_local > 0.0f))
1389 continue;
1390
1391 const float peak_v = sky_work(cy, cx);
1392 const float amp = peak_v - bg_local;
1393 if (!(std::isfinite(amp) && amp > 0.0f))
1394 continue;
1395
1396 const float sigma_px = std::clamp(
1397 0.35f * std::sqrt(static_cast<float>(std::max(1, comp.area)) / kPi),
1398 1.2f, 8.0f);
1399 const float sigma2 = sigma_px * sigma_px;
1400 const int radius =
1401 std::clamp(static_cast<int>(std::ceil(3.0f * sigma_px +
1402 0.5f * comp.dilation_radius)),
1403 4, 28);
1404 const int r2_max = radius * radius;
1405
1406 const int y0 = std::max(0, cy - radius);
1407 const int y1 = std::min(H - 1, cy + radius);
1408 const int x0 = std::max(0, cx - radius);
1409 const int x1 = std::min(W - 1, cx + radius);
1410 for (int y = y0; y <= y1; ++y) {
1411 for (int x = x0; x <= x1; ++x) {
1412 const int dx = x - cx;
1413 const int dy = y - cy;
1414 const int r2 = dx * dx + dy * dy;
1415 if (r2 > r2_max)
1416 continue;
1417 const float model = amp * std::exp(-0.5f * static_cast<float>(r2) /
1418 std::max(sigma2, 1.0e-6f));
1419 const float v = sky_work(y, x);
1420 if (!std::isfinite(v))
1421 continue;
1422 sky_work(y, x) = std::max(bg_local, v - model);
1423 }
1424 }
1425 ++modeled;
1426 }
1427
1428 std::cout << "[BGE] modeled_mask_mesh: bright-source models applied="
1429 << modeled << std::endl;
1430}

1436MeshSkyFitResult fit_modeled_mask_mesh_surface(
1437 const Matrix2Df &channel, const Matrix2Df &luma,
1438 const std::vector<uint8_t> &fg_mask,
1439 const std::vector<ForegroundComponent> &components, float low_threshold,
1440 float sigma, const BGEConfig &config, int tile_size_hint) {
1441 MeshSkyFitResult out;
1442 const int H = static_cast<int>(channel.rows());
1443 const int W = static_cast<int>(channel.cols());
1444 if (H <= 0 || W <= 0)
1445 return out;
1446
1447 Matrix2Df sky_work = channel;
1448 subtract_bright_source_models(sky_work, luma, fg_mask, components,
1449 low_threshold, sigma);
1450
1451 const int min_dim = std::min(W, H);
1452 int mesh = min_dim / std::max(8, config.grid.N_g);
1453 const int mesh_min = std::max(32, config.grid.G_min_px / 2);
1454 int mesh_max = std::max(config.grid.G_min_px, min_dim / 6);
1455 if (tile_size_hint > 0)
1456 mesh_max = std::max(mesh_max, tile_size_hint);
1457 mesh = std::clamp(mesh, mesh_min, mesh_max);
1458 out.mesh_size = std::max(16, mesh);
1459
1460 const int nx = (W + out.mesh_size - 1) / out.mesh_size;
1461 const int ny = (H + out.mesh_size - 1) / out.mesh_size;
1462 if (nx <= 0 || ny <= 0)
1463 return out;
1464
1465 std::vector<float> cell_values(static_cast<size_t>(nx * ny),
1466 std::numeric_limits<float>::quiet_NaN());
1467 std::vector<int> cell_counts(static_cast<size_t>(nx * ny), 0);
1468 std::vector<uint8_t> cell_valid(static_cast<size_t>(nx * ny), 0);
1469
1470 for (int cy = 0; cy < ny; ++cy) {
1471 const int y0 = cy * out.mesh_size;
1472 const int y1 = std::min(H, y0 + out.mesh_size);
1473 for (int cx = 0; cx < nx; ++cx) {
1474 const int x0 = cx * out.mesh_size;
1475 const int x1 = std::min(W, x0 + out.mesh_size);
1476 const int area = std::max(1, (y1 - y0) * (x1 - x0));
1477 const int min_required = std::max(48, area / 16);
1478
1479 std::vector<float> vals;
1480 vals.reserve(static_cast<size_t>(area / 2));
1481 for (int y = y0; y < y1; ++y) {
1482 for (int x = x0; x < x1; ++x) {
1483 const size_t idx = static_cast<size_t>(y * W + x);
1484 if (idx < fg_mask.size() && fg_mask[idx] != 0)
1485 continue;
1486 const float v = sky_work(y, x);
1487 if (std::isfinite(v) && v > 0.0f)
1488 vals.push_back(v);
1489 }
1490 }
1491
1492 const size_t i = static_cast<size_t>(cy * nx + cx);
1493 cell_counts[i] = static_cast<int>(vals.size());
1494 if (cell_counts[i] < min_required)
1495 continue;
1496
1497 const float bg = robust_mesh_background_estimate(std::move(vals));
1498 if (std::isfinite(bg) && bg > 0.0f) {
1499 cell_values[i] = bg;
1500 cell_valid[i] = 1;
1501 ++out.n_valid_cells;
1502 }
1503 }
1504 }
1505
1506 if (out.n_valid_cells < 8)
1507 return out;
1508
1509 std::vector<int> valid_idx;
1510 valid_idx.reserve(static_cast<size_t>(out.n_valid_cells));
1511 for (int i = 0; i < nx * ny; ++i) {
1512 if (cell_valid[static_cast<size_t>(i)] != 0)
1513 valid_idx.push_back(i);
1514 }
1515
1516 for (int cy = 0; cy < ny; ++cy) {
1517 for (int cx = 0; cx < nx; ++cx) {
1518 const size_t i = static_cast<size_t>(cy * nx + cx);
1519 if (cell_valid[i] != 0)
1520 continue;
1521 float best_d2 = std::numeric_limits<float>::infinity();
1522 float best_v = std::numeric_limits<float>::quiet_NaN();
1523 for (int vi : valid_idx) {
1524 const int vy = vi / nx;
1525 const int vx = vi - vy * nx;
1526 const float dx = static_cast<float>(cx - vx);
1527 const float dy = static_cast<float>(cy - vy);
1528 const float d2 = dx * dx + dy * dy;
1529 if (d2 < best_d2) {
1530 best_d2 = d2;
1531 best_v = cell_values[static_cast<size_t>(vi)];
1532 }
1533 }
1534 if (std::isfinite(best_v)) {
1535 cell_values[i] = best_v;
1536 }
1537 }
1538 }
1539
1540 out.model = Matrix2Df::Zero(H, W);
1541 for (int y = 0; y < H; ++y) {
1542 const float gy =
1543 (static_cast<float>(y) + 0.5f) / static_cast<float>(out.mesh_size) -
1544 0.5f;
1545 int y0 = static_cast<int>(std::floor(gy));
1546 float ty = gy - static_cast<float>(y0);
1547 y0 = std::clamp(y0, 0, ny - 1);
1548 const int y1 = std::clamp(y0 + 1, 0, ny - 1);
1549 if (y1 == y0)
1550 ty = 0.0f;
1551
1552 for (int x = 0; x < W; ++x) {
1553 const float gx =
1554 (static_cast<float>(x) + 0.5f) / static_cast<float>(out.mesh_size) -
1555 0.5f;
1556 int x0 = static_cast<int>(std::floor(gx));
1557 float tx = gx - static_cast<float>(x0);
1558 x0 = std::clamp(x0, 0, nx - 1);
1559 const int x1 = std::clamp(x0 + 1, 0, nx - 1);
1560 if (x1 == x0)
1561 tx = 0.0f;
1562
1563 const float v00 = cell_values[static_cast<size_t>(y0 * nx + x0)];
1564 const float v10 = cell_values[static_cast<size_t>(y0 * nx + x1)];
1565 const float v01 = cell_values[static_cast<size_t>(y1 * nx + x0)];
1566 const float v11 = cell_values[static_cast<size_t>(y1 * nx + x1)];
1567 const float v0 = (1.0f - tx) * v00 + tx * v10;
1568 const float v1 = (1.0f - tx) * v01 + tx * v11;
1569 out.model(y, x) = (1.0f - ty) * v0 + ty * v1;
1570 }
1571 }
1572
1573 double sum_sq = 0.0;
1574 int n_r = 0;
1575 out.grid_cells.reserve(static_cast<size_t>(nx * ny));
1576 for (int cy = 0; cy < ny; ++cy) {
1577 for (int cx = 0; cx < nx; ++cx) {
1578 const size_t i = static_cast<size_t>(cy * nx + cx);
1579 GridCell gc;
1580 gc.cell_x = cx;
1581 gc.cell_y = cy;
1582 gc.center_x = std::min(static_cast<float>(W - 1),
1583 (static_cast<float>(cx) + 0.5f) * out.mesh_size);
1584 gc.center_y = std::min(static_cast<float>(H - 1),
1585 (static_cast<float>(cy) + 0.5f) * out.mesh_size);
1586 gc.bg_value = cell_values[i];
1587 gc.weight = 1.0f;
1588 gc.n_samples = cell_counts[i];
1589 gc.valid = (cell_valid[i] != 0);
1590 out.grid_cells.push_back(gc);
1591
1592 if (gc.valid && std::isfinite(gc.bg_value)) {
1593 const int px =
1594 std::clamp(static_cast<int>(std::lround(gc.center_x)), 0, W - 1);
1595 const int py =
1596 std::clamp(static_cast<int>(std::lround(gc.center_y)), 0, H - 1);
1597 const float r = gc.bg_value - out.model(py, px);
1598 sum_sq += static_cast<double>(r) * static_cast<double>(r);
1599 ++n_r;
1600 }
1601 }
1602 }
1603
1604 if (n_r > 0) {
1605 out.rms_residual =
1606 static_cast<float>(std::sqrt(sum_sq / static_cast<double>(n_r)));
1607 }
1608 out.success = out.model.size() > 0 && out.model.allFinite();
1609 return out;
1610}

1682// Compute adaptive grid spacing (v3.3 §6.3.8)
1687int compute_grid_spacing(int image_width, int image_height, int tile_size,
1688 const BGEConfig &config) {
1689 int min_dim = std::min(image_width, image_height);
1690
1691 // Spec intent: G must never drop below max(2*T, G_min).
1692 // If G_max_fraction would violate this, clamp G_max up to the required floor.
1693 int G_from_tiles = 2 * tile_size;
1694 int G_from_resolution = min_dim / std::max(1, config.grid.N_g);
1695 const int G_floor = std::max(G_from_tiles, config.grid.G_min_px);
1696 int G_max = static_cast<int>(min_dim * config.grid.G_max_fraction);
1697 G_max = std::max(G_max, G_floor);
1698
1699 int G = std::max(G_floor, G_from_resolution);
1700 G = std::min(G, G_max);
1701
1702 // §6.3.9 compact-tile warning: when tile size forces G >> resolution-based
1703 // estimate, the BGE grid is coarser than ideal (compact-tile mode).
1704 if (G_from_tiles > G_from_resolution && G_from_resolution > 0) {
1705 std::cout << "[BGE] Warning: compact-tile mode detected (2*T=" << G_from_tiles
1706 << " > G_res=" << G_from_resolution
1707 << "); grid spacing forced to G=" << G
1708 << " (§6.3.9). BGE accuracy may be reduced."
1709 << std::endl;
1710 }
1711
1712 return G;
1713}

1758 if (tile_metrics.size() < tile_grid.tiles.size()) {

1909 continue;

1916 continue;
1917 }

1920 robust_quantile_inplace(scratch.finite_values, 0.999f);

1926 box_blur_subregion(channel, x0, y0, tw, th, r_large, &scratch.blur_integral,

1934 dog_med + 3.0f * std::max(1.4826f * dog_mad, 1.0e-6f);

1936 robust_quantile_inplace(scratch.finite_values, 0.80f);
1937 for (int yy = 0; yy < th; ++yy) {
1938 for (int xx = 0; xx < tw; ++xx) {
1939 const size_t i = static_cast<size_t>(yy * tw + xx);
1940 const float v = tile_value(yy, xx);
1941 if (std::isfinite(v) && v >= bright_thresh &&
1942 scratch.dog_vals[i] > dog_thresh) {
1943 scratch.star_mask[i] = 1;
1944 }
1945 }
1946 }
1947 int star_dilate_px = std::max(0, config.mask.star_dilate_px);
1948 if (std::isfinite(tm.fwhm) && tm.fwhm > 0.0f) {
1949 const int add = static_cast<int>(std::lround(0.25f * tm.fwhm));
1950 star_dilate_px = std::clamp(star_dilate_px + std::max(0, add),
1951 star_dilate_px, star_dilate_px + 8);
1952 }
1953 dilate_mask_in_place(&scratch.star_mask, tw, th, star_dilate_px,
1954 &scratch.dilate_scratch);
1955
1956 for (int yy = 0; yy < th; ++yy) {
1957 for (int xx = 0; xx < tw; ++xx) {
1958 const size_t i = static_cast<size_t>(yy * tw + xx);
1959 const float v = tile_value(yy, xx);
1960 if (std::isfinite(v) && v >= sat_level)
1961 scratch.sat_mask[i] = 1;
1962 }
1963 }
1964 dilate_mask_in_place(&scratch.sat_mask, tw, th,
1965 std::max(0, config.mask.sat_dilate_px),
1966 &scratch.dilate_scratch);
1967
1968 scratch.supported_gradients.clear();
1969 for (int yy = 0; yy < th; ++yy) {
1970 const int ym = std::max(0, yy - 1);
1971 const int yp = std::min(th - 1, yy + 1);
1972 for (int xx = 0; xx < tw; ++xx) {
1973 const int xm = std::max(0, xx - 1);
1974 const int xp = std::min(tw - 1, xx + 1);
1975 const float gx = std::abs(tile_value(yy, xp) - tile_value(yy, xm));
1976 const float gy = std::abs(tile_value(yp, xx) - tile_value(ym, xx));
1977 const float grad = gx + gy;
1978 scratch.tile_gradients[static_cast<size_t>(yy * tw + xx)] = grad;
1979 if (scratch.tile_common_support[static_cast<size_t>(yy * tw + xx)] !=
1980 0) {
1981 scratch.supported_gradients.push_back(grad);
1982 }
1983 }
1984 }
1985 std::vector<float> &gradient_source =
1986 scratch.supported_gradients.empty() ? scratch.tile_gradients
1987 : scratch.supported_gradients;
1988 const float grad_thresh = robust_quantile_inplace(
1989 gradient_source, config.structure_thresh_percentile);
1990
1991 scratch.tile_pixels.clear();
1992 for (int yy = 0; yy < th; ++yy) {
1993 for (int xx = 0; xx < tw; ++xx) {
1994 const size_t i = static_cast<size_t>(yy * tw + xx);
1995 if (scratch.tile_common_support[i] == 0) {
1996 continue;
1997 }
1998 const float v = tile_value(yy, xx);
1999 const bool structure_bad = scratch.tile_gradients[i] > grad_thresh;
2000 const bool masked =
2001 (scratch.star_mask[i] != 0) || (scratch.sat_mask[i] != 0) ||
2002 structure_bad;
2003 if (!masked && std::isfinite(v)) {
2004 scratch.tile_pixels.push_back(v);
2005 }
2006 }
2007 }
2008
2009 if (scratch.tile_pixels.empty()) {
2010 prepared_samples[t] = std::move(prepared);
2011 continue;
2012 }
2013
2014 const float usable_fraction =
2015 static_cast<float>(scratch.tile_pixels.size()) /
2016 static_cast<float>(std::max(1, supported_px));
2017 if (!(std::isfinite(usable_fraction)) ||
2018 usable_fraction < kMinUsableTileFraction) {
2019 prepared_samples[t] = std::move(prepared);
2020 continue;
2021 }
2022
2023 std::sort(scratch.tile_pixels.begin(), scratch.tile_pixels.end());
2024 if (!std::isfinite(scratch.tile_pixels.front())) {
2025 prepared_samples[t] = std::move(prepared);
2026 continue;
2027 }
2028
2029 const float masked_fraction = 1.0f - usable_fraction;
2030 const float structure_noise =
2031 estimate_structure_noise_scale(tm, scratch.tile_pixels);
2032 const float structure_score = compute_structure_score_from_background_mask(
2033 channel, x0, y0, tw, th, grad_thresh, structure_noise, scratch,
2034 &scratch);
2035 prepared.weight =
2036 std::exp(-config.tile_weight_lambda_structure * structure_score) *
2037 (1.0f - masked_fraction);
2038 prepared.weight = std::clamp(prepared.weight, 0.0f, 1.0f);
2039 if (!(std::isfinite(prepared.weight) && prepared.weight > 0.0f)) {
2040 prepared_samples[t] = std::move(prepared);
2041 continue;
2042 }
2043 prepared.sorted_pixels = scratch.tile_pixels;
2044 prepared.valid = true;
2045 prepared_samples[t] = std::move(prepared);
2046 }
2047 }
2048
2049 return prepared_samples;
2050}

2052// Aggregate tiles to coarse grid (v3.3 §6.3.3)
2053std::vector<GridCell>
2058aggregate_to_coarse_grid(const std::vector<TileBGSample> &tile_samples,
2059 int image_width, int image_height, int grid_spacing,
2060 const BGEConfig &config) {
2061
2062 // Compute grid dimensions
2063 int n_cells_x = (image_width + grid_spacing - 1) / grid_spacing;
2064 int n_cells_y = (image_height + grid_spacing - 1) / grid_spacing;
2065
2066 // Initialize grid cells
2067 std::vector<std::vector<GridCell>> grid(n_cells_y,
2068 std::vector<GridCell>(n_cells_x));
2069 for (int cy = 0; cy < n_cells_y; ++cy) {
2070 for (int cx = 0; cx < n_cells_x; ++cx) {
2071 auto &cell = grid[cy][cx];
2072 cell.cell_x = cx;
2073 cell.cell_y = cy;
2074 cell.center_x = (cx + 0.5f) * grid_spacing;
2075 cell.center_y = (cy + 0.5f) * grid_spacing;
2076 cell.n_samples = 0;
2077 cell.valid = false;
2078 }
2079 }
2080
2081 const int total_cells = n_cells_x * n_cells_y;
2082 std::vector<std::vector<TileBGSample>> cell_samples(
2083 static_cast<size_t>(total_cells));
2084 std::vector<int> sample_cell_indices(tile_samples.size(), -1);
2085 std::vector<int> cell_counts(static_cast<size_t>(total_cells), 0);
2086
2087 // Robustly suppress globally implausible tile background samples before
2088 // cell aggregation (e.g., bright-object contamination, near-zero artifacts).
2089 std::vector<float> valid_bg_values;
2090 valid_bg_values.reserve(tile_samples.size());
2091 for (const auto &s : tile_samples) {
2092 if (s.valid && std::isfinite(s.bg_value) &&
2093 s.bg_value >= config.min_sample_bg_value) {
2094 valid_bg_values.push_back(s.bg_value);
2095 }
2096 }
2097
2098 float bg_med = 0.0f;
2099 float bg_sigma = 0.0f;
2100 bool have_bg_guard = false;
2101 if (valid_bg_values.size() >= 16) {
2102 bg_med = robust_median_inplace(valid_bg_values);
2103 const float mad = robust_mad(valid_bg_values, bg_med);
2104 bg_sigma = 1.4826f * mad;
2105 have_bg_guard = std::isfinite(bg_sigma) && bg_sigma > kTiny;
2106 }
2107
2108 int n_rejected_global_outliers = 0;
2109 for (size_t sample_idx = 0; sample_idx < tile_samples.size(); ++sample_idx) {
2110 const auto &sample = tile_samples[sample_idx];
2111 if (!sample.valid)
2112 continue;
2113 if (!std::isfinite(sample.bg_value))
2114 continue;
2115 if (sample.bg_value < config.min_sample_bg_value)
2116 continue;
2117
2118 if (have_bg_guard) {
2119 const float lo = bg_med - 5.0f * bg_sigma;
2120 const float hi = bg_med + 4.0f * bg_sigma;
2121 if (sample.bg_value < lo || sample.bg_value > hi) {
2122 ++n_rejected_global_outliers;
2123 continue;
2124 }
2125 }
2126
2127 int cx = static_cast<int>(sample.x / grid_spacing);
2128 int cy = static_cast<int>(sample.y / grid_spacing);
2129
2130 if (cx >= 0 && cx < n_cells_x && cy >= 0 && cy < n_cells_y) {
2131 const int cell_idx = cy * n_cells_x + cx;
2132 sample_cell_indices[sample_idx] = cell_idx;
2133 ++cell_counts[static_cast<size_t>(cell_idx)];
2134 }
2135 }
2136
2137 for (int cell_idx = 0; cell_idx < total_cells; ++cell_idx) {
2138 cell_samples[static_cast<size_t>(cell_idx)].resize(
2139 static_cast<size_t>(cell_counts[static_cast<size_t>(cell_idx)]));
2140 }
2141 std::vector<int> cell_write_offsets(static_cast<size_t>(total_cells), 0);
2142 for (size_t sample_idx = 0; sample_idx < tile_samples.size(); ++sample_idx) {
2143 const int cell_idx = sample_cell_indices[sample_idx];
2144 if (cell_idx < 0)
2145 continue;
2146 auto &samples = cell_samples[static_cast<size_t>(cell_idx)];
2147 const size_t write_offset =
2148 static_cast<size_t>(cell_write_offsets[static_cast<size_t>(cell_idx)]++);
2149 samples[write_offset] = tile_samples[sample_idx];
2150 }
2151
2152 if (n_rejected_global_outliers > 0) {
2153 std::cout << "[BGE] Rejected global sample outliers: "
2154 << n_rejected_global_outliers << std::endl;
2155 }
2156
2157 // Aggregate per cell (v3.3 §6.3.3c)
2158 const int parallel_workers = bge_parallel_worker_count(total_cells, 8);
2159#pragma omp parallel num_threads(parallel_workers) if(parallel_workers > 1)
2160 {
2161 std::vector<float> bg_values;
2162 std::vector<float> weights;
2163#pragma omp for schedule(static)
2164 for (int cell_idx = 0; cell_idx < total_cells; ++cell_idx) {
2165 const int cy = cell_idx / n_cells_x;
2166 const int cx = cell_idx - cy * n_cells_x;
2167 auto &cell = grid[cy][cx];
2168 const auto &samples = cell_samples[static_cast<size_t>(cell_idx)];
2169
2170 cell.n_samples = static_cast<int>(samples.size());
2171
2172 if (cell.n_samples < config.min_tiles_per_cell) {
2173 cell.valid = false;
2174 continue;
2175 }
2176
2177 bg_values.clear();
2178 weights.clear();
2179 bg_values.reserve(samples.size());
2180 weights.reserve(samples.size());
2181
2182 for (const auto &s : samples) {
2183 bg_values.push_back(s.bg_value);
2184 weights.push_back(s.weight);
2185 }
2186
2187 cell.bg_value = robust_median_inplace(bg_values);
2188 cell.weight = robust_median_inplace(weights);
2189 cell.valid = true;
2190 }
2191 }
2192
2193 // Deterministic handling of insufficient cells (v3.3 §6.3.3d)
2194 auto is_valid = [&](int cx, int cy) {
2195 return cy >= 0 && cy < n_cells_y && cx >= 0 && cx < n_cells_x &&
2196 grid[cy][cx].valid;
2197 };
2198
2199 if (config.grid.insufficient_cell_strategy == "nearest") {
2200 for (int cy = 0; cy < n_cells_y; ++cy) {
2201 for (int cx = 0; cx < n_cells_x; ++cx) {
2202 if (grid[cy][cx].valid)
2203 continue;
2204 float best_d2 = std::numeric_limits<float>::infinity();
2205 int best_x = -1;
2206 int best_y = -1;
2207 for (int sy = 0; sy < n_cells_y; ++sy) {
2208 for (int sx = 0; sx < n_cells_x; ++sx) {
2209 if (!grid[sy][sx].valid)
2210 continue;
2211 const float dx = grid[cy][cx].center_x - grid[sy][sx].center_x;
2212 const float dy = grid[cy][cx].center_y - grid[sy][sx].center_y;
2213 const float d2 = dx * dx + dy * dy;
2214 if (d2 < best_d2 ||
2215 (std::abs(d2 - best_d2) <= 1.0e-6f &&
2216 (sy < best_y || (sy == best_y && sx < best_x)))) {
2217 best_d2 = d2;
2218 best_x = sx;
2219 best_y = sy;
2220 }
2221 }
2222 }
2223 if (best_x >= 0 && best_y >= 0) {
2224 grid[cy][cx].bg_value = grid[best_y][best_x].bg_value;
2225 grid[cy][cx].weight = grid[best_y][best_x].weight;
2226 grid[cy][cx].valid = true;
2227 grid[cy][cx].n_samples = std::max(1, grid[best_y][best_x].n_samples);
2228 }
2229 }
2230 }
2231 } else if (config.grid.insufficient_cell_strategy == "radius_expand") {
2232 const int max_radius = std::max(n_cells_x, n_cells_y);
2233 for (int cy = 0; cy < n_cells_y; ++cy) {
2234 for (int cx = 0; cx < n_cells_x; ++cx) {
2235 if (grid[cy][cx].valid)
2236 continue;
2237
2238 std::vector<float> bg_vals;
2239 std::vector<float> wt_vals;
2240 int total_samples = 0;
2241 for (int r = 1; r <= max_radius; ++r) {
2242 for (int sy = std::max(0, cy - r);
2243 sy <= std::min(n_cells_y - 1, cy + r); ++sy) {
2244 for (int sx = std::max(0, cx - r);
2245 sx <= std::min(n_cells_x - 1, cx + r); ++sx) {
2246 if (std::max(std::abs(sy - cy), std::abs(sx - cx)) != r)
2247 continue;
2248 const int sidx = sy * n_cells_x + sx;
2249 for (const auto &s : cell_samples[sidx]) {
2250 bg_vals.push_back(s.bg_value);
2251 wt_vals.push_back(s.weight);
2252 ++total_samples;
2253 }
2254 }
2255 }
2256 if (total_samples >= config.min_tiles_per_cell) {
2257 grid[cy][cx].bg_value = robust_median_inplace(bg_vals);
2258 grid[cy][cx].weight = robust_median_inplace(wt_vals);
2259 grid[cy][cx].valid = true;
2260 grid[cy][cx].n_samples = total_samples;
2261 break;
2262 }
2263 }
2264 }
2265 }
2266 }
2267
2268 // Flatten to vector of valid cells
2269 std::vector<GridCell> valid_cells;
2270 for (int cy = 0; cy < n_cells_y; ++cy) {
2271 for (int cx = 0; cx < n_cells_x; ++cx) {
2272 if (is_valid(cx, cy)) {
2273 valid_cells.push_back(grid[cy][cx]);
2274 }
2275 }
2276 }
2277
2278 return valid_cells;
2279}

2317float catmull_rom(float p0, float p1, float p2, float p3, float t) {
2318 const float t2 = t * t;
2319 const float t3 = t2 * t;
2320 return 0.5f * ((2.0f * p1) + (-p0 + p2) * t +
2321 (2.0f * p0 - 5.0f * p1 + 4.0f * p2 - p3) * t2 +
2322 (-p0 + 3.0f * p1 - 3.0f * p2 + p3) * t3);

2326 int image_width, int image_height,

2330 const int ny = std::max(1, (image_height + grid_spacing - 1) / grid_spacing);
2331 std::vector<float> mesh(static_cast<size_t>(nx * ny),
2332 std::numeric_limits<float>::quiet_NaN());

2334 for (const auto &gc : grid_cells) {
2335 if (!gc.valid || !std::isfinite(gc.bg_value))
2336 continue;
2337 if (gc.cell_x < 0 || gc.cell_x >= nx || gc.cell_y < 0 || gc.cell_y >= ny)
2338 continue;

2340 mesh[idx] = gc.bg_value;
2341 valid[idx] = 1;
2342 }
2343
2344 std::vector<int> valid_idx;
2345 valid_idx.reserve(grid_cells.size());
2346 for (int i = 0; i < nx * ny; ++i) {

2348 valid_idx.push_back(i);

2351 return Matrix2Df();

2353 for (int cy = 0; cy < ny; ++cy) {
2354 for (int cx = 0; cx < nx; ++cx) {

2362 const int vx = vi - vy * nx;

2364 const float dy = static_cast<float>(cy - vy);
2365 const float d2 = dx * dx + dy * dy;
2366 if (d2 < best_d2) {
2367 best_d2 = d2;

2372 }
2373 }

2376 x = std::clamp(x, 0, nx - 1);
2377 y = std::clamp(y, 0, ny - 1);

2382 const int total_pixels = std::max(0, image_width * image_height);
2383 const int parallel_workers = bge_parallel_worker_count(total_pixels, 32768);
2384#pragma omp parallel for collapse(2) num_threads(parallel_workers) \
2385 if(parallel_workers > 1)

2400 const float p0 = mesh_at(iy + ky, ix - 1);
2401 const float p1 = mesh_at(iy + ky, ix);
2402 const float p2 = mesh_at(iy + ky, ix + 1);
2403 const float p3 = mesh_at(iy + ky, ix + 2);
2404 row_vals[ky + 1] = catmull_rom(p0, p1, p2, p3, tx);

2407 catmull_rom(row_vals[0], row_vals[1], row_vals[2], row_vals[3], ty);

2411 if (out_rms != nullptr) {

2414 for (const auto &gc : grid_cells) {
2415 if (!gc.valid)
2416 continue;

2418 std::clamp(static_cast<int>(std::lround(gc.center_x)), 0,
2419 image_width - 1);
2420 const int py =
2421 std::clamp(static_cast<int>(std::lround(gc.center_y)), 0,
2422 image_height - 1);
2423 const float pred = surface(py, px);
2424 if (!std::isfinite(pred))
2425 continue;
2426 const double r = static_cast<double>(gc.bg_value - pred);
2427 sum_sq += r * r;
2428 ++n;
2429 }
2430 *out_rms = (n > 0) ? static_cast<float>(
2431 std::sqrt(sum_sq / static_cast<double>(n)))
2432 : std::numeric_limits<float>::infinity();
2433 }
2434 return surface;
2435}

2441bool solve_rbf_model(const std::vector<GridCell> &grid_cells, int grid_spacing,
2442 const BGEConfig &config, RBFModelState *out) {
2443 const int M = grid_cells.size();
2444 if (M < 3) {
2445 std::cout << "[BGE] Too few grid cells for RBF: " << M << std::endl;
2446 return false;
2447 }
2448
2449 // Compute mu (shape parameter, v3.3 §6.3.7)
2450 const int G = std::max(1, grid_spacing);
2451 const float mu =
2452 std::max(kTiny, config.fit.rbf_mu_factor * static_cast<float>(G));
2453 const float epsilon = std::max(kTiny, config.fit.rbf_epsilon);
2454 const float lambda_base = std::max(kTiny, config.fit.rbf_lambda);
2455 const RBFKernelType kernel = resolve_rbf_kernel_type(config.fit.rbf_phi);
2456
2457 // Build RBF matrix Phi (M x M)
2458 Eigen::MatrixXf Phi_base(M, M);
2459 for (int i = 0; i < M; ++i) {
2460 Phi_base(i, i) = evaluate_rbf_kernel(kernel, 0.0f, mu, epsilon);
2461 for (int j = i + 1; j < M; ++j) {
2462 float dx = grid_cells[i].center_x - grid_cells[j].center_x;
2463 float dy = grid_cells[i].center_y - grid_cells[j].center_y;
2464 float d = std::sqrt(dx * dx + dy * dy);
2465 const float phi = evaluate_rbf_kernel(kernel, d, mu, epsilon);
2466 Phi_base(i, j) = phi;
2467 Phi_base(j, i) = phi;
2468 }
2469 }
2470
2471 // Build target vector b (M x 1)
2472 Eigen::VectorXf b(M);
2473 Eigen::VectorXf w_rel(M);
2474 std::vector<float> bg_values;
2475 bg_values.reserve(static_cast<size_t>(M));
2476 const auto robust_weight_fn =
2477 resolve_robust_weight_fn(config.fit.robust_loss);
2478 float min_x = std::numeric_limits<float>::infinity();
2479 float max_x = -std::numeric_limits<float>::infinity();
2480 float min_y = std::numeric_limits<float>::infinity();
2481 float max_y = -std::numeric_limits<float>::infinity();
2482 for (int i = 0; i < M; ++i) {
2483 b(i) = grid_cells[i].bg_value;
2484 w_rel(i) = std::max(1.0e-3f, grid_cells[i].weight);
2485 bg_values.push_back(grid_cells[i].bg_value);
2486 min_x = std::min(min_x, grid_cells[i].center_x);
2487 max_x = std::max(max_x, grid_cells[i].center_x);
2488 min_y = std::min(min_y, grid_cells[i].center_y);
2489 max_y = std::max(max_y, grid_cells[i].center_y);
2490 }
2491 const float poly_x_scale =
2492 2.0f / std::max(1.0f, max_x - min_x + static_cast<float>(G));
2493 const float poly_y_scale =
2494 2.0f / std::max(1.0f, max_y - min_y + static_cast<float>(G));
2495
2496 auto solve_rbf_coeffs = [&](float lambda, Eigen::VectorXf *out_u,
2497 Eigen::VectorXf *out_poly,
2498 float *out_rms) -> bool {
2499 Eigen::VectorXf u = Eigen::VectorXf::Zero(M);
2500 Eigen::VectorXf poly = Eigen::VectorXf::Zero(3);
2501 Eigen::VectorXf w_rob = Eigen::VectorXf::Ones(M);
2502
2503 Eigen::MatrixXf P(M, 3);
2504 for (int i = 0; i < M; ++i) {
2505 P(i, 0) = 1.0f;
2506 P(i, 1) = grid_cells[i].center_x * poly_x_scale;
2507 P(i, 2) = grid_cells[i].center_y * poly_y_scale;
2508 }
2509
2510 for (int iter = 0; iter < std::max(1, config.fit.irls_max_iterations);
2511 ++iter) {
2512 Eigen::VectorXf w = w_rel.cwiseProduct(w_rob).cwiseMax(1.0e-6f);
2513
2514 Eigen::MatrixXf lhs = Eigen::MatrixXf::Zero(M + 3, M + 3);
2515 lhs.block(0, 0, M, M) = Phi_base;
2516 for (int i = 0; i < M; ++i) {
2517 lhs(i, i) += lambda / std::max(w(i), 1.0e-6f);
2518 }
2519 lhs.block(0, M, M, 3) = P;
2520 lhs.block(M, 0, 3, M) = P.transpose();
2521
2522 Eigen::VectorXf rhs = Eigen::VectorXf::Zero(M + 3);
2523 rhs.segment(0, M) = b;
2524 Eigen::VectorXf solution = lhs.fullPivLu().solve(rhs);
2525
2526 if (!solution.allFinite()) {
2527 return false;
2528 }
2529
2530 Eigen::VectorXf u_new = solution.segment(0, M);
2531 Eigen::VectorXf poly_new = solution.segment(M, 3);
2532 const float step = (u_new - u).norm() + (poly_new - poly).norm();
2533 const float scale = 1.0f + u.norm() + poly.norm();
2534 u = std::move(u_new);
2535 poly = std::move(poly_new);
2536
2537 Eigen::VectorXf residual = b - (Phi_base * u + P * poly);
2538 update_robust_weights(residual, config.fit.huber_delta, robust_weight_fn,
2539 &w_rob);
2540
2541 if (step <= config.fit.irls_tolerance * scale)
2542 break;
2543 }
2544
2545 Eigen::VectorXf residual = b - (Phi_base * u + P * poly);
2546 const float rms = std::sqrt(residual.squaredNorm() / static_cast<float>(M));
2547 if (!(std::isfinite(rms))) {
2548 return false;
2549 }
2550
2551 *out_u = std::move(u);
2552 *out_poly = std::move(poly);
2553 *out_rms = rms;
2554 return true;
2555 };
2556
2557 // Dynamic lambda adaptation: test/adjust/test and prefer the smoothest
2558 // (highest lambda) model that still fits grid samples well enough.
2559 const float bg_med = robust_median_inplace(bg_values);
2560 const float bg_sigma = 1.4826f * robust_mad(bg_values, bg_med);
2561 const float residual_limit =
2562 std::max(0.15f, 0.20f * std::max(bg_sigma, kTiny));
2563
2564 std::vector<float> lambda_trials;
2565 lambda_trials.reserve(6);
2566 float l = lambda_base;
2567 for (int i = 0; i < 6; ++i) {
2568 lambda_trials.push_back(std::clamp(l, 1.0e-8f, 1.0e-1f));
2569 l *= 3.0f;
2570 }
2571 std::sort(lambda_trials.begin(), lambda_trials.end());
2572 lambda_trials.erase(std::unique(lambda_trials.begin(), lambda_trials.end()),
2573 lambda_trials.end());
2574
2575 float best_lambda = lambda_base;
2576 float best_rms = std::numeric_limits<float>::infinity();
2577 Eigen::VectorXf best_u = Eigen::VectorXf::Zero(M);
2578 Eigen::VectorXf best_poly = Eigen::VectorXf::Zero(3);
2579 bool have_best = false;
2580
2581 float accepted_lambda = -1.0f;
2582 float accepted_rms = std::numeric_limits<float>::infinity();
2583 Eigen::VectorXf accepted_u = Eigen::VectorXf::Zero(M);
2584 Eigen::VectorXf accepted_poly = Eigen::VectorXf::Zero(3);
2585
2586 for (float lambda_try : lambda_trials) {
2587 Eigen::VectorXf u_try = Eigen::VectorXf::Zero(M);
2588 Eigen::VectorXf poly_try = Eigen::VectorXf::Zero(3);
2589 float rms_try = std::numeric_limits<float>::infinity();
2590 const bool ok =
2591 solve_rbf_coeffs(lambda_try, &u_try, &poly_try, &rms_try);
2592 if (!ok) {
2593 std::cout << "[BGE] RBF lambda=" << lambda_try << " fit failed"
2594 << std::endl;
2595 continue;
2596 }
2597
2598 std::cout << "[BGE] RBF lambda=" << lambda_try << " trial RMS=" << rms_try
2599 << std::endl;
2600
2601 if (!have_best || rms_try < best_rms) {
2602 best_lambda = lambda_try;
2603 best_rms = rms_try;
2604 best_u = u_try;
2605 best_poly = poly_try;
2606 have_best = true;
2607 }
2608
2609 if (rms_try <= residual_limit) {
2610 accepted_lambda = lambda_try;
2611 accepted_rms = rms_try;
2612 accepted_u = u_try;
2613 accepted_poly = poly_try;
2614 }
2615 }
2616
2617 if (!have_best) {
2618 std::cerr << "[BGE] RBF solve failed for all lambda trials" << std::endl;
2619 return false;
2620 }
2621
2622 float lambda = best_lambda;
2623 float rms_selected = best_rms;
2624 Eigen::VectorXf u = best_u;
2625 Eigen::VectorXf poly = best_poly;
2626 if (accepted_lambda > 0.0f) {
2627 lambda = accepted_lambda;
2628 rms_selected = accepted_rms;
2629 u = accepted_u;
2630 poly = accepted_poly;
2631 }
2632
2633 std::cout << "[BGE] RBF selected lambda=" << lambda
2634 << " (limit=" << residual_limit << ", rms=" << rms_selected << ")"
2635 << std::endl;
2636
2637 out->grid_cells = &grid_cells;
2638 out->coeffs = std::move(u);
2639 out->poly_coeffs = std::move(poly);
2640 out->kernel = kernel;
2641 out->mu = mu;
2642 out->epsilon = epsilon;
2643 out->lambda = lambda;
2644 out->poly_x_scale = poly_x_scale;
2645 out->poly_y_scale = poly_y_scale;
2646 out->train_rms = rms_selected;
2647 out->success = true;
2648 return true;
2649}

2655float eval_rbf_model_at(const RBFModelState &state, float x, float y) {
2656 if (!state.success || state.grid_cells == nullptr)
2657 return std::numeric_limits<float>::quiet_NaN();

2660 for (int i = 0; i < state.coeffs.size(); ++i) {

2662 const float dy = y - grid_cells[static_cast<size_t>(i)].center_y;

2665 evaluate_rbf_kernel(state.kernel, d, state.mu, state.epsilon);

2669 state.poly_coeffs(2) * y * state.poly_y_scale;
2670 }
2671 return sum;
2672}

2694// Polynomial surface fitting (v3.3 §6.3.7)
2699bool solve_polynomial_model(const std::vector<GridCell> &grid_cells,
2700 int image_width, int image_height,
2701 const BGEConfig &config,
2702 PolynomialModelState *out) {
2703 const int M = grid_cells.size();
2704 const int order = config.fit.polynomial_order;
2705
2706 // Number of polynomial terms: (order+1)*(order+2)/2
2707 int n_terms = 0;
2708 for (int m = 0; m <= order; ++m) {
2709 for (int n = 0; n <= order - m; ++n) {
2710 ++n_terms;
2711 }
2712 }
2713
2714 if (M < n_terms) {
2715 std::cout << "[BGE] Too few grid cells for polynomial order " << order
2716 << ": " << M << " < " << n_terms << std::endl;
2717 return false;
2718 }
2719
2720 // Normalize coordinates to [-1, 1]
2721 float x_scale = 2.0f / image_width;
2722 float y_scale = 2.0f / image_height;
2723 float x_offset = -1.0f;
2724 float y_offset = -1.0f;
2725
2726 // Build design matrix A (M x n_terms)
2727 Eigen::MatrixXf A(M, n_terms);
2728 Eigen::VectorXf b(M);
2729 Eigen::VectorXf w_rel(M);
2730 std::vector<float> x_pows(static_cast<size_t>(order + 1), 1.0f);
2731 std::vector<float> y_pows(static_cast<size_t>(order + 1), 1.0f);
2732
2733 for (int i = 0; i < M; ++i) {
2734 float x_norm = grid_cells[i].center_x * x_scale + x_offset;
2735 float y_norm = grid_cells[i].center_y * y_scale + y_offset;
2736
2737 x_pows[0] = 1.0f;
2738 y_pows[0] = 1.0f;
2739 for (int p = 1; p <= order; ++p) {
2740 x_pows[static_cast<size_t>(p)] =
2741 x_pows[static_cast<size_t>(p - 1)] * x_norm;
2742 y_pows[static_cast<size_t>(p)] =
2743 y_pows[static_cast<size_t>(p - 1)] * y_norm;
2744 }
2745
2746 int col = 0;
2747 for (int m = 0; m <= order; ++m) {
2748 for (int n = 0; n <= order - m; ++n) {
2749 A(i, col) =
2750 x_pows[static_cast<size_t>(m)] * y_pows[static_cast<size_t>(n)];
2751 ++col;
2752 }
2753 }
2754
2755 b(i) = grid_cells[i].bg_value;
2756 w_rel(i) = std::max(1.0e-3f, grid_cells[i].weight);
2757 }
2758 const auto robust_weight_fn =
2759 resolve_robust_weight_fn(config.fit.robust_loss);
2760
2761 // Robust IRLS polynomial fitting (v3.3 §6.3.7)
2762 Eigen::VectorXf coeffs = Eigen::VectorXf::Zero(n_terms);
2763 Eigen::VectorXf w_rob = Eigen::VectorXf::Ones(M);
2764 for (int iter = 0; iter < std::max(1, config.fit.irls_max_iterations);
2765 ++iter) {
2766 Eigen::VectorXf w = w_rel.cwiseProduct(w_rob).cwiseMax(1.0e-6f);
2767 Eigen::MatrixXf W = w.asDiagonal();
2768 Eigen::MatrixXf lhs = A.transpose() * W * A;
2769 Eigen::VectorXf rhs = A.transpose() * W * b;
2770 Eigen::VectorXf coeffs_new = lhs.ldlt().solve(rhs);
2771
2772 const float step = (coeffs_new - coeffs).norm();
2773 const float scale = 1.0f + coeffs.norm();
2774 coeffs = coeffs_new;
2775
2776 Eigen::VectorXf residual = b - A * coeffs;
2777 update_robust_weights(residual, config.fit.huber_delta, robust_weight_fn,
2778 &w_rob);
2779
2780 if (step <= config.fit.irls_tolerance * scale) {
2781 break;
2782 }
2783 }
2784
2785 const Eigen::VectorXf residual = b - A * coeffs;
2786 const float train_rms =
2787 std::sqrt(residual.squaredNorm() / static_cast<float>(M));
2788 if (!(std::isfinite(train_rms))) {
2789 return false;
2790 }
2791
2792 out->coeffs = std::move(coeffs);
2793 out->order = order;
2794 out->x_scale = x_scale;
2795 out->y_scale = y_scale;
2796 out->x_offset = x_offset;
2797 out->y_offset = y_offset;
2798 out->train_rms = train_rms;
2799 out->success = true;
2800 return true;
2801}
2802
2807float eval_polynomial_model_at(const PolynomialModelState &state, float x,
2808 float y, std::vector<float> *x_pows,
2809 std::vector<float> *y_pows) {
2810 if (!state.success)
2811 return std::numeric_limits<float>::quiet_NaN();
2812 const float x_norm = x * state.x_scale + state.x_offset;
2813 const float y_norm = y * state.y_scale + state.y_offset;
2814 x_pows->assign(static_cast<size_t>(state.order + 1), 1.0f);
2815 y_pows->assign(static_cast<size_t>(state.order + 1), 1.0f);
2816 for (int p = 1; p <= state.order; ++p) {
2817 (*x_pows)[static_cast<size_t>(p)] =
2818 (*x_pows)[static_cast<size_t>(p - 1)] * x_norm;
2819 (*y_pows)[static_cast<size_t>(p)] =
2820 (*y_pows)[static_cast<size_t>(p - 1)] * y_norm;
2821 }
2822
2823 float sum = 0.0f;
2824 int col = 0;
2825 for (int m = 0; m <= state.order; ++m) {
2826 for (int n = 0; n <= state.order - m; ++n) {
2827 sum += state.coeffs(col) * (*x_pows)[static_cast<size_t>(m)] *
2828 (*y_pows)[static_cast<size_t>(n)];
2829 ++col;
2830 }
2831 }
2832 return sum;
2833}

2853 for (int p = 1; p <= state.order; ++p) {

2868#pragma omp parallel for collapse(2) num_threads(parallel_workers) \
2869 if(parallel_workers > 1)
2870 for (int y = 0; y < image_height; ++y) {
2871 for (int x = 0; x < image_width; ++x) {
2872 const auto &xp = x_power_table[static_cast<size_t>(x)];
2873 const auto &yp = y_power_table[static_cast<size_t>(y)];
2874
2875 float sum = 0.0f;
2876 int col = 0;
2877 for (int m = 0; m <= state.order; ++m) {
2878 for (int n = 0; n <= state.order - m; ++n) {
2879 sum += state.coeffs(col) * xp[static_cast<size_t>(m)] *
2880 yp[static_cast<size_t>(n)];
2881 ++col;
2882 }
2883 }
2884 surface(y, x) = sum;
2885 }
2886 }
2887
2888 return surface;

3176static float eval_model_flatness(const Matrix2Df &model, int step) {
3177 step = std::max(1, step);
3178 std::vector<float> grad_energy;
3179 grad_energy.reserve(static_cast<size_t>((model.rows() / step + 1) *
3180 (model.cols() / step + 1)));
3181 for (int y = 0; y + step < model.rows(); y += step) {
3182 for (int x = 0; x + step < model.cols(); x += step) {
3183 const float v = model(y, x);
3184 const float vx = model(y, x + step);
3185 const float vy = model(y + step, x);
3186 if (!(std::isfinite(v) && std::isfinite(vx) && std::isfinite(vy)))
3187 continue;
3188 const float gx = (vx - v) / static_cast<float>(step);
3189 const float gy = (vy - v) / static_cast<float>(step);
3190 grad_energy.push_back(gx * gx + gy * gy);
3191 }
3192 }
3193 if (grad_energy.size() < 8)
3194 return std::numeric_limits<float>::infinity();

3202static float eval_model_roughness(const Matrix2Df &model, int step) {
3203 step = std::max(1, step);
3204 std::vector<float> curvature_energy;
3205 curvature_energy.reserve(static_cast<size_t>((model.rows() / step + 1) *
3206 (model.cols() / step + 1)));
3207 for (int y = step; y + step < model.rows(); y += step) {
3208 for (int x = step; x + step < model.cols(); x += step) {
3209 const float c = model(y, x);
3210 const float xp = model(y, x + step);
3211 const float xm = model(y, x - step);
3212 const float yp = model(y + step, x);
3213 const float ym = model(y - step, x);
3214 const float xyp = model(y + step, x + step);
3215 const float xym = model(y - step, x + step);
3216 const float yxp = model(y + step, x - step);
3217 const float yxm = model(y - step, x - step);
3218 if (!(std::isfinite(c) && std::isfinite(xp) && std::isfinite(xm) &&
3219 std::isfinite(yp) && std::isfinite(ym) && std::isfinite(xyp) &&
3220 std::isfinite(xym) && std::isfinite(yxp) && std::isfinite(yxm))) {
3221 continue;
3222 }
3223 const float h = static_cast<float>(step);
3224 const float dxx = (xp - 2.0f * c + xm) / (h * h);
3225 const float dyy = (yp - 2.0f * c + ym) / (h * h);
3226 const float dxy = (xyp - xym - yxp + yxm) / (4.0f * h * h);
3227 curvature_energy.push_back(
3228 std::sqrt(dxx * dxx + dyy * dyy + 2.0f * dxy * dxy));
3229 }
3230 }
3231 if (curvature_energy.size() < 8)
3232 return std::numeric_limits<float>::infinity();
