45bool estimate_paired_fwhm(const Matrix2Df& image, const std::vector<uint8_t>& valid,
46 int x, int y, double& fwhm) {
47 constexpr int patch_radius = kMomentPatchRadius;
48 constexpr int aperture_sq = kMomentApertureRadius * kMomentApertureRadius;
49 if (x < patch_radius || y < patch_radius || x + patch_radius >= image.cols() ||
50 y + patch_radius >= image.rows()) return false;
51 std::vector<double> annulus;
52 annulus.reserve(128);
53 int aperture_total = 0, aperture_valid = 0, annulus_total = 0;
54 for (int dy = -patch_radius; dy <= patch_radius; ++dy)
55 for (int dx = -patch_radius; dx <= patch_radius; ++dx) {
56 const int r2 = dx * dx + dy * dy;
57 const size_t i = static_cast<size_t>(y + dy) * image.cols() + x + dx;
58 if (r2 <= aperture_sq) {
59 ++aperture_total;
60 if (valid[i]) ++aperture_valid;
61 }
62 if (r2 >= 64 && r2 <= patch_radius * patch_radius) {
63 ++annulus_total;
64 if (valid[i]) annulus.push_back(image(y + dy, x + dx));
65 }
66 }
67 if (static_cast<double>(aperture_valid) < kMinFinitePatchFraction * aperture_total ||
68 static_cast<double>(annulus.size()) < kMinFinitePatchFraction * annulus_total)
69 return false;
70 const double background = median(annulus);
71 double sum = 0.0, mx = 0.0, my = 0.0;
72 for (int dy = -kMomentApertureRadius; dy <= kMomentApertureRadius; ++dy)
73 for (int dx = -kMomentApertureRadius; dx <= kMomentApertureRadius; ++dx) {
74 if (dx * dx + dy * dy > aperture_sq) continue;
75 const size_t i = static_cast<size_t>(y + dy) * image.cols() + x + dx;
76 if (!valid[i]) continue;
77 const double weight = std::max(0.0, static_cast<double>(image(y + dy, x + dx)) - background);
78 sum += weight;
79 mx += weight * dx;
80 my += weight * dy;
81 }
82 if (!(sum > 0.0) || !std::isfinite(sum)) return false;
83 mx /= sum;
84 my /= sum;
85 double cxx = 0.0, cyy = 0.0;
86 for (int dy = -kMomentApertureRadius; dy <= kMomentApertureRadius; ++dy)
87 for (int dx = -kMomentApertureRadius; dx <= kMomentApertureRadius; ++dx) {
88 if (dx * dx + dy * dy > aperture_sq) continue;
89 const size_t i = static_cast<size_t>(y + dy) * image.cols() + x + dx;
90 if (!valid[i]) continue;
91 const double weight = std::max(0.0, static_cast<double>(image(y + dy, x + dx)) - background);
92 cxx += weight * (dx - mx) * (dx - mx);
93 cyy += weight * (dy - my) * (dy - my);
94 }
95 const double sigma_sq = (cxx + cyy) / (2.0 * sum);
96 if (!(sigma_sq > 0.0) || !std::isfinite(sigma_sq)) return false;
97 fwhm = 2.3548200450309493 * std::sqrt(sigma_sq);
98 return std::isfinite(fwhm) && fwhm > 0.2 && fwhm < 50.0;
99}

101double median(std::vector<double> values) {
102 if (values.empty()) return std::numeric_limits<double>::quiet_NaN();
103 std::sort(values.begin(), values.end());
104 const size_t n = values.size();
105 return n % 2 ? values[n / 2] : 0.5 * (values[n / 2 - 1] + values[n / 2]);
106}

108double percentile(std::vector<double> values, double p) {
109 if (values.empty()) return std::numeric_limits<double>::quiet_NaN();
110 std::sort(values.begin(), values.end());
111 const double pos = std::clamp(p, 0.0, 1.0) * (values.size() - 1);
112 const size_t lo = static_cast<size_t>(std::floor(pos));
113 const size_t hi = std::min(lo + 1, values.size() - 1);
114 return values[lo] + (values[hi] - values[lo]) * (pos - lo);
115}

117std::array<double, 2> bootstrap_star_median_ci(const std::vector<double>& ratios) {
118 std::mt19937_64 rng(kBootstrapSeed);
119 std::uniform_int_distribution<size_t> draw(0, ratios.size() - 1);
120 std::vector<double> medians;
121 medians.reserve(kBootstrapSamples);
122 std::vector<double> sample(ratios.size());
123 for (int i = 0; i < kBootstrapSamples; ++i) {
124 for (double& value : sample) value = ratios[draw(rng)];
125 medians.push_back(median(sample));
126 }
127 return {percentile(medians, 0.025), percentile(medians, 0.975)};
128}

132nlohmann::json measure_post_run_star_shape(const Matrix2Df& before,
133 const Matrix2Df& after,
134 const std::vector<uint8_t>& common_valid_mask) {
135 if (before.rows() < 32 || before.cols() < 32 || before.rows() != after.rows() ||
136 before.cols() != after.cols()) return unavailable("raster_mismatch_or_too_small");
137 const size_t pixels = static_cast<size_t>(before.rows()) * before.cols();
138 if (common_valid_mask.size() != pixels) return unavailable("valid_mask_mismatch");
139
140 std::vector<uint8_t> valid(pixels, 0);
141 cv::Mat1f safe(before.rows(), before.cols(), 0.0f);
142 cv::Mat1b detector_mask(before.rows(), before.cols(), static_cast<uint8_t>(0));
143 size_t valid_count = 0;
144 for (int y = 0; y < before.rows(); ++y) {
145 for (int x = 0; x < before.cols(); ++x) {
146 const size_t i = static_cast<size_t>(y) * before.cols() + x;
147 const float a = before(y, x), b = after(y, x);
148 if (!common_valid_mask[i] || !std::isfinite(a) || !std::isfinite(b)) continue;
149 valid[i] = 1;
150 detector_mask(y, x) = 255;
151 safe(y, x) = a;
152 ++valid_count;
153 }
154 }
155 if (valid_count < static_cast<size_t>(32 * 32)) return unavailable("insufficient_common_support");
156
157 // Match the established frame-star detector's broad-background subtraction while keeping all
158 // star coordinates sourced from the selected reconstruction only.
159 cv::Mat background, residual;
160 cv::blur(safe, background, cv::Size(31, 31), cv::Point(-1, -1), cv::BORDER_REFLECT_101);
161 cv::subtract(safe, background, residual);
162 for (int y = 0; y < residual.rows; ++y)
163 for (int x = 0; x < residual.cols; ++x)
164 if (!detector_mask(y, x)) residual.at<float>(y, x) = 0.0f;
165
166 std::vector<cv::Point2f> corners;
167 try {
168 cv::goodFeaturesToTrack(residual, corners, kMaxControlStars, 0.001, 8.0,
169 detector_mask, 3, false, 0.04);
170 } catch (...) {
171 return unavailable("star_detection_failed");
172 }
173 if (corners.empty()) return unavailable("no_control_stars");
174
175 std::vector<double> ratios;
176 ratios.reserve(corners.size());
177 for (const auto& corner : corners) {
178 const int x = static_cast<int>(std::lround(corner.x));
179 const int y = static_cast<int>(std::lround(corner.y));
180 double fwhm_before = 0.0, fwhm_after = 0.0;
181 if (!estimate_paired_fwhm(before, valid, x, y, fwhm_before) ||
182 !estimate_paired_fwhm(after, valid, x, y, fwhm_after)) continue;
183 const double ratio = fwhm_after / fwhm_before;
184 if (std::isfinite(ratio) && ratio > 0.0) ratios.push_back(ratio);
185 }
186
187 if (ratios.size() < kMinMatchedStars)
188 return unavailable("too_few_matched_stars", static_cast<int>(corners.size()),
189 static_cast<int>(ratios.size()));
190 const auto ci = bootstrap_star_median_ci(ratios);
191 const double median_ratio = median(ratios);
192 const double p90_ratio = percentile(ratios, 0.90);
193 if (!std::isfinite(median_ratio) || !std::isfinite(p90_ratio) ||
194 !std::isfinite(ci[0]) || !std::isfinite(ci[1]))
195 return unavailable("non_finite_result", static_cast<int>(corners.size()),
196 static_cast<int>(ratios.size()));
197
198 return {{"schema_version", "pi.post-run-star-shape.v1"},
199 {"status", "measured"},
200 {"method", "fixed_control_positions_annulus_aperture_moments_v1"},
201 {"before_stage", "selected_reconstruction"},
202 {"after_stage", "final_hms_rgb_g"},
203 {"metric", "matched_star_fwhm_ratio_after_over_before"},
204 {"median_ratio", median_ratio},
205 {"p90_ratio", p90_ratio},
206 {"ci95_star_bootstrap", {ci[0], ci[1]}},
207 {"detected_control_stars", static_cast<int>(corners.size())},
208 {"matched_stars", static_cast<int>(ratios.size())},
209 {"min_matched_stars", kMinMatchedStars},
210 {"unit", "ratio"},
211 {"interpretation", "descriptive_measurement_only"}};
212}
