24AffineInverse invert_source_to_canvas(
25 const registration::FrameSamplingTransform &frame) {
26 const auto &m = frame.source_to_canvas;
27 const double a = m(0, 0), b = m(0, 1), c = m(0, 2);
28 const double d = m(1, 0), e = m(1, 1), f = m(1, 2);
29 const double det = a * e - b * d;
30 if (!frame.source_to_canvas_affine_valid || !std::isfinite(det) ||
31 std::abs(det) <= std::numeric_limits<double>::epsilon())
32 throw std::invalid_argument("FORWARD_DRIZZLE_V2_INVALID_AFFINE");
33 const double id = 1.0 / det;
34 AffineInverse out;
35 out.a = e * id;
36 out.b = -b * id;
37 out.d = -d * id;
38 out.e = a * id;
39 out.c = -(out.a * c + out.b * f);
40 out.f = -(out.d * c + out.e * f);
41 return out;
42}

75 const SourceImageProvider &source_of,
76 const config::ReconstructionDrizzleConfig &cfg, int y_begin, int rows,
77 const ForwardDrizzleSubdivisionParams &subdivision) {

91 ForwardDrizzleV2UniformResult result;
92 result.accum = make_accum(width, rows, channels);
93 result.stats.target_cells = static_cast<std::uint64_t>(width) * rows;
94 result.stats.workspace_bytes =
95 static_cast<std::size_t>(channels) * width * rows * 3 * sizeof(double);

104 const AffineInverse inv = invert_source_to_canvas(*frame);

107 throw std::invalid_argument("FORWARD_DRIZZLE_V2_SOURCE_SHAPE");

120 static_cast<double>(ty + cy) / scale);

122 max_x = std::max(max_x, p[0]);
123 min_y = std::min(min_y, p[1]);
124 max_y = std::max(max_y, p[1]);
125 }
126
127 // Source samples are centred at (sx+0.5, sy+0.5). Expand by one
128 // additional integer on both sides: the exact polygon test removes
129 // false positives, while roundoff at a tangency cannot create a false
130 // negative in this search bound.
131 const int sx0 = std::max(
132 0, static_cast<int>(std::floor(min_x - half - 0.5)) - 1);
133 const int sx1 = std::min(
134 plan.source_width,
135 static_cast<int>(std::ceil(max_x + half - 0.5)) + 2);
136 const int sy0 = std::max(
137 0, static_cast<int>(std::floor(min_y - half - 0.5)) - 1);
138 const int sy1 = std::min(
139 plan.source_height,
140 static_cast<int>(std::ceil(max_y + half - 0.5)) + 2);
141
142 const std::size_t i = static_cast<std::size_t>(ly) * width + tx;
143 std::array<double, 3> A{0.0, 0.0, 0.0};
144 std::array<double, 3> B{0.0, 0.0, 0.0};
145 for (int sy = sy0; sy < sy1; ++sy) {
146 for (int sx = sx0; sx < sx1; ++sx) {
147 ++result.stats.source_candidates;
148 const double v = static_cast<double>(src(sy, sx));
149 if (!std::isfinite(v)) continue;
150 if (!sample_leaves(plan, *frame, sx, sy, scale, cfg.pixfrac,
151 subdivision, leaves))
152 continue;
153 const int c = channel_for(plan, sx, sy);
154 for (const auto &leaf : leaves) {
155 ++result.stats.leaves_tested;
156 const double k = polygon_rectangle_intersection_area(
157 leaf.x, leaf.y, tx, ty, tx + 1.0, ty + 1.0);
158 if (!(k > 0.0)) continue;
159 ++result.stats.positive_overlaps;
160 A[c] += k * v;
161 B[c] += k;
162 }
163 }
164 }
165 for (int c = 0; c < channels; ++c) {
166 if (!(B[c] > 0.0)) continue;
167 result.accum.wx[c][i] += A[c];
168 result.accum.w[c][i] += B[c];
169 result.accum.w2[c][i] += B[c] * B[c];
170 }
171 }
172 }
173 }
174 return result;
175}

177ForwardDrizzleV2FoldResult fold_native_pixel_v2(
178 std::span<const ForwardDrizzleV2FrameSubpixel> entries,
179 std::size_t frame_count, std::span<const double> area) {
180 if (frame_count == 0 || area.empty() ||
181 frame_count > std::numeric_limits<std::size_t>::max() / area.size() ||
182 entries.size() != frame_count * area.size())
183 throw std::invalid_argument("FORWARD_DRIZZLE_V2_INVALID_FOLD_SHAPE");
184 double total_area = 0.0;
185 for (double a : area) {
186 if (!std::isfinite(a) || a < 0.0)
187 throw std::invalid_argument("FORWARD_DRIZZLE_V2_INVALID_FOLD_AREA");
188 total_area += a;
189 }
190 if (!(total_area > 0.0))
191 throw std::invalid_argument("FORWARD_DRIZZLE_V2_INVALID_FOLD_AREA");
192
193 ForwardDrizzleV2FoldResult out;
194 std::vector<std::uint8_t> geometry_supported(area.size(), 0u);
195 std::vector<std::uint8_t> source_supported(area.size(), 0u);
196 std::vector<std::uint8_t> estimator_supported(area.size(), 0u);
197 std::vector<std::uint8_t> profile_supported(area.size(), 0u);
198 for (std::size_t f = 0; f < frame_count; ++f) {
199 double af = 0.0;
200 double bf = 0.0;
201 for (std::size_t j = 0; j < area.size(); ++j) {
202 const auto &v = entries[f * area.size() + j];
203 if (v.frame_order != f)
204 throw std::invalid_argument("FORWARD_DRIZZLE_V2_FOLD_FRAME_ORDER");
205 const auto valid_weight = [](double w) {
206 return std::isfinite(w) && w >= 0.0;
207 };
208 if (!valid_weight(v.geometry_b) || !valid_weight(v.source_b) ||
209 !valid_weight(v.estimator_b) || !valid_weight(v.b))
210 throw std::invalid_argument("FORWARD_DRIZZLE_V2_FOLD_INVALID_WEIGHT");
211 if ((v.source_b > 0.0 && !(v.geometry_b > 0.0)) ||
212 (v.estimator_b > 0.0 && !(v.source_b > 0.0)) ||
213 (v.b > 0.0 && !(v.estimator_b > 0.0)))
214 throw std::invalid_argument("FORWARD_DRIZZLE_V2_FOLD_SUPPORT_ORDER");
215 if (v.b > 0.0 && !std::isfinite(v.a))
216 throw std::invalid_argument("FORWARD_DRIZZLE_V2_FOLD_NONFINITE_VALUE");
217 geometry_supported[j] |= static_cast<std::uint8_t>(v.geometry_b > 0.0);
218 source_supported[j] |= static_cast<std::uint8_t>(v.source_b > 0.0);
219 estimator_supported[j] |=
220 static_cast<std::uint8_t>(v.estimator_b > 0.0);
221 profile_supported[j] |= static_cast<std::uint8_t>(v.b > 0.0);
222 if (!(v.b > 0.0)) continue;
223 af += area[j] * v.a;
224 bf += area[j] * v.b;
225 }
226 if (bf > 0.0) {
227 out.a += af;
228 out.b += bf;
229 out.b2 += bf * bf;
230 }
231 }
232 auto area_fraction = [&](const std::vector<std::uint8_t> &supported) {
233 double sum = 0.0;
234 for (std::size_t j = 0; j < area.size(); ++j)
235 if (supported[j]) sum += area[j];
236 return std::clamp(sum / total_area, 0.0, 1.0);
237 };
238 out.geometry_area_fraction = area_fraction(geometry_supported);
239 out.source_area_fraction = area_fraction(source_supported);
240 out.estimator_area_fraction = area_fraction(estimator_supported);
241 out.profile_area_fraction = area_fraction(profile_supported);
242 out.geometry_support = out.geometry_area_fraction > 0.0;
243 out.source_support = out.source_area_fraction > 0.0;
244 out.estimator_support = out.estimator_area_fraction > 0.0;
245 out.profile_support = out.b > 0.0 && std::isfinite(out.a) &&
246 std::isfinite(out.b) && std::isfinite(out.b2);
247 if (out.profile_support) {
248 out.value = out.a / out.b;
249 out.n_eff = out.b2 > 0.0 ? out.b * out.b / out.b2 : 0.0;
250 }
251 return out;
252}

316ForwardDrizzleV2RobustResult robust_reduce_v2(
317 const ForwardDrizzleV2CandidateReplay &replay,
318 const ForwardDrizzleV2RobustConfig &cfg) {
319 if (!replay || cfg.groups < 3 || (cfg.groups % 2) == 0 ||
320 cfg.min_candidates < 1 || cfg.min_groups < 1 ||
321 cfg.min_groups > cfg.groups || !std::isfinite(cfg.winsor_sigma) ||
322 cfg.winsor_sigma <= 0.0)
323 throw std::invalid_argument("FORWARD_DRIZZLE_V2_INVALID_ROBUST_CONFIG");
324
325 struct Group {
326 double a = 0.0;
327 double b = 0.0;
328 };
329 std::vector<Group> groups(static_cast<std::size_t>(cfg.groups));
330 ForwardDrizzleV2RobustResult out;
331 double uniform_a = 0.0, uniform_b = 0.0, uniform_b2 = 0.0;
332 replay([&](const ForwardDrizzleV2RobustCandidate &v) {
333 if (!std::isfinite(v.x) || !std::isfinite(v.b) || !(v.b > 0.0)) return;
334 ++out.candidates;
335 uniform_a += v.b * v.x;
336 uniform_b += v.b;
337 uniform_b2 += v.b * v.b;
338 Group &g = groups[v.frame_order % groups.size()];
339 g.a += v.b * v.x;
340 g.b += v.b;
341 });
342
343 if (!(uniform_b > 0.0)) return out;
344 if (out.candidates < static_cast<std::uint64_t>(cfg.min_candidates)) {
345 out.state = ForwardDrizzleV2RobustState::too_few_candidates_fallback;
346 finish_uniform_fallback(out, uniform_a, uniform_b, uniform_b2);
347 out.conf_b = uniform_b;
348 finalize_confidence(out);
349 return out;
350 }
351
352 std::vector<double> means;
353 means.reserve(groups.size());
354 for (const auto &g : groups)
355 if (g.b > 0.0) means.push_back(g.a / g.b);
356 out.groups_used = means.size();
357 if (means.size() < static_cast<std::size_t>(cfg.min_groups)) {
358 out.state = ForwardDrizzleV2RobustState::too_few_groups_fallback;
359 finish_uniform_fallback(out, uniform_a, uniform_b, uniform_b2);
360 out.conf_b = uniform_b;
361 finalize_confidence(out);
362 return out;
363 }
364
365 out.center = median_sorted(means);
366 std::vector<double> deviations;
367 deviations.reserve(means.size());
368 for (double x : means) deviations.push_back(std::abs(x - out.center));
369 out.scale = 1.482602218505602 * median_sorted(std::move(deviations));
370 if (!(out.scale > 0.0) || !std::isfinite(out.scale)) {
371 out.state = ForwardDrizzleV2RobustState::degenerate_scale;
372 // A zero group-MAD means at least half of the valid groups agree exactly.
373 // Returning Uniform here would give an arbitrarily large isolated sample
374 // full influence. The mathematically defined degenerate solution is the
375 // common robust centre with the original geometric support weights.
376 finish_uniform_fallback(out, out.center * uniform_b, uniform_b, uniform_b2);
377 out.conf_b = uniform_b;
378 finalize_confidence(out);
379 return out;
380 }
381
382 const double lo = out.center - cfg.winsor_sigma * out.scale;
383 const double hi = out.center + cfg.winsor_sigma * out.scale;
384 replay([&](const ForwardDrizzleV2RobustCandidate &v) {
385 if (!std::isfinite(v.x) || !std::isfinite(v.b) || !(v.b > 0.0)) return;
386 const double x = std::clamp(v.x, lo, hi);
387 out.a += v.b * x;
388 out.b += v.b;
389 out.b2 += v.b * v.b;
390 });
391 out.value = out.a / out.b;
392 out.n_eff = out.b2 > 0.0 ? out.b * out.b / out.b2 : 0.0;
393 out.state = ForwardDrizzleV2RobustState::primary_winsorized_mom;
394 out.conf_b = out.b;
395 finalize_confidence(out);
396 return out;
397}

416double weighted_median_of(std::span<const double> values,
417 std::span<const double> weights) {
418 // Deterministic (value, index) order; cumulative weight reaching half of
419 // the total selects the value, matching the production clip semantics.
420 std::vector<std::size_t> order(values.size());
421 std::iota(order.begin(), order.end(), std::size_t{0});
422 std::sort(order.begin(), order.end(), [&](std::size_t i, std::size_t j) {
423 if (values[i] != values[j]) return values[i] < values[j];
424 return i < j;
425 });
426 double total_w = 0.0;
427 for (double w : weights) total_w += w;
428 if (!(total_w > 0.0)) return values.empty() ? 0.0 : values[order.back()];
429 double cum = 0.0;
430 for (std::size_t i : order) {
431 cum += weights[i];
432 if (cum >= total_w / 2.0) return values[i];
433 }
434 return values[order.back()];
435}

439std::vector<std::uint64_t> forward_drizzle_v2_selected_frame_orders(
440 std::uint64_t stream_length, int reservoir_size,
441 std::uint64_t reservoir_seed) {
442 if (stream_length == 0 || reservoir_size < 1)
443 throw std::invalid_argument(
444 "FORWARD_DRIZZLE_V2_INVALID_RESERVOIR_STREAM");
445 const bool keep_all =
446 stream_length <= static_cast<std::uint64_t>(reservoir_size);
447 const std::uint64_t threshold =
448 keep_all ? 0 : static_cast<std::uint64_t>(
449 (static_cast<unsigned __int128>(reservoir_size)
450 << 64) /
451 stream_length);
452 std::vector<std::uint64_t> selected;
453 selected.reserve(keep_all ? static_cast<std::size_t>(stream_length)
454 : static_cast<std::size_t>(reservoir_size) + 8);
455 for (std::uint64_t order = 0; order < stream_length; ++order) {
456 if (keep_all || splitmix64(order ^ reservoir_seed) < threshold)
457 selected.push_back(order);
458 }
459 return selected;
460}

489ForwardDrizzleV2RobustResult robust_frame_oracle_v2(
490 std::span<const ForwardDrizzleV2RobustCandidate> candidates,
491 int min_clip_contributors, int robust_passes, double sigma_low,
492 double sigma_high, std::span<const double> candidate_sigma2) {
493 if (min_clip_contributors < 1 || robust_passes < 1 ||
494 !std::isfinite(sigma_low) || !std::isfinite(sigma_high) ||
495 sigma_low <= 0.0 || sigma_high <= 0.0 ||
496 (!candidate_sigma2.empty() &&
497 candidate_sigma2.size() != candidates.size()))

499 ForwardDrizzleV2RobustResult out;
500 std::vector<ForwardDrizzleV2RobustCandidate> valid;
501 std::vector<std::size_t> valid_index;
502 for (std::size_t i = 0; i < candidates.size(); ++i) {

505 valid.push_back(v);
506 valid_index.push_back(i);
507 }
508 out.candidates = valid.size();
509 if (valid.empty()) return out;
510
511 const std::size_t n = valid.size();
512 std::vector<std::uint8_t> accepted(n, std::uint8_t{1});
513 if (n >= static_cast<std::size_t>(min_clip_contributors)) {
514 std::vector<std::size_t> order(n);
515 std::iota(order.begin(), order.end(), std::size_t{0});
516 std::sort(order.begin(), order.end(), [&](std::size_t i, std::size_t j) {

520 std::vector<std::size_t> active;
521 std::vector<std::size_t> dev_order;
522 for (int pass = 0; pass < robust_passes; ++pass) {
523 active.clear();
524 for (std::size_t idx : order)
525 if (accepted[idx]) active.push_back(idx);
526 if (active.empty()) break;
527 double total_w = 0.0;
528 for (std::size_t idx : active) total_w += valid[idx].b;
529 double median = valid[active.back()].x;
530 if (total_w > 0.0) {
531 double cum = 0.0;
532 for (std::size_t idx : active) {
533 cum += valid[idx].b;
534 if (cum >= total_w / 2.0) {
535 median = valid[idx].x;
536 break;
537 }
538 }
539 }
540 dev_order = active;
541 std::sort(dev_order.begin(), dev_order.end(),
542 [&](std::size_t i, std::size_t j) {
543 const double di = std::abs(valid[i].x - median);
544 const double dj = std::abs(valid[j].x - median);
545 if (di != dj) return di < dj;
546 return valid[i].frame_order < valid[j].frame_order;
547 });
548 double mad = std::abs(valid[dev_order.back()].x - median);
549 if (total_w > 0.0) {
550 double cum = 0.0;
551 for (std::size_t idx : dev_order) {
552 cum += valid[idx].b;
553 if (cum >= total_w / 2.0) {
554 mad = std::abs(valid[idx].x - median);
555 break;
556 }
557 }
558 }
559 const double lower = median - sigma_low * mad;
560 const double upper = median + sigma_high * mad;
561 bool changed = false;
562 for (std::size_t idx : active) {
563 const double x = valid[idx].x;
564 if (!(x >= lower && x <= upper)) {
565 accepted[idx] = 0;
566 changed = true;
567 }
568 }
569 if (!changed) break;
570 }
571 }
572
573 double a = 0.0, b = 0.0, b2 = 0.0;
574 for (std::size_t i = 0; i < n; ++i) {
575 if (!accepted[i]) continue;
576 a += valid[i].b * valid[i].x;
577 b += valid[i].b;
578 b2 += valid[i].b * valid[i].b;
579 out.conf_b += valid[i].b;
580 const double s2 =
581 conf_sigma2_of(candidate_sigma2, valid_index[i], out.conf_degraded);
582 if (s2 > 0.0) {
583 out.conf_s += valid[i].b * std::sqrt(s2);
584 out.conf_c += valid[i].b * valid[i].b * s2;
585 }
586 }
587 out.state = ForwardDrizzleV2RobustState::oracle_sigma_clip;
588 out.center = b > 0.0 ? a / b : 0.0;
589 finish_uniform_fallback(out, a, b, b2);
590 out.state = ForwardDrizzleV2RobustState::oracle_sigma_clip;
591 finalize_confidence(out);
592 return out;
593}
594
595ForwardDrizzleV2RobustResult robust_reduce_candidates_v2(
596 std::span<const ForwardDrizzleV2RobustCandidate> candidates,
597 ForwardDrizzleV2Estimator estimator, const ForwardDrizzleV2RobustConfig &cfg,
598 std::uint64_t stream_length, std::span<const double> candidate_sigma2) {
599 if (estimator == ForwardDrizzleV2Estimator::two_pass_winsorized_frames ||
600 cfg.groups < 3 || (cfg.groups % 2) == 0 || cfg.min_candidates < 1 ||
601 cfg.min_groups < 1 || cfg.min_groups > cfg.groups ||
602 !std::isfinite(cfg.winsor_sigma) || cfg.winsor_sigma <= 0.0 ||
603 cfg.reservoir_size < 1 ||
604 (!candidate_sigma2.empty() &&
605 candidate_sigma2.size() != candidates.size()))
606 throw std::invalid_argument("FORWARD_DRIZZLE_V2_INVALID_ROBUST_CONFIG");
607
608 if (estimator == ForwardDrizzleV2Estimator::reservoir_sigma_clip) {
609 // Single pass: the hash keep predicate only needs the known stream
610 // length, so a production stream can push kept frames into the reservoir
611 // while accumulating the uniform totals; no replay is required.
612 const std::uint64_t n_stream =
613 stream_length > 0 ? stream_length
614 : static_cast<std::uint64_t>(candidates.size());
615 const bool keep_all =
616 n_stream <= static_cast<std::uint64_t>(cfg.reservoir_size);
617 const std::uint64_t threshold = keep_all ? 0 : static_cast<std::uint64_t>(
618 (static_cast<unsigned __int128>(cfg.reservoir_size) << 64) / n_stream);
619 ForwardDrizzleV2RobustResult out;
620 std::vector<ForwardDrizzleV2RobustCandidate> reservoir;
621 std::vector<double> reservoir_sigma2;
622 if (keep_all) reservoir.reserve(candidates.size());
623 else reservoir.reserve(static_cast<std::size_t>(cfg.reservoir_size) +
624 static_cast<std::size_t>(cfg.reservoir_size) / 4);
625 double uniform_a = 0.0, uniform_b = 0.0, uniform_b2 = 0.0;
626 double stream_s = 0.0, stream_c = 0.0;
627 for (std::size_t i = 0; i < candidates.size(); ++i) {
628 const auto &v = candidates[i];
629 if (!std::isfinite(v.x) || !std::isfinite(v.b) || !(v.b > 0.0)) continue;
630 ++out.candidates;
631 uniform_a += v.b * v.x;
632 uniform_b += v.b;
633 uniform_b2 += v.b * v.b;
634 const double s2 =
635 conf_sigma2_of(candidate_sigma2, i, out.conf_degraded);
636 if (s2 > 0.0) {
637 stream_s += v.b * std::sqrt(s2);
638 stream_c += v.b * v.b * s2;
639 }
640 if (keep_all || splitmix64(v.frame_order ^ cfg.reservoir_seed) <
641 threshold) {
642 reservoir.push_back(v);
643 reservoir_sigma2.push_back(s2);
644 }
645 }
646 if (!(uniform_b > 0.0)) return out;
647 if (out.candidates < static_cast<std::uint64_t>(cfg.min_candidates) ||
648 reservoir.size() <
649 static_cast<std::size_t>(cfg.oracle_min_clip_contributors)) {
650 out.state = ForwardDrizzleV2RobustState::too_few_candidates_fallback;
651 finish_uniform_fallback(out, uniform_a, uniform_b, uniform_b2);
652 out.conf_b = uniform_b;
653 out.conf_s = stream_s;
654 out.conf_c = stream_c;
655 finalize_confidence(out);
656 return out;
657 }
658 const auto clipped = robust_frame_oracle_v2(
659 reservoir, cfg.oracle_min_clip_contributors, cfg.oracle_passes,
660 cfg.oracle_sigma_low, cfg.oracle_sigma_high, reservoir_sigma2);
661 // The reservoir decides the value; B/B2 cover the full stream so support
662 // and effective-N reflect every contributing frame. Confidence uses the
663 // clip-accepted reservoir members (its sigma inputs were already
664 // validated above, so the oracle's degraded count stays zero there).
665 out.state = ForwardDrizzleV2RobustState::primary_reservoir_sigma_clip;
666 out.center = clipped.center;
667 out.scale = clipped.scale;
668 out.groups_used = reservoir.size();
669 finish_uniform_fallback(out, clipped.value * uniform_b, uniform_b,
670 uniform_b2);
671 out.conf_b = clipped.conf_b;
672 out.conf_s = clipped.conf_s;
673 out.conf_c = clipped.conf_c;
674 finalize_confidence(out);
675 return out;
676 }
677
678 const std::size_t k = static_cast<std::size_t>(cfg.groups);
679 std::vector<GroupSlot> groups(k);
680 ForwardDrizzleV2RobustResult out;
681 double uniform_a = 0.0, uniform_b = 0.0, uniform_b2 = 0.0;
682 double stream_s = 0.0, stream_c = 0.0;
683 for (std::size_t i = 0; i < candidates.size(); ++i) {
684 const auto &v = candidates[i];
685 if (!std::isfinite(v.x) || !std::isfinite(v.b) || !(v.b > 0.0)) continue;
686 ++out.candidates;
687 uniform_a += v.b * v.x;
688 uniform_b += v.b;
689 uniform_b2 += v.b * v.b;
690 const double s2 = conf_sigma2_of(candidate_sigma2, i, out.conf_degraded);
691 const double s = s2 > 0.0 ? std::sqrt(s2) : 0.0;
692 stream_s += v.b * s;
693 stream_c += v.b * v.b * s2;
694 GroupSlot &g = groups[v.frame_order % k];
695 g.a += v.b * v.x;
696 g.b += v.b;
697 g.b2 += v.b * v.b;
698 g.s += v.b * s;
699 g.c += v.b * v.b * s2;
700 }
701
702 if (!(uniform_b > 0.0)) return out;
703 if (out.candidates < static_cast<std::uint64_t>(cfg.min_candidates)) {
704 out.state = ForwardDrizzleV2RobustState::too_few_candidates_fallback;
705 finish_uniform_fallback(out, uniform_a, uniform_b, uniform_b2);
706 out.conf_b = uniform_b;
707 out.conf_s = stream_s;
708 out.conf_c = stream_c;
709 finalize_confidence(out);
710 return out;
711 }
712
713 std::vector<double> means, weights, group_b2, group_s, group_c;
714 means.reserve(k);
715 weights.reserve(k);
716 group_b2.reserve(k);
717 group_s.reserve(k);
718 group_c.reserve(k);
719 for (const auto &g : groups) {
720 if (!(g.b > 0.0)) continue;
721 means.push_back(g.a / g.b);
722 weights.push_back(g.b);
723 group_b2.push_back(g.b2);
724 group_s.push_back(g.s);
725 group_c.push_back(g.c);
726 }
727 out.groups_used = means.size();
728 if (means.size() < static_cast<std::size_t>(cfg.min_groups)) {
729 out.state = ForwardDrizzleV2RobustState::too_few_groups_fallback;
730 finish_uniform_fallback(out, uniform_a, uniform_b, uniform_b2);
731 out.conf_b = uniform_b;
732 out.conf_s = stream_s;
733 out.conf_c = stream_c;
734 finalize_confidence(out);
735 return out;
736 }
737 out.center = median_sorted(means);
738 std::vector<double> deviations;
739 deviations.reserve(means.size());
740 for (double x : means) deviations.push_back(std::abs(x - out.center));
741 out.scale = 1.482602218505602 * median_sorted(std::move(deviations));
742
743 switch (estimator) {
744 case ForwardDrizzleV2Estimator::uniform:
745 out.state = ForwardDrizzleV2RobustState::primary_uniform;
746 finish_uniform_fallback(out, uniform_a, uniform_b, uniform_b2);
747 out.conf_b = uniform_b;
748 out.conf_s = stream_s;
749 out.conf_c = stream_c;
750 finalize_confidence(out);
751 return out;
752
753 case ForwardDrizzleV2Estimator::mom_median: {
754 out.state = ForwardDrizzleV2RobustState::primary_mom_median;
755 const double value = weighted_median_of(means, weights);
756 // No frame weight is dropped: B/B2 keep the uniform totals while the
757 // value is the weighted median of the group means.
758 finish_uniform_fallback(out, value * uniform_b, uniform_b, uniform_b2);
759 out.center = value;
760 out.conf_b = uniform_b;
761 out.conf_s = stream_s;
762 out.conf_c = stream_c;
763 finalize_confidence(out);
764 return out;
765 }
766
767 case ForwardDrizzleV2Estimator::mom_winsorized_groups:
768 case ForwardDrizzleV2Estimator::mom_trimmed_groups: {
769 if (!(out.scale > 0.0) || !std::isfinite(out.scale)) {
770 out.state = ForwardDrizzleV2RobustState::degenerate_scale;
771 finish_uniform_fallback(out, out.center * uniform_b, uniform_b,
772 uniform_b2);
773 out.conf_b = uniform_b;
774 out.conf_s = stream_s;
775 out.conf_c = stream_c;
776 finalize_confidence(out);
777 return out;
778 }
779 const double lo = out.center - cfg.winsor_sigma * out.scale;
780 const double hi = out.center + cfg.winsor_sigma * out.scale;
781 const bool trimmed =
782 estimator == ForwardDrizzleV2Estimator::mom_trimmed_groups;
783 double a = 0.0, b = 0.0, b2 = 0.0, cb = 0.0, cs = 0.0, cc = 0.0;
784 for (std::size_t g = 0; g < means.size(); ++g) {
785 if (trimmed) {
786 if (!(means[g] >= lo && means[g] <= hi)) continue;
787 a += means[g] * weights[g];
788 } else {
789 a += std::clamp(means[g], lo, hi) * weights[g];
790 }
791 b += weights[g];
792 b2 += group_b2[g];
793 cb += weights[g];
794 cs += group_s[g];
795 cc += group_c[g];
796 }
797 if (!(b > 0.0)) {
798 out.state = ForwardDrizzleV2RobustState::degenerate_scale;
799 finish_uniform_fallback(out, out.center * uniform_b, uniform_b,
800 uniform_b2);
801 out.conf_b = uniform_b;
802 out.conf_s = stream_s;
803 out.conf_c = stream_c;
804 finalize_confidence(out);
805 return out;
806 }
807 out.state = trimmed
808 ? ForwardDrizzleV2RobustState::primary_mom_trimmed_groups
809 : ForwardDrizzleV2RobustState::primary_mom_winsorized_groups;
810 finish_uniform_fallback(out, a, b, b2);
811 out.conf_b = cb;
812 out.conf_s = cs;
813 out.conf_c = cc;
814 finalize_confidence(out);
815 return out;
816 }
817
818 case ForwardDrizzleV2Estimator::two_pass_winsorized_frames:
819 break;
820 }
821 throw std::invalid_argument("FORWARD_DRIZZLE_V2_INVALID_ROBUST_CONFIG");
822}

824ForwardDrizzleV2MemoryPlan plan_forward_drizzle_v2_memory(
825 const ForwardDrizzleV2MemoryInputs &in) {
826 ForwardDrizzleV2MemoryPlan out;
827 const auto &r = in.roles;
828 if (in.target_width <= 0 || in.target_height <= 0 ||
829 (in.channels != 1 && in.channels != 3) || in.frame_count <= 0 ||
830 in.internal_scale < 1 ||
831 r.reservoir_size < 1 || r.reservoir_slot_factor < 1 ||
832 r.reservoir_candidate_bytes == 0 ||
833 r.frame_plane_doubles < 0 || r.robust_doubles < 0 ||
834 r.confidence_doubles < 0 || r.centre_scale_doubles < 0 ||
835 r.fold_staging_doubles < 0 || in.source_device_slots < 0 ||
836 in.quality_device_slots < 0 || in.pinned_source_slots < 0 ||
837 in.pinned_quality_slots < 0 || in.halo_rows_per_band_edge < 0 ||
838 !std::isfinite(in.max_source_read_amplification) ||
839 in.max_source_read_amplification < 1.0 ||
840 !std::isfinite(in.max_quality_read_amplification) ||
841 in.max_quality_read_amplification < 1.0)
842 return out;
843 // A nonzero per-row byte bound requires at least one resident slot.
844 if ((in.pinned_source_bytes_per_frame_row > 0 &&
845 in.pinned_source_slots < 1) ||
846 (in.pinned_quality_bytes_per_frame_row > 0 &&
847 in.pinned_quality_slots < 1) ||
848 (in.source_slot_device_bytes > 0 && in.source_device_slots < 1) ||
849 (in.quality_slot_device_bytes > 0 && in.quality_device_slots < 1))
850 return out;
851
852 // Per-native-pixel-per-channel accumulator bytes from the frozen role
853 // table, plus frame planes at internal resolution (scale^2 subpixels).
854 std::size_t per_channel = 0, tmp = 0, pixel_bytes = 0;
855 const std::size_t acc_doubles =
856 static_cast<std::size_t>(r.robust_doubles) +
857 static_cast<std::size_t>(r.confidence_doubles) +
858 static_cast<std::size_t>(r.centre_scale_doubles) +
859 static_cast<std::size_t>(r.fold_staging_doubles);
860 const std::size_t frame_planes =
861 static_cast<std::size_t>(r.frame_plane_doubles) +
862 (r.sigma2_plane ? std::size_t{1} : std::size_t{0});
863 std::size_t internal_bytes = 0;
864 const std::size_t scale_sq =
865 static_cast<std::size_t>(in.internal_scale) *
866 static_cast<std::size_t>(in.internal_scale);
867 // Gate-9 amendment: profiled mode adds the float4 quality side array per
868 // reservoir slot and the profile result record (per channel), plus the five
869 // quality frame planes at internal resolution.
870 const std::size_t slot_bytes =
871 r.reservoir_candidate_bytes +
872 (in.emit_profiles ? r.reservoir_quality_bytes_per_slot
873 : std::size_t{0});
874 const std::size_t all_frame_planes =
875 frame_planes +
876 (in.emit_profiles
877 ? static_cast<std::size_t>(r.quality_frame_plane_doubles)
878 : std::size_t{0});
879 std::size_t reservoir_slots = 0;
880 if (!checked_mul(static_cast<std::size_t>(r.reservoir_size),
881 static_cast<std::size_t>(r.reservoir_slot_factor),
882 reservoir_slots) ||
883 !checked_mul(reservoir_slots, slot_bytes, per_channel) ||
884 !checked_add(per_channel, r.reservoir_kept_bytes, per_channel) ||
885 !checked_mul(acc_doubles, sizeof(double), tmp) ||
886 !checked_add(per_channel, tmp, per_channel) ||
887 !checked_add(per_channel, r.confidence_counter_bytes, per_channel) ||
888 !checked_add(per_channel, r.coverage_bytes_per_channel, per_channel) ||
889 !checked_add(per_channel, r.result_record_bytes, per_channel) ||
890 (in.emit_profiles &&
891 !checked_add(per_channel, r.profile_result_bytes, per_channel)) ||
892 !checked_mul(per_channel, static_cast<std::size_t>(in.channels),
893 pixel_bytes) ||
894 !checked_add(pixel_bytes, r.support_bytes_per_pixel, pixel_bytes) ||
895 !checked_mul(all_frame_planes, sizeof(double), internal_bytes) ||
896 !checked_mul(internal_bytes, static_cast<std::size_t>(in.channels),
897 internal_bytes) ||
898 !checked_mul(internal_bytes, scale_sq, internal_bytes) ||
899 !checked_add(pixel_bytes, internal_bytes, pixel_bytes))
900 return out;
901 out.device_bytes_per_target_pixel = pixel_bytes;
902
903 // Device fixed roles: pipeline slots, frame-meta table and reserve.
904 std::size_t device_fixed = 0;
905 if (!checked_mul(static_cast<std::size_t>(in.source_device_slots),
906 in.source_slot_device_bytes, device_fixed) ||
907 !checked_mul(in.frame_meta_slots, r.frame_meta_bytes_per_slot, tmp) ||
908 !checked_add(device_fixed, tmp, device_fixed) ||
909 !checked_mul(static_cast<std::size_t>(in.quality_device_slots),
910 in.quality_slot_device_bytes, tmp) ||
911 !checked_add(device_fixed, tmp, device_fixed) ||
912 !checked_add(device_fixed, in.device_reserve_bytes, device_fixed) ||
913 device_fixed >= in.device_budget_bytes)
914 return out;
915 const std::size_t device_usable = in.device_budget_bytes - device_fixed;
916 const std::size_t max_pixels = device_usable / pixel_bytes;
917
918 // Minimum padded band height: one core row plus halo on both edges.
919 std::size_t min_padded = 0;
920 if (!checked_mul(static_cast<std::size_t>(in.halo_rows_per_band_edge), 2,
921 min_padded) ||
922 !checked_add(min_padded, std::size_t{1}, min_padded))
923 return out;
924
925 // Full-width first; X-tiling only when one padded full-width row does not
926 // fit. tile_cols never depends on frame_count.
927 const std::size_t max_cols_for_band = max_pixels / min_padded;
928 out.tile_cols = static_cast<int>(std::min<std::size_t>(
929 static_cast<std::size_t>(in.target_width), max_cols_for_band));
930 if (out.tile_cols <= 0) return out;
931 out.x_tiled = out.tile_cols < in.target_width;
932 const std::size_t device_rows = max_pixels / static_cast<std::size_t>(out.tile_cols);
933
934 // Host pinned bytes per padded band row, bounded by slot counts.
935 std::size_t pinned_per_row = 0;
936 if (!checked_mul(static_cast<std::size_t>(in.pinned_source_slots),
937 in.pinned_source_bytes_per_frame_row, pinned_per_row) ||
938 !checked_mul(static_cast<std::size_t>(in.pinned_quality_slots),
939 in.pinned_quality_bytes_per_frame_row, tmp) ||
940 !checked_add(pinned_per_row, tmp, pinned_per_row))
941 return out;
942 std::size_t host_rows = std::numeric_limits<std::size_t>::max();
943 if (pinned_per_row > 0) {
944 if (in.host_fixed_bytes >= in.host_budget_bytes) return out;
945 host_rows = (in.host_budget_bytes - in.host_fixed_bytes) / pinned_per_row;
946 } else if (in.host_fixed_bytes > in.host_budget_bytes) {
947 return out;
948 }
949
950 const std::size_t halo2 =
951 2 * static_cast<std::size_t>(in.halo_rows_per_band_edge);
952 std::size_t needed_rows = 0;
953 if (!checked_add(static_cast<std::size_t>(in.target_height), halo2,
954 needed_rows))
955 return out;
956 const std::size_t padded_rows =
957 std::min({device_rows, host_rows, needed_rows});
958 if (padded_rows < min_padded ||
959 padded_rows > static_cast<std::size_t>(
960 std::numeric_limits<int>::max()))
961 return out;
962 out.band_rows = static_cast<int>(padded_rows);
963 out.band_core_rows = static_cast<int>(padded_rows - halo2);
964 out.band_count = static_cast<int>(
965 (static_cast<std::size_t>(in.target_height) +
966 static_cast<std::size_t>(out.band_core_rows) - 1) /
967 static_cast<std::size_t>(out.band_core_rows));
968 // Conservative bound: every band is charged its full halo even though the
969 // lifetime contract reuses shared edge rows.
970 out.predicted_read_amplification =
971 (static_cast<double>(in.target_height) +
972 2.0 * static_cast<double>(in.halo_rows_per_band_edge) *
973 static_cast<double>(out.band_count)) /
974 static_cast<double>(in.target_height);
975
976 std::size_t pixels = 0;
977 if (!checked_mul(static_cast<std::size_t>(out.tile_cols), padded_rows,
978 pixels) ||
979 !checked_mul(pixels, pixel_bytes, out.device_dynamic_bytes) ||
980 !checked_add(device_fixed, out.device_dynamic_bytes,
981 out.device_peak_bytes) ||
982 !checked_mul(pinned_per_row, padded_rows, out.host_pinned_bytes) ||
983 !checked_add(in.host_fixed_bytes, out.host_pinned_bytes,
984 out.host_peak_bytes))
985 return ForwardDrizzleV2MemoryPlan{};
986 out.feasible = out.device_peak_bytes <= in.device_budget_bytes &&
987 out.host_peak_bytes <= in.host_budget_bytes &&
988 out.predicted_read_amplification <=
989 in.max_source_read_amplification &&
990 out.predicted_read_amplification <=
991 in.max_quality_read_amplification;
992 return out;
993}

997ForwardDrizzleV2ProfileCandidate forward_drizzle_v2_fold_profile_candidate(
998 std::size_t frame_order, double a, double b,
999 const ForwardDrizzleV2QualityFold &q,
1000 const ForwardDrizzleV2QualityFold &q0,
1001 const ForwardDrizzleV2QualityFold &q1,
1002 const ForwardDrizzleV2QualityFold &qa) {
1003 if (!std::isfinite(a) || !std::isfinite(b) || b < 0.0 ||
1004 !std::isfinite(q.sum) || !std::isfinite(q0.sum) ||
1005 !std::isfinite(q1.sum) || !std::isfinite(qa.sum) ||
1006 !std::isfinite(q.weight_sum) || !std::isfinite(qa.weight_sum))
1007 throw std::invalid_argument("FORWARD_DRIZZLE_V2_PROFILE_FOLD");
1008 ForwardDrizzleV2ProfileCandidate out;
1009 out.frame_order = frame_order;
1010 out.b = b;
1011 if (b > 0.0) out.x = a / b;
1012 out.q = q.present && b > 0.0 ? q.sum / b : 1.0;
1013 out.q0 = q0.present && b > 0.0 ? q0.sum / b : 1.0;
1014 out.q1 = q1.present && b > 0.0 ? q1.sum / b : 1.0;

1016 out.qa_has_data = qa.present && qa.weight_sum > 0.0;

1032ForwardDrizzleV2ProfileResult forward_drizzle_v2_profile_reduce(
1033 std::span<const ForwardDrizzleV2ProfileCandidate> candidates,
1034 std::span<const std::uint8_t> accepted,
1035 std::span<const ForwardDrizzleV2FrameMeta> frame_meta,
1036 const ForwardDrizzleV2ProfileConfig &cfg, double confidence,
1037 bool sigma_fully_degraded, ForwardDrizzleV2ProfileScratch &scratch) {
1038 if (accepted.size() != candidates.size())
1039 throw std::invalid_argument("FORWARD_DRIZZLE_V2_PROFILE_MASK_SIZE");
1040 if (!std::isfinite(cfg.fine_quality_exponent) ||
1041 !std::isfinite(cfg.medium_quality_exponent) ||
1042 cfg.fine_quality_exponent < 0.0f || cfg.medium_quality_exponent < 0.0f)
1043 throw std::invalid_argument("FORWARD_DRIZZLE_V2_PROFILE_EXPONENT");
1044
1045 ForwardDrizzleV2ProfileResult out;
1046 struct Accum {
1047 double wx = 0.0, w = 0.0, w2 = 0.0;
1048 } au, ar, af, am;
1049 auto add = [](Accum &a, double w, double x) {
1050 a.wx += w * x;
1051 a.w += w;
1052 a.w2 += w * w;
1053 };
1054 std::vector<AlphaFactorContribution> &contribs = scratch.contribs;
1055 contribs.clear();
1056 for (std::size_t k = 0; k < candidates.size(); ++k) {
1057 if (!accepted[k]) continue;
1058 const auto &cd = candidates[k];
1059 if (cd.frame_order >= frame_meta.size())
1060 throw std::invalid_argument("FORWARD_DRIZZLE_V2_PROFILE_META_RANGE");
1061 const auto &fm = frame_meta[cd.frame_order];
1062 const double g = static_cast<double>(fm.g_eff);
1063 if (!std::isfinite(g) || g < 0.0)
1064 throw std::invalid_argument("FORWARD_DRIZZLE_V2_PROFILE_GEFF");
1065 if (!std::isfinite(cd.x) || !std::isfinite(cd.b) || cd.b < 0.0 ||
1066 !std::isfinite(cd.q) || cd.q < 0.0 || !std::isfinite(cd.q0) ||
1067 cd.q0 < 0.0 || !std::isfinite(cd.q1) || cd.q1 < 0.0 ||
1068 (cd.qa_has_data && !std::isfinite(cd.qa)))
1069 throw std::invalid_argument("FORWARD_DRIZZLE_V2_PROFILE_CANDIDATE");
1070 add(au, cd.b, cd.x);
1071 add(ar, cd.b * g * cd.q, cd.x);
1072 add(af, cd.b * g * std::pow(cd.q0, cfg.fine_quality_exponent), cd.x);
1073 add(am, cd.b * g * std::pow(cd.q1, cfg.medium_quality_exponent), cd.x);
1074 if (cfg.emit_alpha_confidence) {
1075 contribs.push_back(
1076 {cd.b, cd.q,
1077 cd.qa_has_data ? cd.qa
1078 : std::numeric_limits<double>::quiet_NaN(),
1079 fm.is_direct != 0, static_cast<double>(fm.residual_factor)});
1080 }
1081 }
1082 auto write = [](ForwardDrizzleV2ProfileOutput &p, const Accum &a) {
1083 if (!(a.w > 0.0)) return;
1084 p.value = static_cast<float>(a.wx / a.w);
1085 p.weight_sum = static_cast<float>(a.w);
1086 p.n_eff = static_cast<float>(a.w2 > 0.0 ? (a.w * a.w) / a.w2 : 0.0);
1087 p.support = 1;
1088 };
1089 write(out.uniform, au);
1090 write(out.raw, ar);
1091 write(out.fine, af);
1092 write(out.medium, am);
1093 if (cfg.emit_alpha_confidence) {
1094 const auto fac =
1095 compute_alpha_confidence_channel(contribs, cfg.alpha, scratch.alpha);
1096 // sigma_fully_degraded: every accepted contributor carried an invalid
1097 // sigma2, so no noise model underwrites fine-scale separation.
1098 out.a_separation = static_cast<float>(
1099 sigma_fully_degraded
1100 ? 0.0
1101 : std::isfinite(confidence)
1102 ? std::clamp(confidence, 0.0, 1.0)
1103 : fac.a_separation);
1104 out.a_artifact = static_cast<float>(fac.a_artifact);
1105 out.a_registration = static_cast<float>(fac.a_registration);
1106 out.artifact_applicable = fac.artifact_applicable;
1107 }
1108 return out;
1109}

1128ForwardDrizzleV2StreamingRA forward_drizzle_v2_streaming_ra(
1129 std::uint64_t total_rows, std::uint64_t row_bytes, std::uint64_t halo_rows,
1130 std::uint64_t band_count) {
1131 if (total_rows == 0 || row_bytes == 0 || band_count == 0)
1132 throw std::invalid_argument("FORWARD_DRIZZLE_V2_RA_ARGS");
1133 ForwardDrizzleV2StreamingRA out;
1134 const auto mul = [](std::uint64_t a, std::uint64_t b) {
1135 if (b && a > std::numeric_limits<std::uint64_t>::max() / b)
1136 throw std::invalid_argument("FORWARD_DRIZZLE_V2_RA_OVERFLOW");
1137 return a * b;
1138 };
1139 out.logical_bytes = mul(total_rows, row_bytes);
1140 // Without reuse every band pays both edge pads; with the ring, each pad
1141 // row is read once (for the first band needing it) and reused by the
1142 // next: the union of all padded windows is total + 2*halo rows.
1143 const std::uint64_t pad = mul(halo_rows, row_bytes);
1144 out.application_bytes_no_reuse =
1145 out.logical_bytes + mul(mul(2, band_count), pad);
1146 out.application_bytes_reuse = out.logical_bytes + mul(2, pad);
1147 out.reused_bytes =
1148 out.application_bytes_no_reuse - out.application_bytes_reuse;
1149 out.ra_no_reuse = static_cast<double>(out.application_bytes_no_reuse) /
1150 static_cast<double>(out.logical_bytes);
1151 out.ra_reuse = static_cast<double>(out.application_bytes_reuse) /
1152 static_cast<double>(out.logical_bytes);
1153 return out;

1261// --- Gate-9 ring state machine ------------------------------------------------
1262//
1263// Input band i covers rows [i*core, min((i+1)*core, H)) of the v2 output
1264// grid; the fusion window of output band b is [b*core - h, (b+1)*core + h)
1265// clamped to the canvas. Core b becomes immutable once every input band
1266// intersecting its window is finalized; input band i stays resident until
1267// every core whose window intersects it is immutable. The read model counts
1268// each band's padded window once and credits the rows already covered by
1269// resident windows as reused.
1270
1271struct ForwardDrizzleV2MultibandRing::Impl {
1272 ForwardDrizzleV2RingPlan plan;
1273 std::uint64_t src_row_bytes = 0;
1274 std::uint64_t q_row_bytes = 0;
1275 int nbands = 0;
1276 std::vector<char> finalized;
1277 std::vector<char> immutable;
1278 std::vector<char> resident;
1279 // Covered read windows: per-row coverage flags so out-of-order finalize
1280 // still counts only rows never resident before.
1281 std::vector<char> covered;
1282 std::uint64_t src_reused = 0, q_reused = 0;
1283 std::uint64_t src_read = 0, q_read = 0;
1284
1285 int win_lo(int band) const {
1286 return std::max(0, band * plan.band_core_rows - plan.halo_rows);
1287 }
1288 int win_hi(int band) const {
1289 return std::min(plan.canvas_rows,
1290 (band + 1) * plan.band_core_rows + plan.halo_rows);
1291 }
1292 // Input bands whose row extent intersects [lo, hi).
1293 int input_lo(int lo) const {
1294 return std::max(0, lo / plan.band_core_rows);
1295 }
1296 int input_hi(int hi) const {
1297 // Last band index whose start row is < hi.
1298 return std::min(nbands - 1, (hi - 1) / plan.band_core_rows);
1299 }
1300};

1322std::vector<int> ForwardDrizzleV2MultibandRing::finalize_band(int band) {
1323 Impl &im = *impl_;
1324 if (band < 0 || band >= im.nbands || im.finalized[band])
1325 throw std::invalid_argument("FORWARD_DRIZZLE_V2_RING_FINALIZE");
1326 im.finalized[band] = 1;
1327
1328 // Read accounting: this band's padded window is read once; rows already
1329 // covered by a resident predecessor window are reused, not re-read.
1330 const int lo = im.win_lo(band), hi = im.win_hi(band);
1331 std::uint64_t fresh = 0, reused = 0;
1332 for (int r = lo; r < hi; ++r) {
1333 if (im.covered[r]) ++reused;
1334 else ++fresh;
1335 im.covered[r] = 1;
1336 }
1337 im.src_read += fresh * im.src_row_bytes;
1338 im.q_read += fresh * im.q_row_bytes;
1339 im.src_reused += reused * im.src_row_bytes;
1340 im.q_reused += reused * im.q_row_bytes;
1341 im.resident[band] = 1;
1342
1343 std::vector<int> newly;
1344 // A core becomes immutable once every input band overlapping its halo
1345 // window is finalized.
1346 for (int b = 0; b < im.nbands; ++b) {
1347 if (im.immutable[b]) continue;
1348 const int ilo = im.input_lo(im.win_lo(b));
1349 const int ihi = im.input_hi(im.win_hi(b));
1350 bool ready = true;
1351 for (int i = ilo; i <= ihi; ++i)
1352 if (!im.finalized[i]) {
1353 ready = false;
1354 break;
1355 }
1356 if (ready) {
1357 im.immutable[b] = 1;
1358 newly.push_back(b);
1359 }
1360 }
1361 // Evict input bands no core still needs: every output band whose window
1362 // overlaps band i must already be immutable.
1363 for (int i = 0; i < im.nbands; ++i) {
1364 if (!im.resident[i]) continue;
1365 const int i_lo = i * im.plan.band_core_rows;
1366 const int i_hi =
1367 std::min(im.plan.canvas_rows, (i + 1) * im.plan.band_core_rows);
1368 const int clo = im.input_lo(std::max(0, i_lo - im.plan.halo_rows));
1369 const int chi =
1370 im.input_hi(std::min(im.plan.canvas_rows, i_hi + im.plan.halo_rows));
1371 bool needed = false;
1372 for (int b = clo; b <= chi; ++b)
1373 if (!im.immutable[b]) {
1374 needed = true;
1375 break;
1376 }
1377 if (!needed) im.resident[i] = 0;
1378 }
1379 return newly;
1380}
