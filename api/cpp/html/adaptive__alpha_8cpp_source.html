11double alpha_smoothstep(double edge0, double edge1, double x) {
12 if (!(edge1 > edge0)) return x >= edge1 ? 1.0 : 0.0;
13 const double t = std::clamp((x - edge0) / (edge1 - edge0), 0.0, 1.0);
14 return t * t * (3.0 - 2.0 * t);
