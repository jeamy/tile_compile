25static std::string sanitize_yaml_windows_paths(const std::string& yaml) {
26 std::string out;
27 out.reserve(yaml.size());
28 bool in_dq = false;
29 for (size_t i = 0; i < yaml.size(); ++i) {
30 char c = yaml[i];
31 if (!in_dq) {
32 if (c == '"') in_dq = true;
33 out.push_back(c);
34 } else {
35 if (c == '\\' && i + 1 < yaml.size()) {
36 char next = yaml[i + 1];
37 // Keep valid YAML escape sequences (e.g. \n \t \\ \" \/ \uXXXX)
38 static const char valid[] = "\"\\0abtnvfrNLP_e/xuU";
39 bool valid_esc = false;
40 for (char v : valid) { if (next == v) { valid_esc = true; break; } }
41 if (!valid_esc) {
42 // Treat as literal backslash in a Windows path → forward slash
43 out.push_back('/');
44 } else {
45 out.push_back(c);
46 }
47 } else {
48 if (c == '"') in_dq = false;
49 out.push_back(c);
50 }
51 }
52 }
53 return out;
54}

110std::string trim_trailing_zeros(std::string text) {
111 const auto dot = text.find('.');
112 if (dot == std::string::npos) {
113 return text;
114 }
115 while (!text.empty() && text.back() == '0') {
116 text.pop_back();
117 }
118 if (!text.empty() && text.back() == '.') {
119 text.pop_back();
120 }
121 if (text == "-0") {
122 return "0";
123 }
124 return text.empty() ? "0" : text;

131std::string format_config_float_scalar(double value) {
132 if (!std::isfinite(value)) {
133 return "0";
134 }
135
136 const double rounded = std::round(value * 100.0) / 100.0;
137 if (rounded == 0.0 && value != 0.0 && std::fabs(value) < 0.01) {
138 std::ostringstream oss;
139 oss << std::scientific << std::setprecision(2) << value;
140 return oss.str();

143 std::ostringstream oss;
144 oss << std::fixed << std::setprecision(2) << rounded;
145 return trim_trailing_zeros(oss.str());

152void round_yaml_numeric_scalars_inplace(YAML::Node node) {
153 if (!node || node.IsNull()) {
154 return;
155 }
156 if (node.IsMap()) {
157 for (auto it = node.begin(); it != node.end(); ++it) {
158 round_yaml_numeric_scalars_inplace(it->second);
159 }
160 return;
161 }
162 if (node.IsSequence()) {
163 for (auto it = node.begin(); it != node.end(); ++it) {
164 round_yaml_numeric_scalars_inplace(*it);
165 }
166 return;
167 }
168 if (!node.IsScalar()) {
169 return;
170 }
171
172 const std::string raw = node.Scalar();
173 if (!scalar_looks_like_float(raw)) {
174 return;
175 }
176
177 char *end = nullptr;
178 errno = 0;
179 const double value = std::strtod(raw.c_str(), &end);
180 if (errno != 0 || end == raw.c_str() || (end && *end != '\0') ||
181 !std::isfinite(value)) {
182 return;
183 }
184 node = format_config_float_scalar(value);
185}
