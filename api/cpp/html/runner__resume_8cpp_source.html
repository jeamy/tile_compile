447 }
448
449 int row_idx = 0;
450 for (auto &kv : y_to_row)
451 kv.second = row_idx++;
452 int col_idx = 0;
453 for (auto &kv : x_to_col)
454 kv.second = col_idx++;
455
456 for (auto &t : out.tiles) {
457 t.row = y_to_row[t.y];
458 t.col = x_to_col[t.x];
459 }
460 out.rows = static_cast<int>(y_to_row.size());
461 out.cols = static_cast<int>(x_to_col.size());
462 return true;
463 } catch (const std::exception &e) {
464 error_out = std::string("tile_grid parse failed: ") + e.what();
465 return false;

473bool load_aggregated_tile_metrics(const fs::path &local_metrics_path,
474 std::vector<TileMetrics> &out,
475 std::string &error_out) {
476 if (!fs::exists(local_metrics_path)) {
477 error_out = "missing local_metrics.json";
478 return false;
479 }
480 try {
481 const auto j = tile_compile::core::json::parse(
482 tile_compile::core::read_text(local_metrics_path));
483 if (!j.contains("tile_metrics") || !j["tile_metrics"].is_array() ||
484 j["tile_metrics"].empty()) {
485 error_out = "local_metrics.json missing tile_metrics[][]";
486 return false;
487 }
488
489 const auto &all_frames = j["tile_metrics"];
490 size_t n_tiles = 0;
491 if (all_frames.front().is_array()) {
492 n_tiles = all_frames.front().size();
493 }
494 if (n_tiles == 0) {
495 error_out = "local_metrics.json has zero tiles";
496 return false;
497 }
498
499 const bool consistent = std::all_of(
500 all_frames.begin(), all_frames.end(),
501 [n_tiles](const auto &fm) { return fm.is_array() && fm.size() == n_tiles; });
502
503 if (!consistent) {
504 out.clear();
505 out.reserve(n_tiles);
506 for (size_t ti = 0; ti < n_tiles; ++ti) {
507 out.push_back(parse_tile_metrics_json(all_frames.front()[ti]));
508 }
509 return !out.empty();
510 }
511
512 auto median_or_zero = [](std::vector<float> vals) -> float {
513 if (vals.empty()) return 0.0f;
514 return tile_compile::core::median_of(vals);
515 };
516
517 out.assign(n_tiles, TileMetrics{});
518 for (size_t ti = 0; ti < n_tiles; ++ti) {
519 std::vector<float> fwhm_vals;
520 std::vector<float> round_vals;
521 std::vector<float> contrast_vals;
522 std::vector<float> sharp_vals;
523 std::vector<float> bg_vals;
524 std::vector<float> noise_vals;
525 std::vector<float> grad_vals;
526 std::vector<float> q_vals;
527 std::vector<float> star_count_vals;
528 int star_votes = 0;
529 int structure_votes = 0;
530
531 fwhm_vals.reserve(all_frames.size());
532 round_vals.reserve(all_frames.size());
533 contrast_vals.reserve(all_frames.size());
534 sharp_vals.reserve(all_frames.size());
535 bg_vals.reserve(all_frames.size());
536 noise_vals.reserve(all_frames.size());
537 grad_vals.reserve(all_frames.size());
538 q_vals.reserve(all_frames.size());
539 star_count_vals.reserve(all_frames.size());
540

542 const TileMetrics tm = parse_tile_metrics_json(fm[ti]);
543 if (std::isfinite(tm.fwhm)) fwhm_vals.push_back(tm.fwhm);
544 if (std::isfinite(tm.roundness)) round_vals.push_back(tm.roundness);
545 if (std::isfinite(tm.contrast)) contrast_vals.push_back(tm.contrast);
546 if (std::isfinite(tm.sharpness)) sharp_vals.push_back(tm.sharpness);
547 if (std::isfinite(tm.background)) bg_vals.push_back(tm.background);
548 if (std::isfinite(tm.noise)) noise_vals.push_back(tm.noise);

550 if (std::isfinite(tm.quality_score)) q_vals.push_back(tm.quality_score);
551 star_count_vals.push_back(static_cast<float>(tm.star_count));
552 if (tm.type == TileType::STAR) {
553 ++star_votes;
554 } else {
555 ++structure_votes;
556 }
557 }
558
559 TileMetrics agg{};
560 agg.fwhm = median_or_zero(std::move(fwhm_vals));
561 agg.roundness = median_or_zero(std::move(round_vals));
562 agg.contrast = median_or_zero(std::move(contrast_vals));
563 agg.sharpness = median_or_zero(std::move(sharp_vals));
564 agg.background = median_or_zero(std::move(bg_vals));
565 agg.noise = median_or_zero(std::move(noise_vals));
566 agg.gradient_energy = median_or_zero(std::move(grad_vals));
567 agg.quality_score = median_or_zero(std::move(q_vals));
568 agg.star_count = static_cast<int>(
569 std::lround(median_or_zero(std::move(star_count_vals))));
570 agg.type = (star_votes >= structure_votes) ? TileType::STAR
571 : TileType::STRUCTURE;
572 out[ti] = agg;
573 }
574 return true;
575 } catch (const std::exception &e) {
576 error_out = std::string("local_metrics parse failed: ") + e.what();
577 return false;
578 }
