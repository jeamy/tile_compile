48 throw std::invalid_argument("SOURCE_QUALITY_INVALID_CONFIG");
49 for (float v : {cfg.w_bg,cfg.w_noise,cfg.w_grad,cfg.w_fwhm,cfg.w_roundness,

52 if (cfg.clamp_lo>cfg.clamp_hi) throw std::invalid_argument("SOURCE_QUALITY_INVALID_CONFIG");

54 // image-metric scratch. No frame-count-multiplied full-image allocation.

56 geometry.canvas_width_native=sampling.source_width;
57 geometry.canvas_height_native=sampling.source_height;

73 throw std::runtime_error("SOURCE_QUALITY_MEMORY_BUDGET");

263// Combine up to 3 channel planes into the fixed working luminance
264// (kWorkingLumaDefinition) over an n-pixel block: MONO -> channel 0 as-is;
265// OSC -> 0.25 R + 0.5 G + 0.25 B. Luma-supported only where EVERY active
266// channel is finite and supported.
267void combine_luma(const std::vector<const std::vector<float> *> &vals,
268 const std::vector<const std::vector<uint8_t> *> &sups,int nch,
269 std::size_t n,std::vector<float> &luma,
270 std::vector<uint8_t> &sup) {
271 luma.assign(n,std::numeric_limits<float>::quiet_NaN());
272 sup.assign(n,0u);
273 const double wgt[3]={nch==1?1.0:kWorkingLumaWeightsOsc[0],
274 kWorkingLumaWeightsOsc[1],kWorkingLumaWeightsOsc[2]};
275 for (std::size_t i=0;i<n;++i) {
276 double acc=0.0; bool ok=true;
277 for (int c=0;c<nch;++c) {
278 if (!(*sups[c])[i] || !std::isfinite((*vals[c])[i])) { ok=false; break; }
279 acc+=wgt[c]*(*vals[c])[i];
280 }
281 if (ok) { luma[i]=static_cast<float>(acc); sup[i]=1u; }
282 }
283}

292MultibandFusionMemoryPlan plan_multiband_fusion_memory(
293 int width,int height,int nch,int levels,int chunk_rows,int halo_rows,
294 bool with_candidate_luma,bool with_candidate_spool,std::size_t budget_bytes) {
295 MultibandFusionMemoryPlan p;
296 p.budget_bytes=budget_bytes;
297 if (width<=0||height<=0||nch<=0||levels<=0||chunk_rows<=0) return p; // !fits
298 const auto sat_mul=[](std::size_t a,std::size_t b)->std::size_t{
299 if (a!=0 && b>std::numeric_limits<std::size_t>::max()/a)
300 return std::numeric_limits<std::size_t>::max();
301 return a*b;
302 };
303 const auto sat_add=[](std::size_t a,std::size_t b)->std::size_t{
304 return (b>std::numeric_limits<std::size_t>::max()-a)
305 ? std::numeric_limits<std::size_t>::max() : a+b;
306 };
307 const std::size_t n=sat_mul(static_cast<std::size_t>(width),
308 static_cast<std::size_t>(height));
309 const std::size_t nc=static_cast<std::size_t>(nch);
310 const std::size_t lv=static_cast<std::size_t>(levels);
311 const std::size_t stripe_rows=static_cast<std::size_t>(chunk_rows)+
312 2u*static_cast<std::size_t>(std::max(0,halo_rows));
313 const std::size_t stripe_px=sat_mul(stripe_rows,
314 static_cast<std::size_t>(width));
315
316 // Final X_out held whole: nch planes * 4 bytes.
317 p.final_image_bytes=sat_mul(sat_mul(nc,n),4u);
318 // One resident fusion stripe: U/R/F/M (value+support, all nch) + the fused
319 // `sub` (value+support+levels alpha) + 3 alpha maps + combine_luma scratch.
320 // Generous per-pixel upper bound: nch*40 + levels*8 + 64 bytes.
321 p.stripe_working_bytes=sat_mul(stripe_px,
322 sat_add(sat_add(sat_mul(nc,40u),sat_mul(lv,8u)),64u));
323 // Candidate working-luminance buffers held whole (plan 15 selection needs the
324 // full field): 3 luma (4) + support (1) + levels alpha (4).
325 p.candidate_luma_bytes=with_candidate_luma
326 ? sat_mul(n,sat_add(13u,sat_mul(lv,4u))) : 0u;
327 // Spool write path is zero-copy from the stripe buffers; only OS write
328 // buffering is transient. Budget up to 64 rows per (candidate, channel).
329 p.spool_stripe_bytes=with_candidate_spool
330 ? sat_mul(sat_mul(sat_mul(3u,nc),
331 sat_mul(static_cast<std::size_t>(width),
332 std::min<std::size_t>(64u,stripe_rows))),4u)
333 : 0u;
334 // Delivery: the runner reads back one spooled plane at a time for FITS export.
335 p.delivery_readback_bytes=with_candidate_spool ? sat_mul(n,4u) : 0u;
336
337 std::size_t sum=0;
338 for (std::size_t v : {p.final_image_bytes,p.stripe_working_bytes,
339 p.candidate_luma_bytes,p.spool_stripe_bytes,
340 p.delivery_readback_bytes})
341 sum=sat_add(sum,v);
342 // 5% of the working set, with a modest absolute floor that is itself capped
343 // at the working set so a tiny image under a tiny explicit budget is not
344 // rejected by the margin alone.
345 p.margin_bytes=std::max<std::size_t>(
346 sum/20u,std::min<std::size_t>(std::size_t(64)<<20,sum));
347 p.estimated_peak_bytes=sat_add(sum,p.margin_bytes);
348 p.fits=p.estimated_peak_bytes<=budget_bytes;
349
350 // Plan 11.13(2) + 11.11 temp space: the spool writes 3 * nch * N * 4 bytes to
351 // the temp filesystem. `required_free_temp` here uses the 2 GiB floor; the
352 // caller refines it with the real filesystem capacity and fills
353 // available_temp_bytes / temp_space_ok.
354 if (with_candidate_spool) {
355 p.spool_temp_bytes=sat_mul(sat_mul(sat_mul(3u,nc),n),4u);
356 p.required_free_temp_bytes=sat_add(
357 static_cast<std::size_t>(static_cast<long double>(p.spool_temp_bytes)*1.20L),
358 std::size_t(2)<<30);
359 }
360 return p;
361}

412 fcfg.levels=plan.multiband_levels;
413 const int W=plan.native_width, H=plan.native_height;

423 // (channel-major ProfileResult buffer over chunk + 2*halo rows).
424 MultibandFusionMemoryPlan mem_plan=plan_multiband_fusion_memory(
425 W,H,nch_plan,fcfg.levels,chunk,halo,
426 candidates_out!=nullptr,spool_out!=nullptr,
427 static_cast<std::size_t>(budget)*1024*1024);
428 const std::size_t window_rows=
429 static_cast<std::size_t>(chunk)+2u*static_cast<std::size_t>(halo);
430 const std::size_t window_record_bytes=
431 window_rows*static_cast<std::size_t>(W)*
432 static_cast<std::size_t>(nch_plan)*rec_bytes;
433 mem_plan.stripe_working_bytes+=window_record_bytes;
434 // robust_state veto (see band_robust_states below): a retained 1 B/record
435 // window (~1/64 of window_record_bytes) plus one transient full-size
436 // ForwardDrizzleV2PixelResult band decode (freed once its states are
437 // extracted) sized by the store's own band_rows, not the fusion chunk.
438 const std::size_t robust_state_window_bytes=
439 window_rows*static_cast<std::size_t>(W)*static_cast<std::size_t>(nch_plan);
440 const std::size_t one_band_pixel_result_bytes=
441 static_cast<std::size_t>(std::max(plan.band_rows,1))*
442 static_cast<std::size_t>(W)*static_cast<std::size_t>(nch_plan)*
443 sizeof(ForwardDrizzleV2PixelResult);
