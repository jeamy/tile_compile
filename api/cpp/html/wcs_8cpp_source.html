43static double parse_fits_double(const std::string &val_str) {
44 std::string s = val_str;
45 // Remove trailing comment (after /)
46 auto slash = s.find('/');
47 if (slash != std::string::npos) s = s.substr(0, slash);
48 // Trim whitespace
49 while (!s.empty() && (s.front() == ' ' || s.front() == '\t')) s.erase(s.begin());
50 while (!s.empty() && (s.back() == ' ' || s.back() == '\t' || s.back() == '\r')) s.pop_back();
51 // Remove surrounding quotes if present
52 if (s.size() >= 2 && s.front() == '\'' && s.back() == '\'') {
53 s = s.substr(1, s.size() - 2);
54 while (!s.empty() && s.back() == ' ') s.pop_back();
55 }
56
57 std::stringstream ss(s);
58 ss.imbue(std::locale::classic());
59 double result = 0.0;
60 ss >> result;
61 if (ss.fail()) return 0.0;
62 return result;
63}
