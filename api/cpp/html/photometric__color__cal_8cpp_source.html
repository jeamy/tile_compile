32static PlaneFit fit_sky_plane_huber(const std::vector<double> &xs,
33 const std::vector<double> &ys,
34 const std::vector<double> &vs,
35 double huber_delta,
36 int max_iters) {
37 PlaneFit pf;
38 const size_t n = vs.size();
39 if (n < 20) return pf;
40
41 for (size_t i = 0; i < n; ++i) {
42 pf.mx += xs[i];
43 pf.my += ys[i];
44 }
45 pf.mx /= static_cast<double>(n);
46 pf.my /= static_cast<double>(n);
47
48 std::vector<double> w(n, 1.0);

50 auto solve_wls = [&](double &a, double &b, double &c) -> bool {
51 double S00 = 0.0, S01 = 0.0, S02 = 0.0, S11 = 0.0, S12 = 0.0, S22 = 0.0;
52 double T0 = 0.0, T1 = 0.0, T2 = 0.0;
53 for (size_t i = 0; i < n; ++i) {
54 const double wi = w[i];
55 const double X = xs[i] - pf.mx;
56 const double Y = ys[i] - pf.my;
57 S00 += wi;
58 S01 += wi * X;
59 S02 += wi * Y;
60 S11 += wi * X * X;
61 S12 += wi * X * Y;
62 S22 += wi * Y * Y;
63 const double vi = vs[i];
64 T0 += wi * vi;
65 T1 += wi * vi * X;
66 T2 += wi * vi * Y;
67 }
68
69 double A[3][4] = {
70 {S00, S01, S02, T0},
71 {S01, S11, S12, T1},
72 {S02, S12, S22, T2},
73 };
74
75 for (int r = 0; r < 3; ++r) {
76 int piv = r;
77 for (int rr = r + 1; rr < 3; ++rr) {
78 if (std::fabs(A[rr][r]) > std::fabs(A[piv][r])) piv = rr;
79 }
80 if (std::fabs(A[piv][r]) < 1e-12) return false;
81 if (piv != r) {
82 for (int c0 = r; c0 < 4; ++c0) std::swap(A[r][c0], A[piv][c0]);
83 }
84
85 const double div = A[r][r];
86 for (int c0 = r; c0 < 4; ++c0) A[r][c0] /= div;
87 for (int rr = 0; rr < 3; ++rr) {
88 if (rr == r) continue;
89 const double f = A[rr][r];
90 for (int c0 = r; c0 < 4; ++c0) A[rr][c0] -= f * A[r][c0];
91 }
92 }
93
94 a = A[0][3];
95 b = A[1][3];
96 c = A[2][3];
97 return true;
98 };
99
100 double a = 0.0, b = 0.0, c = 0.0;
101 if (!solve_wls(a, b, c)) return pf;
102
103 for (int it = 0; it < std::max(1, max_iters); ++it) {
104 std::vector<double> r(n, 0.0);
105 for (size_t i = 0; i < n; ++i) {
106 const double X = xs[i] - pf.mx;
107 const double Y = ys[i] - pf.my;
108 r[i] = vs[i] - (a + b * X + c * Y);
109 }
110
111 std::vector<double> absr = r;
112 for (double &v : absr) v = std::fabs(v);
113 std::sort(absr.begin(), absr.end());
114 double mad = absr[absr.size() / 2] * 1.4826;
115 mad = std::max(1e-9, mad);
116 const double delta = std::max(1e-6, huber_delta * mad);
117
118 for (size_t i = 0; i < n; ++i) {
119 const double ar = std::fabs(r[i]);
120 w[i] = (ar <= delta) ? 1.0 : (delta / ar);
121 }
122
123 const double a_prev = a;
124 const double b_prev = b;
125 const double c_prev = c;
126 if (!solve_wls(a, b, c)) return pf;
127
128 const double step = std::fabs(a - a_prev) + std::fabs(b - b_prev) + std::fabs(c - c_prev);
129 if (step < 1e-8) break;
130 }
131
132 pf.a = a;
133 pf.b = b;
134 pf.c = c;
135 pf.ok = true;
136 return pf;
137}

177// ─── Teff estimation from XP spectrum ───────────────────────────────────
178//
179// Estimate effective temperature from the Gaia XP sampled spectrum by
180// fitting the ratio of blue-band to red-band integrated flux to a
181// Planck function. This avoids the need for filter curves entirely.
182
187static double estimate_teff_from_xp(const std::vector<float> &xp_flux) {
188 if (xp_flux.size() != XPSAMPLED_LEN) return 0.0;
189
190 // Integrate flux in blue (400-500nm) and red (600-700nm) bands
191 double blue_sum = 0, red_sum = 0;
192 int blue_n = 0, red_n = 0;
193 for (int i = 0; i < XPSAMPLED_LEN; ++i) {
194 double wl = XPSAMPLED_WL_START + i * XPSAMPLED_WL_STEP;
195 double f = static_cast<double>(xp_flux[i]);
196 if (f <= 0) continue;
197 if (wl >= 400 && wl <= 500) { blue_sum += f; ++blue_n; }
198 if (wl >= 600 && wl <= 700) { red_sum += f; ++red_n; }
199 }
200 if (blue_n == 0 || red_n == 0 || red_sum <= 0) return 0.0;
201
202 double ratio = (blue_sum / blue_n) / (red_sum / red_n);
203
204 // Map blue/red ratio to Teff using Planck function lookup.
205 // Precomputed: for Planck B(λ,T), ratio of mean flux in 400-500nm
206 // to mean flux in 600-700nm:
207 // T=3000K → ~0.10, T=5000K → ~0.55, T=7000K → ~1.05
208 // T=10000K → ~1.55, T=20000K → ~2.20, T=40000K → ~2.60
209 // Use bisection on Planck function for accuracy.
210 auto planck_ratio = [](double T) -> double {
211 constexpr double h = 6.62607015e-34;
212 constexpr double c = 2.99792458e8;
213 constexpr double k = 1.380649e-23;
214 double b_sum = 0, r_sum = 0;
215 for (double wl = 400; wl <= 500; wl += 2.0) {
216 double wl_m = wl * 1e-9;
217 b_sum += (1.0 / (wl_m * wl_m * wl_m * wl_m * wl_m)) /
218 (std::exp(h * c / (wl_m * k * T)) - 1.0);
219 }
220 for (double wl = 600; wl <= 700; wl += 2.0) {
221 double wl_m = wl * 1e-9;
222 r_sum += (1.0 / (wl_m * wl_m * wl_m * wl_m * wl_m)) /
223 (std::exp(h * c / (wl_m * k * T)) - 1.0);
224 }
225 return (r_sum > 0) ? b_sum / r_sum : 0.0;
226 };
227
228 // Bisection: find T such that planck_ratio(T) ≈ ratio
229 double T_lo = 2000, T_hi = 50000;
230 for (int i = 0; i < 40; ++i) {
231 double T_mid = 0.5 * (T_lo + T_hi);
232 if (planck_ratio(T_mid) < ratio)
233 T_lo = T_mid;
234 else
235 T_hi = T_mid;
236 }
237 double teff = 0.5 * (T_lo + T_hi);
238 return (teff >= 2000 && teff <= 50000) ? teff : 0.0;
239}

241// ─── Blackbody temperature to linear sRGB ───────────────────────────────
242//
243// Converts a color temperature to linear sRGB (r,g,b) normalized so
244// max(r,g,b) = 1. Uses the CIE 1931 2° standard observer + Planck
245// function → XYZ → linear sRGB (D65).
246//
247// This is the same approach as Siril's TempK2rgb() but without lcms2.
248
253static void teff_to_rgb(double T, double &r, double &g, double &b) {
254 r = g = b = 0;
255 if (T < 1000 || T > 50000) return;
256
257 // CIE 1931 2° color matching functions (5nm steps, 380-780nm)
258 // Tabulated x̄, ȳ, z̄ — we use a simplified analytical approximation
259 // (multi-lobe Gaussian fit by Wyman et al. 2013)
260 auto xFit = [](double wl) -> double {
261 double t1 = (wl - 442.0) * ((wl < 442.0) ? 0.0624 : 0.0374);
262 double t2 = (wl - 599.8) * ((wl < 599.8) ? 0.0264 : 0.0323);
263 double t3 = (wl - 501.1) * ((wl < 501.1) ? 0.0490 : 0.0382);
264 return 0.362 * std::exp(-0.5*t1*t1) + 1.056 * std::exp(-0.5*t2*t2)
265 - 0.065 * std::exp(-0.5*t3*t3);
266 };
267 auto yFit = [](double wl) -> double {
268 double t1 = (wl - 568.8) * ((wl < 568.8) ? 0.0213 : 0.0247);
269 double t2 = (wl - 530.9) * ((wl < 530.9) ? 0.0613 : 0.0322);
270 return 0.821 * std::exp(-0.5*t1*t1) + 0.286 * std::exp(-0.5*t2*t2);
271 };
272 auto zFit = [](double wl) -> double {
273 double t1 = (wl - 437.0) * ((wl < 437.0) ? 0.0845 : 0.0278);
274 double t2 = (wl - 459.0) * ((wl < 459.0) ? 0.0385 : 0.0725);
275 return 1.217 * std::exp(-0.5*t1*t1) + 0.681 * std::exp(-0.5*t2*t2);
276 };
277
278 constexpr double h = 6.62607015e-34;
279 constexpr double c = 2.99792458e8;
280 constexpr double k = 1.380649e-23;
281
282 double X = 0, Y = 0, Z = 0;
283 for (double wl = 380; wl <= 780; wl += 5.0) {
284 double wl_m = wl * 1e-9;
285 double planck = (2.0 * h * c * c / (wl_m * wl_m * wl_m * wl_m * wl_m)) /
286 (std::exp(h * c / (wl_m * k * T)) - 1.0);
287 X += planck * xFit(wl);
288 Y += planck * yFit(wl);
289 Z += planck * zFit(wl);
290 }
291
292 // CIE XYZ → linear sRGB (D65 illuminant)
293 // sRGB matrix (IEC 61966-2-1)
294 r = 3.2404542 * X - 1.5371385 * Y - 0.4985314 * Z;
295 g = -0.9692660 * X + 1.8760108 * Y + 0.0415560 * Z;
296 b = 0.0556434 * X - 0.2040259 * Y + 1.0572252 * Z;
297
298 // Clamp negatives, then normalize so sum = 1.
299 // Sum-normalization makes cat values comparable across different Teff
300 // (unlike max=1 which changes the reference channel per star).
301 r = std::max(0.0, r);
302 g = std::max(0.0, g);
303 b = std::max(0.0, b);
304 double s = r + g + b;
305 if (s > 0) { r /= s; g /= s; b /= s; }
306}

314static double aperture_flux(const Matrix2Df &img, double cx, double cy,
315 double r_ap, double r_ann_in, double r_ann_out,
316 const std::string &background_model,
317 const std::vector<uint8_t> *support_mask = nullptr) {
318 int rows = img.rows();
319 int cols = img.cols();
320 const bool use_support_mask =
321 (support_mask != nullptr &&
322 support_mask->size() == static_cast<size_t>(rows * cols));
323
324 int x0 = std::max(0, static_cast<int>(cx - r_ann_out - 1));
325 int x1 = std::min(cols - 1, static_cast<int>(cx + r_ann_out + 1));
326 int y0 = std::max(0, static_cast<int>(cy - r_ann_out - 1));
327 int y1 = std::min(rows - 1, static_cast<int>(cy + r_ann_out + 1));
328
329 double r_ap2 = r_ap * r_ap;
330 double r_in2 = r_ann_in * r_ann_in;
331 double r_out2 = r_ann_out * r_ann_out;
332
333 // Collect sky annulus pixels for background estimation
334 std::vector<float> sky_pixels;
335 std::vector<double> sky_x;
336 std::vector<double> sky_y;
337 sky_x.reserve(static_cast<size_t>(std::max(1, (x1 - x0 + 1) * (y1 - y0 + 1) / 2)));
338 sky_y.reserve(sky_x.capacity());
339 for (int y = y0; y <= y1; ++y) {
340 for (int x = x0; x <= x1; ++x) {
341 double dx = x - cx;
342 double dy = y - cy;
343 double d2 = dx * dx + dy * dy;
344 if (d2 >= r_in2 && d2 <= r_out2) {

346 if (use_support_mask && (*support_mask)[idx] == 0) {

351 sky_pixels.push_back(v);
352 sky_x.push_back(static_cast<double>(x));
353 sky_y.push_back(static_cast<double>(y));
354 }
355 }
356 }
357 }
358
359 if (sky_pixels.size() < 10) return -1.0; // not enough sky pixels
360
361 // Median sky background (fallback and noise guard reference)
362 std::sort(sky_pixels.begin(), sky_pixels.end());
363 double sky_bg_median = sky_pixels[sky_pixels.size() / 2];
364 PlaneFit plane_fit{};
365 bool use_plane_bg = false;
366
367 if (background_model == "plane") {
368 std::vector<double> sky_vals;
369 sky_vals.reserve(sky_pixels.size());
370 for (float v : sky_pixels) sky_vals.push_back(static_cast<double>(v));
371 plane_fit = fit_sky_plane_huber(sky_x, sky_y, sky_vals, 1.5, 8);
372 use_plane_bg = plane_fit.ok;
373 if (use_plane_bg) {
374 sky_bg_median = plane_fit.a + plane_fit.b * (cx - plane_fit.mx) +
375 plane_fit.c * (cy - plane_fit.my);
376 }
377 }
378
379 if (!(std::isfinite(sky_bg_median) && sky_bg_median > 0.0)) return -1.0;
380 const size_t n_sky = sky_pixels.size();
381 const float q1 = sky_pixels[(n_sky * 1) / 4];
382 const float q3 = sky_pixels[(n_sky * 3) / 4];
383 const double iqr = static_cast<double>(q3) - static_cast<double>(q1);
384 if (!(std::isfinite(iqr) && iqr >= 0.0)) return -1.0;
385 if (iqr > 0.35 * sky_bg_median) return -1.0;
386
387 // Sum aperture flux minus background.
388 // For model=plane subtract the local plane value at each aperture pixel.
389 double total = 0.0;
390 int n_ap = 0;
391 for (int y = y0; y <= y1; ++y) {
392 for (int x = x0; x <= x1; ++x) {
393 double dx = x - cx;
394 double dy = y - cy;
395 if (dx * dx + dy * dy <= r_ap2) {
396 const size_t idx = static_cast<size_t>(y * cols + x);
397 if (use_support_mask && (*support_mask)[idx] == 0) {
398 continue;
399 }
400 float v = img(y, x);
401 if (std::isfinite(v) && v > 0.0f) {
402 double bg_here = sky_bg_median;
403 if (use_plane_bg) {
404 bg_here = plane_fit.a + plane_fit.b * (x - plane_fit.mx) +
405 plane_fit.c * (y - plane_fit.my);
406 }
407 if (!(std::isfinite(bg_here) && bg_here > 0.0)) continue;
408 total += static_cast<double>(v) - bg_here;
409 ++n_ap;
410 }
411 }
412 }
413 }
414
415 return (n_ap > 0) ? total : -1.0;
416}

491 int rows, int cols) {

503 std::vector<int> current;
504 current.reserve(4096);

506 for (int y = 0; y < rows; ++y) {
507 for (int x = 0; x < cols; ++x) {

518 const int idx = q.front();
519 q.pop();
520 current.push_back(idx);
521
522 const int cy = idx / cols;
523 const int cx = idx - cy * cols;
524 for (int k = 0; k < 8; ++k) {
525 const int nx = cx + dx8[k];
526 const int ny = cy + dy8[k];
527 if (nx < 0 || ny < 0 || nx >= cols || ny >= rows) continue;
528 const int nidx = ny * cols + nx;
529 const size_t npos = static_cast<size_t>(nidx);
530 if (mask[npos] == 0 || visited[npos] != 0) continue;
531 visited[npos] = 1;
532 q.push(nidx);
533 }
534 }
535
536 if (current.size() > best_component.size()) {
537 best_component = current;
538 }
539 }
540 }
541
542 if (best_component.empty()) {
543 std::fill(mask.begin(), mask.end(), static_cast<uint8_t>(0));
544 return 0;
545 }
546
547 std::vector<uint8_t> out(mask.size(), 0);
548 for (int idx : best_component) {
549 out[static_cast<size_t>(idx)] = 1;
550 }

591static ColorMatrix blend_matrix_with_identity_per_channel(
592 const ColorMatrix &m, double alpha_r, double alpha_b) {
593 alpha_r = std::clamp(alpha_r, 0.0, 1.0);
594 alpha_b = std::clamp(alpha_b, 0.0, 1.0);
595 // Use a neutral diagonal anchor at the current green scale (m[1][1]).
596 // This preserves color-neutral fallback behavior in linear mode:
597 // alpha=0 => diag(kg, kg, kg) instead of [1, kg, 1], avoiding red/magenta bias.
598 const double kg = (std::isfinite(m[1][1]) && m[1][1] > 0.0) ? m[1][1] : 1.0;
599 ColorMatrix out = {{{kg, 0, 0}, {0, kg, 0}, {0, 0, kg}}};
600 for (int r = 0; r < 3; ++r) {
601 for (int c = 0; c < 3; ++c) {
602 double ident = 0.0;
603 if (r == 0 && c == 0) {
604 ident = kg;
605 } else if (r == 1 && c == 1) {
606 ident = kg;
607 } else if (r == 2 && c == 2) {
608 ident = kg;
609 }
610 double a = 1.0;
611 if (r == 0) {
612 a = alpha_r;
613 } else if (r == 2) {
614 a = alpha_b;
615 }
616 out[r][c] = ident + a * (m[r][c] - ident);
617 }
618 }
619 return out;
620}

870// ─── Color matrix fitting (Siril SPCC-inspired) ─────────────────────────
871//
872// Inspired by Siril's get_spcc_white_balance_coeffs:
873// 1. Per star: measure image flux ratios (r/g, b/g) and catalog flux ratios
874// 2. Robust linear fit: image_rg = a + b * catalog_rg (repeated median)
875// 3. Evaluate fit at a white reference
876// 4. Normalize with green anchor (kg=1)
877//
878// White reference selection (critical for field-independent color correction):
879// 1. Prefer white/neutral stars (Teff 5500-7000K, Sun-like) from catalog
880// 2. Fallback to broader Teff range (4500-8000K) if insufficient
881// 3. Last resort: use all stars but clamp to [0.8, 1.2] to prevent bias
882// 4. Final fallback: neutral 1.0
883//
884// Why this matters:
885// - Using all-star median biases correction on red fields (IC434) → green cast
886// - Using fixed 1.0 ignores field-specific systematic shifts → color errors
887// - Using catalog-white stars gives field-independent neutral reference
888//
889// The repeated median fit (Siegel 1982) is breakdown-point 0.5 and
890// handles both slope and intercept robustly unlike simple ratio medians.
891
896static double weighted_median(const std::vector<double> &vals,
897 const std::vector<double> &weights) {
898 if (vals.empty() || vals.size() != weights.size()) return 0.0;
899 std::vector<std::pair<double, double>> vw;
900 vw.reserve(vals.size());
901 double w_sum = 0.0;
902 for (size_t i = 0; i < vals.size(); ++i) {
903 const double w = std::max(1.0e-9, weights[i]);
904 vw.emplace_back(vals[i], w);
905 w_sum += w;
906 }
907 if (w_sum <= 0.0) return 0.0;
908
909 std::sort(vw.begin(), vw.end(),
910 [](const auto &a, const auto &b) { return a.first < b.first; });
911 const double half = 0.5 * w_sum;
912 double acc = 0.0;
913 for (const auto &p : vw) {
914 acc += p.second;
915 if (acc >= half) return p.first;
916 }
917 return vw.back().first;
918}

924static double simple_median(std::vector<double> v) {
925 if (v.empty()) return 0.0;
926 const size_t n = v.size();
927 std::nth_element(v.begin(), v.begin() + n / 2, v.end());
928 if (n % 2 == 1) return v[n / 2];
929 const double hi = v[n / 2];
930 std::nth_element(v.begin(), v.begin() + n / 2 - 1, v.end());
931 return 0.5 * (v[n / 2 - 1] + hi);
932}

934// Siegel's repeated median estimator: robust linear fit y = a + b*x.
935// Breakdown point 0.5. Returns false if insufficient data.
936// deviation_out = MAD of residuals after fit.
941static bool repeated_median_fit(const std::vector<double> &x,
942 const std::vector<double> &y,
943 double &a_out, double &b_out,
944 double &deviation_out) {
945 const size_t n = x.size();
946 deviation_out = 0.0;
947 if (n < 3 || x.size() != y.size()) return false;
948
949 // For each i: median over j≠i of (y[j]-y[i])/(x[j]-x[i])
950 std::vector<double> slopes_i;
951 slopes_i.reserve(n);
952 for (size_t i = 0; i < n; ++i) {
953 std::vector<double> slopes_j;
954 slopes_j.reserve(n - 1);
955 for (size_t j = 0; j < n; ++j) {
956 if (j == i) continue;
957 const double dx = x[j] - x[i];
958 if (std::abs(dx) < 1.0e-15) continue;
959 slopes_j.push_back((y[j] - y[i]) / dx);
960 }
961 if (slopes_j.empty()) continue;
962 slopes_i.push_back(simple_median(slopes_j));
963 }
964 if (slopes_i.size() < 3) return false;
965
966 b_out = simple_median(slopes_i);
967
968 // Intercept: median of (y[i] - b*x[i])
969 std::vector<double> intercepts;
970 intercepts.reserve(n);
971 for (size_t i = 0; i < n; ++i)
972 intercepts.push_back(y[i] - b_out * x[i]);
973 a_out = simple_median(intercepts);
974
975 // MAD of residuals
976 std::vector<double> resid;
977 resid.reserve(n);
978 for (size_t i = 0; i < n; ++i)
979 resid.push_back(std::abs(y[i] - (a_out + b_out * x[i])));
980 deviation_out = 1.4826 * simple_median(resid);
981 return true;
982}

984// Weighted robust location estimator with iterative sigma clipping around
985// weighted median / weighted MAD.
990static double robust_mean_weighted(std::vector<double> &data,
991 std::vector<double> &weights,
992 double sigma,
993 double &deviation_out) {
994 deviation_out = 0.0;
995 if (data.empty() || data.size() != weights.size()) return 0.0;
996
997 for (int iter = 0; iter < 5; ++iter) {
998 const int n = static_cast<int>(data.size());
999 if (n < 3) break;
1000 const double center = weighted_median(data, weights);
1001
1002 std::vector<double> devs;
1003 devs.reserve(data.size());
1004 for (double v : data) devs.push_back(std::abs(v - center));
1005 const double mad = 1.4826 * weighted_median(devs, weights);
1006 if (mad < 1.0e-15) break;
1007
1008 const double threshold = sigma * mad;
1009 std::vector<double> kept_data;
1010 std::vector<double> kept_weights;
1011 kept_data.reserve(data.size());
1012 kept_weights.reserve(weights.size());
1013 for (size_t i = 0; i < data.size(); ++i) {
1014 if (std::abs(data[i] - center) <= threshold) {
1015 kept_data.push_back(data[i]);
1016 kept_weights.push_back(weights[i]);
1017 }
1018 }
1019 if (static_cast<int>(kept_data.size()) < 3) break;
1020 if (kept_data.size() == data.size()) break;
1021 data.swap(kept_data);
1022 weights.swap(kept_weights);
1023 }
1024
1025 const double center = weighted_median(data, weights);
1026 double dev_sum = 0.0;
1027 double w_sum = 0.0;
1028 for (size_t i = 0; i < data.size(); ++i) {
1029 const double w = std::max(1.0e-9, weights[i]);
1030 dev_sum += w * std::abs(data[i] - center);
1031 w_sum += w;
1032 }
1033 deviation_out = (w_sum > 0.0) ? (dev_sum / w_sum) : 0.0;
1034 return center;
1035}

1037// Recompute robust log-chroma residuals for the actually applied PCC matrix.
1042static double recompute_residual_rms_for_matrix(const std::vector<StarPhotometry> &stars,
1043 const ColorMatrix &matrix,
1044 double sigma_clip,
1045 int *n_used_out) {
1046 if (n_used_out != nullptr) *n_used_out = 0;
1047
1048 std::vector<double> d_rg_vec;
1049 std::vector<double> d_bg_vec;
1050 std::vector<double> w_rg_vec;
1051 std::vector<double> w_bg_vec;
1052 d_rg_vec.reserve(stars.size());
1053 d_bg_vec.reserve(stars.size());
1054 w_rg_vec.reserve(stars.size());
1055 w_bg_vec.reserve(stars.size());
1056
1057 for (const auto &s : stars) {
1058 if (!(s.valid &&
1059 s.flux_r > 0.0 && s.flux_g > 0.0 && s.flux_b > 0.0 &&
1060 s.cat_r > 0.0 && s.cat_g > 0.0 && s.cat_b > 0.0)) {
1061 continue;
1062 }
1063
1064 const double fr = s.flux_r;
1065 const double fg = s.flux_g;
1066 const double fb = s.flux_b;
1067 const double mr = matrix[0][0] * fr + matrix[0][1] * fg + matrix[0][2] * fb;
1068 const double mg = matrix[1][0] * fr + matrix[1][1] * fg + matrix[1][2] * fb;
1069 const double mb = matrix[2][0] * fr + matrix[2][1] * fg + matrix[2][2] * fb;
1070 if (!(std::isfinite(mr) && std::isfinite(mg) && std::isfinite(mb) &&
1071 mr > 0.0 && mg > 0.0 && mb > 0.0)) {
1072 continue;
1073 }
1074
1075 const double meas_rg = mr / mg;
1076 const double meas_bg = mb / mg;
1077 const double cat_rg = s.cat_r / s.cat_g;
1078 const double cat_bg = s.cat_b / s.cat_g;
1079 if (!(meas_rg > 0.0 && meas_bg > 0.0 && cat_rg > 0.0 && cat_bg > 0.0)) {
1080 continue;
1081 }
1082
1083 const double w = std::clamp(s.quality_weight, 1.0e-3, 10.0);
1084 d_rg_vec.push_back(std::log(cat_rg) - std::log(meas_rg));
1085 d_bg_vec.push_back(std::log(cat_bg) - std::log(meas_bg));
1086 w_rg_vec.push_back(w);
1087 w_bg_vec.push_back(w);
1088 }
1089
1090 if (d_rg_vec.size() < 3 || d_bg_vec.size() < 3) {
1091 return std::numeric_limits<double>::infinity();
1092 }
1093
1094 double dev_rg = 0.0;
1095 double dev_bg = 0.0;
1096 (void)robust_mean_weighted(d_rg_vec, w_rg_vec, sigma_clip, dev_rg);
1097 (void)robust_mean_weighted(d_bg_vec, w_bg_vec, sigma_clip, dev_bg);
1098 if (n_used_out != nullptr) {
1099 *n_used_out = static_cast<int>(std::min(d_rg_vec.size(), d_bg_vec.size()));
1100 }
1101 return std::max(dev_rg, dev_bg);
1102}

1109 const PCCConfig &config) {

1111 res.success = false;
1112 res.matrix = {{{1, 0, 0}, {0, 1, 0}, {0, 0, 1}}}; // identity default
1113 res.n_stars_matched = 0;
1114 res.n_stars_used = 0;
1115 res.residual_rms = 0.0;
1116 res.determinant = 1.0;
1117 res.condition_number = 1.0;
1118
1119 // Collect valid stars
1120 std::vector<const StarPhotometry*> valid;
1121 for (const auto &s : stars) {
1122 if (s.valid && s.flux_r > 0 && s.flux_g > 0 && s.flux_b > 0 &&
1123 s.cat_r > 0 && s.cat_g > 0 && s.cat_b > 0)
1124 valid.push_back(&s);
1125 }
1126
1127 res.n_stars_matched = static_cast<int>(valid.size());
1128
1129 if (static_cast<int>(valid.size()) < config.min_stars) {
1130 res.error_message = "Not enough valid stars (" +
1131 std::to_string(valid.size()) + " < " +
1132 std::to_string(config.min_stars) + ")";
1133 return res;
1134 }
1135
1136 // Siril-like robust ratio fit:
1137 // fit image ratios vs catalog ratios, then evaluate at an adaptive
1138 // white reference derived from the catalog colors present in this frame.
1139 std::vector<double> cat_rg_vec, img_rg_vec;
1140 std::vector<double> cat_bg_vec, img_bg_vec;
1141 std::vector<double> w_vec;
1142 cat_rg_vec.reserve(valid.size());
1143 img_rg_vec.reserve(valid.size());
1144 cat_bg_vec.reserve(valid.size());
1145 img_bg_vec.reserve(valid.size());
1146 w_vec.reserve(valid.size());
1147
1148 for (const auto *s : valid) {
1149 if (!(s->flux_r > 0.0 && s->flux_g > 0.0 && s->flux_b > 0.0 &&
1150 s->cat_r > 0.0 && s->cat_g > 0.0 && s->cat_b > 0.0)) {
1151 continue;
1152 }
1153 const double meas_rg = s->flux_r / s->flux_g;
1154 const double meas_bg = s->flux_b / s->flux_g;
1155 const double c_rg = s->cat_r / s->cat_g;
1156 const double c_bg = s->cat_b / s->cat_g;
1157 if (!(std::isfinite(meas_rg) && meas_rg > 0.0 &&
1158 std::isfinite(meas_bg) && meas_bg > 0.0 &&
1159 std::isfinite(c_rg) && c_rg > 0.0 &&
1160 std::isfinite(c_bg) && c_bg > 0.0)) {
1161 continue;
1162 }
1163 const double w = std::clamp(s->quality_weight, 1.0e-3, 10.0);
1164 cat_rg_vec.push_back(c_rg);
1165 img_rg_vec.push_back(meas_rg);
1166 cat_bg_vec.push_back(c_bg);
1167 img_bg_vec.push_back(meas_bg);
1168 w_vec.push_back(w);
1169 }
1170
1171 if (cat_rg_vec.size() < static_cast<size_t>(config.min_stars)) {
1172 res.error_message = "Not enough valid stars for adaptive-ratio PCC fit";
1173 return res;
1174 }
1175
1176 double a_rg = 0.0, b_rg = 1.0, dev_rg = 0.0;
1177 double a_bg = 0.0, b_bg = 1.0, dev_bg = 0.0;
1178 if (!repeated_median_fit(cat_rg_vec, img_rg_vec, a_rg, b_rg, dev_rg)) {
1179 res.error_message = "repeated_median_fit failed for R/G";
1180 return res;
1181 }
1182 if (!repeated_median_fit(cat_bg_vec, img_bg_vec, a_bg, b_bg, dev_bg)) {
1183 res.error_message = "repeated_median_fit failed for B/G";
1184 return res;
1185 }
1186
1187 // Select white reference from catalog-neutral stars (Teff ≈ Sun-like).
1188 // This avoids bias from fields dominated by red (IC434) or blue stars.
1189 // Fallback: use all stars with clamping, then neutral 1.0 as last resort.
1190 std::vector<double> white_cat_rg, white_cat_bg, white_weights;
1191 white_cat_rg.reserve(valid.size());
1192 white_cat_bg.reserve(valid.size());
1193 white_weights.reserve(valid.size());
1194
1195 // Pass 1: Select white stars (Teff 5500-7000K, Sun-like)
1196 // Note: teff=0 means unknown temperature, skip those stars in this pass
1197 for (const auto *s : valid) {
1198 const double teff = static_cast<double>(s->teff);
1199
1200 if (teff >= 5500.0 && teff <= 7000.0) {
1201 const double c_rg = s->cat_r / s->cat_g;
1202 const double c_bg = s->cat_b / s->cat_g;
1203 if (std::isfinite(c_rg) && c_rg > 0.0 &&
1204 std::isfinite(c_bg) && c_bg > 0.0) {
1205 white_cat_rg.push_back(c_rg);
1206 white_cat_bg.push_back(c_bg);
1207 white_weights.push_back(std::clamp(s->quality_weight, 1.0e-3, 10.0));
1208 }
1209 }
1210 }
1211
1212 double wrg = 1.0, wbg = 1.0;
1213 std::string ref_method = "neutral_default";
1214
1215 if (white_cat_rg.size() >= 5) {
1216 // Sufficient white stars found
1217 wrg = weighted_median(white_cat_rg, white_weights);
1218 wbg = weighted_median(white_cat_bg, white_weights);
1219 ref_method = "white_stars_5500_7000K";
1220 } else if (!valid.empty()) {
1221 // Pass 2: Fallback to broader Teff range (4500-8000K)
1222 white_cat_rg.clear();
1223 white_cat_bg.clear();
1224 white_weights.clear();
1225 for (const auto *s : valid) {
1226 const double teff = static_cast<double>(s->teff);
1227
1228 if (teff >= 4500.0 && teff <= 8000.0) {
1229 const double c_rg = s->cat_r / s->cat_g;
1230 const double c_bg = s->cat_b / s->cat_g;
1231 if (std::isfinite(c_rg) && c_rg > 0.0 &&
1232 std::isfinite(c_bg) && c_bg > 0.0) {
1233 white_cat_rg.push_back(c_rg);
1234 white_cat_bg.push_back(c_bg);
1235 white_weights.push_back(std::clamp(s->quality_weight, 1.0e-3, 10.0));
1236 }
1237 }
1238 }
1239
1240 if (white_cat_rg.size() >= 3) {
1241 wrg = weighted_median(white_cat_rg, white_weights);
1242 wbg = weighted_median(white_cat_bg, white_weights);
1243 ref_method = "white_stars_broad_4500_8000K";
1244 } else {
1245 // Pass 3: Use all stars but clamp to prevent extreme bias
1246 wrg = weighted_median(cat_rg_vec, w_vec);
1247 wbg = weighted_median(cat_bg_vec, w_vec);
1248 wrg = std::clamp(wrg, 0.80, 1.20);
1249 wbg = std::clamp(wbg, 0.80, 1.20);
1250 ref_method = "all_stars_clamped";
1251 }
1252 }
1253
1254 // Ensure valid reference point
1255 if (!(std::isfinite(wrg) && wrg > 0.0)) wrg = 1.0;
1256 if (!(std::isfinite(wbg) && wbg > 0.0)) wbg = 1.0;
1257
1258 double kw_r = 1.0 / (a_rg + b_rg * wrg);
1259 double kw_g = 1.0;
1260 double kw_b = 1.0 / (a_bg + b_bg * wbg);
1261 if (!std::isfinite(kw_r) || kw_r <= 0.0 ||
1262 !std::isfinite(kw_b) || kw_b <= 0.0) {
1263 res.error_message = "Degenerate PCC fit: non-positive adaptive-ratio scale factor";
1264 return res;
1265 }
1266
1267 const double raw_k_max = std::max(1.0, static_cast<double>(config.k_max));
1268 kw_r = std::min(kw_r, raw_k_max);
1269 kw_g = std::min(kw_g, raw_k_max);
1270 kw_b = std::min(kw_b, raw_k_max);
1271
1272 // Anchor gains to green (kg = 1) instead of max-normalizing all channels.
1273 // Max-normalization can push m11 away from 1 and, in combination with
1274 // strong damping, collapse PCC to a near-gray global scale.
1275 // Green anchoring keeps the PCC matrix photometrically interpretable:
1276 // measured_rg * kw_r -> target_rg
1277 // measured_bg * kw_b -> target_bg
1278 // while preserving kg as the chromatic reference channel.
1279 const double kg_anchor = std::max(1.0e-6, kw_g);
1280 kw_r = std::max(1.0e-3, kw_r / kg_anchor);
1281 kw_g = 1.0;
1282 kw_b = std::max(1.0e-3, kw_b / kg_anchor);
1283 kw_r = std::min(kw_r, raw_k_max);
1284 kw_b = std::min(kw_b, raw_k_max);
1285
1286 std::cout << "[PCC] White reference: wrg=" << wrg
1287 << " wbg=" << wbg
1288 << " method=" << ref_method
1289 << " white_stars=" << white_cat_rg.size() << std::endl;
1290 std::cout << "[PCC] Repeated-median fit R/G: a=" << a_rg
1291 << " b=" << b_rg << " dev=" << dev_rg << std::endl;
1292 std::cout << "[PCC] Repeated-median fit B/G: a=" << a_bg
1293 << " b=" << b_bg << " dev=" << dev_bg << std::endl;
1294 std::cout << "[PCC] Scale factors after green-anchor norm: R=" << kw_r
1295 << " G=" << kw_g << " B=" << kw_b
1296 << " (raw_k_max=" << raw_k_max << ")" << std::endl;
1297
1298 res.matrix = {{{kw_r, 0, 0}, {0, kw_g, 0}, {0, 0, kw_b}}};
1299 res.n_stars_used = static_cast<int>(cat_rg_vec.size());
1300 int residual_used = 0;
1301 res.residual_rms =
1302 recompute_residual_rms_for_matrix(stars, res.matrix, config.sigma_clip, &residual_used);
1303 if (residual_used > 0) {
1304 res.n_stars_used = residual_used;
1305 }
1306 res.determinant = kw_r * kw_g * kw_b;
1307 const double s0 = std::abs(kw_r);
1308 const double s1 = std::abs(kw_g);
1309 const double s2 = std::abs(kw_b);
1310 const double s_max = std::max({s0, s1, s2});
1311 const double s_min = std::min({s0, s1, s2});
1312 res.condition_number =
1313 (s_min > 1.0e-12) ? (s_max / s_min) : std::numeric_limits<double>::infinity();
1314
1315 std::cout << "[PCC] Matrix stability: det=" << res.determinant
1316 << " cond=" << res.condition_number
1317 << " residual=" << res.residual_rms << std::endl;
1318
1319 if (!std::isfinite(res.determinant) || res.determinant <= 0.0) {
1320 res.error_message = "Unstable PCC matrix: non-positive determinant";
1321 return res;
1322 }
1323 if (!std::isfinite(res.condition_number) ||
1324 res.condition_number > config.max_condition_number) {
1325 res.error_message = "Unstable PCC matrix: condition number " +
1326 std::to_string(res.condition_number) + " exceeds " +
1327 std::to_string(config.max_condition_number);
1328 return res;
1329 }
1330 if (!std::isfinite(res.residual_rms) ||
1331 res.residual_rms > config.max_residual_rms) {
1332 res.error_message = "PCC residual RMS " + std::to_string(res.residual_rms) +
1333 " exceeds " + std::to_string(config.max_residual_rms);
1334 return res;
1335 }
1336
1337 res.success = true;
1338 std::cout << "[PCC] Scale factors: R=" << kw_r
1339 << " G=" << kw_g << " B=" << kw_b << std::endl;
1340 return res;
1341}

1350static float estimate_background(const Matrix2Df &img,
1351 const std::vector<uint8_t> *valid_mask = nullptr) {
1352 int rows = img.rows();
1353 int cols = img.cols();
1354 const bool use_mask =
1355 (valid_mask != nullptr &&
1356 valid_mask->size() == static_cast<size_t>(rows * cols));
1357 // Sample every 8th pixel for speed
1358 std::vector<float> samples;
1359 samples.reserve((rows / 8 + 1) * (cols / 8 + 1));
1360 for (int y = 0; y < rows; y += 8)
1361 for (int x = 0; x < cols; x += 8)
1362 {
1363 const size_t idx = static_cast<size_t>(y * cols + x);
1364 if (use_mask && (*valid_mask)[idx] == 0) {
1365 continue;
1366 }
1367 float v = img(y, x);
1368 if (std::isfinite(v) && v > 0.0f) {
1369 samples.push_back(v);
1370 }
1371 }
1372 if (samples.empty())
1373 return 0.0f;
1374 std::sort(samples.begin(), samples.end());
1375 return samples[samples.size() / 2];
1376}

1382static float percentile_sorted(const std::vector<float> &sorted, float q) {
1383 if (sorted.empty()) return 0.0f;

1397static std::vector<float> box_blur_with_valid_mask(const std::vector<float> &src,
1398 const std::vector<uint8_t> &valid,
1399 int rows, int cols,
1400 int radius) {
1401 std::vector<float> out(src.size(), 0.0f);
1402 if (rows <= 0 || cols <= 0 || src.size() != valid.size() ||
1403 src.size() != static_cast<size_t>(rows * cols)) {
1404 return out;
1405 }
1406 radius = std::max(0, radius);
1407 if (radius == 0) return src;
1408
1409 const int stride = cols + 1;
1410 std::vector<double> isum(static_cast<size_t>((rows + 1) * (cols + 1)), 0.0);
1411 std::vector<int> icnt(static_cast<size_t>((rows + 1) * (cols + 1)), 0);
1412 auto idx = [stride](int y, int x) { return static_cast<size_t>(y * stride + x); };
1413
1414 for (int y = 0; y < rows; ++y) {
1415 double row_sum = 0.0;
1416 int row_cnt = 0;
1417 for (int x = 0; x < cols; ++x) {
1418 const size_t p = static_cast<size_t>(y * cols + x);
1419 if (valid[p] != 0) {
1420 row_sum += static_cast<double>(src[p]);
1421 ++row_cnt;
1422 }
1423 const size_t q = idx(y + 1, x + 1);
1424 isum[q] = isum[idx(y, x + 1)] + row_sum;
1425 icnt[q] = icnt[idx(y, x + 1)] + row_cnt;
1426 }
1427 }
1428
1429 for (int y = 0; y < rows; ++y) {
1430 const int y0 = std::max(0, y - radius);
1431 const int y1 = std::min(rows - 1, y + radius);
1432 for (int x = 0; x < cols; ++x) {
1433 const int x0 = std::max(0, x - radius);
1434 const int x1 = std::min(cols - 1, x + radius);
1435 const size_t p11 = idx(y1 + 1, x1 + 1);
1436 const size_t p01 = idx(y0, x1 + 1);
1437 const size_t p10 = idx(y1 + 1, x0);
1438 const size_t p00 = idx(y0, x0);
1439 const double sum = isum[p11] - isum[p01] - isum[p10] + isum[p00];
1440 const int cnt = icnt[p11] - icnt[p01] - icnt[p10] + icnt[p00];
1441 const size_t p = static_cast<size_t>(y * cols + x);
1442 if (cnt > 0) {
1443 out[p] = static_cast<float>(sum / static_cast<double>(cnt));
1444 } else {
1445 out[p] = src[p];
1446 }
1447 }
1448 }
1449 return out;
1450}

1472static PCCAttenuationContext build_pcc_attenuation_context(const Matrix2Df &R,
1473 const Matrix2Df &G,
1474 const Matrix2Df &B,
1475 const std::vector<uint8_t> *valid_mask = nullptr,
1476 float shadow_floor = kShadowAttenFloor,
1477 float highlight_floor = kHighlightAttenFloor) {
1478 PCCAttenuationContext ctx;
1479 ctx.rows = R.rows();
1480 ctx.cols = R.cols();
1481 ctx.shadow_floor = shadow_floor;
1482 ctx.highlight_floor = highlight_floor;
1483 if (ctx.rows <= 0 || ctx.cols <= 0) return ctx;
1484 const bool use_mask =
1485 (valid_mask != nullptr &&
1486 valid_mask->size() == static_cast<size_t>(ctx.rows * ctx.cols));
1487
1488 ctx.bg_r = estimate_background(R, valid_mask);
1489 ctx.bg_g = estimate_background(G, valid_mask);
1490 ctx.bg_b = estimate_background(B, valid_mask);
1491 ctx.bg_out = (ctx.bg_r + ctx.bg_g + ctx.bg_b) / 3.0f;
1492
1493 std::vector<float> signal_luma;
1494 signal_luma.reserve((ctx.rows / 4 + 1) * (ctx.cols / 4 + 1));
1495 for (int y = 0; y < ctx.rows; y += 4) {
1496 for (int x = 0; x < ctx.cols; x += 4) {
1497 const size_t idx = static_cast<size_t>(y * ctx.cols + x);
1498 if (use_mask && (*valid_mask)[idx] == 0) {
1499 continue;
1500 }
1501 const float r0 = R(y, x);
1502 const float g0 = G(y, x);
1503 const float b0 = B(y, x);
1504 if (!(std::isfinite(r0) && r0 > 0.0f &&
1505 std::isfinite(g0) && g0 > 0.0f &&
1506 std::isfinite(b0) && b0 > 0.0f)) {
1507 continue;
1508 }
1509 const float dr = r0 - ctx.bg_out;
1510 const float dg = g0 - ctx.bg_out;
1511 const float db = b0 - ctx.bg_out;
1512 const float luma =
1513 std::max(0.0f, 0.2126f * dr + 0.7152f * dg + 0.0722f * db);
1514 if (luma > 0.0f) signal_luma.push_back(luma);
1515 }
1516 }
1517
1518 if (signal_luma.size() >= 32) {
1519 std::sort(signal_luma.begin(), signal_luma.end());
1520 ctx.shadow_lo = percentile_sorted(signal_luma, 0.01f);
1521 ctx.shadow_hi = percentile_sorted(signal_luma, 0.20f);
1522 ctx.blend_lo = percentile_sorted(signal_luma, 0.90f);
1523 ctx.blend_hi = percentile_sorted(signal_luma, 0.995f);
1524 if (!(std::isfinite(ctx.shadow_lo) && std::isfinite(ctx.shadow_hi) &&
1525 ctx.shadow_hi > ctx.shadow_lo)) {
1526 ctx.shadow_lo = 0.0f;
1527 ctx.shadow_hi = 0.0f;
1528 }
1529 if (!(std::isfinite(ctx.blend_lo) && std::isfinite(ctx.blend_hi) &&
1530 ctx.blend_hi > ctx.blend_lo)) {
1531 ctx.blend_lo = 0.0f;
1532 ctx.blend_hi = 0.0f;
1533 }
1534 }
1535
1536 std::vector<float> luma_raw(static_cast<size_t>(ctx.rows * ctx.cols), 0.0f);
1537 std::vector<uint8_t> luma_valid(static_cast<size_t>(ctx.rows * ctx.cols), 0);
1538 for (int y = 0; y < ctx.rows; ++y) {
1539 for (int x = 0; x < ctx.cols; ++x) {
1540 const size_t p = static_cast<size_t>(y * ctx.cols + x);
1541 if (use_mask && (*valid_mask)[p] == 0) {
1542 continue;
1543 }
1544 const float r0 = R(y, x);
1545 const float g0 = G(y, x);
1546 const float b0 = B(y, x);
1547 if (!(std::isfinite(r0) && r0 > 0.0f &&
1548 std::isfinite(g0) && g0 > 0.0f &&
1549 std::isfinite(b0) && b0 > 0.0f)) {
1550 continue;
1551 }
1552 const float dr = r0 - ctx.bg_out;
1553 const float dg = g0 - ctx.bg_out;
1554 const float db = b0 - ctx.bg_out;
1555 luma_raw[p] = std::max(0.0f, 0.2126f * dr + 0.7152f * dg + 0.0722f * db);
1556 luma_valid[p] = 1;
1557 }
1558 }
1559 ctx.luma_smooth = box_blur_with_valid_mask(luma_raw, luma_valid, ctx.rows, ctx.cols, 12);
1560 return ctx;
1561}

1567static float attenuation_from_luma(float luma, const PCCAttenuationContext &ctx) {
1568 float atten_shadows = 1.0f;
1569 if (ctx.shadow_hi > ctx.shadow_lo && luma < ctx.shadow_hi) {
1570 float t = (luma - ctx.shadow_lo) / (ctx.shadow_hi - ctx.shadow_lo);
1571 t = std::clamp(t, 0.0f, 1.0f);
1572 const float s = t * t * (3.0f - 2.0f * t);
1573 atten_shadows = ctx.shadow_floor + (1.0f - ctx.shadow_floor) * s;
1574 }
1575
1576 float atten_highlights = 1.0f;
1577 if (ctx.blend_hi > ctx.blend_lo && luma > ctx.blend_lo) {
1578 float t = (luma - ctx.blend_lo) / (ctx.blend_hi - ctx.blend_lo);
1579 t = std::clamp(t, 0.0f, 1.0f);
1580 const float s = t * t * (3.0f - 2.0f * t);
1581 atten_highlights = 1.0f - (1.0f - ctx.highlight_floor) * s;
1582 }
1583 return std::min(atten_shadows, atten_highlights);
1584}

1590static inline void apply_color_matrix_to_deltas(const ColorMatrix &m, float atten,
1591 float dr, float dg, float db,
1592 float *nr, float *ng, float *nb) {
1593 const float m00 = static_cast<float>(1.0 + atten * (m[0][0] - 1.0));
1594 const float m01 = static_cast<float>(atten * m[0][1]);
1595 const float m02 = static_cast<float>(atten * m[0][2]);
1596 const float m10 = static_cast<float>(atten * m[1][0]);
1597 const float m11 = static_cast<float>(1.0 + atten * (m[1][1] - 1.0));
1598 const float m12 = static_cast<float>(atten * m[1][2]);
1599 const float m20 = static_cast<float>(atten * m[2][0]);
1600 const float m21 = static_cast<float>(atten * m[2][1]);
1601 const float m22 = static_cast<float>(1.0 + atten * (m[2][2] - 1.0));
1602 *nr = m00 * dr + m01 * dg + m02 * db;
1603 *ng = m10 * dr + m11 * dg + m12 * db;
1604 *nb = m20 * dr + m21 * dg + m22 * db;
1605}

1709 double bg_std = std::numeric_limits<double>::infinity();

1716static PCCBackgroundStdPair sampled_background_std_after_matrix(
1717 const std::vector<PCCBackgroundSample> &samples,
1718 const ColorMatrix &matrix) {
1719 PCCBackgroundStdPair out;
1720 if (samples.empty()) return out;
1721
1722 size_t n_rg = 0;
1723 size_t n_bg = 0;
1724 double mean_rg = 0.0;
1725 double mean_bg = 0.0;
1726 double m2_rg = 0.0;
1727 double m2_bg = 0.0;
1728
1729 for (const auto &s : samples) {
1730 float nr = 0.0f;
1731 float ng = 0.0f;
1732 float nb = 0.0f;
1733 apply_color_matrix_to_deltas(matrix, s.atten, s.dr, s.dg, s.db, &nr, &ng, &nb);
1734
1735 const double rv = static_cast<double>(s.bg_out + nr);
1736 const double gv = static_cast<double>(s.bg_out + ng);
1737 const double bv = static_cast<double>(s.bg_out + nb);
1738 if (!(std::isfinite(gv) && gv > 0.0)) continue;
1739
1740 if (std::isfinite(rv) && rv > 0.0) {
1741 const double v = std::log(rv / gv);
1742 ++n_rg;
1743 const double d = v - mean_rg;
1744 mean_rg += d / static_cast<double>(n_rg);
1745 m2_rg += d * (v - mean_rg);
1746 }
1747 if (std::isfinite(bv) && bv > 0.0) {
1748 const double v = std::log(bv / gv);
1749 ++n_bg;
1750 const double d = v - mean_bg;
1751 mean_bg += d / static_cast<double>(n_bg);
1752 m2_bg += d * (v - mean_bg);
1753 }
1754 }
1755
1756 if (n_rg >= 512) out.rg_std = std::sqrt(m2_rg / static_cast<double>(n_rg));
1757 if (n_bg >= 512) out.bg_std = std::sqrt(m2_bg / static_cast<double>(n_bg));
1758 return out;
1759}

1765static float robust_sigma_from_samples(std::vector<float> values) {
1766 if (values.size() < 32) return 0.0f;
1767 std::sort(values.begin(), values.end());
1768 const float median = core::percentile_from_sorted(values, 50.0f);
1769 for (float &v : values) v = std::abs(v - median);
1770 std::sort(values.begin(), values.end());
1771 const float mad = core::percentile_from_sorted(values, 50.0f);
1772 return 1.4826f * mad;
1773}

1803static PCCBackgroundNeutralizationDecision decide_pcc_background_neutralization(
1804 const Matrix2Df &R, const Matrix2Df &G, const Matrix2Df &B,
1805 const std::vector<uint8_t> &analysis_mask,
1806 const std::string &mode) {
1807 PCCBackgroundNeutralizationDecision out;
1808 if (analysis_mask.size() != static_cast<size_t>(R.rows() * R.cols())) {
1809 out.reason = "invalid_analysis_mask";
1810 return out;
1811 }
1812 if (mode == "off") {
1813 out.reason = "disabled_by_config";
1814 return out;
1815 }
1816
1817 const std::vector<uint8_t> bg_mask =
1818 image::build_chroma_background_mask_from_rgb(R, G, B, analysis_mask);
1819 if (bg_mask.empty()) {
1820 out.reason = "empty_background_mask";
1821 return out;
1822 }
1823
1824 if (!estimate_channel_background_median(R, bg_mask, &out.bg_r) ||
1825 !estimate_channel_background_median(G, bg_mask, &out.bg_g) ||
1826 !estimate_channel_background_median(B, bg_mask, &out.bg_b)) {
1827 out.reason = "insufficient_background_samples";
1828 return out;
1829 }
1830
1831 size_t analysis_count = 0;
1832 size_t bg_count = 0;
1833 std::vector<float> luma_samples;
1834 luma_samples.reserve(bg_mask.size() / 2);
1835 for (int y = 0; y < R.rows(); ++y) {
1836 for (int x = 0; x < R.cols(); ++x) {
1837 const size_t idx = static_cast<size_t>(y * R.cols() + x);
1838 if (analysis_mask[idx] != 0) ++analysis_count;
1839 if (bg_mask[idx] == 0) continue;
1840 ++bg_count;
1841 const float rv = R(y, x);
1842 const float gv = G(y, x);
1843 const float bv = B(y, x);
1844 if (!(std::isfinite(rv) && std::isfinite(gv) && std::isfinite(bv))) continue;
1845 luma_samples.push_back(0.2126f * rv + 0.7152f * gv + 0.0722f * bv);
1846 }
1847 }
1848
1849 out.bg_ref = (out.bg_r + out.bg_g + out.bg_b) / 3.0;
1850 const float dr = static_cast<float>(out.bg_ref - out.bg_r);
1851 const float dg = static_cast<float>(out.bg_ref - out.bg_g);
1852 const float db = static_cast<float>(out.bg_ref - out.bg_b);
1853 out.max_abs_shift = std::max({std::abs(dr), std::abs(dg), std::abs(db)});
1854 out.bg_luma_sigma = robust_sigma_from_samples(std::move(luma_samples));
1855 out.shift_sigma_ratio =
1856 out.max_abs_shift / std::max(out.bg_luma_sigma, 1.0e-3f);
1857 out.bg_mask_fraction =
1858 (analysis_count > 0)
1859 ? static_cast<float>(bg_count) / static_cast<float>(analysis_count)
1860 : 0.0f;
1861 out.rg_std = image::log_chroma_std_background(R, G, bg_mask);
1862 out.bg_std = image::log_chroma_std_background(B, G, bg_mask);
1863
1864 if (mode == "always") {
1865 out.apply = true;
1866 out.strength = 1.0f;
1867 out.reason = "forced_by_config";
1868 return out;
1869 }
1870
1871 const float coverage_factor =
1872 (out.bg_mask_fraction <= 0.10f)
1873 ? 0.0f
1874 : std::clamp((out.bg_mask_fraction - 0.10f) / 0.25f, 0.0f, 1.0f);
1875 const float chroma_std = std::max(
1876 std::isfinite(out.rg_std) ? out.rg_std : 0.0f,
1877 std::isfinite(out.bg_std) ? out.bg_std : 0.0f);
1878 const bool stable_low_chroma_background =
1879 std::isfinite(chroma_std) && chroma_std <= 0.0030f;
1880 const float shift_factor =
1881 stable_low_chroma_background
1882 ? 1.0f
1883 : ((out.shift_sigma_ratio <= 3.0f)
1884 ? 1.0f
1885 : std::sqrt(std::clamp(3.0f / out.shift_sigma_ratio, 0.0f, 1.0f)));
1886 const float chroma_factor =
1887 (chroma_std <= 0.015f)
1888 ? 1.0f
1889 : std::clamp(0.03f / chroma_std, 0.0f, 1.0f);
1890
1891 out.strength = coverage_factor * shift_factor * chroma_factor;
1892 if (out.strength >= 0.999f) {
1893 out.apply = true;
1894 out.strength = 1.0f;
1895 out.reason = "auto_full";
1896 } else if (out.strength >= 0.05f) {
1897 out.apply = true;
1898 out.reason = "auto_attenuated";
1899 } else {
1900 out.apply = false;
1901 out.strength = 0.0f;
1902 out.reason = "auto_suppressed_nebulous_background";
1903 }
1904 return out;
1905}
