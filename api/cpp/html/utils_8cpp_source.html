87std::vector<fs::path> discover_frames(const fs::path& input_dir, const std::string& pattern) {
88 std::vector<fs::path> frames;
89
90 if (!fs::exists(input_dir) || !fs::is_directory(input_dir)) {
91 return frames;
92 }
93
94 std::vector<std::string> patterns;
95 {
96 std::string token;
97 for (char c : pattern) {
98 if (c == ';' || c == ',') {
99 // trim token
100 size_t b = token.find_first_not_of(" \t");
101 if (b != std::string::npos) {
102 size_t e = token.find_last_not_of(" \t");
103 patterns.push_back(token.substr(b, e - b + 1));
104 }
105 token.clear();
106 } else {
107 token.push_back(c);
108 }
109 }
110 size_t b = token.find_first_not_of(" \t");
111 if (b != std::string::npos) {
112 size_t e = token.find_last_not_of(" \t");
113 patterns.push_back(token.substr(b, e - b + 1));
114 }
115 }
116 if (patterns.empty()) {
117 patterns.push_back(pattern.empty() ? "*" : pattern);
118 }
119
120 for (const auto& entry : fs::directory_iterator(input_dir)) {
121 if (entry.is_regular_file()) {
122 std::string filename = entry.path().filename().string();
123 for (const auto& pat : patterns) {
124 if (glob_match(pat, filename)) {
125 frames.push_back(entry.path());
126 break;
127 }
128 }
129 }
130 }
131
132 std::sort(frames.begin(), frames.end());
133 return frames;
134}

197std::string sha256_file(const fs::path& path) {
198 std::ifstream file(path, std::ios::binary);
199 if (!file) {
200 throw IOError("Cannot open file for SHA256: " + path.string());
201 }
202
203 std::unique_ptr<EVP_MD_CTX, decltype(&EVP_MD_CTX_free)> context(
204 EVP_MD_CTX_new(), &EVP_MD_CTX_free);
205 if (!context || EVP_DigestInit_ex(context.get(), EVP_sha256(), nullptr) != 1) {
206 throw IOError("Cannot initialize SHA256 for: " + path.string());
207 }
208
209 std::vector<char> buffer(1024u * 1024u);
210 while (file) {
211 file.read(buffer.data(), static_cast<std::streamsize>(buffer.size()));
212 const std::streamsize count = file.gcount();
213 if (count > 0 &&
214 EVP_DigestUpdate(context.get(), buffer.data(),
215 static_cast<size_t>(count)) != 1) {
216 throw IOError("Cannot update SHA256 for: " + path.string());
217 }
218 }
219 if (!file.eof()) {
220 throw IOError("Cannot read file for SHA256: " + path.string());
221 }
222
223 unsigned char hash[EVP_MAX_MD_SIZE];
224 unsigned int hash_size = 0;
225 if (EVP_DigestFinal_ex(context.get(), hash, &hash_size) != 1) {
226 throw IOError("Cannot finalize SHA256 for: " + path.string());
227 }
228 std::ostringstream oss;
229 for (unsigned int i = 0; i < hash_size; ++i) {
230 oss << std::hex << std::setfill('0') << std::setw(2)
231 << static_cast<int>(hash[i]);
232 }
233 return oss.str();
234}

355float median_of(std::vector<float> v) {
356 if (v.empty()) return 0.0f;
357 const size_t n = v.size();
358 const size_t mid = n / 2;
359 std::nth_element(v.begin(), v.begin() + static_cast<std::ptrdiff_t>(mid), v.end());
360 const float hi = v[mid];
361 if ((n % 2) == 1) return hi;
362 std::nth_element(v.begin(), v.begin() + static_cast<std::ptrdiff_t>(mid - 1), v.end());
363 const float lo = v[mid - 1];
364 return 0.5f * (lo + hi);
365}

394float stddev_of(const std::vector<float>& v) {
395 if (v.size() < 2) return 0.0f;
396 double sum = 0.0;
397 for (float x : v) sum += static_cast<double>(x);
398 const double mean = sum / static_cast<double>(v.size());
399 double var = 0.0;
400 for (float x : v) {
401 const double d = static_cast<double>(x) - mean;
402 var += d * d;
403 }
404 var /= static_cast<double>(v.size());
405 return (var > 0.0) ? static_cast<float>(std::sqrt(var)) : 0.0f;
406}

444 if (values.empty()) return 0.0f;

461float estimate_background_sigma_clip(std::vector<float> pixels) {
462 if (pixels.empty()) return 0.0f;
463 for (int iter = 0; iter < 5; ++iter) {
464 float mu = median_of(pixels);
465 float sigma = stddev_of(pixels);
466 if (!(sigma > 0.0f)) break;
467
468 std::vector<float> clipped;
469 clipped.reserve(pixels.size());
470 const float thr = 3.0f * sigma;
471 for (float x : pixels) {
472 if (std::fabs(x - mu) < thr) {
473 clipped.push_back(x);
474 }
475 }
476 if (clipped.size() == pixels.size() || clipped.empty()) break;
477 pixels.swap(clipped);
478 }
479 return median_of(pixels);
480}

487 std::vector<size_t> out;

498 if (out.empty() || out.back() != idx) {
499 out.push_back(idx);
500 }
501 }
502 return out;

537Matrix2Df downsample2x2_mean(const Matrix2Df& in) {
538 const int h = static_cast<int>(in.rows());
539 const int w = static_cast<int>(in.cols());
540 if (h <= 0 || w <= 0) return Matrix2Df(0, 0);
541 const int h2 = h - (h % 2);
542 const int w2 = w - (w % 2);
543 // `std::max(1, ...)` keeps the pre-existing shape contract every caller
544 // (global_registration.cpp's pyramid loops included) already depends on:
545 // a shrinking pyramid never collapses to a 0-sized matrix mid-loop, even
546 // for a 1-row/1-col input (h2/w2 == 0). But reading row/col `sy+1`/`sx+1`
547 // unconditionally is only valid when h2/w2 >= 2; for h2/w2 == 0 that would
548 // read one row/column past the end of `in` (h<2 means row index 1 does
549 // not exist). Clamp each of the 4 taps to the last valid row/col instead
550 // of changing the output shape --- for every h>=2,w>=2 input (everything
551 // any current caller/test actually reaches) sy+1<h and sx+1<w already, so
552 // the clamp is a no-op and the result is unchanged; only the previously
553 // out-of-bounds h<2-or-w<2 case changes, from UB to a defined "duplicate
554 // the single row/col" mean.
555 const int out_h = std::max(1, h2 / 2);
556 const int out_w = std::max(1, w2 / 2);
557 Matrix2Df out(out_h, out_w);
558 for (int y = 0; y < out_h; ++y) {
559 for (int x = 0; x < out_w; ++x) {
560 const int sy = std::min(y * 2, h - 1);
561 const int sx = std::min(x * 2, w - 1);
562 const int sy1 = std::min(sy + 1, h - 1);
563 const int sx1 = std::min(sx + 1, w - 1);
564 const float a = in(sy, sx);
565 const float b = in(sy, sx1);
566 const float c = in(sy1, sx);
567 const float d = in(sy1, sx1);
568 out(y, x) = 0.25f * (a + b + c + d);
569 }
570 }
571 return out;
572}

576StretchResult stretch_to_u16_linear_from_zero_inplace(Matrix2Df& img) {
577 StretchResult result;
578 std::vector<float> finite_positive;
579 finite_positive.reserve(static_cast<size_t>(img.size()) / 2);
580 for (Eigen::Index i = 0; i < img.size(); ++i) {
581 const float v = img.data()[i];
582 if (std::isfinite(v) && v > 0.0f)
583 finite_positive.push_back(v);
584 }
585
586 result.sample_count = finite_positive.size();
587 if (finite_positive.empty()) return result;
588
589 // Compute p1 as floor and p99.9 as robust maximum.
590 // The stretch maps [floor, robust_max] -> [0, 65535]. Using p1 (not p10
591 // or the background itself) as the floor keeps below-background pixels
592 // visible: the sky background lands at a small positive value instead of
593 // consuming >60% of the range, and only the darkest 1% of noise clips to 0.
594 const size_t idx_1 = static_cast<size_t>(
595 std::clamp(0.01, 0.0, 1.0) * static_cast<double>(finite_positive.size() - 1));
596 std::nth_element(finite_positive.begin(),
597 finite_positive.begin() + idx_1,
598 finite_positive.end());
599 const float floor_level = finite_positive[idx_1];
600
601 const size_t idx_999 = static_cast<size_t>(
602 0.999 * static_cast<double>(finite_positive.size() - 1));
603 std::nth_element(finite_positive.begin(),
604 finite_positive.begin() + idx_999,
605 finite_positive.end());
606 const float robust_max = finite_positive[idx_999];
607
608 result.low = floor_level;
609 result.high = robust_max;
610 const float range = robust_max - floor_level;
611 if (!(range > 1.0e-6f)) return result;
612
613 const float scale = 65535.0f / range;
614 for (Eigen::Index i = 0; i < img.size(); ++i) {
615 const float v = img.data()[i];
616 if (std::isfinite(v)) {
617 const float stretched = (v - floor_level) * scale;
618 img.data()[i] = std::clamp(stretched, 0.0f, 65535.0f);
619 } else {
620 img.data()[i] = 0.0f;
621 }
622 }
623
624 result.applied = true;
625 return result;
626}

640StretchResult stretch_rgb_to_u32_linear_from_zero_inplace(
641 Matrix2Df& r,
642 Matrix2Df& g,
643 Matrix2Df& b,
644 const std::vector<uint8_t>& statistics_mask) {
645 StretchResult result;
646 if (r.rows() != g.rows() || r.rows() != b.rows() ||
647 r.cols() != g.cols() || r.cols() != b.cols() ||
648 (!statistics_mask.empty() &&
649 statistics_mask.size() != static_cast<size_t>(r.size()))) {
650 return result;
651 }
652
653 // Collect all finite positive values to compute a robust maximum.
654 // Using p99.9 ensures that star cores (typically <0.1% of pixels) don't
655 // define the scale, while all diffuse structure is faithfully represented.
656 std::vector<float> all_values;
657 all_values.reserve(static_cast<size_t>(r.size()) * 3 / 4);
658 for (Matrix2Df* ch : {&r, &g, &b}) {
659 for (Eigen::Index i = 0; i < ch->size(); ++i) {
660 if (!statistics_mask.empty() &&
661 statistics_mask[static_cast<size_t>(i)] == 0) {
662 continue;
663 }
664 const float v = ch->data()[i];
665 if (std::isfinite(v) && v > 0.0f)
666 all_values.push_back(v);
667 }
668 }
669
670 result.sample_count = all_values.size();
671 if (all_values.empty()) return result;
672
673 // Compute p1 as floor and p99.9 as robust maximum.
674 // The stretch maps [floor, robust_max] -> [0, target]. Using p1 (not p10
675 // or the background itself) as the floor keeps below-background pixels
676 // visible: the sky background lands at a small positive value instead of
677 // consuming >60% of the range, and only the darkest 1% of noise clips to 0.
678 const size_t idx_1 = static_cast<size_t>(
679 std::clamp(0.01, 0.0, 1.0) * static_cast<double>(all_values.size() - 1));
680 std::nth_element(all_values.begin(), all_values.begin() + idx_1,
681 all_values.end());
682 const float floor_level = all_values[idx_1];
683
684 const size_t idx_999 = static_cast<size_t>(
685 std::clamp(0.999, 0.0, 1.0) * static_cast<double>(all_values.size() - 1));
686 std::nth_element(all_values.begin(), all_values.begin() + idx_999,
687 all_values.end());
688 const float robust_max = all_values[idx_999];
689
690 result.low = floor_level;
691 result.high = robust_max;
692 const float range = robust_max - floor_level;
693 if (!(range > 1.0e-6f)) return result;
694
695 constexpr float target = 4294967295.0f;
696 const float scale = target / range;
697 for (Matrix2Df* ch : {&r, &g, &b}) {
698 for (Eigen::Index i = 0; i < ch->size(); ++i) {
699 const float v = ch->data()[i];
700 if (std::isfinite(v)) {
701 const float stretched = (v - floor_level) * scale;
702 ch->data()[i] = std::clamp(stretched, 0.0f, target);
703 } else {
704 ch->data()[i] = 0.0f;
705 }
706 }
707 }
708
709 result.applied = true;
710 return result;
711}
