14// overlap of interval [a, b] with [0, 1]
15double overlap_unit(double a, double b) {
16 return std::max(0.0, std::min(b, 1.0) - std::max(a, 0.0));
17}
18// K(0, x0) for a top-hat of side d centred at x0, integrated over cell [0,1]
19double kernel_overlap(double x0, double d) {
20 return overlap_unit(x0 - d / 2.0, x0 + d / 2.0);
21}

24std::vector<double> kernel_noise_autocorrelation_1d(float pixfrac, int internal_scale,
25 int max_lag) {
26 if (!(pixfrac > 0.0f) || pixfrac > 1.0f || (internal_scale != 1 && internal_scale != 2) ||
27 max_lag < 0)
28 throw std::invalid_argument("KERNEL_NOISE_INVALID_ARGS");
29 const double d = static_cast<double>(pixfrac) * internal_scale;
30 // Input samples at j + 0.5; a top-hat of side d reaches at most
31 // ceil(d/2)+1 cells either way, so this range is generous.
32 const int jr = static_cast<int>(std::ceil(d)) + max_lag + 3;
33
34 auto s_at_lag = [&](int lag) {
35 double s = 0.0;
36 for (int j = -jr; j <= jr; ++j) {
37 const double x0 = j + 0.5;
38 s += kernel_overlap(x0, d) * kernel_overlap(x0 - lag, d);
39 }
40 return s;
41 };
42 const double s0 = s_at_lag(0);
43 std::vector<double> rho(static_cast<size_t>(max_lag) + 1, 0.0);
44 if (s0 <= 0.0) {
45 rho[0] = 1.0;
46 return rho;
47 }
48 for (int lag = 0; lag <= max_lag; ++lag) rho[static_cast<size_t>(lag)] = s_at_lag(lag) / s0;
49 return rho;

52double kernel_noise_correlation_sigma_factor(float pixfrac, int internal_scale) {
53 if (!(pixfrac > 0.0f) || pixfrac > 1.0f || (internal_scale != 1 && internal_scale != 2))
54 throw std::invalid_argument("KERNEL_NOISE_INVALID_ARGS");
55 const double d = static_cast<double>(pixfrac) * internal_scale;
56 const int jr = static_cast<int>(std::ceil(d)) + 4;
57 double s0 = 0.0;
58 for (int j = -jr; j <= jr; ++j) {
59 const double k = kernel_overlap(j + 0.5, d);
60 s0 += k * k;
61 }
62 // W = sum_k K(k, x0) = d for a top-hat of side d.
63 return (s0 > 0.0) ? d / std::sqrt(s0) : 1.0;
64}
