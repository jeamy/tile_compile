7CfaWarpDims compute_cfa_warp_dims(int h, int w, int out_height, int out_width) {
8 CfaWarpDims d;
9 d.h2 = h - (h % 2);
10 d.w2 = w - (w % 2);
11 d.out_h = (out_height > 0) ? out_height : h;
12 d.out_w = (out_width > 0) ? out_width : w;
13 d.out_h2 = d.out_h - (d.out_h % 2);
14 d.out_w2 = d.out_w - (d.out_w % 2);
15 d.sub_h = d.h2 / 2;
16 d.sub_w = d.w2 / 2;
17 d.out_h_sub = std::max(1, d.out_h2 / 2);
18 d.out_w_sub = std::max(1, d.out_w2 / 2);
19 return d;
20}

22CfaSubplanes extract_cfa_subplanes(const Matrix2Df& mosaic, const CfaWarpDims& dims) {

28 for (int y = 0; y < dims.sub_h; ++y) {

30 s.a(y, x) = mosaic(y * 2, x * 2);

39cv::Mat make_cfa_subplane_warp(const WarpMatrix& warp, float dx, float dy) {
