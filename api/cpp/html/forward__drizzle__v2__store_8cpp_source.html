25// Self-describing band artifact header, host byte order like the record
26// payload it precedes (trusted-run derived artifact, same-binary reader).
27struct BandHeader {
28 std::uint32_t magic = kBandMagic;
29 std::uint32_t schema_version = 1;
30 std::int32_t band_index = 0;
31 std::int32_t y_begin = 0;
32 std::int32_t rows = 0;
33 std::int32_t native_cols = 0;
34 std::int32_t channels = 0;
35 std::uint32_t record_bytes = sizeof(ForwardDrizzleV2PixelResult);
36 std::uint64_t record_count = 0;
37 std::uint64_t dense_overlap_count = 0;
38 std::uint64_t reserved0 = 0;
39 std::uint64_t reserved1 = 0;
40};
41static_assert(sizeof(BandHeader) == kBandHeaderBytes);
42static_assert(sizeof(ForwardDrizzleV2PixelResult) == 64);

177void validate_plan_fields(const ForwardDrizzleV2RunPlan &p) {
178 const bool bad =
179 p.schema_version != ForwardDrizzleV2RunPlan::kSchemaVersion ||
180 p.pipeline_contract.empty() || p.contract_version != 1 ||
181 p.source_identity_hash.empty() || p.sampling_plan_hash.empty() ||
182 p.config_snapshot_hash.empty() || p.enumeration.empty() ||
183 p.estimator.empty() || p.reservoir_size <= 0 ||
184 p.min_clip_contributors <= 0 || p.min_candidates <= 0 ||
185 p.robust_passes <= 0 || !(p.sigma_low > 0.0) || !(p.sigma_high > 0.0) ||
186 p.support_fold_contract.empty() || p.numerics.empty() ||
187 p.local_warp_representation.empty() || p.native_width <= 0 ||
188 p.native_height <= 0 || (p.channels != 1 && p.channels != 3) ||
189 (p.internal_scale != 1 && p.internal_scale != 2) ||
190 (p.channels == 1) != (p.color_mode == "MONO") ||
191 (p.channels == 3) != (p.color_mode == "OSC") || !(p.pixfrac > 0.0) ||
192 p.pixfrac > 1.0 || p.bayer_pattern < 0 || p.cfa_origin_x < 0 ||
193 p.multiband_levels < 0 || p.multiband_levels > 4 ||
194 (!p.emit_profiles && p.multiband_levels != 0) ||
195 !(p.fine_quality_exponent > 0.0f) ||
196 !(p.medium_quality_exponent > 0.0f) ||
197 p.cfa_origin_y < 0 || p.frame_count == 0 || p.band_rows <= 0 ||
198 p.band_count <= 0 || p.halo_rows < 0 ||
199 // Bands tile [0, native_height) contiguously; the last band is
200 // allowed to be short.
201 static_cast<std::int64_t>(p.band_rows) * (p.band_count - 1) >=
202 p.native_height ||
203 static_cast<std::int64_t>(p.band_rows) * p.band_count <
204 p.native_height ||
205 (p.x_tiled ? (p.tile_cols <= 0 || p.tile_cols >= p.native_width)
206 : (p.tile_cols != 0 && p.tile_cols != p.native_width));
207 if (bad) throw std::invalid_argument("FDV2_STORE_INVALID_PLAN");

282// Verifies one committed band artifact on disk: existence, size and
283// content hash against its checkpoint entry. Returns false with `error`
284// set on any failure (fail closed, never throws for store state).
285bool verify_band_artifact(const fs::path &generation,
286 const ForwardDrizzleV2BandCommit &b,
287 std::string &error) {
288 std::error_code ec;
289 const fs::path path = generation / b.artifact;
290 if (!fs::is_regular_file(fs::symlink_status(path, ec))) {
291 error = "missing band artifact " + b.artifact;
292 return false;
293 }
294 const auto size = fs::file_size(path, ec);
295 if (ec || size != b.bytes) {
296 error = "size mismatch on " + b.artifact;
297 return false;
298 }
299 if (core::sha256_file(path) != b.sha256) {
300 error = "sha256 mismatch on " + b.artifact;
301 return false;
302 }
303 if (b.profiles_artifact.empty()) return true;
304 const fs::path ppath = generation / b.profiles_artifact;
305 if (!fs::is_regular_file(fs::symlink_status(ppath, ec))) {
306 error = "missing band artifact " + b.profiles_artifact;
307 return false;
308 }
309 const auto psize = fs::file_size(ppath, ec);
310 if (ec || psize != b.profiles_bytes) {
311 error = "size mismatch on " + b.profiles_artifact;
312 return false;
313 }
314 if (core::sha256_file(ppath) != b.profiles_sha256) {
315 error = "sha256 mismatch on " + b.profiles_artifact;
316 return false;
317 }
318 return true;
319}

321// Verifies a checkpoint's band list forms a contiguous prefix whose
322// artifacts are all intact. `expected_cols` is the band width the plan
323// implies. Returns the verified prefix length via `bands` (already
324// parsed).
325bool verify_checkpoint_bands(const fs::path &generation,
326 const ForwardDrizzleV2Checkpoint &c,
327 int expected_cols, int native_height,
328 bool emit_profiles, std::string &error) {
329 int y = 0;
330 for (std::size_t i = 0; i < c.bands.size(); ++i) {
331 const auto &b = c.bands[i];
332 const bool profiles_expected =
333 emit_profiles && !b.profiles_artifact.empty() &&
334 b.profiles_bytes > 0 && !b.profiles_sha256.empty() &&
335 b.profiles_artifact ==
336 band_profiles_artifact_name(b.band_index);
337 const bool profiles_absent = b.profiles_artifact.empty() &&
338 b.profiles_bytes == 0 &&
339 b.profiles_sha256.empty();
340 if (b.band_index != static_cast<int>(i) || b.y_begin != y ||
341 b.rows <= 0 || b.native_cols != expected_cols ||
342 b.channels <= 0 || b.artifact != band_artifact_name(b.band_index) ||
343 b.sha256.empty() || y + b.rows > native_height ||
344 (emit_profiles ? !profiles_expected : !profiles_absent)) {
345 error = "non-contiguous or malformed checkpoint band " +
346 std::to_string(i);
347 return false;
348 }
349 if (!verify_band_artifact(generation, b, error)) return false;
350 y += b.rows;
351 }
352 return true;
353}

405void ForwardDrizzleV2StoreWriter::begin() {
406 if (begun_) throw std::runtime_error("FDV2_STORE_BEGIN_TWICE");
407 fs::create_directories(root_);
408 static std::atomic<std::uint64_t> sequence{0};
409 for (int attempt = 0; attempt < 64; ++attempt) {
410 auto dir =
411 root_ / (kGenerationPrefix +
412 std::to_string(std::chrono::steady_clock::now()
413 .time_since_epoch()
414 .count()) +
415 "-" + std::to_string(sequence++));
416 if (fs::create_directory(dir)) {
417 generation_ = dir;
418 break;
419 }
420 }
421 if (generation_.empty())
422 throw std::runtime_error("FDV2_STORE_GENERATION_FAILED");
423 core::write_text_atomic(generation_ / "plan.json",
424 serialize_forward_drizzle_v2_plan(plan_));
425 ForwardDrizzleV2Checkpoint checkpoint;
426 checkpoint.plan_hash = plan_.plan_hash;
427 checkpoint.band_count = plan_.band_count;
428 write_checkpoint(generation_, checkpoint);
429 begun_ = true;
430}

432void ForwardDrizzleV2StoreWriter::adopt(
433 const fs::path &generation, int resume_from_band,
434 std::vector<ForwardDrizzleV2BandCommit> committed) {
435 if (begun_) throw std::runtime_error("FDV2_STORE_BEGIN_TWICE");
436 if (resume_from_band != static_cast<int>(committed.size()) ||
437 resume_from_band > plan_.band_count)
438 throw std::invalid_argument("FDV2_STORE_ADOPT_MISMATCH");
439 std::error_code ec;
440 if (!fs::is_directory(fs::symlink_status(generation, ec)))
441 throw std::runtime_error("FDV2_STORE_ADOPT_MISSING_GENERATION");
442 // The adopted generation must provably belong to this plan even when the
443 // caller fabricated the inspection state.
444 ForwardDrizzleV2RunPlan stored;
445 std::string parse_error;
446 if (!parse_forward_drizzle_v2_plan(
447 read_small_json(generation / "plan.json").dump(), stored,
448 parse_error) ||
449 stored.plan_hash != plan_.plan_hash ||
450 stored.plan_hash != digest(plan_to_json(stored, false)))
451 throw std::runtime_error("FDV2_STORE_ADOPT_MISMATCH");
452 int y = 0;
453 for (std::size_t i = 0; i < committed.size(); ++i) {
454 const auto &b = committed[i];
455 if (b.band_index != static_cast<int>(i) || b.y_begin != y || b.rows <= 0)
456 throw std::invalid_argument("FDV2_STORE_ADOPT_MISMATCH");
457 if (plan_.emit_profiles !=
458 (!b.profiles_artifact.empty() && b.profiles_bytes > 0 &&
459 !b.profiles_sha256.empty()))
460 throw std::invalid_argument("FDV2_STORE_ADOPT_MISMATCH");
461 y += b.rows;
462 }
463 if (y > plan_.native_height)
464 throw std::invalid_argument("FDV2_STORE_ADOPT_MISMATCH");
465 generation_ = generation;
466 bands_ = std::move(committed);
467 next_y_ = y;
468 begun_ = true;
469}

471void ForwardDrizzleV2StoreWriter::commit_band(
472 int band_index, int y_begin, int rows,
473 std::span<const ForwardDrizzleV2PixelResult> results,
474 std::span<const ForwardDrizzleV2ProfileResult> profiles,
475 std::uint64_t dense_overlap_count) {
476 if (!begun_ || publish_attempted_)
477 throw std::runtime_error("FDV2_STORE_BAND_ORDER");
478 const int expected_cols = plan_.x_tiled ? plan_.tile_cols : plan_.native_width;
479 if (band_index != static_cast<int>(bands_.size()) ||
480 band_index >= plan_.band_count || y_begin != next_y_ || rows <= 0 ||
481 y_begin + rows > plan_.native_height)
482 throw std::runtime_error("FDV2_STORE_BAND_ORDER");
483 const std::uint64_t record_count =
484 static_cast<std::uint64_t>(rows) * expected_cols * plan_.channels;
485 if (results.size() != record_count ||
486 record_count > (std::numeric_limits<std::uint64_t>::max() -
487 kBandHeaderBytes) /
488 sizeof(ForwardDrizzleV2PixelResult))
489 throw std::runtime_error("FDV2_STORE_INVALID_BAND");
490 if (plan_.emit_profiles != (profiles.size() == record_count))
491 throw std::runtime_error("FDV2_STORE_INVALID_BAND");
492
493 const auto name = band_artifact_name(band_index);
494 core::AtomicOutput output(generation_ / name);
495 {
496 BandHeader header;
497 header.band_index = band_index;
498 header.y_begin = y_begin;
499 header.rows = rows;
500 header.native_cols = expected_cols;
501 header.channels = plan_.channels;
502 header.record_count = record_count;
503 header.dense_overlap_count = dense_overlap_count;
504 std::ofstream file(output.path(), std::ios::binary | std::ios::trunc);
505 file.write(reinterpret_cast<const char *>(&header), sizeof(header));
506 file.write(reinterpret_cast<const char *>(results.data()),
507 static_cast<std::streamsize>(record_count *
508 sizeof(results[0])));
509 file.flush();
510 if (!file) throw std::runtime_error("FDV2_STORE_BAND_WRITE_FAILED");
511 }
512 output.commit(); // fsync + rename + dir fsync: artifact durable first
513
514 ForwardDrizzleV2BandCommit commit;
515 commit.band_index = band_index;
516 commit.y_begin = y_begin;
517 commit.rows = rows;
518 commit.native_cols = expected_cols;
519 commit.channels = plan_.channels;
520 commit.dense_overlap_count = dense_overlap_count;
521 commit.artifact = name;
522 commit.bytes = kBandHeaderBytes +
523 record_count * sizeof(ForwardDrizzleV2PixelResult);
524 commit.sha256 = core::sha256_file(generation_ / name);
525 if (plan_.emit_profiles) {
526 const auto pname = band_profiles_artifact_name(band_index);
527 core::AtomicOutput poutput(generation_ / pname);
528 {
529 BandHeader header;
530 header.band_index = band_index;
531 header.y_begin = y_begin;
532 header.rows = rows;
533 header.native_cols = expected_cols;
534 header.channels = plan_.channels;
535 header.record_bytes = sizeof(ForwardDrizzleV2ProfileResult);
536 header.record_count = record_count;
537 header.dense_overlap_count = dense_overlap_count;
538 std::ofstream file(poutput.path(), std::ios::binary | std::ios::trunc);
539 file.write(reinterpret_cast<const char *>(&header), sizeof(header));
540 file.write(reinterpret_cast<const char *>(profiles.data()),
541 static_cast<std::streamsize>(
542 record_count * sizeof(profiles[0])));
543 file.flush();
544 if (!file) throw std::runtime_error("FDV2_STORE_BAND_WRITE_FAILED");
545 }
546 poutput.commit();
547 commit.profiles_artifact = pname;
548 commit.profiles_bytes =
549 kBandHeaderBytes +
550 record_count * sizeof(ForwardDrizzleV2ProfileResult);
551 commit.profiles_sha256 =
552 core::sha256_file(generation_ / pname);
553 }
554 bands_.push_back(std::move(commit));
555 next_y_ += rows;
556
557 // The commit mark is the last durable step of the band transaction.
558 ForwardDrizzleV2Checkpoint checkpoint;
559 checkpoint.plan_hash = plan_.plan_hash;
560 checkpoint.band_count = plan_.band_count;
561 checkpoint.bands = bands_;
562 write_checkpoint(generation_, checkpoint);
563}

565fs::path ForwardDrizzleV2StoreWriter::finish(
566 const ForwardDrizzleV2CommitGate &gate) {
567 if (!begun_ || publish_attempted_)
568 throw std::runtime_error("FDV2_STORE_FINISH_ORDER");
569 if (static_cast<int>(bands_.size()) != plan_.band_count ||
570 next_y_ != plan_.native_height)
571 throw std::runtime_error("FDV2_STORE_INCOMPLETE_BANDS");
572 if (gate.nonfinite_pixels_inside_source_support != 0 ||
573 gate.bands_processed != static_cast<std::uint64_t>(plan_.band_count))
574 throw std::runtime_error("FDV2_STORE_COMMIT_GATE_FAILED");
575 std::string error;
576 for (const auto &b : bands_)
577 if (!verify_band_artifact(generation_, b, error))
578 throw std::runtime_error("FDV2_STORE_CORRUPT_PLANES: " + error);
579
580 ForwardDrizzleV2Checkpoint checkpoint;
581 checkpoint.plan_hash = plan_.plan_hash;
582 checkpoint.band_count = plan_.band_count;
583 checkpoint.bands = bands_;
584 write_checkpoint(generation_, checkpoint);
585
586 json telemetry = json::object();
587 if (!gate.telemetry_json.empty())
588 telemetry = json::parse(gate.telemetry_json);
589 json commit = {
590 {"schema_version", 1},
591 {"generation", generation_.filename().string()},
592 {"plan_hash", plan_.plan_hash},
593 {"checkpoint_hash", checkpoint.checkpoint_hash},
594 {"band_count", plan_.band_count},
595 {"commit_gate",
596 {{"nonfinite_pixels_inside_source_support",
597 gate.nonfinite_pixels_inside_source_support},
598 {"bands_processed", gate.bands_processed}}},
599 {"telemetry", telemetry}};
600 commit["commit_hash"] = digest(commit);
601 core::write_text_atomic(generation_ / "commit.json", commit.dump(2));
602
603 // If the rename succeeded but the directory sync failed, current.json
604 // may already name this generation: never delete it afterwards.
605 publish_attempted_ = true;
606 core::write_text_atomic(root_ / "current.json", commit.dump(2));
607 return generation_;
608}

610bool load_forward_drizzle_v2_published_plan(
611 const fs::path &root, ForwardDrizzleV2RunPlan &out, std::string &error) {
612 error.clear();
613 std::error_code ec;
614 const fs::path current_path = root / "current.json";
615 if (!fs::is_regular_file(fs::symlink_status(current_path, ec)))
616 return false;
617 try {
618 const json current = read_small_json(current_path);
619 const auto generation_name = current.value("generation", std::string{});
620 if (generation_name.empty())
621 throw std::runtime_error("malformed current.json");
622 const fs::path generation = root / generation_name;
623 std::string parse_error;
624 if (!parse_forward_drizzle_v2_plan(
625 read_small_json(generation / "plan.json").dump(), out,
626 parse_error))
627 throw std::runtime_error("unparseable plan.json: " + parse_error);
628 } catch (const std::exception &e) {
629 throw std::runtime_error(std::string("FDV2_STORE_PLAN_LOAD: ") + e.what());
630 }
631 // The parse round-trip is structural only; the identity is verified by a
632 // following inspect() call (plan_hash recompute + commit chain).
633 ForwardDrizzleV2RunPlan reparsed;
634 std::string re_error;
635 if (!parse_forward_drizzle_v2_plan(
636 serialize_forward_drizzle_v2_plan(out), reparsed, re_error) ||
637 reparsed.plan_hash != out.plan_hash)
638 throw std::runtime_error("FDV2_STORE_PLAN_LOAD_ROUNDTRIP");
639 return true;
640}

642std::vector<ForwardDrizzleV2PixelResult> read_forward_drizzle_v2_band(
643 const fs::path &generation, const ForwardDrizzleV2BandCommit &commit) {
644 const fs::path path = generation / commit.artifact;
645 std::error_code ec;
646 if (!fs::is_regular_file(fs::symlink_status(path, ec)) ||
647 fs::file_size(path, ec) != commit.bytes)
648 throw std::runtime_error("FDV2_STORE_BAND_MISSING_OR_SIZED");
649 if (core::sha256_file(path) != commit.sha256)
650 throw std::runtime_error("FDV2_STORE_BAND_HASH_MISMATCH");
651 std::ifstream file(path, std::ios::binary);
652 BandHeader header;
653 file.read(reinterpret_cast<char *>(&header), sizeof(header));
654 if (!file || header.magic != kBandMagic ||
655 header.schema_version != 1 || header.band_index != commit.band_index ||
656 header.y_begin != commit.y_begin || header.rows != commit.rows ||
657 header.native_cols != commit.native_cols ||
658 header.channels != commit.channels ||
659 header.record_bytes != sizeof(ForwardDrizzleV2PixelResult) ||
660 header.record_count !=
661 static_cast<std::uint64_t>(commit.rows) * commit.native_cols *
662 commit.channels ||
663 header.dense_overlap_count != commit.dense_overlap_count)
664 throw std::runtime_error("FDV2_STORE_BAND_HEADER_MISMATCH");
665 std::vector<ForwardDrizzleV2PixelResult> records(header.record_count);
666 file.read(reinterpret_cast<char *>(records.data()),
667 static_cast<std::streamsize>(header.record_count *
668 sizeof(records[0])));
669 if (!file) throw std::runtime_error("FDV2_STORE_BAND_READ_FAILED");
670 return records;
671}

706ForwardDrizzleV2StoreInspection inspect_forward_drizzle_v2_store(
707 const fs::path &root, const ForwardDrizzleV2RunPlan &expected_plan) {
708 if (expected_plan.plan_hash.empty())
709 throw std::invalid_argument("FDV2_STORE_PLAN_NOT_FINALIZED");
710 ForwardDrizzleV2StoreInspection out;
711 const auto fail = [&out](std::string error) {
712 out.status = ForwardDrizzleV2StoreStatus::corrupt;
713 out.error = std::move(error);
714 return out;
715 };
716 std::error_code ec;
717 if (!fs::is_directory(fs::symlink_status(root, ec))) return out; // fresh
718
719 const int expected_cols = expected_plan.x_tiled ? expected_plan.tile_cols
720 : expected_plan.native_width;
721 const fs::path current_path = root / "current.json";
722 if (fs::is_regular_file(fs::symlink_status(current_path, ec))) {
723 json current;
724 try {
725 current = read_small_json(current_path);
726 } catch (const std::exception &e) {
727 return fail(std::string("unreadable current.json: ") + e.what());
728 }
729 const auto generation_name =
730 current.value("generation", std::string{});
731 const auto plan_hash = current.value("plan_hash", std::string{});
732 const auto commit_hash = current.value("commit_hash", std::string{});
733 if (current.value("schema_version", 0) != 1 || generation_name.empty() ||
734 commit_hash.empty())
735 return fail("malformed current.json");
736 if (plan_hash != expected_plan.plan_hash)
737 return fail("current.json plan_hash mismatch");
738 const fs::path generation = root / generation_name;
739 if (!fs::is_directory(fs::symlink_status(generation, ec)))
740 return fail("current.json points at missing generation");
741 json committed;
742 try {
743 committed = read_small_json(generation / "commit.json");
744 } catch (const std::exception &e) {
745 return fail(std::string("unreadable commit.json: ") + e.what());
746 }
747 committed.erase("commit_hash");
748 if (digest(committed) != commit_hash ||
749 committed.value("plan_hash", std::string{}) != plan_hash)
750 return fail("commit.json hash mismatch");
751 // The final checkpoint.json is hash-chained into commit.json via
752 // checkpoint_hash: parse and bind it so a complete-store consumer gets
753 // the same verified band list a resumable inspection exposes. Every
754 // band read still re-checks its sha256 against the commit entry.
755 ForwardDrizzleV2Checkpoint final_checkpoint;
756 std::string fc_parse_error;
757 try {
758 if (!parse_forward_drizzle_v2_checkpoint(
759 read_small_json(generation / "checkpoint.json").dump(),
760 final_checkpoint, fc_parse_error))
761 return fail("unparseable final checkpoint.json: " + fc_parse_error);
762 } catch (const std::exception &e) {
763 return fail(std::string("unreadable final checkpoint.json: ") +
764 e.what());
765 }
766 if (final_checkpoint.schema_version !=
767 ForwardDrizzleV2Checkpoint::kSchemaVersion ||
768 digest(checkpoint_to_json(final_checkpoint, false)) !=
769 final_checkpoint.checkpoint_hash ||
770 final_checkpoint.plan_hash != expected_plan.plan_hash ||
771 final_checkpoint.band_count != expected_plan.band_count ||
772 final_checkpoint.checkpoint_hash !=
773 committed.value("checkpoint_hash", std::string{}))
774 return fail("final checkpoint.json binding mismatch");
775 out.status = ForwardDrizzleV2StoreStatus::complete;
776 out.generation = generation;
777 out.commit_hash = commit_hash;
778 out.committed = std::move(final_checkpoint.bands);
779 return out;
780 }
781
782 std::vector<fs::path> generations;
783 for (const auto &entry : fs::directory_iterator(root, ec)) {
784 if (ec) break;
785 if (entry.is_directory(ec) &&
786 entry.path().filename().string().rfind(kGenerationPrefix, 0) == 0)
787 generations.push_back(entry.path());
788 }
789 if (generations.empty()) return out; // fresh
790 if (generations.size() > 1)
791 return fail("ambiguous: multiple unpublished generations");
792 const fs::path generation = generations.front();
793
794 ForwardDrizzleV2RunPlan stored_plan;
795 std::string parse_error;
796 try {
797 if (!parse_forward_drizzle_v2_plan(
798 read_small_json(generation / "plan.json").dump(), stored_plan,
799 parse_error))
800 return fail("unparseable plan.json: " + parse_error);
801 } catch (const std::exception &e) {
802 return fail(std::string("unreadable plan.json: ") + e.what());
803 }
804 if (stored_plan.schema_version != ForwardDrizzleV2RunPlan::kSchemaVersion)
805 return fail("plan schema_version mismatch");
806 const std::string recomputed = digest(plan_to_json(stored_plan, false));
807 if (stored_plan.plan_hash != recomputed)
808 return fail("plan.json hash mismatch (tampered)");
809 if (stored_plan.plan_hash != expected_plan.plan_hash)
810 return fail("plan context mismatch");
811
812 ForwardDrizzleV2Checkpoint checkpoint;
813 try {
814 if (!parse_forward_drizzle_v2_checkpoint(
815 read_small_json(generation / "checkpoint.json").dump(), checkpoint,
816 parse_error))
817 return fail("unparseable checkpoint.json: " + parse_error);
818 } catch (const std::exception &e) {
819 return fail(std::string("unreadable checkpoint.json: ") + e.what());
820 }
821 if (checkpoint.schema_version != ForwardDrizzleV2Checkpoint::kSchemaVersion)
822 return fail("checkpoint schema_version mismatch");
823 if (digest(checkpoint_to_json(checkpoint, false)) !=
824 checkpoint.checkpoint_hash)
825 return fail("checkpoint hash mismatch (tampered)");
826 if (checkpoint.plan_hash != expected_plan.plan_hash ||
827 checkpoint.band_count != expected_plan.band_count)
828 return fail("checkpoint context mismatch");
829
830 std::string verify_error;
831 if (!verify_checkpoint_bands(generation, checkpoint, expected_cols,
832 expected_plan.native_height,
833 expected_plan.emit_profiles, verify_error))
834 return fail(verify_error);
835
836 out.status = ForwardDrizzleV2StoreStatus::resumable;
837 out.next_band = static_cast<int>(checkpoint.bands.size());
838 out.generation = generation;
839 out.committed = std::move(checkpoint.bands);
840 return out;
841}
