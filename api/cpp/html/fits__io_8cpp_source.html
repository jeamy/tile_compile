71 auto try_read_znaxis = [&](int& out_naxis, long out_naxes[3]) -> bool {
72 int st = 0;
73 long znaxis = 0;
74 long znaxis1 = 0;
75 long znaxis2 = 0;
76 fits_read_key(fptr, TLONG, const_cast<char*>("ZNAXIS"), &znaxis, nullptr, &st);
77 if (st || znaxis < 2) {
78 return false;
79 }
80 st = 0;
81 fits_read_key(fptr, TLONG, const_cast<char*>("ZNAXIS1"), &znaxis1, nullptr, &st);
82 if (st || znaxis1 <= 0) {
83 return false;
84 }
85 st = 0;
86 fits_read_key(fptr, TLONG, const_cast<char*>("ZNAXIS2"), &znaxis2, nullptr, &st);
87 if (st || znaxis2 <= 0) {
88 return false;
89 }
90
91 out_naxis = static_cast<int>(znaxis);
92 out_naxes[0] = znaxis1;
93 out_naxes[1] = znaxis2;
94 out_naxes[2] = 0;
95 return true;
96 };
97
98 status = 0;
99 fits_get_img_param(fptr, 3, &bitpix, &naxis, naxes, &status);
100 if (status == 0 && has_valid_image_shape(naxis, naxes)) {
101 return true;
102 }
103 if (try_read_znaxis(naxis, naxes)) {
104 status = 0;
105 return true;
106 }
107
108 status = 0;
109 int nhdus = 0;
110 fits_get_num_hdus(fptr, &nhdus, &status);
111 if (status) {
112 return false;
113 }
114
115 for (int hdu = 2; hdu <= nhdus; ++hdu) {
116 int hdu_type = 0;
117 status = 0;
118 fits_movabs_hdu(fptr, hdu, &hdu_type, &status);
119 if (status) {
120 continue;
121 }
122
123 status = 0;
124 fits_get_img_param(fptr, 3, &bitpix, &naxis, naxes, &status);
125 if (status == 0 && has_valid_image_shape(naxis, naxes)) {
126 return true;
127 }
128 if (try_read_znaxis(naxis, naxes)) {
129 status = 0;
130 return true;
131 }
132 }
133
134 return false;
135}

141FitsHeader read_current_header(fitsfile* fptr, int& status) {
142 FitsHeader header;
143
144 char card[FLEN_CARD];
145 int nkeys = 0;
146 fits_get_hdrspace(fptr, &nkeys, nullptr, &status);
147
148 for (int i = 1; i <= nkeys; ++i) {
149 fits_read_record(fptr, i, card, &status);
150 if (status) {
151 status = 0;
152 continue;
153 }
154
155 char keyname[FLEN_KEYWORD];
156 char value[FLEN_VALUE];
157 char comment[FLEN_COMMENT];
158 int keylen = 0;
159
160 fits_get_keyname(card, keyname, &keylen, &status);
161 if (status) {
162 status = 0;
163 continue;
164 }
165
166 const std::string key(keyname);
167 if (key.empty() || key == "COMMENT" || key == "HISTORY" || key == "END") {
168 continue;
169 }
170
171 char dtype;
172 fits_get_keytype(card, &dtype, &status);
173 if (status) {
174 status = 0;
175 continue;
176 }
177
178 fits_parse_value(card, value, comment, &status);
179 if (status) {
180 status = 0;
181 continue;
182 }
183
184 std::string val_str(value);
185 val_str.erase(0, val_str.find_first_not_of(" '"));
186 val_str.erase(val_str.find_last_not_of(" '") + 1);
187
188 switch (dtype) {
189 case 'C':
190 header.set(key, val_str);
191 break;
192 case 'L':
193 header.set(key, val_str == "T" || val_str == "1");
194 break;
195 case 'I':
196 try {
197 header.set(key, std::stoi(val_str));
198 } catch (...) {
199 header.set(key, val_str);
200 }
201 break;
202 case 'F':
203 try {
204 header.set(key, std::stod(val_str));
205 } catch (...) {
206 header.set(key, val_str);
207 }
208 break;
209 default:
210 header.set(key, val_str);
211 break;
212 }
213 }
214
215 status = 0;
216 return header;
217}

239void write_header_keywords(fitsfile* fptr, const FitsHeader& header,
240 bool is_rgb_image, int& status) {
241 for (const auto& [key, value] : header.string_values) {
242 if (key.size() <= 8 && !should_skip_header_key(key, is_rgb_image)) {
243 fits_update_key(fptr, TSTRING, key.c_str(),
244 const_cast<char*>(value.c_str()), nullptr, &status);
245 if (status) status = 0;
246 }
247 }
248
249 for (const auto& [key, value] : header.numeric_values) {
250 if (key.size() <= 8 && !should_skip_header_key(key, is_rgb_image)) {
251 double val = value;
252 fits_update_key(fptr, TDOUBLE, key.c_str(), &val, nullptr, &status);
253 if (status) status = 0;
254 }
255 }
256
257 for (const auto& [key, value] : header.int_values) {
258 if (key.size() <= 8 && !should_skip_header_key(key, is_rgb_image)) {
259 int val = value;
260 fits_update_key(fptr, TINT, key.c_str(), &val, nullptr, &status);
261 if (status) status = 0;
262 }
263 }
264
265 for (const auto& [key, value] : header.bool_values) {
266 if (key.size() <= 8 && !should_skip_header_key(key, is_rgb_image)) {
267 int val = value ? 1 : 0;
268 fits_update_key(fptr, TLOGICAL, key.c_str(), &val, nullptr, &status);
269 if (status) status = 0;
270 }
271 }

561Matrix2Df read_fits_region_float(const fs::path& path, int x0, int y0, int width, int height) {
562 fitsfile* fptr = nullptr;
563 int status = 0;
564
565 if (fits_open_file(&fptr, path.string().c_str(), READONLY, &status)) {
566 throw FitsError("Cannot open FITS file: " + path.string());
567 }
568
569 int naxis = 0;
570 long naxes[3] = {0, 0, 0};
571 int bitpix = 0;
572
573 if (!move_to_first_image_hdu(fptr, bitpix, naxis, naxes, status)) {
574 fits_close_file(fptr, &status);
575 throw FitsError("Cannot read FITS image parameters: " + path.string());
576 }
577
578 int img_w = static_cast<int>(naxes[0]);
579 int img_h = static_cast<int>(naxes[1]);
580
581 int rx0 = std::max(0, x0);
582 int ry0 = std::max(0, y0);
583 int rx1 = std::min(img_w, x0 + width);
584 int ry1 = std::min(img_h, y0 + height);
585
586 int rw = std::max(0, rx1 - rx0);
587 int rh = std::max(0, ry1 - ry0);
588
589 if (rw <= 0 || rh <= 0) {
590 fits_close_file(fptr, &status);
591 return Matrix2Df();
592 }
593
594 Matrix2Df data(rh, rw);
595
596 if (naxis >= 3) {
597 // RGB cube: read ROI from first plane to keep scalar ROI API contract.
598 long fpixel[3] = {static_cast<long>(rx0 + 1), static_cast<long>(ry0 + 1), 1};
599 long lpixel[3] = {static_cast<long>(rx0 + rw), static_cast<long>(ry0 + rh), 1};
600 long inc[3] = {1, 1, 1};
601 fits_read_subset(fptr, TFLOAT, fpixel, lpixel, inc, nullptr,
602 data.data(), nullptr, &status);
603 } else {
604 long fpixel[2] = {static_cast<long>(rx0 + 1), static_cast<long>(ry0 + 1)};
605 long lpixel[2] = {static_cast<long>(rx0 + rw), static_cast<long>(ry0 + rh)};
606 long inc[2] = {1, 1};
607 fits_read_subset(fptr, TFLOAT, fpixel, lpixel, inc, nullptr,
608 data.data(), nullptr, &status);
609 }
610 fits_close_file(fptr, &status);
611
612 if (status) {
613 throw FitsError("Cannot read FITS ROI pixel data: " + path.string() +
614 " (cfitsio_status=" + std::to_string(status) +
615 ", reason=\"" + cfitsio_status_text(status) + "\")");
616 }
617
618 return data;
619}

764BayerPattern detect_bayer_pattern(const FitsHeader& header) {
765 auto bayerpat = header.get_string("BAYERPAT");
766 if (bayerpat) {
767 return string_to_bayer_pattern(*bayerpat);
768 }
769
770 auto colortyp = header.get_string("COLORTYP");
771 if (colortyp) {
772 std::string ct = core::to_lower(*colortyp);
773 if (ct.find("rggb") != std::string::npos) return BayerPattern::RGGB;
774 if (ct.find("bggr") != std::string::npos) return BayerPattern::BGGR;
775 if (ct.find("grbg") != std::string::npos) return BayerPattern::GRBG;
776 if (ct.find("gbrg") != std::string::npos) return BayerPattern::GBRG;
777 }
778

797ColorMode detect_color_mode(const FitsHeader& header, int naxis) {
