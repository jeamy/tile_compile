199bool warp_is_identity(const WarpMatrix &warp) {
200 const float eps = 1.0e-6f;
201 return std::fabs(warp(0, 0) - 1.0f) < eps &&
202 std::fabs(warp(0, 1)) < eps && std::fabs(warp(1, 0)) < eps &&
203 std::fabs(warp(1, 1) - 1.0f) < eps && std::fabs(warp(0, 2)) < eps &&
204 std::fabs(warp(1, 2)) < eps;
205}

225void write_valid_outputs_from_mask(const cv::Mat &mask,
226 std::vector<uint8_t> *valid_mask_out,
227 bool *has_data_out) {
228 if (valid_mask_out == nullptr && has_data_out == nullptr) {
229 return;
230 }
231
232 const int rows = std::max(0, mask.rows);
233 const int cols = std::max(0, mask.cols);
234 const size_t pixel_count =
235 static_cast<size_t>(rows) * static_cast<size_t>(cols);
236 if (valid_mask_out != nullptr) {
237 valid_mask_out->assign(pixel_count, 0u);
238 }
239
240 bool has_data = false;
241 for (int y = 0; y < rows; ++y) {
242 const uchar *row = mask.ptr<uchar>(y);
243 const size_t row_off = static_cast<size_t>(y) * static_cast<size_t>(cols);
244 for (int x = 0; x < cols; ++x) {
245 if (row[x] == 0) {
246 continue;
247 }
248 has_data = true;
249 if (valid_mask_out != nullptr) {
250 (*valid_mask_out)[row_off + static_cast<size_t>(x)] = 1u;
251 }
252 }
253 }
254
255 if (has_data_out != nullptr) {
256 *has_data_out = has_data;
257 }
258}

260void invalidate_matrix_outside_support(Matrix2Df &matrix,
261 const cv::Mat &support_mask) {
262 if (matrix.rows() != support_mask.rows ||
263 matrix.cols() != support_mask.cols || support_mask.type() != CV_8U) {
264 return;
265 }
266 const float invalid = std::numeric_limits<float>::quiet_NaN();
267 for (int y = 0; y < support_mask.rows; ++y) {

269 for (int x = 0; x < support_mask.cols; ++x) {

281bool build_warped_support_mask(const cv::Mat &warp_matrix, int src_rows,
282 int src_cols, cv::Size output_size,
283 cv::Mat &support_mask) {
284 if (src_rows <= 0 || src_cols <= 0 || output_size.width <= 0 ||
285 output_size.height <= 0) {
286 support_mask.release();
287 return false;
288 }
289
290 cv::Mat ones(src_rows, src_cols, CV_32F, cv::Scalar(1.0f));
291 cv::Mat warped_support;
292 cv::warpAffine(ones, warped_support, warp_matrix, output_size,
293 cv::INTER_NEAREST | cv::WARP_INVERSE_MAP,
294 cv::BORDER_CONSTANT, cv::Scalar(0.0f));
295 cv::compare(warped_support, 0.5f, support_mask, cv::CMP_GT);
296 return !support_mask.empty();
297}

321bool cuda_warp_affine_impl(const cv::Mat &src, const cv::Mat &warp_matrix,
322 cv::Size output_size, cv::Mat &dst,
323 int interpolation_flag, cv::cuda::Stream *stream) {
324 try {
325 auto &buffers = cuda_warp_buffers();
326 cv::cuda::GpuMat &d_src = buffers.src[0];
327 cv::cuda::GpuMat &d_dst = buffers.dst[0];
328 cv::cuda::Stream &cuda_stream =
329 stream ? *stream : cv::cuda::Stream::Null();
330 d_src.upload(src, cuda_stream);
331 cv::cuda::warpAffine(d_src, d_dst, warp_matrix, output_size,
332 interpolation_flag | cv::WARP_INVERSE_MAP,
333 cv::BORDER_CONSTANT, cv::Scalar(0), cuda_stream);
334 d_dst.download(dst, cuda_stream);
335 cuda_stream.waitForCompletion();
336 return !dst.empty();
337 } catch (...) {
338 return false;
339 }

382bool cuda_warp_cfa_mosaic(const Matrix2Df &mosaic, const WarpMatrix &warp,
383 int out_height, int out_width, Matrix2Df &out,
384 int interpolation_flag, cv::cuda::Stream *stream) {
385 const int h = static_cast<int>(mosaic.rows());
386 const int w = static_cast<int>(mosaic.cols());
387 const auto dims = compute_cfa_warp_dims(h, w, out_height, out_width);
388 auto sub = extract_cfa_subplanes(mosaic, dims);
389 const auto warps = make_all_cfa_subplane_warps(warp);
390
391 cv::Mat a_cv(dims.sub_h, dims.sub_w, CV_32F, sub.a.data());
392 cv::Mat b_cv(dims.sub_h, dims.sub_w, CV_32F, sub.b.data());
393 cv::Mat c_cv(dims.sub_h, dims.sub_w, CV_32F, sub.c.data());
394 cv::Mat d_cv(dims.sub_h, dims.sub_w, CV_32F, sub.d.data());
395 cv::Mat a_w, b_w, c_w, d_w;
396 const cv::Size out_size(dims.out_w_sub, dims.out_h_sub);
397 if (stream) {
398 // Keep all four CFA planes in one stream and synchronize once. Calling the
399 // single-plane wrapper here would force four upload/warp/download barriers.
400 try {
401 auto &buffers = cuda_warp_buffers();
402 std::array<cv::cuda::GpuMat, 4> &d_src = buffers.src;
403 std::array<cv::cuda::GpuMat, 4> &d_dst = buffers.dst;
404 const std::array<cv::Mat, 4> src = {a_cv, b_cv, c_cv, d_cv};
405 std::array<cv::Mat *, 4> dst = {&a_w, &b_w, &c_w, &d_w};
406 const std::array<cv::Mat, 4> warp_arr = {warps.a, warps.b, warps.c, warps.d};
407 for (size_t i = 0; i < src.size(); ++i) {
408 d_src[i].upload(src[i], *stream);
409 cv::cuda::warpAffine(d_src[i], d_dst[i], warp_arr[i], out_size,
410 interpolation_flag | cv::WARP_INVERSE_MAP,
411 cv::BORDER_CONSTANT, cv::Scalar(0), *stream);
412 d_dst[i].download(*dst[i], *stream);
413 }
414 stream->waitForCompletion();
415 } catch (...) {
416 return false;
417 }
418 } else if (!cuda_warp_affine_impl(a_cv, warps.a, out_size, a_w,
419 interpolation_flag, nullptr) ||
420 !cuda_warp_affine_impl(b_cv, warps.b, out_size, b_w,
421 interpolation_flag, nullptr) ||
422 !cuda_warp_affine_impl(c_cv, warps.c, out_size, c_w,
423 interpolation_flag, nullptr) ||
424 !cuda_warp_affine_impl(d_cv, warps.d, out_size, d_w,
425 interpolation_flag, nullptr)) {
426 return false;
427 }
428
429 out = reassemble_cfa_subplanes(a_w, b_w, c_w, d_w, dims);
430 return out.size() > 0;
431}
432#endif

522AccelerationBackend choose_auto_backend(AccelerationPhase phase,
523 bool tile_compile_with_cuda,
524 bool opencv_cuda_runtime,
525 bool opencv_opencl_runtime) {
526 const bool opencv_cuda_headers = opencv_cuda_headers_available(phase);
527 const bool opencv_opencl_headers = TILE_COMPILE_HAS_OPENCV_OPENCL != 0;
528 const AccelerationBackend candidates[] = {
529 AccelerationBackend::cuda,
530 AccelerationBackend::opencv_cuda,
531 AccelerationBackend::opencv_opencl,
532 AccelerationBackend::cpu,
533 };
534 for (AccelerationBackend candidate : candidates) {
535 if (!phase_supports_backend(phase, candidate)) {
536 continue;
537 }
538 if (missing_backend_reason(candidate, tile_compile_with_cuda,
539 opencv_cuda_headers,
540 opencv_cuda_runtime,
541 opencv_opencl_headers,
542 opencv_opencl_runtime)
543 .empty()) {
544 return candidate;
545 }
546 }
547 return AccelerationBackend::cpu;
548}

550// Keep stateless selection and run-scoped selection on one code path. The
551// caller supplies either freshly probed or context-snapshotted capabilities.
552AccelerationSelection select_with_capabilities(
553 const std::string &requested_backend_name, AccelerationPhase phase,
554 const AccelerationCapabilities &capabilities) {
555 AccelerationSelection selection;
556 selection.phase = phase;
557 selection.tile_compile_with_cuda = capabilities.tile_compile_with_cuda;
558 selection.opencv_cuda_headers = opencv_cuda_headers_available(phase);
559 selection.opencv_cuda_runtime = capabilities.opencv_cuda_runtime;
560 selection.opencv_opencl_headers = capabilities.opencv_opencl_headers;
561 selection.opencv_opencl_runtime = capabilities.opencv_opencl_runtime;
562 selection.requested_name = core::to_lower(requested_backend_name);
563 if (selection.requested_name.empty()) selection.requested_name = "auto";
564 selection.auto_requested = auto_backend_requested(selection.requested_name);
565
566 if (selection.auto_requested) {
567 selection.selected = choose_auto_backend(
568 phase, selection.tile_compile_with_cuda,
569 selection.opencv_cuda_runtime, selection.opencv_opencl_runtime);
570 selection.requested = selection.selected;
571 selection.gpu_requested = selection.selected != AccelerationBackend::cpu;
572 selection.using_gpu = selection.gpu_requested;
573 return selection;
574 }
575
576 AccelerationBackend requested = AccelerationBackend::cpu;
577 if (!parse_acceleration_backend(selection.requested_name, requested)) {
578 selection.request_honored = false;
579 selection.fallback_reason = "invalid_requested_backend";
580 return selection;
581 }
582 selection.requested = requested;
583 selection.selected = requested;
584 selection.gpu_requested = requested != AccelerationBackend::cpu;
585
586 const std::string missing = missing_backend_reason(
587 requested, selection.tile_compile_with_cuda,
588 selection.opencv_cuda_headers, selection.opencv_cuda_runtime,
589 selection.opencv_opencl_headers, selection.opencv_opencl_runtime);
590 if (!missing.empty()) {
591 selection.selected = AccelerationBackend::cpu;
592 selection.request_honored = false;
593 selection.fallback_reason = missing;
594 return selection;
595 }
596 if (!phase_supports_backend(phase, requested)) {
597 selection.selected = AccelerationBackend::cpu;
598 selection.request_honored = false;
599 selection.fallback_reason = unsupported_phase_reason(requested, phase);
600 return selection;
601 }
602 selection.using_gpu = selection.gpu_requested;
603 return selection;
604}

644bool parse_acceleration_backend(const std::string &name,
645 AccelerationBackend &backend_out) {
646 const std::string normalized = core::to_lower(name);
647 if (normalized == "cpu") {
648 backend_out = AccelerationBackend::cpu;
649 return true;
650 }
651 if (normalized == "opencv_cuda") {
652 backend_out = AccelerationBackend::opencv_cuda;
653 return true;
654 }
655 if (normalized == "opencv_opencl" || normalized == "opencl") {
656 backend_out = AccelerationBackend::opencv_opencl;
657 return true;
658 }
659 if (normalized == "cuda") {
660 backend_out = AccelerationBackend::cuda;
661 return true;
662 }
663 return false;

939bool AccelerationOps::warp_affine_frame(Matrix2Df img, const WarpMatrix &warp,
940 ColorMode mode, int canvas_height,
941 int canvas_width, int offset_x,
942 int offset_y, Matrix2Df &warped_out,
943 std::vector<uint8_t> *valid_mask_out,
944 bool *has_data_out,
945 cv::cuda::Stream *stream) const {
946 auto update_valid_outputs_from_rect = [&](int dst_y, int dst_x, int copy_h,
947 int copy_w) {
948 cv::Mat mask(canvas_height, canvas_width, CV_8U, cv::Scalar(0));
949 if (copy_h > 0 && copy_w > 0 && dst_y >= 0 && dst_x >= 0 &&
950 (dst_y + copy_h) <= canvas_height &&
951 (dst_x + copy_w) <= canvas_width) {
952 mask(cv::Rect(dst_x, dst_y, copy_w, copy_h)).setTo(cv::Scalar(255));
953 }
954 invalidate_matrix_outside_support(warped_out, mask);
955 write_valid_outputs_from_mask(mask, valid_mask_out, has_data_out);
956 };
957
958 auto update_valid_outputs_from_warp = [&](int src_height,
959 int src_width) {
960 cv::Mat support_mask;
961 if (!build_warped_support_mask(warp_matrix_to_cv(warp), src_height,
962 src_width,
963 cv::Size(canvas_width, canvas_height),
964 support_mask)) {
965 support_mask = cv::Mat(canvas_height, canvas_width, CV_8U,
966 cv::Scalar(0));
967 }
968 invalidate_matrix_outside_support(warped_out, support_mask);
969 write_valid_outputs_from_mask(support_mask, valid_mask_out, has_data_out);
970 };
971
972 if (img.size() <= 0) {
973 warped_out.resize(0, 0);
974 if (valid_mask_out != nullptr) {
975 valid_mask_out->clear();
976 }
977 if (has_data_out != nullptr) {
978 *has_data_out = false;
979 }
980 return false;
981 }
982
983 const int src_height = static_cast<int>(img.rows());
984 const int src_width = static_cast<int>(img.cols());
985 const int interpolation_flag =
986 interpolation_flag_from_name(prewarp_interpolation_);
987 if (warp_is_identity(warp)) {
988 int dst_y = 0;
989 int dst_x = 0;
990 int copy_h = src_height;
991 int copy_w = src_width;
992 if (canvas_width > src_width || canvas_height > src_height) {
993 warped_out = Matrix2Df::Zero(canvas_height, canvas_width);
994 dst_y = std::max(0, offset_y);
995 dst_x = std::max(0, offset_x);
996 copy_h = std::min(src_height, canvas_height - dst_y);
997 copy_w = std::min(src_width, canvas_width - dst_x);
998 if (copy_h > 0 && copy_w > 0) {
999 warped_out.block(dst_y, dst_x, copy_h, copy_w) =
1000 img.block(0, 0, copy_h, copy_w);
1001 }
1002 } else {
1003 warped_out = std::move(img);
1004 copy_h = std::min(src_height, canvas_height);
1005 copy_w = std::min(src_width, canvas_width);
1006 }
1007 update_valid_outputs_from_rect(dst_y, dst_x, copy_h, copy_w);
1008 return warped_out.size() > 0;
1009 }
1010
1011#if TILE_COMPILE_HAS_OPENCV_CUDA_HEADERS && TILE_COMPILE_HAS_OPENCV_CUDA_WARPING
1012 if (selection_.selected == AccelerationBackend::opencv_cuda &&
1013 selection_.phase == AccelerationPhase::prewarp) {
1014 if (mode == ColorMode::OSC) {
1015 if (cuda_warp_cfa_mosaic(img, warp, canvas_height, canvas_width,
1016 warped_out, interpolation_flag, stream)) {
1017 update_valid_outputs_from_warp(src_height, src_width);
1018 return true;
1019 }
1020 } else {
1021 const cv::Mat src(static_cast<int>(img.rows()), static_cast<int>(img.cols()),
1022 CV_32F, const_cast<float *>(img.data()));
1023 const cv::Mat warp_matrix = warp_matrix_to_cv(warp);
1024 cv::Mat dst;
1025 if (cuda_warp_affine_impl(src, warp_matrix,
1026 cv::Size(canvas_width, canvas_height), dst,
1027 interpolation_flag, stream)) {
1028 warped_out.resize(canvas_height, canvas_width);
1029 std::memcpy(warped_out.data(), dst.data,
1030 static_cast<size_t>(warped_out.size()) * sizeof(float));
1031 update_valid_outputs_from_warp(src_height, src_width);
1032 return true;
1033 }
1034 }
1035 }
1036#endif
1037
1038#if TILE_COMPILE_HAS_OPENCV_OPENCL
1039 if (selection_.selected == AccelerationBackend::opencv_opencl &&
1040 selection_.phase == AccelerationPhase::prewarp) {
1041 if (mode == ColorMode::OSC) {
1042 if (opencl_warp_cfa_mosaic(img, warp, canvas_height, canvas_width,
1043 warped_out, interpolation_flag)) {
1044 update_valid_outputs_from_warp(src_height, src_width);
1045 return true;
1046 }
1047 } else {
1048 const cv::Mat src(static_cast<int>(img.rows()), static_cast<int>(img.cols()),
1049 CV_32F, const_cast<float *>(img.data()));
1050 const cv::Mat warp_matrix = warp_matrix_to_cv(warp);
1051 cv::Mat dst;
1052 if (opencl_warp_affine_impl(src, warp_matrix,
1053 cv::Size(canvas_width, canvas_height), dst,
1054 interpolation_flag)) {
1055 warped_out.resize(canvas_height, canvas_width);
1056 std::memcpy(warped_out.data(), dst.data,
1057 static_cast<size_t>(warped_out.size()) * sizeof(float));
1058 update_valid_outputs_from_warp(src_height, src_width);
1059 return true;
1060 }
1061 }
1062 }
1063#endif
1064
1065 warped_out =
1066 image::apply_global_warp(img, warp, mode, canvas_height, canvas_width,
1067 prewarp_interpolation_);
1068 update_valid_outputs_from_warp(src_height, src_width);
1069 return warped_out.size() > 0;
1070}

1073bool AccelerationOps::warp_affine_rgb_frame(
1074 Matrix2Df img_r, Matrix2Df img_g, Matrix2Df img_b,
1075 const WarpMatrix &warp, int canvas_height, int canvas_width,
1076 int offset_x, int offset_y,
1077 Matrix2Df &warped_r_out, Matrix2Df &warped_g_out, Matrix2Df &warped_b_out,
1078 std::vector<uint8_t> *valid_mask_out,
1079 bool *has_data_out,
1080 cv::cuda::Stream *stream) const {
1081 if (img_r.size() <= 0 || img_g.size() <= 0 || img_b.size() <= 0) {
1082 warped_r_out.resize(0, 0);
1083 warped_g_out.resize(0, 0);
1084 warped_b_out.resize(0, 0);
1085 if (valid_mask_out != nullptr) valid_mask_out->clear();
1086 if (has_data_out != nullptr) *has_data_out = false;
1087 return false;
1088 }
1089
1090 const int src_height = static_cast<int>(img_r.rows());
1091 const int src_width = static_cast<int>(img_r.cols());
1092 const int interpolation_flag =
1093 interpolation_flag_from_name(prewarp_interpolation_);
1094
1095 auto update_valid_outputs_from_warp = [&](int sh, int sw) {
1096 cv::Mat support_mask;
1097 if (!build_warped_support_mask(warp_matrix_to_cv(warp), sh, sw,
1098 cv::Size(canvas_width, canvas_height),
1099 support_mask)) {
1100 support_mask = cv::Mat(canvas_height, canvas_width, CV_8U, cv::Scalar(0));
1101 }
1102 invalidate_matrix_outside_support(warped_r_out, support_mask);
1103 invalidate_matrix_outside_support(warped_g_out, support_mask);
1104 invalidate_matrix_outside_support(warped_b_out, support_mask);
1105 write_valid_outputs_from_mask(support_mask, valid_mask_out, has_data_out);
1106 };
1107
1108#if TILE_COMPILE_HAS_OPENCV_CUDA_HEADERS && TILE_COMPILE_HAS_OPENCV_CUDA_WARPING
1109 if (selection_.selected == AccelerationBackend::opencv_cuda &&
1110 selection_.phase == AccelerationPhase::prewarp) {
1111 const cv::Mat src_r(src_height, src_width, CV_32F, const_cast<float *>(img_r.data()));
1112 const cv::Mat src_g(src_height, src_width, CV_32F, const_cast<float *>(img_g.data()));
1113 const cv::Mat src_b(src_height, src_width, CV_32F, const_cast<float *>(img_b.data()));
1114 const cv::Mat warp_matrix = warp_matrix_to_cv(warp);
1115 cv::Mat dst_r, dst_g, dst_b;
1116 if (cuda_warp_affine_rgb_impl(src_r, src_g, src_b, warp_matrix,
1117 cv::Size(canvas_width, canvas_height),
1118 dst_r, dst_g, dst_b,
1119 interpolation_flag, stream)) {
1120 warped_r_out.resize(canvas_height, canvas_width);
1121 warped_g_out.resize(canvas_height, canvas_width);
1122 warped_b_out.resize(canvas_height, canvas_width);
1123 const size_t byte_count = static_cast<size_t>(canvas_height * canvas_width) * sizeof(float);
1124 std::memcpy(warped_r_out.data(), dst_r.data, byte_count);
1125 std::memcpy(warped_g_out.data(), dst_g.data, byte_count);
1126 std::memcpy(warped_b_out.data(), dst_b.data, byte_count);
1127 update_valid_outputs_from_warp(src_height, src_width);
1128 return true;
1129 }
1130 }
1131#endif
1132
1133 // Fallback: per-channel affine warp
1134 bool r_ok = warp_affine_frame(std::move(img_r), warp, ColorMode::MONO,
1135 canvas_height, canvas_width, offset_x, offset_y,
1136 warped_r_out, valid_mask_out, has_data_out, stream);
1137 bool g_ok = warp_affine_frame(std::move(img_g), warp, ColorMode::MONO,
1138 canvas_height, canvas_width, offset_x, offset_y,
1139 warped_g_out, nullptr, nullptr, stream);
1140 bool b_ok = warp_affine_frame(std::move(img_b), warp, ColorMode::MONO,
1141 canvas_height, canvas_width, offset_x, offset_y,
1142 warped_b_out, nullptr, nullptr, stream);
1143 return r_ok && g_ok && b_ok;
1144}
