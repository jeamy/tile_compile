14double median_inplace(std::vector<float> &v) {
15 if (v.empty()) return std::numeric_limits<double>::quiet_NaN();
16 const size_t m = v.size() / 2;
17 std::nth_element(v.begin(), v.begin() + m, v.end());
18 const double hi = v[m];
19 if (v.size() % 2 == 1) return hi;
20 const double lo = *std::max_element(v.begin(), v.begin() + m);
21 return 0.5 * (lo + hi);
22}

27// D1 remainder: mad_sigma's by-value signature copies its input because the
28// in-place median permutes it. Callers that own a mutable scratch buffer and
29// no longer need its ordering (e.g. the bisection's rebuilt `mix`) avoid both
30// the copy and the per-call `dev` allocation through this helper.
31double mad_sigma_mutate(std::vector<float> &values, std::vector<float> &dev) {
32 if (values.empty()) return std::numeric_limits<double>::quiet_NaN();
33 const double med = median_inplace(values);
34 dev.resize(values.size());
35 for (std::size_t k = 0; k < values.size(); ++k)
36 dev[k] = static_cast<float>(std::abs(values[k] - med));
37 return 1.4826 * median_inplace(dev);
38}

42double mad_sigma(std::vector<float> values) {
43 if (values.empty()) return std::numeric_limits<double>::quiet_NaN();
44 const double med = median_inplace(values);
45 std::vector<float> dev;
46 dev.reserve(values.size());
47 for (float v : values) dev.push_back(static_cast<float>(std::abs(v - med)));
48 return 1.4826 * median_inplace(dev);
49}

56std::vector<float> apply_energy_guard(const std::vector<float> &alpha_pre,
57 const std::vector<float> &d_r_luma,
58 const std::vector<float> &d_profile_luma,
59 const std::vector<uint8_t> &support,
60 int width, int height, int window_radius,
61 double background_floor,
62 const EnergyGuardParams &params) {
63 if (width <= 0 || height <= 0)
64 throw std::invalid_argument("ENERGY_GUARD_DIMENSIONS");
65 const std::size_t n = static_cast<std::size_t>(width) * height;
66 for (const auto *v :
67 {&alpha_pre, &d_r_luma, &d_profile_luma})
68 if (v->size() != n) throw std::invalid_argument("ENERGY_GUARD_SIZE");
69 if (support.size() != n) throw std::invalid_argument("ENERGY_GUARD_SIZE");
70 if (window_radius < 1) throw std::invalid_argument("ENERGY_GUARD_RADIUS");
71 if (!(background_floor >= 0.0))
72 throw std::invalid_argument("ENERGY_GUARD_FLOOR");
73 if (params.bisection_iters < 1)
74 throw std::invalid_argument("ENERGY_GUARD_ITERS");
75
76 std::vector<float> out(n, 0.0f);
77 // Every output pixel depends only on its own read-only window, so rows are
78 // independent: row-parallel with thread-private scratch, identical per-
79 // pixel arithmetic and no shared mutable state besides out[i] (written by
80 // exactly one iteration). Dynamic schedule because the per-row cost
81 // follows the support/alpha_pre distribution.
82#pragma omp parallel
83 {
84 // Hoisted out of the per-pixel loop: dr_vals/dp_vals were previously fresh
85 // std::vector allocations every pixel, and `mix` was reallocated on every
86 // ratio_at() call within the bisection (1 + bisection_iters times per
87 // pixel). Reusing capacity via .clear()/.resize() instead of re-declaring
88 // avoids that malloc/free churn.
89 std::vector<float> dr_vals, dp_vals, mix, mad_dev;
90#pragma omp for schedule(dynamic, 4)
91 for (int y = 0; y < height; ++y) {
92 for (int x = 0; x < width; ++x) {
93 const std::size_t i = static_cast<std::size_t>(y) * width + x;
94 if (!support[i]) continue;
95 const double a_pre =
96 std::isfinite(alpha_pre[i]) ? std::clamp<double>(alpha_pre[i], 0.0, 1.0)
97 : 0.0;
98 if (a_pre <= 0.0) continue; // out stays 0
99
100 dr_vals.clear();
101 dp_vals.clear();
102 for (int yy = std::max(0, y - window_radius);
103 yy <= std::min(height - 1, y + window_radius); ++yy)
104 for (int xx = std::max(0, x - window_radius);
105 xx <= std::min(width - 1, x + window_radius); ++xx) {
106 const std::size_t j = static_cast<std::size_t>(yy) * width + xx;
107 if (!support[j] || !std::isfinite(d_r_luma[j]) ||
108 !std::isfinite(d_profile_luma[j]))
109 continue;
110 dr_vals.push_back(d_r_luma[j]);
111 dp_vals.push_back(d_profile_luma[j]);
112 }
113 if (static_cast<int>(dr_vals.size()) < params.min_window_pixels) continue;
114
115 const double mad_r = mad_sigma(dr_vals);
116 const double scale_raw = std::max(mad_r, background_floor);
117 auto ratio_at = [&](double a) -> double {
118 // Reuses the hoisted `mix`/`mad_dev` buffers' capacity across
119 // bisection iterations (resize only grows/shrinks the logical size,
120 // no reallocation once the largest window size seen is reached).
121 // mad_sigma_mutate permutes `mix` in place for the median — legal
122 // here because `mix` is rebuilt from dr_vals/dp_vals on every call.
123 // dr_vals keeps the by-value mad_sigma call: its ordering must
124 // survive to stay paired with dp_vals.
125 mix.resize(dr_vals.size());
126 for (std::size_t k = 0; k < dr_vals.size(); ++k)
127 mix[k] = static_cast<float>(dr_vals[k] +
128 a * (dp_vals[k] - dr_vals[k]));
129 const double mad_mix = mad_sigma_mutate(mix, mad_dev);
130 return scale_raw > 0.0 ? mad_mix / scale_raw
131 : (mad_mix > 0.0 ? std::numeric_limits<double>::infinity()
132 : 0.0);
133 };
134
135 if (ratio_at(a_pre) <= params.energy_limit) {
136 out[i] = static_cast<float>(a_pre);
137 continue;
138 }
139 // Bisection: lo always feasible (ratio_at(0) = mad_r/scale_raw <= 1).
140 double lo = 0.0, hi = a_pre;
141 for (int it = 0; it < params.bisection_iters; ++it) {
142 const double mid = 0.5 * (lo + hi);
143 if (ratio_at(mid) <= params.energy_limit)
144 lo = mid;
145 else
146 hi = mid;
147 }
148 out[i] = static_cast<float>(lo);
149 }
150 }
151 } // omp parallel
152 return out;
153}

155std::vector<float> smooth_alpha_b3(const std::vector<float> &alpha_guarded,
156 const std::vector<uint8_t> &support,
157 int width, int height) {
158 if (width <= 0 || height <= 0)
159 throw std::invalid_argument("ALPHA_SMOOTH_DIMENSIONS");
160 const std::size_t n = static_cast<std::size_t>(width) * height;
161 if (alpha_guarded.size() != n || support.size() != n)
162 throw std::invalid_argument("ALPHA_SMOOTH_SIZE");
163
164 // 4-connected component labels of the support (plan 14.7: no convolution
165 // across separate support islands).
166 std::vector<int> label(n, -1);
167 int next = 0;
168 std::vector<std::size_t> stack;
169 // Hoisted out of the per-stack-pop loop body (redundant-reload analysis
170 // C8): these are compile-time-constant per BFS, not per pop.
171 const int dx[4] = {-1, 1, 0, 0};
172 const int dy[4] = {0, 0, -1, 1};
173 for (std::size_t s = 0; s < n; ++s) {
174 if (!support[s] || label[s] >= 0) continue;
175 label[s] = next;
176 stack.push_back(s);
177 while (!stack.empty()) {
178 const std::size_t p = stack.back();
179 stack.pop_back();
180 const int px = static_cast<int>(p % width);
181 const int py = static_cast<int>(p / width);
182 for (int k = 0; k < 4; ++k) {
183 const int nx = px + dx[k], ny = py + dy[k];
184 if (nx < 0 || ny < 0 || nx >= width || ny >= height) continue;
185 const std::size_t q = static_cast<std::size_t>(ny) * width + nx;
186 if (support[q] && label[q] < 0) {
187 label[q] = next;
188 stack.push_back(q);
189 }
190 }
191 }
192 ++next;
193 }
194
195 // B3: the numerator (alpha*support) and denominator (support) convolutions
196 // share the same component-restricted stencil, so one fused separable pass
197 // accumulates both instead of running the machinery twice. Unlabelled taps
198 // (label < 0) can never match label[i] >= 0, so they contribute nothing —
199 // identical to the previous src-prefill form.
200 std::vector<double> hx_num(n, 0.0), hx_den(n, 0.0);
201 for (int y = 0; y < height; ++y)
202 for (int x = 0; x < width; ++x) {
203 const std::size_t i = static_cast<std::size_t>(y) * width + x;
204 if (label[i] < 0) continue;
205 double acc_n = 0.0, acc_d = 0.0;
206 for (int t = -2; t <= 2; ++t) {
207 const int xx = x + t;
208 if (xx < 0 || xx >= width) continue;
209 const std::size_t j = static_cast<std::size_t>(y) * width + xx;
210 if (label[j] != label[i]) continue;
211 acc_n += kB3[t + 2] * std::max(0.0f, alpha_guarded[j]);
212 acc_d += kB3[t + 2];
213 }
214 hx_num[i] = acc_n;
215 hx_den[i] = acc_d;
216 }
217 std::vector<double> num(n, 0.0), den(n, 0.0);
218 for (int y = 0; y < height; ++y)
219 for (int x = 0; x < width; ++x) {
220 const std::size_t i = static_cast<std::size_t>(y) * width + x;
221 if (label[i] < 0) continue;
222 double acc_n = 0.0, acc_d = 0.0;
223 for (int t = -2; t <= 2; ++t) {
224 const int yy = y + t;
225 if (yy < 0 || yy >= height) continue;
226 const std::size_t j = static_cast<std::size_t>(yy) * width + x;
227 if (label[j] != label[i]) continue;
228 acc_n += kB3[t + 2] * hx_num[j];
229 acc_d += kB3[t + 2] * hx_den[j];
230 }
231 num[i] = acc_n;
232 den[i] = acc_d;
233 }
234
235 std::vector<float> final_(n, nanf_());
236 for (std::size_t i = 0; i < n; ++i) {
237 if (!support[i]) continue;
238 const double blur = den[i] > 0.0 ? num[i] / den[i] : 0.0;
239 const double g = std::max(0.0f, alpha_guarded[i]);
240 final_[i] = static_cast<float>(std::min(g, blur)); // mandatory min cap
241 }
242 return final_;
243}
