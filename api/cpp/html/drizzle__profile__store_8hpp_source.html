19// M6: the full multiband contract that enters `multiband_config_hash`
20// (plan 16.4: profile hashes plus levels, alpha, energy, support, downsample
21// and validation contract). `enabled == false` => a plain uniform_raw store,
22// byte-identical to a pre-M6 store. The versioned a-trous constants
23// (kAtrousDenMinFraction, kAtrousDecompositionVersion) are folded in by
24// make_drizzle_store_identity itself, not carried here.
25struct MultibandStoreContract {
26 bool enabled = false;
27 int levels = 3; // [1,4]; medium plane emitted only for levels >= 2
28 float fine_quality_exponent = 4.0f;
29 float medium_quality_exponent = 2.0f;
30 AdaptiveAlphaParams alpha{};
31 EnergyGuardParams guard{};
32 AlphaConfidenceParams confidence{};
33};

35struct DrizzleStoreIdentity {
