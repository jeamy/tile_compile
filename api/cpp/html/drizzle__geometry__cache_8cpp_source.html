1// Plan section 11.14 P1/P2 --- see the header.
2//
3// On-disk layout (committed generations are immutable and never deleted; the
4// live one is named by <root>/current.json):
5// <root>/current.json {schema, generation, variants[]}
6// <root>/generation-<hash>-<uid>/manifest.json identities + per-frame stats
7// + per-file sha256 (written LAST)
8// <root>/generation-.../v<vi>_f<idx>.rows source_height x RowEntry (32 B)
9// <root>/generation-.../v<vi>_f<idx>.leaves leaves x LeafRecord (72 B)
10//
11// Build streams: each completed SOURCE ROW's leaf records are appended to the
12// .leaves file and freed before the next row starts (bounded staging buffer,
13// capped by memory_budget_bytes). Discards are counted directly from
14// sample_leaves' return value --- never from the optional geomstats registry.
15//
16// Reader loads only the .rows indices; enumerate_stripe() seek-reads the row
17// blocks intersecting the requested stripe. Leaf records are never fully
18// resident. Every read replays the SAME (source_y, source_x, leaf_order, y, x)
19// canonical order and the SAME bbox clamp the stripe enumerator uses.
20//
21// Commit is crash-safe: unique generation + staging directory names, the old
22// generation is left untouched, all payload is fsync'd before the manifest,
23// the manifest is fsync'd before the directory rename, and current.json is
24// swapped last via a temp-file rename. An abort at any point leaves the
25// previous committed generation fully usable.

110// Append-writes a byte span to an open fd, retrying short writes.
111void write_all(int fd, const void *data, std::size_t n) {
112 const auto *p = static_cast<const char *>(data);
113 while (n) {
114 const std::ptrdiff_t w = write_fd(fd, p, n);
115 if (w < 0) {
116 if (errno == EINTR) continue;
117 throw std::runtime_error(std::string("DRIZZLE_GEOMETRY_CACHE_WRITE: ") +
118 std::strerror(errno));
119 }
120 p += w;
121 n -= static_cast<std::size_t>(w);
122 }
123}
124// Portable open/fsync shims: POSIX on Unix, _wopen/_commit on Windows
125// (path::value_type is wchar_t there, and O_DIRECTORY/fsync do not exist;
126// directory fsync is a no-op on Windows anyway).
127int open_trunc_for_write(const fs::path &p) {
128#if defined(_WIN32)
129 return ::_wopen(p.c_str(), _O_WRONLY | _O_CREAT | _O_TRUNC | _O_BINARY,
130 0644);
131#else
132 return ::open(p.c_str(), O_WRONLY | O_CREAT | O_TRUNC, 0644);
133#endif
134}
135int fsync_fd(int fd) {
136#if defined(_WIN32)
137 return ::_commit(fd);
138#else
139 return ::fsync(fd);
140#endif
141}
142void fsync_path(const fs::path &p, bool dir) {
143#if defined(_WIN32)
144 if (dir) return; // no directory fsync on Windows
145 const int fd = ::_wopen(p.c_str(), _O_RDONLY | _O_BINARY);
146#else
147 const int flags = dir ? (O_RDONLY | O_DIRECTORY) : O_RDONLY;
148 const int fd = ::open(p.c_str(), flags);
149#endif
150 if (fd < 0) {
151 if (dir) return; // some filesystems reject O_DIRECTORY fsync; best effort
152 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_FSYNC_OPEN: " + p.string());
153 }
154 const int r = fsync_fd(fd);
155 close_fd(fd);
156 if (r != 0 && !dir)
157 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_FSYNC: " + p.string());
158}

168std::string fresh_uid() {
169 std::random_device rd;
170 std::uniform_int_distribution<std::uint64_t> d;
171 std::uint64_t a = (static_cast<std::uint64_t>(rd()) << 32) ^ rd() ^
172 static_cast<std::uint64_t>(::getpid());
173 std::uint64_t b = (static_cast<std::uint64_t>(rd()) << 32) ^ rd() ^ d(rd);
174 char buf[33];
175 std::snprintf(buf, sizeof(buf), "%016llx%016llx",
176 static_cast<unsigned long long>(a),
177 static_cast<unsigned long long>(b));
178 return std::string(buf);
179}

181// One independent (variant, frame) build task. No shared mutable state: its own
182// sample_leaves sweep, its own .rows/.leaves files, its own SHA contexts. Safe
183// to run concurrently for distinct (vi, source_index). geomstats must be
184// disabled by the caller across the (possibly parallel) region.
185struct FrameBuildOut {
186 bool present = false; // false => (vi, fidx) was not a local frame
187 json fj;
188 GeometryCacheFrameStats stats;
189 std::uint64_t record_bytes = 0;
190 std::uint64_t index_bytes = 0;
191 bool excluded = false;
192 std::string excluded_frame_id;
193 double excluded_rate = 0.0;
194 double t_compute = 0.0, t_write = 0.0;
195};

197FrameBuildOut build_one_frame(
198 const fs::path &staging, std::size_t vi, float pixfrac,
199 const RegistrationSamplingPlan &plan,
200 const registration::FrameSamplingTransform &f,
201 const ForwardDrizzleSubdivisionParams &subdivision, int scale,
202 int internal_w, int internal_h, std::uint64_t source_pixels) {
203 using bclk = std::chrono::steady_clock;
204 FrameBuildOut out;
205 if (!f.valid || !f.has_smooth_local_model) return out;
206 if (!f.source_to_canvas_affine_valid || !f.source_to_canvas.allFinite())
207 throw std::invalid_argument("DRIZZLE_GEOMETRY_CACHE_BAD_TRANSFORM");
208 out.present = true;
209
210 const fs::path rp = staging / rows_name(vi, f.source_index);
211 const fs::path lp = staging / leaves_name(vi, f.source_index);
212 const int lfd = open_trunc_for_write(lp);
213 if (lfd < 0)
214 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_OPEN: " + lp.string());
215 // T1: no SHA-256 streaming in the hot path (trusted run).
216 std::vector<Leaf> leaves;
217 leaves.reserve(16);
218 std::vector<LeafRecord> row_buf;
219 std::vector<RowEntry> rows(static_cast<std::size_t>(plan.source_height));
220 std::uint64_t running = 0, leaves_total = 0, discarded = 0;
221
222 for (int sy = 0; sy < plan.source_height; ++sy) {
223 row_buf.clear();
224 double ymin = std::numeric_limits<double>::infinity();
225 double ymax = -std::numeric_limits<double>::infinity();
226 const std::uint64_t row_start = running;
227 const auto row_c0 = bclk::now();
228 for (int sx = 0; sx < plan.source_width; ++sx) {
229 if (!sample_leaves(plan, f, sx, sy, scale, pixfrac, subdivision, leaves)) {
230 ++discarded;
231 continue;
232 }
233 std::uint16_t channel = 0;
234 if (plan.color_mode == ColorMode::OSC) {
235 const auto ch = cfa_channel_for_source_pixel(
236 sx, sy, plan.bayer_pattern, plan.cfa_origin_x, plan.cfa_origin_y);
237 channel = ch == CfaChannel::R ? 0 : ch == CfaChannel::G ? 1 : 2;
238 }
239 for (std::size_t li = 0; li < leaves.size(); ++li) {
240 LeafRecord rec{};
241 rec.source_x = static_cast<std::uint32_t>(sx);
242 rec.channel = channel;
243 rec.leaf_order = static_cast<std::uint16_t>(li);
244 for (int i = 0; i < 4; ++i) {
245 rec.x[i] = leaves[li].x[i];
246 rec.y[i] = leaves[li].y[i];
247 ymin = std::min(ymin, leaves[li].y[i]);
248 ymax = std::max(ymax, leaves[li].y[i]);
249 }
250 row_buf.push_back(rec);
251 }
252 }
253 const auto row_w0 = bclk::now();
254 out.t_compute += std::chrono::duration<double>(row_w0 - row_c0).count();
255 if (!row_buf.empty()) {
256 const std::size_t nb = row_buf.size() * sizeof(LeafRecord);
257 write_all(lfd, row_buf.data(), nb);
258 running += nb;
259 leaves_total += row_buf.size();
260 }
261 out.t_write += std::chrono::duration<double>(bclk::now() - row_w0).count();
262 RowEntry &re = rows[static_cast<std::size_t>(sy)];
263 re.record_offset = row_start;
264 re.record_count = (running - row_start) / sizeof(LeafRecord);
265 re.canvas_ymin = re.record_count ? ymin : 0.0;
266 re.canvas_ymax = re.record_count ? ymax : 0.0;
267 }
268
269 const auto tail_w0 = bclk::now();
270 if (fsync_fd(lfd) != 0) {
271 close_fd(lfd);
272 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_FSYNC: " + lp.string());
273 }
274 close_fd(lfd);
275 // T1: no SHA-256 in the hot path.
276 {
277 const int rfd = open_trunc_for_write(rp);
278 if (rfd < 0)
279 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_OPEN: " + rp.string());
280 write_all(rfd, rows.data(), rows.size() * sizeof(RowEntry));
281 if (fsync_fd(rfd) != 0) {
282 close_fd(rfd);
283 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_FSYNC: " + rp.string());
284 }
285 close_fd(rfd);
286 }
287 out.t_write += std::chrono::duration<double>(bclk::now() - tail_w0).count();
288
289 const double rate = source_pixels
290 ? static_cast<double>(discarded) /
291 static_cast<double>(source_pixels)
292 : 0.0;
293 const bool excluded = rate > subdivision.per_frame_inversion_error_rate_max;
294
295 out.fj = json{{"source_index", f.source_index},
296 {"frame_id", f.frame_id},
297 {"source_width", plan.source_width},
298 {"source_height", plan.source_height},
299 {"internal_width", internal_w},
300 {"internal_height", internal_h},
301 {"samples_total", source_pixels},
302 {"samples_discarded", discarded},
303 {"leaves", leaves_total},
304 {"leaves_bytes", running},
305 {"subdivision_error_rate", rate},
306 {"excluded", excluded},
307 {"rows_file", rows_name(vi, f.source_index)},
308 {"leaves_file", leaves_name(vi, f.source_index)},
309 {"rows_bytes", rows.size() * sizeof(RowEntry)},
310 {"leaves_bytes", running}};
311 out.stats.source_index = f.source_index;
312 out.stats.samples_total = source_pixels;
313 out.stats.top_level_sample_leaves_calls = source_pixels;
314 out.stats.leaves_written = leaves_total;
315 out.stats.samples_discarded = discarded;
316 out.stats.subdivision_error_rate = rate;
317 out.stats.excluded = excluded;
318 out.record_bytes = running;
319 out.index_bytes = rows.size() * sizeof(RowEntry);
320 out.excluded = excluded;
321 out.excluded_frame_id = f.frame_id;
322 out.excluded_rate = rate;
323 return out;
324}

332GeometryCacheIdentity make_geometry_cache_identity(
333 const RegistrationSamplingPlan &plan,
334 const config::ReconstructionDrizzleConfig &cfg,
335 const GeometryVariant &variant,
336 const ForwardDrizzleSubdivisionParams &subdivision) {
337 config::ReconstructionDrizzleConfig vcfg = cfg;
338 vcfg.pixfrac = variant.pixfrac;
339 const std::string base = registration::compute_coverage_geometry_hash(
340 plan, vcfg, /*common_fraction=*/0.0f);
341 std::string s = "drizzle-geometry-cache:schema" +
342 std::to_string(kSchemaVersion) + ":algo" +
343 std::to_string(kAlgorithmVersion) +
344 ":edge-centers:exact-polygons:" + base;
345 std::vector<std::uint8_t> bytes(s.begin(), s.end());
346 auto u32 = [&](std::uint32_t v) {
347 for (int i = 0; i < 4; ++i)
348 bytes.push_back(static_cast<std::uint8_t>(v >> (8 * i)));
349 };
350 auto f32 = [&](float v) {
351 std::uint32_t u;
352 std::memcpy(&u, &v, 4);
353 u32(u);
354 };
355 const registration::LocalInversionParams inv{};
356 u32(static_cast<std::uint32_t>(inv.max_iter));
357 f32(inv.tol_px);
358 f32(inv.safety_margin_px);
359 u32(static_cast<std::uint32_t>(subdivision.max_subdivision_depth));
360 f32(subdivision.position_epsilon_internal_px);
361 f32(subdivision.area_relative_epsilon);
362 f32(subdivision.per_frame_inversion_error_rate_max);
363 u32(static_cast<std::uint32_t>(plan.canvas_width_native));
364 u32(static_cast<std::uint32_t>(plan.canvas_height_native));
365 u32(static_cast<std::uint32_t>(cfg.internal_scale));
366 u32(static_cast<std::uint32_t>(plan.bayer_pattern));
367 u32(static_cast<std::uint32_t>(plan.cfa_origin_x));
368 u32(static_cast<std::uint32_t>(plan.cfa_origin_y));
369 u32(static_cast<std::uint32_t>(plan.color_mode));
370
371 GeometryCacheIdentity id;
372 id.geometry_hash = core::sha256_bytes(bytes);
373 id.sampling_plan_hash = plan.plan_hash;
374 id.canvas_width_native = plan.canvas_width_native;
375 id.canvas_height_native = plan.canvas_height_native;
376 id.internal_scale = cfg.internal_scale;
377 id.pixfrac = variant.pixfrac;
378 id.color_mode = plan.color_mode;
379 return id;
380}

386GeometryCacheBuildResult build_drizzle_geometry_cache(
387 const fs::path &root, const RegistrationSamplingPlan &plan,
388 const config::ReconstructionDrizzleConfig &cfg,
389 const std::vector<GeometryVariant> &variants_in,
390 const ForwardDrizzleSubdivisionParams &subdivision,
391 std::uint64_t memory_budget_bytes, int max_workers,
392 const GeometryCacheProgressFn &on_progress) {
393 if (plan.source_width <= 0 || plan.source_height <= 0)
394 throw std::invalid_argument("DRIZZLE_GEOMETRY_CACHE_BAD_PLAN");
395 if (static_cast<long long>(plan.source_width) > 0xFFFFFFFF)
396 throw std::invalid_argument("DRIZZLE_GEOMETRY_CACHE_SOURCE_TOO_WIDE");
397 if (cfg.internal_scale < 1 || cfg.internal_scale > 2)
398 throw std::invalid_argument("DRIZZLE_GEOMETRY_CACHE_BAD_SCALE");
399
400 std::vector<GeometryVariant> variants;
401 for (const auto &v : variants_in) {
402 bool dup = false;
403 for (const auto &e : variants) dup = dup || (e.pixfrac == v.pixfrac);
404 if (!dup) variants.push_back(v);
405 }
406 if (variants.empty())
407 throw std::invalid_argument("DRIZZLE_GEOMETRY_CACHE_NO_VARIANT");
408
409 GeometryCacheBuildResult result;
410 for (const auto &v : variants)
411 result.identities.push_back(
412 make_geometry_cache_identity(plan, cfg, v, subdivision));
413
414 const int scale = cfg.internal_scale;
415 const int internal_w = plan.canvas_width_native * scale;
416 const int internal_h = plan.canvas_height_native * scale;
417 const std::uint64_t source_pixels =
418 static_cast<std::uint64_t>(plan.source_width) * plan.source_height;
419
420 // The staging buffer is ONE completed source row of leaf records --- a
421 // bounded batch (<= source_width * max_leaves_per_sample * 72 B, independent
422 // of frame count and canvas height, satisfying plan 11.14.3). It is written
423 // and freed before the next row. memory_budget_bytes is a floor sanity check
424 // only: a single row must never exceed it.
425 const std::uint64_t max_leaves_per_sample = 1ull
426 << (2 * static_cast<unsigned>(std::max(0, subdivision.max_subdivision_depth)));
427 const std::uint64_t row_bytes_bound =
428 static_cast<std::uint64_t>(plan.source_width) * max_leaves_per_sample *
429 sizeof(LeafRecord);
430 if (memory_budget_bytes && row_bytes_bound > memory_budget_bytes)
431 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_ROW_EXCEEDS_BUDGET");
432
433 const std::string uid = fresh_uid();
434 const std::string gen_id =
435 "generation-" + result.identities.front().geometry_hash.substr(0, 16) +
436 "-" + uid.substr(0, 12);
437 const fs::path staging = root / (".staging-" + uid);
438 std::error_code ec;
439 fs::create_directories(staging, ec);
440 if (ec) throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_STAGING: " + ec.message());
441
442 json manifest;
443 manifest["schema"] = kSchemaVersion;
444 manifest["algorithm_version"] = kAlgorithmVersion;
445 manifest["source_width"] = plan.source_width;
446 manifest["source_height"] = plan.source_height;
447 manifest["internal_width"] = internal_w;
448 manifest["internal_height"] = internal_h;
449 manifest["variants"] = json::array();
450
451 // Plan 11.14.5 P3: one independent task per (variant, frame). Assemble the
452 // manifest in deterministic (variant, source_index) order afterwards, so the
453 // committed store is byte-identical regardless of worker count / scheduling.
454 struct Task { std::size_t vi, fidx; };
455 std::vector<Task> tasks;
456 for (std::size_t vi = 0; vi < variants.size(); ++vi)
457 for (std::size_t fidx = 0; fidx < plan.frames.size(); ++fidx)
458 tasks.push_back({vi, fidx});
459 std::vector<FrameBuildOut> outs(tasks.size());
460
461 const int workers = std::max(1, max_workers);
462 const auto wall0 = std::chrono::steady_clock::now();
463
464 // Progress reporting: count only tasks that will actually run the
465 // expensive per-sample sweep (has_smooth_local_model). The many trivial
466 // (non-local) tasks in `tasks` finish instantly and are not reported ---
467 // reporting all of them would flood the log for no benefit.
468 std::size_t local_frame_count = 0;
469 for (const auto &f : plan.frames)
470 if (f.valid && f.has_smooth_local_model) ++local_frame_count;
471 const std::size_t total_present = local_frame_count * variants.size();
472 std::atomic<std::size_t> tasks_present_done{0};
473 std::mutex progress_mutex;
474 if (on_progress && total_present > 0) {
475 std::lock_guard<std::mutex> lk(progress_mutex);
476 on_progress(0, total_present,
477 "geometry_cache_build starting, " +
478 std::to_string(total_present) + " (variant,frame) tasks, " +
479 std::to_string(workers) + " workers");
480 }
481
482 // The sample_leaves path touches the process-global geomstats counters when
483 // enabled --- a data race under parallelism, and the build is not a consumer
484 // anyway. Disable across the region, restore after (single-threaded here).
485 auto &greg = geomstats::registry();
486 const bool prev_geom_enabled = greg.enabled;
487 greg.enabled = false;
488
489 std::exception_ptr eptr;
490#ifdef _OPENMP
491 if (workers > 1) omp_set_num_threads(workers);
492#endif
493#pragma omp parallel for schedule(dynamic, 1) if (workers > 1)
494 for (long long k = 0; k < static_cast<long long>(tasks.size()); ++k) {
495 if (eptr) continue;
496 try {
497 const std::size_t vi = tasks[static_cast<std::size_t>(k)].vi;
498 const std::size_t fidx = tasks[static_cast<std::size_t>(k)].fidx;
499 outs[static_cast<std::size_t>(k)] = build_one_frame(
500 staging, vi, variants[vi].pixfrac, plan, plan.frames[fidx],
501 subdivision, scale, internal_w, internal_h, source_pixels);
502 if (on_progress && outs[static_cast<std::size_t>(k)].present) {
503 const std::size_t done =
504 tasks_present_done.fetch_add(1, std::memory_order_relaxed) + 1;
505 std::lock_guard<std::mutex> lk(progress_mutex);
506 on_progress(done, total_present,
507 "geom_cache " + std::to_string(done) + "/" +
508 std::to_string(total_present) + " frame=" +
509 plan.frames[fidx].frame_id + " pixfrac=" +
510 std::to_string(variants[vi].pixfrac));
511 }
512 } catch (...) {
513#pragma omp critical
514 { if (!eptr) eptr = std::current_exception(); }
515 }
516 }
517 greg.enabled = prev_geom_enabled;
518 if (eptr) std::rethrow_exception(eptr);
519
520 double t_compute = 0.0, t_write = 0.0;
521 for (std::size_t vi = 0; vi < variants.size(); ++vi) {
522 json vjson;
523 vjson["pixfrac"] = variants[vi].pixfrac;
524 vjson["geometry_hash"] = result.identities[vi].geometry_hash;
525 vjson["sampling_plan_hash"] = plan.plan_hash;
526 vjson["canvas_width_native"] = plan.canvas_width_native;
527 vjson["canvas_height_native"] = plan.canvas_height_native;
528 vjson["internal_scale"] = scale;
529 vjson["color_mode"] = plan.color_mode == ColorMode::MONO ? "MONO" : "OSC";
530 vjson["frames"] = json::array();
531 // tasks were pushed in (vi, fidx) order == source_index order.
532 for (std::size_t k = 0; k < tasks.size(); ++k) {
533 if (tasks[k].vi != vi) continue;
534 const auto &o = outs[k];
535 if (!o.present) continue;
536 vjson["frames"].push_back(o.fj);
537 result.frames.push_back(o.stats);
538 result.total_leaves += o.stats.leaves_written;
539 result.total_record_bytes += o.record_bytes;
540 result.index_bytes += o.index_bytes;
541 if (o.excluded)
542 result.excluded_frames.emplace_back(o.excluded_frame_id, o.excluded_rate);
543 t_compute += o.t_compute;
544 t_write += o.t_write;
545 }
546 manifest["variants"].push_back(vjson);
547 }

672 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_BAD_DIMS");

697 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_PIXFRAC_MISMATCH");
698 if (vj.at("canvas_width_native").get<int>() * exp.internal_scale != m_iw)
699 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_DIM_CONSISTENCY");

707 for (const auto &fj : vj.at("frames")) {
708 Impl::FrameData fd;
709 const std::size_t sidx = fj.at("source_index").get<std::size_t>();
710 fd.source_height = fj.at("source_height").get<int>();
711 if (fd.source_height != m_sh ||

716 fd.excluded = fj.at("excluded").get<bool>();
717 fd.samples_total = fj.at("samples_total").get<std::uint64_t>();
718 fd.samples_discarded = fj.at("samples_discarded").get<std::uint64_t>();
719 fd.subdivision_error_rate = fj.at("subdivision_error_rate").get<double>();
720 fd.leaves_bytes = fj.at("leaves_bytes").get<std::uint64_t>();
721 if (fd.samples_total !=
722 static_cast<std::uint64_t>(m_sw) * static_cast<std::uint64_t>(m_sh))
723 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_SAMPLES_TOTAL");
724 if (fd.samples_discarded > fd.samples_total)
725 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_DISCARD_RANGE");
726 {
727 const double want_rate =
728 fd.samples_total ? static_cast<double>(fd.samples_discarded) /
729 static_cast<double>(fd.samples_total)
730 : 0.0;
731 if (std::abs(want_rate - fd.subdivision_error_rate) > 1e-12)
732 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_RATE_CONSISTENCY");
733 }
734
735 const fs::path rp = gen / fj.at("rows_file").get<std::string>();
736 const fs::path lp = gen / fj.at("leaves_file").get<std::string>();
737 if (!fs::is_regular_file(rp) || !fs::is_regular_file(lp))
738 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_MISSING_FILE");
739 // T1: no SHA-256 verification in the hot path (trusted run). v1 manifests
740 // with rows_sha256/leaves_sha256 are accepted without re-hashing; the
741 // size checks below catch truncation and corruption.
742
743 const auto rsz = fs::file_size(rp);
744 const auto lsz = fs::file_size(lp);
745 if (rsz % sizeof(RowEntry) ||
746 rsz / sizeof(RowEntry) != static_cast<std::uint64_t>(fd.source_height))
747 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_ROW_COUNT");
748 if (lsz != fd.leaves_bytes || lsz % sizeof(LeafRecord))
749 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_LEAVES_SIZE");

751 fd.rows.resize(static_cast<std::size_t>(fd.source_height));
752 {
753 std::ifstream in(rp, std::ios::binary);
754 in.read(reinterpret_cast<char *>(fd.rows.data()),
755 static_cast<std::streamsize>(rsz));
756 if (!in) throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_ROWS_READ");
757 }
758 // Offsets must tile [0, lsz) exactly, in order, aligned.
759 std::uint64_t expect_off = 0;
760 std::uint64_t leaf_sum = 0;
761 for (const auto &re : fd.rows) {
762 if (re.record_offset != expect_off ||
763 re.record_offset % sizeof(LeafRecord) != 0)
764 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_ROW_OFFSET");
765 const std::uint64_t block = re.record_count * sizeof(LeafRecord);
766 if (re.record_offset + block > lsz)
767 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_ROW_BOUNDS");
768 expect_off += block;
769 leaf_sum += re.record_count;
770 if (re.record_count &&
771 !(re.canvas_ymin <= re.canvas_ymax))
772 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_ROW_YRANGE");
773 }
774 if (expect_off != lsz ||
775 leaf_sum != fj.at("leaves").get<std::uint64_t>())
776 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_LEAF_SUM");
777
778 fd.leaves_path = lp;
779 if (fd.excluded)
780 impl_->excluded.emplace_back(fj.at("frame_id").get<std::string>(),
781 fd.subdivision_error_rate);
782 impl_->resident += fd.rows.size() * sizeof(RowEntry) + sizeof(fd);
783 got.push_back(sidx);
784 if (!vd.frames.emplace(sidx, std::move(fd)).second)
785 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_DUP_FRAME");
786 }
787 std::sort(got.begin(), got.end());
788 if (got != want)
789 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_POPULATION");
790 impl_->variants.push_back(std::move(vd));
791 }

820void DrizzleGeometryCacheReader::enumerate_stripe(
821 float pixfrac, std::size_t source_index, int scale, int y_begin, int rows,
822 const DrizzleLeafCellSink &sink) const {
823 impl_->enumerate_calls.fetch_add(1, std::memory_order_relaxed);
824 const auto *v = impl_->find_variant(pixfrac);
825 if (!v) throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_UNKNOWN_VARIANT");
826 auto it = v->frames.find(source_index);
827 if (it == v->frames.end())
828 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_UNKNOWN_FRAME");
829 const auto &fd = it->second;
830 if (scale != v->internal_scale)
831 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_SCALE_MISMATCH");
832 const int Wi = v->canvas_width_native * scale;
833 const double y_lo = static_cast<double>(y_begin);
834 const double y_hi = static_cast<double>(y_begin + rows);
835
836 std::ifstream in(fd.leaves_path, std::ios::binary);
837 if (!in) throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_LEAVES_OPEN");
838 std::vector<LeafRecord> block; // reused per row
839
840 for (int sy = 0; sy < fd.source_height; ++sy) {
841 const RowEntry &re = fd.rows[static_cast<std::size_t>(sy)];
842 if (re.record_count == 0) continue;
843 if (!(re.canvas_ymin < y_hi && re.canvas_ymax > y_lo)) continue;
844
845 block.resize(static_cast<std::size_t>(re.record_count));
846 in.seekg(static_cast<std::streamoff>(re.record_offset));
847 in.read(reinterpret_cast<char *>(block.data()),
848 static_cast<std::streamsize>(re.record_count * sizeof(LeafRecord)));
849 if (!in) throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_STRIPE_READ");
850
851 for (const LeafRecord &rec : block) {
852 const double xmin = *std::min_element(rec.x, rec.x + 4);
853 const double xmax = *std::max_element(rec.x, rec.x + 4);
854 const double ymin = *std::min_element(rec.y, rec.y + 4);
855 const double ymax = *std::max_element(rec.y, rec.y + 4);
856 const int x0 = static_cast<int>(
857 std::clamp(std::floor(xmin), 0.0, static_cast<double>(Wi)));
858 const int x1 = static_cast<int>(
859 std::clamp(std::ceil(xmax), 0.0, static_cast<double>(Wi)));
860 const int cy0 = static_cast<int>(std::clamp(
861 std::floor(ymin), static_cast<double>(y_begin),
862 static_cast<double>(y_begin + rows)));
863 const int cy1 = static_cast<int>(std::clamp(
864 std::ceil(ymax), static_cast<double>(y_begin),
865 static_cast<double>(y_begin + rows)));
866 for (int y = cy0; y < cy1; ++y)
867 for (int x = x0; x < x1; ++x)
868 sink(static_cast<int>(rec.source_x), sy, rec.channel, rec.leaf_order,
869 x, y, rec.x, rec.y);
870 }
871 }

874void DrizzleGeometryCacheReader::read_stripe_leaves_into(
875 float pixfrac, std::size_t source_index, int scale, int y_begin, int rows,
876 DrizzleCachedLeafWindow &out) const {
877 impl_->enumerate_calls.fetch_add(1, std::memory_order_relaxed);
878 const auto *v = impl_->find_variant(pixfrac);
879 if (!v) throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_UNKNOWN_VARIANT");
880 auto it = v->frames.find(source_index);
881 if (it == v->frames.end())
882 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_UNKNOWN_FRAME");
883 const auto &fd = it->second;
884 if (scale != v->internal_scale)
885 throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_SCALE_MISMATCH");
886 const double y_lo = static_cast<double>(y_begin);
887 const double y_hi = static_cast<double>(y_begin + rows);
888
889 out.leaves.clear();
890 out.source_x0 = out.source_y0 = out.source_x1 = out.source_y1 = 0;
891 out.unique_source_samples = 0;
892
893 std::ifstream in(fd.leaves_path, std::ios::binary);
894 if (!in) throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_LEAVES_OPEN");
895 // Raw on-disk row blocks stage through out.io_scratch; each record is
896 // memcpy'd into an aligned LeafRecord before its fields are read.
897 std::vector<std::uint8_t> &block = out.io_scratch;
898
899 std::uint64_t prev_sx = 0, prev_sy = 0;
900 bool have_any = false;
901 std::uint64_t bytes_read = 0;
902 for (int sy = 0; sy < fd.source_height; ++sy) {
903 const RowEntry &re = fd.rows[static_cast<std::size_t>(sy)];
904 if (re.record_count == 0) continue;
905 if (!(re.canvas_ymin < y_hi && re.canvas_ymax > y_lo)) continue;
906
907 block.resize(static_cast<std::size_t>(re.record_count) *
908 sizeof(LeafRecord));
909 in.seekg(static_cast<std::streamoff>(re.record_offset));
910 in.read(reinterpret_cast<char *>(block.data()),
911 static_cast<std::streamsize>(re.record_count * sizeof(LeafRecord)));
912 if (!in) throw std::runtime_error("DRIZZLE_GEOMETRY_CACHE_STRIPE_READ");
913 bytes_read += re.record_count * sizeof(LeafRecord);
914
915 for (std::uint64_t ri = 0; ri < re.record_count; ++ri) {
916 LeafRecord rec;
917 static_assert(std::is_trivially_copyable_v<LeafRecord>);
918 std::memcpy(&rec, block.data() + ri * sizeof(LeafRecord),
919 sizeof(LeafRecord));
920 const double ymin = *std::min_element(rec.y, rec.y + 4);
921 const double ymax = *std::max_element(rec.y, rec.y + 4);
922 if (!(ymin < y_hi && ymax > y_lo)) continue;
923 DrizzleCachedLeaf leaf;
924 leaf.source_x = rec.source_x;
925 leaf.source_y = static_cast<std::uint32_t>(sy);
926 leaf.channel = rec.channel;
927 leaf.leaf_order = rec.leaf_order;
928 for (int i = 0; i < 4; ++i) {
929 leaf.x[i] = rec.x[i];
930 leaf.y[i] = rec.y[i];
931 }
932 out.leaves.push_back(leaf);
933 // Canonical (sy, sx, leaf_order) order: a (sy, sx) transition marks a
934 // new covered source sample.
935 if (!have_any || rec.source_x != prev_sx ||
936 static_cast<std::uint64_t>(sy) != prev_sy) {
937 ++out.unique_source_samples;
938 prev_sx = rec.source_x;
939 prev_sy = static_cast<std::uint64_t>(sy);
940 }
941 have_any = true;
942 out.source_x0 = out.leaves.size() == 1
943 ? static_cast<int>(rec.source_x)
944 : std::min(out.source_x0,
945 static_cast<int>(rec.source_x));
946 out.source_x1 =
947 std::max(out.source_x1, static_cast<int>(rec.source_x) + 1);
948 out.source_y0 = out.leaves.size() == 1
949 ? sy
950 : std::min(out.source_y0, sy);
951 out.source_y1 = std::max(out.source_y1, sy + 1);
952 }
953 }
954 if (!have_any) {
955 out.source_x0 = out.source_y0 = out.source_x1 = out.source_y1 = 0;
956 out.unique_source_samples = 0;
957 }
958 impl_->leaf_record_bytes.fetch_add(bytes_read, std::memory_order_relaxed);
959}

961std::uint64_t DrizzleGeometryCacheReader::max_stripe_leaf_records(
962 float pixfrac, int stripe_rows) const {
963 const auto *v = impl_->find_variant(pixfrac);
964 if (!v || stripe_rows <= 0) return 0;
965 std::uint64_t bound = 0;
966 for (const auto &[sidx, fd] : v->frames) {
967 (void)sidx;
968 if (fd.excluded) continue;
969 // A stripe [y0, y0+S) intersects a source row iff
970 // row.canvas_ymin < y0+S && row.canvas_ymax > y0, i.e. iff the stripe
971 // start lies in (row.ymin - S, row.ymax). The intersecting set only
972 // changes when y0 crosses one of those endpoints, so sweeping them
973 // gives the exact maximum intersecting record count for this frame.
974 std::vector<std::pair<double, std::int64_t>> ev;
975 ev.reserve(fd.rows.size() * 2);
976 for (const RowEntry &re : fd.rows) {
977 if (re.record_count == 0) continue;
978 ev.emplace_back(re.canvas_ymin - stripe_rows,
979 static_cast<std::int64_t>(re.record_count));
980 ev.emplace_back(re.canvas_ymax,
981 -static_cast<std::int64_t>(re.record_count));
982 }
983 // Conservative tie order: opens before closes at equal positions.
984 std::sort(ev.begin(), ev.end(),
985 [](const auto &a, const auto &b) {
986 return a.first < b.first ||
987 (a.first == b.first && a.second > b.second);
988 });
989 std::int64_t cur = 0, best = 0;
990 for (const auto &[pos, delta] : ev) {
991 (void)pos;
992 cur += delta;
993 best = std::max(best, cur);
994 }
995 bound = std::max(bound, static_cast<std::uint64_t>(best));
996 }
997 return bound;
998}
