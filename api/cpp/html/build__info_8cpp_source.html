24nlohmann::json feature_flags_json(const std::string& encoded) {
25 nlohmann::json flags = nlohmann::json::object();
26 std::istringstream stream(encoded);
27 std::string item;
28 while (std::getline(stream, item, '|')) {
29 const auto separator = item.find('=');
30 if (separator == std::string::npos || separator == 0) continue;
31 const std::string key = item.substr(0, separator);
32 const std::string value = item.substr(separator + 1);
33 std::string lower = value;
34 std::transform(lower.begin(), lower.end(), lower.begin(),
35 [](unsigned char c) {
36 return static_cast<char>(std::tolower(c));
37 });
38 if (lower == "on" || lower == "true" || lower == "1") {
39 flags[key] = true;
40 } else if (lower == "off" || lower == "false" || lower == "0") {
41 flags[key] = false;
42 } else {
43 flags[key] = value;
44 }
45 }
46 return flags;
47}

51std::filesystem::path current_executable_path() {
52#ifdef _WIN32
53 std::vector<wchar_t> buffer(32768);
54 const DWORD length = GetModuleFileNameW(
55 nullptr, buffer.data(), static_cast<DWORD>(buffer.size()));
56 if (length > 0 && length < buffer.size()) {
57 return std::filesystem::path(std::wstring(buffer.data(), length));
58 }
59#elif defined(__APPLE__)
60 uint32_t size = 0;
61 _NSGetExecutablePath(nullptr, &size);
62 std::vector<char> buffer(size + 1u, '\0');
63 if (_NSGetExecutablePath(buffer.data(), &size) == 0) {
64 std::error_code ec;
65 auto canonical = std::filesystem::weakly_canonical(buffer.data(), ec);
66 return ec ? std::filesystem::path(buffer.data()) : canonical;
67 }
68#else
69 std::error_code ec;
70 auto path = std::filesystem::read_symlink("/proc/self/exe", ec);
71 if (!ec && !path.empty()) return path;
72#endif
73 return {};
74}

76nlohmann::json binary_provenance_json(const std::filesystem::path& path) {
77 nlohmann::json result = {
78 {"path", path.empty() ? std::string() : path.string()},
79 {"exists", false},
80 {"size_bytes", nullptr},
81 {"sha256", nullptr}
82 };
83 if (path.empty()) {
84 result["error"] = "executable_path_unavailable";
85 return result;
86 }
87
88 std::error_code ec;
89 const bool exists = std::filesystem::is_regular_file(path, ec);
90 result["exists"] = exists && !ec;
91 if (!exists || ec) {
92 result["error"] = ec ? ec.message() : "not_a_regular_file";
93 return result;
94 }
95
96 const auto size = std::filesystem::file_size(path, ec);
97 if (!ec) result["size_bytes"] = size;
98 try {
99 result["sha256"] = sha256_file(path);
100 } catch (const std::exception& e) {
101 result["error"] = e.what();
102 }
103 return result;
104}
