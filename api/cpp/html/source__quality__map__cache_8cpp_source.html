50void write_bin_atomic(const fs::path &target, int storage_w, int storage_h,
51 int divisor, std::size_t source_index,
52 const std::vector<uint16_t> &cells,
53 const std::vector<uint8_t> &veto_cells) {
54 core::AtomicOutput out(target);
55 {
56 std::ofstream f(out.path(), std::ios::binary);
57 if (!f) throw std::runtime_error("SQM_CACHE_BIN_OPEN_FAILED: " +
58 target.string());
59 ByteSink h;
60 h.bytes.insert(h.bytes.end(), kBinMagic, kBinMagic + 4);
61 h.u32(kBinSchema);
62 h.u32(static_cast<uint32_t>(storage_w));
63 h.u32(static_cast<uint32_t>(storage_h));
64 h.u32(static_cast<uint32_t>(divisor));
65 h.u64(static_cast<uint64_t>(source_index));
66 f.write(reinterpret_cast<const char *>(h.bytes.data()),
67 static_cast<std::streamsize>(h.bytes.size()));
68 std::vector<uint8_t> body(cells.size() * 2);
69 for (std::size_t i = 0; i < cells.size(); ++i) {
70 body[2 * i] = static_cast<uint8_t>(cells[i] & 0xff);
71 body[2 * i + 1] = static_cast<uint8_t>((cells[i] >> 8) & 0xff);
72 }
73 f.write(reinterpret_cast<const char *>(body.data()),
74 static_cast<std::streamsize>(body.size()));
75 f.write(reinterpret_cast<const char *>(veto_cells.data()),
76 static_cast<std::streamsize>(veto_cells.size()));
77 if (!f) throw std::runtime_error("SQM_CACHE_BIN_WRITE_FAILED: " +
78 target.string());
79 }
80 out.commit();
81}

83// Fixed .bin header: magic(4) + u32 schema + u32 storage_w + u32 storage_h
84// + u32 divisor + u64 source_index. Cells (storage_w*storage_h uint16 LE,
85// row-major) follow at kBinHeaderBytes; the veto block (one uint8 per cell)
86// follows the cells.
87constexpr std::streamoff kBinHeaderBytes = 4 + 4 * 4 + 8;
88
89struct BinHeader {
90 int storage_w = 0, storage_h = 0, divisor = 0;
91 std::size_t source_index = 0;
92};

94BinHeader read_bin_header(std::ifstream &f, const fs::path &path) {
95 char magic[4];
96 f.read(magic, 4);
97 if (!f || std::memcmp(magic, kBinMagic, 4) != 0)
98 throw std::runtime_error("SQM_CACHE_BIN_BAD_MAGIC: " + path.string());
99 auto rd_u32 = [&]() -> uint32_t {
100 uint8_t b[4];
101 f.read(reinterpret_cast<char *>(b), 4);
102 return static_cast<uint32_t>(b[0]) | (static_cast<uint32_t>(b[1]) << 8) |
103 (static_cast<uint32_t>(b[2]) << 16) |
104 (static_cast<uint32_t>(b[3]) << 24);
105 };
106 if (rd_u32() != kBinSchema)
107 throw std::runtime_error("SQM_CACHE_BIN_BAD_SCHEMA: " + path.string());
108 BinHeader h;
109 h.storage_w = static_cast<int>(rd_u32());
110 h.storage_h = static_cast<int>(rd_u32());
111 h.divisor = static_cast<int>(rd_u32());
112 const uint32_t lo = rd_u32();
113 const uint32_t hi = rd_u32();
114 h.source_index =
115 static_cast<std::size_t>(lo) | (static_cast<std::size_t>(hi) << 32);
116 if (!f || h.storage_w <= 0 || h.storage_h <= 0 || h.divisor <= 0)
117 throw std::runtime_error("SQM_CACHE_BIN_BAD_DIMS: " + path.string());
118 return h;
119}

121// §30.81 step 3a-2b: read ONLY the storage cells covering
122// [cy0, cy1] x [cx0, cx1] (inclusive) --- one contiguous seek+read per cell
123// row, for the value block and the veto block. `win` is
124// (cy1-cy0+1) x (cx1-cx0+1), row-major; caller rebases (cy - cy0, cx - cx0).
125struct BinWindow {
126 int storage_w = 0, storage_h = 0, divisor = 0;
127 int cy0 = 0, cx0 = 0, win_h = 0, win_w = 0;
128 std::vector<uint16_t> cells;
129 std::vector<uint8_t> veto;
130};

132// Same coverage as read_bin_window but fills caller-owned buffers in place
133// (resize, capacity preserved): each storage row is seek-read straight into
134// the cells/veto spans --- no row scratch buffer. Cells are stored LE; on a
135// big-endian host they are byteswapped in place after the read.
136void read_bin_window_into(const fs::path &path, int cy0, int cy1, int cx0,
137 int cx1, int &storage_w, int &storage_h,
138 int &divisor, int &win_cy0, int &win_cx0,
139 int &win_h, int &win_w,
140 std::vector<uint16_t> &cells,
141 std::vector<uint8_t> &veto) {
142 std::ifstream f(path, std::ios::binary);
143 if (!f) throw std::runtime_error("SQM_CACHE_BIN_MISSING: " + path.string());
144 const BinHeader h = read_bin_header(f, path);
145 cy0 = std::clamp(cy0, 0, h.storage_h - 1);
146 cy1 = std::clamp(cy1, cy0, h.storage_h - 1);
147 cx0 = std::clamp(cx0, 0, h.storage_w - 1);
148 cx1 = std::clamp(cx1, cx0, h.storage_w - 1);
149 storage_w = h.storage_w;
150 storage_h = h.storage_h;
151 divisor = h.divisor;
152 win_cy0 = cy0;
153 win_cx0 = cx0;
154 win_h = cy1 - cy0 + 1;
155 win_w = cx1 - cx0 + 1;
156 const std::size_t run = static_cast<std::size_t>(win_w);
157 const std::size_t total = run * static_cast<std::size_t>(win_h);
158 const std::streamoff n_cells = static_cast<std::streamoff>(h.storage_w) *
159 static_cast<std::streamoff>(h.storage_h);
160 const std::streamoff veto_base = kBinHeaderBytes + n_cells * 2;
161 cells.resize(total);
162 veto.resize(total);
163 for (int cy = cy0; cy <= cy1; ++cy) {
164 const std::streamoff first =
165 static_cast<std::streamoff>(cy) * h.storage_w + cx0;
166 const std::size_t out = static_cast<std::size_t>(cy - cy0) * run;
167 f.seekg(kBinHeaderBytes + first * 2, std::ios::beg);
168 f.read(reinterpret_cast<char *>(cells.data() + out),
169 static_cast<std::streamsize>(run * 2));
170 if (static_cast<std::size_t>(f.gcount()) != run * 2)
171 throw std::runtime_error("SQM_CACHE_BIN_TRUNCATED: " + path.string());
172 if constexpr (std::endian::native == std::endian::big)
173 for (std::size_t i = 0; i < run; ++i) {
174 const uint16_t v = cells[out + i];
175 cells[out + i] = static_cast<uint16_t>((v >> 8) | (v << 8));
176 }
177 f.seekg(veto_base + first, std::ios::beg);
178 f.read(reinterpret_cast<char *>(veto.data() + out),
179 static_cast<std::streamsize>(run));
180 if (static_cast<std::size_t>(f.gcount()) != run)
181 throw std::runtime_error("SQM_CACHE_BIN_TRUNCATED_VETO: " + path.string());
182 }

237uint16_t quantize_quality(float v) {
238 if (!finite_f(v) || v <= 0.0f) return 0u;
239 const float clamped = v > 1.0f ? 1.0f : v;
240 const long q = std::lround(static_cast<double>(clamped) * 65535.0);
241 if (q <= 0) return 1u;
242 if (q >= 65535) return 65535u;
243 return static_cast<uint16_t>(q);
244}

246float dequantize_quality(uint16_t q) {
247 if (q == 0u) return nan_value();
248 return static_cast<float>(q) / 65535.0f;
249}

314void SourceQualityMapCacheWriter::put(const std::string &stream,
315 std::size_t source_index,
316 const Matrix2Df &m) {
317 if (m.rows() != source_height_ || m.cols() != source_width_)
318 throw std::invalid_argument("SQM_CACHE_MAP_GEOMETRY_MISMATCH");
319 const int d = cfg_.storage_divisor;
320 const int sw = storage_dim(source_width_, d);
321 const int sh = storage_dim(source_height_, d);
322 const std::size_t ncells = static_cast<std::size_t>(sw) * sh;
323 std::vector<uint16_t> cells(ncells, 0u);
324 std::vector<uint8_t> veto_cells(ncells, 0u);
325
326 // Downsample (plan 13.5): the value cell is the valid-mean over strictly
327 // positive covered source pixels, so good data next to an unsupported
328 // (NaN) border survives. A SEPARATE hard-veto cell is set when ANY covered
329 // source pixel is an exact Q=0 hard veto (finite and <= 0); the read path
330 // then forces NaN there so an exact zero-veto can never resample positive.
331 // Cells with neither positive nor hard-veto data (all NaN = no support)
332 // store value 0 -> decode NaN, veto 0.
333 for (int cy = 0; cy < sh; ++cy) {
334 for (int cx = 0; cx < sw; ++cx) {
335 double sum = 0.0;
336 int count = 0;
337 bool hard_veto = false;
338 for (int y = cy * d; y < std::min(source_height_, (cy + 1) * d); ++y) {
339 for (int x = cx * d; x < std::min(source_width_, (cx + 1) * d); ++x) {
340 const float v = m(y, x);
341 if (!finite_f(v)) continue; // no support -- neither veto nor data
342 if (v <= 0.0f) {
343 hard_veto = true; // explicit Q=0 hard veto
344 } else {
345 sum += v;
346 ++count;
347 }
348 }
349 }
350 const std::size_t idx =
351 static_cast<std::size_t>(cy) * static_cast<std::size_t>(sw) + cx;
352 cells[idx] = count == 0
353 ? 0u
354 : quantize_quality(static_cast<float>(sum / count));
355 veto_cells[idx] = hard_veto ? 1u : 0u;
356 }
357 }
358
359 const std::string name = stream_file_name(stream, source_index);
360 const fs::path target = root_ / name;
361 {
362 // AtomicOutput's staging dir lives under target.parent_path(), so the
363 // stream directory must exist before write_bin_atomic. Concurrent
364 // create_directories on the same path can throw on some libstdc++, so
365 // serialise just this step.
366 std::lock_guard<std::mutex> lk(files_mu_);
367 fs::create_directories(target.parent_path());
368 }
369 // Lock-free: distinct (stream, source_index) -> distinct file, written via
370 // AtomicOutput (its own unique staging dir).
371 write_bin_atomic(target, sw, sh, d, source_index, cells, veto_cells);
372
373 SourceQualityCacheFileEntry e;
374 e.stream = stream;
375 e.source_index = source_index;
376 e.name = name;
377 e.bytes = fs::file_size(target); // T1: size only, no SHA-256
378 {
379 std::lock_guard<std::mutex> lk(files_mu_);
380 files_.erase(std::remove_if(files_.begin(), files_.end(),
381 [&](const SourceQualityCacheFileEntry &fe) {
382 return fe.stream == stream &&
383 fe.source_index == source_index;
384 }),
385 files_.end());
386 files_.push_back(std::move(e));
387 }

555void SourceQualityMapCacheReader::read_packed_rect_into(
556 const std::string &stream, std::size_t source_index, int y0, int y1,
557 int x0, int x1, SourceQualityPackedWindow &out) const {
558 if (!usable_) throw std::runtime_error("SQM_CACHE_NOT_USABLE: " + error_);
559 y0 = std::max(0, y0);
560 y1 = std::min(meta_.source_height, y1);
561 x0 = std::max(0, x0);
562 x1 = std::min(meta_.source_width, x1);
563 out.storage_x_begin = 0;
564 out.storage_y_begin = 0;
565 out.storage_width = 0;
566 out.storage_height = 0;
567 out.storage_divisor = 1;
568 out.source_x_begin = 0;
569 out.source_y_begin = 0;
570 out.source_width = 0;
571 out.source_height = 0;
572 out.cells.clear(); // keep capacity for the caller's reusable window
573 out.veto.clear();
574 if (y1 <= y0 || x1 <= x0) return;
575 const int d = meta_.storage_divisor;
576 // §30.81 step 3a-2b: seek-read only the storage cells covering the rect,
577 // straight into the caller's cells/veto spans.
578 int bin_w = 0, bin_h = 0, bin_d = 0;
579 read_bin_window_into(file_path(stream, source_index), y0 / d, (y1 - 1) / d,
580 x0 / d, (x1 - 1) / d, bin_w, bin_h, bin_d,
581 out.storage_y_begin, out.storage_x_begin,
582 out.storage_height, out.storage_width, out.cells,
583 out.veto);
584 if (bin_d != d || bin_w != storage_dim(meta_.source_width, d) ||
585 bin_h != storage_dim(meta_.source_height, d))
586 throw std::runtime_error("SQM_CACHE_BIN_GEOMETRY_MISMATCH: " + stream);
587 bin_loads_.fetch_add(1, std::memory_order_relaxed);
588 bin_cells_decoded_.fetch_add(
589 static_cast<std::uint64_t>(out.storage_height) *
590 static_cast<std::uint64_t>(out.storage_width),
591 std::memory_order_relaxed);
592 out.storage_divisor = d;
593 out.source_x_begin = x0;
594 out.source_y_begin = y0;
595 out.source_width = x1 - x0;
596 out.source_height = y1 - y0;
597}

599void SourceQualityMapCacheReader::read_packed_samples_into(
600 const std::string &stream, std::size_t source_index,
601 const std::vector<DrizzleAffineSourceSpan> &spans,
602 std::vector<std::uint16_t> &cells, std::vector<std::uint8_t> &veto,
603 SourceQualityPackedWindow &scratch) const {
604 if (!usable_) throw std::runtime_error("SQM_CACHE_NOT_USABLE: " + error_);
605 cells.clear();
606 veto.clear();
607 scratch.cells.clear();
608 scratch.veto.clear();
609 if (spans.empty()) return;
610 const int d = meta_.storage_divisor;
611 const int full_w = storage_dim(meta_.source_width, d);
612 const int full_h = storage_dim(meta_.source_height, d);
613 std::size_t total = 0;
614 int prev_sy = -1;
615 for (const auto &s : spans) {
616 // Canonical contract: strictly increasing source rows, in-extent,
617 // non-empty intervals.
618 if (s.source_y < 0 || s.source_y >= meta_.source_height ||
619 s.source_y <= prev_sy || s.x_begin < 0 ||
620 s.x_end > meta_.source_width || s.x_end <= s.x_begin)
621 throw std::invalid_argument("SQM_CACHE_INVALID_SPAN");
622 prev_sy = s.source_y;
623 total += static_cast<std::size_t>(s.x_end - s.x_begin);
624 }
625 cells.resize(total); // resize keeps capacity for the reusable buffers
626 veto.resize(total);
627 const fs::path path = file_path(stream, source_index);
628 std::ifstream f(path, std::ios::binary);
629 if (!f) throw std::runtime_error("SQM_CACHE_BIN_MISSING: " + path.string());
630 const BinHeader h = read_bin_header(f, path);
631 if (h.divisor != d || h.storage_w != full_w || h.storage_h != full_h)
632 throw std::runtime_error("SQM_CACHE_BIN_GEOMETRY_MISMATCH: " + stream);
633 const std::streamoff n_cells = static_cast<std::streamoff>(h.storage_w) *
634 static_cast<std::streamoff>(h.storage_h);
635 const std::streamoff veto_base = kBinHeaderBytes + n_cells * 2;
636 std::uint64_t cells_read = 0;
637 std::size_t out_off = 0;
638 std::size_t i = 0;
639 while (i < spans.size()) {
640 // Group the consecutive spans sharing one storage row and seek-read its
641 // union cell interval exactly once.
642 const int cy = std::min(full_h - 1, spans[i].source_y / d);
643 int cx0 = std::min(full_w - 1, spans[i].x_begin / d);
644 int cx1 = std::min(full_w - 1, (spans[i].x_end - 1) / d);
645 std::size_t j = i + 1;
646 while (j < spans.size() &&
647 std::min(full_h - 1, spans[j].source_y / d) == cy) {
648 cx0 = std::min(cx0, std::min(full_w - 1, spans[j].x_begin / d));
649 cx1 = std::max(cx1, std::min(full_w - 1, (spans[j].x_end - 1) / d));
650 ++j;
651 }
652 const int run = cx1 - cx0 + 1;
653 scratch.cells.resize(static_cast<std::size_t>(run));
654 scratch.veto.resize(static_cast<std::size_t>(run));
655 const std::streamoff first =
656 static_cast<std::streamoff>(cy) * h.storage_w + cx0;
657 f.seekg(kBinHeaderBytes + first * 2, std::ios::beg);
658 f.read(reinterpret_cast<char *>(scratch.cells.data()),
659 static_cast<std::streamsize>(run) * 2);
660 if (f.gcount() != static_cast<std::streamsize>(run) * 2)
661 throw std::runtime_error("SQM_CACHE_BIN_TRUNCATED: " + path.string());
662 if constexpr (std::endian::native == std::endian::big)
663 for (int k = 0; k < run; ++k) {
664 const uint16_t v = scratch.cells[static_cast<std::size_t>(k)];
665 scratch.cells[static_cast<std::size_t>(k)] =
666 static_cast<uint16_t>((v >> 8) | (v << 8));
667 }
668 f.seekg(veto_base + first, std::ios::beg);
669 f.read(reinterpret_cast<char *>(scratch.veto.data()),
670 static_cast<std::streamsize>(run));
671 if (f.gcount() != static_cast<std::streamsize>(run))
672 throw std::runtime_error("SQM_CACHE_BIN_TRUNCATED_VETO: " +
673 path.string());
674 cells_read += static_cast<std::uint64_t>(run);
675 for (std::size_t k = i; k < j; ++k) {
676 const auto &s = spans[k];
677 for (int sx = s.x_begin; sx < s.x_end; ++sx) {
678 const int cx = std::min(full_w - 1, sx / d);
679 cells[out_off] = scratch.cells[static_cast<std::size_t>(cx - cx0)];
680 veto[out_off] = scratch.veto[static_cast<std::size_t>(cx - cx0)];
681 ++out_off;
682 }
683 }
684 i = j;
685 }
686 bin_loads_.fetch_add(1, std::memory_order_relaxed);
687 bin_cells_decoded_.fetch_add(cells_read, std::memory_order_relaxed);
688}

698Matrix2Df SourceQualityMapCacheReader::read_rect(const std::string &stream,
699 std::size_t source_index,
700 int y0, int y1, int x0,
701 int x1) const {
702 const SourceQualityPackedWindow w =
703 read_packed_rect(stream, source_index, y0, y1, x0, x1);
704 if (w.source_width <= 0 || w.source_height <= 0) return Matrix2Df(0, 0);
705 const int x0c = w.source_x_begin, y0c = w.source_y_begin;
706 const int x1c = x0c + w.source_width, y1c = y0c + w.source_height;
707 const int d = w.storage_divisor;
708 // Full-grid dims for the absolute edge clamp, recomputed from the clamped
709 // window origin (storage cells per row of the full grid).
710 const int full_w = storage_dim(meta_.source_width, d);
711 const int full_h = storage_dim(meta_.source_height, d);
712 expanded_floats_.fetch_add(
713 static_cast<std::uint64_t>(w.source_height) *
714 static_cast<std::uint64_t>(w.source_width),
715 std::memory_order_relaxed);
716
717 Matrix2Df out(w.source_height, w.source_width);
718 for (int y = y0c; y < y1c; ++y) {
719 // Storage cell picked from the ABSOLUTE y (edge-clamp unchanged), then
720 // rebased into the window that was actually read.
721 const int cy = std::min(full_h - 1, y / d);
722 const std::size_t wy =
723 static_cast<std::size_t>(cy - w.storage_y_begin) * w.storage_width;
724 for (int x = x0c; x < x1c; ++x) {
725 const int cx = std::min(full_w - 1, x / d); // ABSOLUTE x
726 const std::size_t wi =
727 wy + static_cast<std::size_t>(cx - w.storage_x_begin);
728 // Hard-veto cell forces NaN regardless of the value cell (plan 13.5).
729 out(y - y0c, x - x0c) =
730 w.veto[wi] ? std::numeric_limits<float>::quiet_NaN()
731 : dequantize_quality(w.cells[wi]);
732 }
733 }
734 return out;
735}

744
745SourceQualityMapsBuildResult build_source_quality_map_cache(
746 const fs::path &cache_root,
747 const registration::RegistrationSamplingPlan &plan,
748 VerifiedNormalizedSourceCache &cache,
749 const config::ReconstructionQualityPyramidConfig &pyramid,
750 SourceQualityMapCacheConfig cache_cfg, int workers) {
751 const std::string normalized_cache_hash = cache.manifest_hash();
752 SourceQualityMapCacheWriter writer(cache_root, plan, normalized_cache_hash,
753 pyramid, cache_cfg);
754
755 SourceQualityMapsBuildResult r;
756 r.source_identity_hash = writer.identity_hash();
757 r.source_quality_config_hash = writer.config_hash();
758
759 std::vector<const registration::FrameSamplingTransform *> valid;
760 for (const auto &f : plan.frames)
761 if (f.valid)
762 valid.push_back(&f);
763 const int nf = static_cast<int>(valid.size());
764 int nw = std::max(1, std::min(workers, nf));
765
766 // T3: per-frame metrics collection (optional, when star_max_corners > 0).
767 const bool compute_metrics = cache_cfg.star_max_corners > 0;
768 std::vector<SourceQualityFrameMetrics> all_metrics;
769 if (compute_metrics) all_metrics.resize(nf);
770
771 int max_scales = 0;
772 // `failed` is the lock-free fast-path check; `worker_error` is only ever
773 // touched inside the sqm_error critical section. A plain read of
774 // std::exception_ptr racing with its assignment would be UB.
775 std::atomic<bool> failed{false};
776 std::exception_ptr worker_error;
777 auto record_error = [&] {
778#pragma omp critical(sqm_error)
779 if (!worker_error)
780 worker_error = std::current_exception();
781 failed.store(true, std::memory_order_relaxed);
782 };
783
784 // One process for each frame: load (own cache clone) -> proxy -> quality
785 // maps, buffered locally; then a short critical section flushes the writer.
786 // The writer sorts its file list at commit(), so completion order does not
787 // affect the committed bytes or source_quality_cache_hash.
788 auto process_frame = [&](int idx, VerifiedNormalizedSourceCache &wc) {
789 const auto &f = *valid[static_cast<size_t>(idx)];
790 const Matrix2Df &source = wc.load(f.source_index);
791 const auto proxy = compute_source_quality_proxy_v1(
792 source, plan.color_mode, plan.bayer_pattern, plan.cfa_origin_x,
793 plan.cfa_origin_y);
794 const Matrix2Df &analysis = proxy.proxy_full;
795
796 std::vector<std::pair<std::string, Matrix2Df>> pending;
797 QualityScaleMapSink sink = [&](int scale_index, int /*downsample_factor*/,
798 const Matrix2Df &psi_source_geom) {
799 pending.emplace_back("scale_" + std::to_string(scale_index),
800 psi_source_geom);
801 };
802 const auto maps = compute_source_quality_maps(
803 analysis, /*source_valid_mask=*/{}, analysis.cols(), analysis.rows(),
804 pyramid, sink);
805 pending.emplace_back("composite", maps.q_map);
806 pending.emplace_back("artifact", maps.artifact_confidence);
807
808 // writer.put is internally thread-safe (plan §30.72 R4): the heavy
809 // downsample/quantize/write runs concurrently; only the file-list
810 // update is briefly locked. Only the tiny scalar reduction needs a
811 // critical here.
812 for (const auto &[stream, mtx] : pending)
813 writer.put(stream, f.source_index, mtx);
814
815 // T3: compute per-frame metrics from the already-computed proxy.
816 // FrameStarMetrics.wfwhm is computed with ref_star_count=0 (=> wfwhm=fwhm);
817 // the weight calculation does not use wfwhm, so this is safe.
818 if (compute_metrics) {
819 const auto fm = metrics::calculate_frame_metrics(analysis);
820 const auto sm = metrics::measure_frame_stars(
821 analysis, /*ref_star_count=*/0, cache_cfg.star_max_corners,
822 cache_cfg.star_patch_radius);
823 SourceQualityFrameMetrics &out = all_metrics[static_cast<size_t>(idx)];
824 out.source_index = f.source_index;
825 out.frame_id = f.frame_id;
826 out.background = fm.background;
827 out.noise = fm.noise;
828 out.gradient_energy = fm.gradient_energy;
829 out.sky_gradient = fm.sky_gradient;
830 out.quality_score = fm.quality_score;
831 out.fwhm = sm.fwhm;
832 out.fwhm_x = sm.fwhm_x;
833 out.fwhm_y = sm.fwhm_y;
834 out.roundness = sm.roundness;
835 out.wfwhm = sm.wfwhm;
836 out.star_count = sm.star_count;
837 }
838
839#pragma omp critical(sqm_reduce)
840 {
841 max_scales = std::max(max_scales, maps.diagnostics.computed_scales);
842 ++r.frames;
843 }
844 };
845
846 if (nw <= 1) {
847 for (int i = 0; i < nf; ++i)
848 process_frame(i, cache);
849 } else {
850#ifdef _OPENMP
851 const size_t worker_mb =
852 cache.frame_byte_size() / (1024 * 1024) + 4; // ~2 frames resident
853#pragma omp parallel num_threads(nw)
854 {
855 // Per-worker cache clone: independent LRU, no shared load() state. The
856 // clone constructor can throw (budget check); a throw escaping the
857 // parallel region would call std::terminate, so catch it here. Every
858 // thread must still reach the `omp for` worksharing region below (a
859 // skipped one deadlocks the team), so a clone failure just sets `failed`
860 // and every iteration becomes a no-op.
861 std::optional<VerifiedNormalizedSourceCache> wc;
862 try {
863 wc.emplace(cache, worker_mb);
864 } catch (...) {
865 record_error();
866 }
867#pragma omp for schedule(dynamic, 1)
868 for (int i = 0; i < nf; ++i) {
869 if (failed.load(std::memory_order_relaxed) || !wc)
870 continue;
871 try {
872 process_frame(i, *wc);
873 } catch (...) {
874 record_error();
875 }
876 }
877 }
878#else
879 for (int i = 0; i < nf; ++i)
880 process_frame(i, cache);
881#endif
882 }
883 if (worker_error)
884 std::rethrow_exception(worker_error);
885
886 const auto meta = writer.commit();
887 r.source_quality_cache_hash = meta.source_quality_cache_hash;
888 r.streams = meta.streams;
889 r.computed_scales = max_scales;
890
891 // T3: write the metrics artifact sorted by source_index.
892 if (compute_metrics) {
893 std::sort(all_metrics.begin(), all_metrics.end(),
894 [](const SourceQualityFrameMetrics &a,
895 const SourceQualityFrameMetrics &b) {
896 return a.source_index < b.source_index;
897 });
898 SourceQualityMetricsArtifact ma;
899 ma.schema_version = 1;
900 ma.source_identity_hash = r.source_identity_hash;
901 ma.source_quality_config_hash = r.source_quality_config_hash;
902 ma.normalized_cache_hash = normalized_cache_hash;
903 ma.frames = std::move(all_metrics);
904 write_source_quality_metrics(cache_root / "source_quality_metrics-v1.json", ma);
905 }
906
907 return r;
908}
