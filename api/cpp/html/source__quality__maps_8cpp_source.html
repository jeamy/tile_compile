56 ++observed_scales;
57 // psi_src_in is `psi` already bilinearly upsampled to source geometry
58 // by compute_source_quality_map (plan: avoid running that identical
59 // interpolation a second time here). This map plus any retained in
60 // result.scale_maps are the only full source-geometry maps held here;
61 // the geometric-mean composite lives inside compute_source_quality_map

63 Matrix2Df psi_src = psi_src_in;
64 ++live_maps;
65 peak_live_maps = std::max(peak_live_maps, live_maps);
66
67 if (!captured_artifact) {
68 // Finest computed scale supplies artifact_confidence (least
69 // smoothed; native-sampling artefact response). 1 = clean.
70 //
71 // Plan 14.4: unlike the legacy diagnostic path, insufficient local
72 // support must be marked NOT APPLICABLE, not full confidence.
73 // Legacy phi_artifact returns exactly 1.0f where its window has < 3
74 // valid high-pass samples; those pixels coincide with where psi is
75 // NaN (psi needs finite z_sharp/z_snr/artifact, i.e. real support).
76 // So mask the artifact map to NaN wherever the finest psi is NaN
77 // before upsampling --- a conservative "not applicable" that keeps
78 // genuinely clean, well-supported pixels at their true value.
79 Matrix2Df artifact_masked = artifact;
80 for (int i = 0; i < artifact_masked.size(); ++i)
81 if (!finite_f(psi.data()[i]))
82 artifact_masked.data()[i] = nan_value();

84 artifact_masked, source_width, source_height, downsample_factor);
85 captured_artifact = true;
86 }

90 } else {

92 m.scale_index = scale_index;
93 m.downsample_factor = downsample_factor;
94 m.psi = std::move(psi_src);
95 result.scale_maps.push_back(std::move(m));
96 }
97 --live_maps;
98 };
