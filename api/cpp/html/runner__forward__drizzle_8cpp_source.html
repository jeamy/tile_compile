42// Process-lifetime peak RSS, in KiB (Linux ru_maxrss is already KiB). This is a
43// LIFETIME maximum -- the difference of two such readings is NOT phase-local
44// growth (plan 11.13(4)); it is reported only as `rss_process_peak_kib`.

52 if (getrusage(RUSAGE_SELF, &ru) != 0) return 0;
53 return static_cast<long long>(ru.ru_maxrss);
54#endif

56// Current / peak resident set size from /proc/self/status, in KiB. `VmRSS` is
57// the live figure used for phase-scoped growth (sampled at phase begin/end);
58// `VmHWM` is the process-lifetime high-water mark. 0 if unreadable.
59long long read_proc_status_kb(const char *key) {
60 std::ifstream st("/proc/self/status");
61 std::string line;
62 const std::string want = std::string(key) + ":";
63 while (std::getline(st, line)) {
64 if (line.rfind(want, 0) != 0) continue;
65 long long kb = 0;
66 bool seen = false;
67 for (std::size_t i = want.size(); i < line.size(); ++i) {
68 const char ch = line[i];
69 if (ch >= '0' && ch <= '9') { kb = kb * 10 + (ch - '0'); seen = true; }
70 else if (seen) break; // stop at the first token boundary after the number
71 }
72 return kb;
73 }
74 return 0;
75}

161// Config sections allowed to differ for a GLOBAL_QUALITY/FORWARD_DRIZZLE
162// resume: the reconstruction predecessors (NORMALIZED_CACHE, SAMPLING_
163// GEOMETRY, COMMON_OVERLAP, SOURCE_QUALITY_MAPS) are reused from cache and
164// never recomputed from the current config on this path -- they are only
165// validated against their OWN recorded checkpoint hashes, not against
166// these sections -- so a change confined to them cannot desync cache from
167// config. Verified by reading every cfg.* access in run_forward_drizzle_
168// stages: only reconstruction, global_metrics (GLOBAL_QUALITY only) and
169// runtime_limits, plus the astrometry/bge/pcc/hypermetric_stretch

189// Per-top-level-key YAML serialisation for the scoped resume comparison,
190// operating directly on YAML text (no config parse/validate needed here) --
191// mirrors the equivalent check the web backend runs as a preflight
192// (web_backend_cpp's downstream_resume_scope_violation).
193std::map<std::string,std::string> yaml_section_map(
194 const std::string &yaml_text) {
195 std::map<std::string,std::string> out;

197 const YAML::Node root = YAML::Load(yaml_text);

199 YAML::Emitter e;

202 }
203 } catch (...) {
204 }

217 // A fresh run (resume_from empty) must always match exactly -- there is
218 // no earlier phase to have legitimately changed config since. A

220 // allowed_reconstruction_resume_sections; anything else must still

229 const auto &allowed = allowed_reconstruction_resume_sections(resume_from);
230 std::set<std::string> keys;
231 for (const auto &[k,v] : before) keys.insert(k);
232 for (const auto &[k,v] : after) keys.insert(k);
233 std::string bad;
234 for (const auto &k : keys) {
235 if (allowed.count(k)) continue;
236 const auto ia = before.find(k), ib = after.find(k);
237 const std::string va =
238 ia==before.end() ? std::string("<absent>") : ia->second;
239 const std::string vb =
240 ib==after.end() ? std::string("<absent>") : ib->second;
241 if (va!=vb) { if (!bad.empty()) bad+=", "; bad+=k; }
242 }
243 if (!bad.empty())
244 throw std::runtime_error(
245 "FORWARD_STAGE_CONFIG_SCOPE_MISMATCH: "+resume_from+
246 " resume allows changes only in sections that do not feed the "
247 "reused reconstruction predecessors; changed outside scope: "+bad);
248 }
249 const auto identity=provenance.at("input_manifest").at("sha256").get<std::string>()+":"+
250 provenance.at("config").at("sha256").get<std::string>();
251 if (plan.source_identity_hash!=core::sha256_bytes(std::vector<uint8_t>(identity.begin(),identity.end())))
252 throw std::runtime_error("FORWARD_STAGE_SOURCE_IDENTITY_MISMATCH");
253 registration::RegistrationSamplingPlan parsed;
254 std::string error;
255 if (!registration::parse_from_json_string(registration::serialize_to_json_string(plan),parsed,error))
256 throw std::runtime_error("FORWARD_STAGE_INVALID_SAMPLING_PLAN: "+error);
257 return provenance;
258}

1408// Downstream-only resume phases handled by run_rgb_downstream directly:
1409// they reuse the persisted reconstruction outputs and never touch the
1410// expensive M1-M3 predecessors (sampling geometry / forward drizzle).
1411bool is_downstream_resume_phase(const std::string &p) {
1412 return p=="ASTROMETRY"||p=="BGE"||p=="PCC"||
1413 p=="HYPERMETRIC_STRETCH"||p=="HMS";
1414}
1415
1416// Config sections allowed to differ for a downstream-only resume: the
1417// sections consumed at-or-after the resume entry, plus non-numeric runtime
1418// limits. Every earlier section must stay identical so the reused
1419// predecessor artifacts remain valid for the effective configuration.
1420const std::set<std::string> &allowed_resume_sections(const std::string &p) {
1421 static const std::set<std::string> hms =
1422 {"hypermetric_stretch","runtime_limits"};
1423 static const std::set<std::string> pcc =
1424 {"pcc","chroma_denoise","hypermetric_stretch","runtime_limits"};
1425 static const std::set<std::string> bge =
1426 {"bge","pcc","chroma_denoise","luma_denoise","hypermetric_stretch",
1427 "runtime_limits"};
1428 static const std::set<std::string> astro =
1429 {"astrometry","bge","pcc","chroma_denoise","luma_denoise",
1430 "hypermetric_stretch","runtime_limits"};
1431 if (p=="HYPERMETRIC_STRETCH"||p=="HMS") return hms;
1432 if (p=="PCC") return pcc;
1433 if (p=="BGE") return bge;
1434 return astro;
1435}

1437// Serialised top-level config sections for the scoped resume comparison.
1438// Emitting each section separately yields a per-section diff so the error
1439// can name exactly which out-of-scope sections changed.
1440std::map<std::string,std::string> config_section_map(
1441 const config::Config &cfg) {
1442 std::map<std::string,std::string> out;
1443 const YAML::Node root = cfg.to_yaml();
1444 for (const auto &kv : root) {
1445 YAML::Emitter e;
1446 e << kv.second;
1447 out[kv.first.as<std::string>()] = e.c_str();
1448 }
1449 return out;

1452int resume_downstream_command(const fs::path &dir, const json &provenance,
1453 const std::string &phase_upper) {
1454 config::ConfigMigrationReport migration;
1455 const auto cfg=config::Config::from_yaml_text_migrated(
1456 core::read_text(dir/"config.yaml"),migration);
1457 cfg.validate();
1458
1459 const std::string orig_yaml = tile_compile::runner::original_run_config_yaml(dir,provenance);
1460 if (orig_yaml.empty())
1461 throw std::runtime_error("FORWARD_STAGE_CONFIG_OR_SCOPE_MISMATCH");
1462 config::ConfigMigrationReport orig_migration;
1463 const auto orig = config::Config::from_yaml_text_migrated(
1464 orig_yaml,orig_migration);
1465 const auto &allowed = allowed_resume_sections(
1466 phase_upper=="HMS" ? "HYPERMETRIC_STRETCH" : phase_upper);
1467 const auto before = config_section_map(orig);
1468 const auto after = config_section_map(cfg);
1469 std::set<std::string> keys;
1470 for (const auto &[k,v] : before) keys.insert(k);
1471 for (const auto &[k,v] : after) keys.insert(k);
1472 std::string bad;
1473 for (const auto &k : keys) {
1474 if (allowed.count(k)) continue;
1475 const auto ia = before.find(k), ib = after.find(k);
1476 const std::string va =
1477 ia==before.end() ? std::string("<absent>") : ia->second;
1478 const std::string vb =
1479 ib==after.end() ? std::string("<absent>") : ib->second;
1480 if (va!=vb) { if (!bad.empty()) bad+=", "; bad+=k; }
1481 }
1482 if (!bad.empty())
1483 throw std::runtime_error(
1484 "FORWARD_STAGE_CONFIG_SCOPE_MISMATCH: "+phase_upper+
1485 " resume allows changes only in the downstream sections; "
1486 "changed outside scope: "+bad);
1487
1488 // Fail closed on missing predecessor artifacts BEFORE touching the log.
1489 const fs::path outputs = dir/"outputs";
1490 const bool rgb_ok = fs::exists(outputs/"stacked_rgb.fits") ||
1491 fs::exists(outputs/"stacked_rgb_solve.fits");
1492 if (!rgb_ok)
1493 throw std::runtime_error(
1494 "FORWARD_STAGE_DOWNSTREAM_INPUT_MISSING: outputs/stacked_rgb.fits");
1495 if ((phase_upper=="HYPERMETRIC_STRETCH"||phase_upper=="HMS") &&
1496 !fs::exists(outputs/"stacked_rgb_pcc.fits"))
1497 throw std::runtime_error(
1498 "FORWARD_STAGE_DOWNSTREAM_INPUT_MISSING: outputs/stacked_rgb_pcc.fits");
1499
1500 std::ofstream log(dir/"logs/run_events.jsonl",std::ios::app);
1501 if (!log) throw std::runtime_error("FORWARD_STAGE_LOG_UNAVAILABLE");
1502 core::EventEmitter emitter;
1503 const std::string run_id = dir.filename().string();
1504 const auto t0 = std::chrono::steady_clock::now();
1505 const auto abort_fn = [&](const std::string &) {
1506 return std::chrono::duration<double>(
1507 std::chrono::steady_clock::now()-t0).count()
1508 > cfg.runtime_limits.hard_abort_hours*3600.0;
1509 };
1510 const int rc = runner::run_rgb_downstream(dir,run_id,cfg,phase_upper,log,
1511 abort_fn);
1512 if (rc==0) {
1513 tile_compile::reconstruction::write_post_run_star_shape_artifact(
1514 dir, cfg.hypermetric_stretch.output_rgb, cfg.hypermetric_stretch.enabled);
1515 // Refresh the checkpoint's downstream markers (same contract as the
1516 // full-run tail).
1517 const auto ckpt = dir/"artifacts/forward_drizzle_checkpoint.json";
1518 std::error_code ec;
1519 if (fs::is_regular_file(ckpt,ec)) {
1520 try {
1521 auto cp = json::parse(core::read_text(ckpt));
1522 cp["downstream_status"]="complete";
1523 if (cfg.hypermetric_stretch.enabled) {
1524 fs::path hms_path(cfg.hypermetric_stretch.output_rgb);
1525 if (hms_path.is_relative()) hms_path=dir/"outputs"/hms_path;
1526 if (fs::exists(hms_path,ec))
1527 cp["final_output"]={{"path",hms_path.string()},
1528 {"bytes",fs::file_size(hms_path,ec)}};
1529 }
1530 core::write_text_atomic(ckpt,cp.dump(2));
1531 } catch (...) {
1532 }
1533 }
1534 }
1535 emitter.run_end(run_id, rc==0, rc==0 ? "final_image_ready" : "error", log,
1536 {{"execution_scope","forward_drizzle_m1_m3"},
1537 {"resume_from",phase_upper}});
1538 return rc;
1539}
