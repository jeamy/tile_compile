108Matrix2Df apply_global_warp(const Matrix2Df &img, const WarpMatrix &warp,
109 ColorMode mode, int out_rows, int out_cols) {
110 if (img.size() <= 0 || out_rows <= 0 || out_cols <= 0) {

113 if (mode == ColorMode::OSC) {

117 // For MONO/RGB: use standard warp with WARP_INVERSE_MAP
118 // (imported via registration header would create circular dep — inline here)
119 cv::Mat cv_img(img.rows(), img.cols(), CV_32F,
120 const_cast<float *>(img.data()));
121 cv::Mat warp_matrix(2, 3, CV_32F);
122 for (int i = 0; i < 2; ++i)
123 for (int j = 0; j < 3; ++j)
124 warp_matrix.at<float>(i, j) = warp(i, j);
125 cv::Mat warped;
126 cv::warpAffine(cv_img, warped, warp_matrix, cv::Size(out_cols, out_rows),
127 cv::INTER_LINEAR | cv::WARP_INVERSE_MAP);
128 Matrix2Df result(out_rows, out_cols);
129 std::memcpy(result.data(), warped.data,
130 static_cast<size_t>(result.size()) * sizeof(float));
131 return result;
132}
