3// Plan section 19.6 --- the deterministic forward-contribution-list rasterizer.
4//
5// This is the CPU REFERENCE for the ordering the plan-19 CUDA path must obey:
6// materialise every positive-area (source-pixel droplet leaf) x (output cell)
7// overlap of one stripe as a record with a canonical key, sort by that key,
8// then reduce each (frame, channel, output cell) segment single-threaded with
9// double accumulators in record order. By construction this reproduces the
10// streaming Uniform reference (stream_forward_drizzle_uniform) bit-for-bit:
11// for a fixed output cell the streaming emission order IS
12// (source_y, source_x, leaf_order) ascending, which is exactly the canonical
13// key order within a segment; and segments are visited in frame-then-cell
14// order, matching the streaming per-frame `wx += A` accumulation.
15//
16// No float atomics, no unordered scatter, no inverse search window (plan 19.3).

30// One forward contribution within a single stripe. `frame_order` is the index
31// into PreparedDrizzleFrames::frames (the plan's canonical frame order);
32// `leaf_order` is 0 for the affine path and 0..n-1 for a subdivided local warp.
33struct DrizzleContribKey {

39 std::uint32_t source_x = 0;
40 std::uint32_t leaf_order = 0;
41};

62// The plan-11.9 Uniform accumulators, produced by the segment reduction. These
63// are exactly what stream_forward_drizzle_uniform accumulates per stripe:
64// wx[c][i] = sum_f ( sum_{contrib in (f,c,i)} k*v )
65// w [c][i] = sum_f ( sum_{contrib in (f,c,i)} k )
66// w2[c][i] = sum_f ( sum_{contrib in (f,c,i)} k )^2
67// with the inner sums in canonical record order and the outer sum in frame
68// order. `value = wx/w` where `w > 0`.
69struct DrizzleUniformAccum {
70 int width = 0;
71 int rows = 0;
72 int channels = 1;
73 std::array<std::vector<double>, 3> wx, w, w2;
74};

91// Production form: because `frame_order` leads the canonical key, no segment
92// ever spans two frames, so the list can be built, sorted and reduced ONE
93// FRAME AT A TIME, accumulating into shared wx/w/w2. The result is bit-identical
94// to reduce_uniform_contrib_list(build_uniform_contrib_list(...)) --- the same
95// `wx[cell] += A_segment` operations in the same (frame, then cell) order ---

100// is preserved.
