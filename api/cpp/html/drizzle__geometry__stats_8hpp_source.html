1// Plan section 11.14 P0 --- forward-drizzle geometry instrumentation.
2//
3// NON-INVASIVE diagnostic counters + coarse sub-phase timers for the
4// CFA-forward-drizzle geometry path (`enumerate_drizzle_stripe_leaf_cells` ->
5// `sample_leaves` -> `subdivide_local` / `local_forward` ->
6// `invert_local_source_to_canvas` -> `smooth_local_basis`) and its callers
7// (coverage geometry, the production uniform/raw stream, the contrib-list count
8// and fill passes, and the hybrid CPU-geometry path).
9//
10// Purpose: measure --- per phase, per frame class and per geometry VARIANT ---
11// how many source samples are visited, how many top-level `sample_leaves`
12// calls run, how many fixed-point inversion iterations / RBF basis evaluations
13// those cost, and how that scales with the stripe count K (chunk height). This
14// is what confirms or refutes the O(frames * source_pixels * stripes *
15// consumers) term named in plan 11.14.1.
16//
17// HARD CONSTRAINT (plan 11.14.2): these counters must not change any
18// compute-hash / pixel behaviour. Every field is a plain integer or double
19// accumulator; increments are guarded by `registry().enabled` (a single
20// well-predicted branch) and never touch floating-point geometry, control flow,
21// reduction order or iteration bounds. Not thread-safe by design --- the
22// forward-drizzle reference path is single-threaded today; deterministic
23// parallelism is P3 and owns the atomics/merge story.

35// A geometry "variant" is one (consumer, pixfrac-class) enumeration context.
36// pixfrac matters (plan 11.14.3): the dense footprint at pixfrac = 1 and the
37// drizzle coverage droplet at pixfrac < 1 are DIFFERENT variants even though
38// they share a call site, and P1/P2 dedup must not conflate them.
39enum class Variant : int {
40 kUnattributed = 0,
41 kPrepareExclusionScan, // prepare_drizzle_frames: upfront full source sweep
42 kCoverageCfa, // sampling_geometry: CFA droplet at cfg.pixfrac
43 kCoverageFootprint, // sampling_geometry: dense footprint at pixfrac = 1
44 kCoverageAccumulatorReset, // T6: per-frame B[c] clear before CFA raster
45 kCoverageAccumulatorReduction, // T6: per-frame w/w2/count fold after CFA
46 kCoverageHoleQuantile, // T6: post-band hole detection + quantile merge
47 kUniformDiagnostic, // stream_forward_drizzle_uniform (non-production diag)
48 kContribFill, // build_frame_records: fill pass
49 kVariantCount
50};

54struct VariantCounters {
55 // Enumeration structure.
56 std::uint64_t enumerate_calls = 0; // enumerate_drizzle_stripe_leaf_cells
57 std::uint64_t source_rows_scanned = 0; // sum of (source_y1 - source_y0)
58 std::uint64_t source_samples_visited = 0; // inner (sx, sy) iterations
59
60 // Per-sample work. `top_level_sample_leaves_calls` is the P2 acceptance
61 // number (plan 11.14.4): it must equal eligible_local_frames * source_pixels
62 // per variant and stay invariant across chunk heights. Counted ONLY at the
63 // top of `sample_leaves`, never in `subdivide_local` recursion.
64 std::uint64_t top_level_sample_leaves_calls = 0;
65 std::uint64_t sample_leaves_discarded = 0; // returned false (whole sample)
66 std::uint64_t leaves_generated = 0; // accepted leaves pushed
67
68 // Local-warp numeric cost.
69 std::uint64_t subdivide_local_calls = 0; // every recursion level
70 std::uint64_t local_forward_calls = 0; // droplet probe warp evaluations
71 std::uint64_t invert_calls = 0; // invert_local_source_to_canvas
72 // Newton steps actually executed. Each step runs exactly one
73 // local_displacement_render_units -> evaluate_smooth_local_displacement ->
74 // smooth_local_basis, i.e. 16 std::exp calls. So basis_evaluations ==
75 // invert_iterations and exp_calls == 16 * invert_iterations.
76 std::uint64_t invert_iterations = 0;
77
78 // Output.
79 std::uint64_t leaf_cells_emitted = 0; // sink() invocations
80
81 double pixfrac = 0.0; // recorded for the variant (footprint = 1.0)
82
83 // Coarse geometry timing for this variant: wall + CPU seconds spent inside
84 // enumerate_drizzle_stripe_leaf_cells calls attributed here (includes the
85 // polygon/rect area work done in the rasterize_drizzle_stripe sink).
86 double geometry_wall_s = 0.0;
87 double geometry_cpu_s = 0.0;
88};

123// RAII: mark the active geometry variant (and its pixfrac) for the enclosed
124// enumeration. Nesting restores the previous variant.
125struct ScopedVariant {
126 Variant prev;
127 bool active;
128 ScopedVariant(Variant variant, double pixfrac) {
129 auto &r = registry();
130 active = r.enabled;
131 prev = r.current;
132 if (active) {
133 r.current = variant;
134 r.v[static_cast<std::size_t>(variant)].pixfrac = pixfrac;
135 }
136 }
137 ~ScopedVariant() {
138 if (active)
139 registry().current = prev;
140 }
141 ScopedVariant(const ScopedVariant &) = delete;
142 ScopedVariant &operator=(const ScopedVariant &) = delete;
143};

145// RAII: accumulate wall + CPU seconds into the current variant's geometry
146// timers. Cheap enough to wrap a whole enumerate call (once per frame per
147// stripe); NEVER put this in a per-sample loop.
148struct ScopedGeometryTimer {
149 bool active;
150 std::chrono::steady_clock::time_point w0;
151 std::clock_t c0;
152 ScopedGeometryTimer() {
153 active = registry().enabled;
154 if (active) {
155 w0 = std::chrono::steady_clock::now();
156 c0 = std::clock();
157 }
158 }
159 ~ScopedGeometryTimer() {
160 if (!active)
161 return;
162 const auto w1 = std::chrono::steady_clock::now();
163 const std::clock_t c1 = std::clock();
164 auto &c = registry().cur();
165 c.geometry_wall_s +=
166 std::chrono::duration<double>(w1 - w0).count();
167 c.geometry_cpu_s +=
168 static_cast<double>(c1 - c0) / static_cast<double>(CLOCKS_PER_SEC);
169 }
170 ScopedGeometryTimer(const ScopedGeometryTimer &) = delete;
171 ScopedGeometryTimer &operator=(const ScopedGeometryTimer &) = delete;
172};
