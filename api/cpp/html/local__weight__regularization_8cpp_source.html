107 ti < (*confidence_scores)[fi].size() &&

109 confidence = std::clamp((*confidence_scores)[fi][ti], 0.0f, 1.0f);
110 }
111 const float lambda_eff = cfg.lambda * (1.0f - confidence);
112 if (!(lambda_eff > 0.0f)) {
113 continue;
114 }
115 double neighbor_sum = 0.0;
116 double affinity_sum = 0.0;
117 for (size_t ni : neighbors[ti]) {

121 }
122 const double affinity =

126 affinity_sum += affinity;

137 }
138 for (size_t ti = 0; ti < tiles.size(); ++ti) {

141 continue;
142 }
143 const float abs_delta = std::fabs(current[ti] - scores[fi][ti]);
144 if (abs_delta > 0.0f) {

147 }

149 }
150 }
