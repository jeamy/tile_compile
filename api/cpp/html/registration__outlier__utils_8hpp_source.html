40// A physical meridian flip changes the orientation of a temporally coherent
41// part of a session. Isolated half-turn solutions are the common 180-degree
42// ambiguity of star-pattern matching and must not be admitted as valid model
43// anchors. Allow small holes so an occasional failed registration inside a
44// real flip segment does not split it.
45inline std::vector<uint8_t> persistent_half_turn_support(
46 const std::vector<uint8_t> &candidates, int radius = 8,
47 int minimum_support = 4) {
48 std::vector<uint8_t> supported(candidates.size(), 0);
49 if (radius < 0 || minimum_support <= 0) {
50 return supported;
51 }
52 for (size_t i = 0; i < candidates.size(); ++i) {
53 if (candidates[i] == 0) {
54 continue;
55 }
56 const size_t lo = i > static_cast<size_t>(radius)
57 ? i - static_cast<size_t>(radius)
58 : 0;
59 const size_t hi = std::min(candidates.size(),
60 i + static_cast<size_t>(radius) + 1);
61 int count = 0;
62 for (size_t j = lo; j < hi; ++j) {
63 count += candidates[j] != 0 ? 1 : 0;
64 }
65 supported[i] = count >= minimum_support ? 1 : 0;
66 }
67 return supported;
68}
