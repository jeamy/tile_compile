14double smoothstep(double e0, double e1, double x) {
15 if (!(e1 > e0)) return x >= e1 ? 1.0 : 0.0;
16 const double t = std::clamp((x - e0) / (e1 - e0), 0.0, 1.0);
17 return t * t * (3.0 - 2.0 * t);
18}

22double weighted_percentile(std::span<const double> values,
23 std::span<const double> weights, double p) {
24 if (values.size() != weights.size())
25 throw std::invalid_argument("WEIGHTED_PERCENTILE_SIZE");
26 if (values.empty()) return std::numeric_limits<double>::quiet_NaN();
27 p = std::clamp(p, 0.0, 1.0);
28
29 std::vector<std::size_t> idx(values.size());
30 std::iota(idx.begin(), idx.end(), std::size_t{0});
31 std::sort(idx.begin(), idx.end(),
32 [&](std::size_t a, std::size_t b) { return values[a] < values[b]; });
33
34 double total = 0.0;
35 for (double w : weights) {
36 if (!(w >= 0.0)) throw std::invalid_argument("WEIGHTED_PERCENTILE_WEIGHT");
37 total += w;
38 }
39 if (!(total > 0.0)) return values[idx.front()];
40
41 // Hazen plotting position: CDF at sample k is (cum_k - w_k/2) / total.
42 double cum = 0.0;
43 double prev_cdf = 0.0;
44 double prev_val = values[idx.front()];
45 for (std::size_t k = 0; k < idx.size(); ++k) {
46 const double w = weights[idx[k]];
47 cum += w;
48 const double cdf = (cum - 0.5 * w) / total;
49 const double val = values[idx[k]];
50 if (p <= cdf) {
51 if (k == 0 || cdf <= prev_cdf) return val;
52 const double frac = (p - prev_cdf) / (cdf - prev_cdf);
53 return prev_val + std::clamp(frac, 0.0, 1.0) * (val - prev_val);
54 }
55 prev_cdf = cdf;
56 prev_val = val;
57 }
58 return values[idx.back()];
59}

61std::vector<double> weighted_percentiles(std::span<const double> values,
62 std::span<const double> weights,
63 std::span<const double> ps) {
64 if (values.size() != weights.size())
65 throw std::invalid_argument("WEIGHTED_PERCENTILE_SIZE");
66 std::vector<double> out(ps.size(), std::numeric_limits<double>::quiet_NaN());
67 if (values.empty()) return out;
68
69 std::vector<std::size_t> idx(values.size());
70 std::iota(idx.begin(), idx.end(), std::size_t{0});
71 std::sort(idx.begin(), idx.end(),
72 [&](std::size_t a, std::size_t b) { return values[a] < values[b]; });
73
74 double total = 0.0;
75 for (double w : weights) {
76 if (!(w >= 0.0)) throw std::invalid_argument("WEIGHTED_PERCENTILE_WEIGHT");
77 total += w;
78 }
79 if (!(total > 0.0)) {
80 std::fill(out.begin(), out.end(), values[idx.front()]);
81 return out;
82 }
83
84 // Resolve requests in ascending p order -- cdf is non-decreasing in k, so
85 // each request's answer is found at or after the previous one's k, exactly
86 // matching what an independent weighted_percentile(values, weights, p)
87 // call would find on its own walk from k=0.
88 std::vector<std::size_t> order(ps.size());
89 std::iota(order.begin(), order.end(), std::size_t{0});
90 std::sort(order.begin(), order.end(),
91 [&](std::size_t a, std::size_t b) { return ps[a] < ps[b]; });
92
93 double cum = 0.0, prev_cdf = 0.0, prev_val = values[idx.front()];
94 std::size_t req = 0;
95 for (std::size_t k = 0; k < idx.size() && req < order.size(); ++k) {
96 const double w = weights[idx[k]];
97 cum += w;
98 const double cdf = (cum - 0.5 * w) / total;
99 const double val = values[idx[k]];
100 while (req < order.size()) {
101 const double p = std::clamp(ps[order[req]], 0.0, 1.0);
102 if (p > cdf) break;
103 if (k == 0 || cdf <= prev_cdf) {
104 out[order[req]] = val;
105 } else {
106 const double frac = (p - prev_cdf) / (cdf - prev_cdf);
107 out[order[req]] = prev_val + std::clamp(frac, 0.0, 1.0) * (val - prev_val);
108 }
109 ++req;
110 }
111 prev_cdf = cdf;
112 prev_val = val;
113 }
114 while (req < order.size()) {
115 out[order[req]] = values[idx.back()];
116 ++req;
117 }
118 return out;
119}

128AlphaConfidenceFactors compute_alpha_confidence_channel(
129 std::span<const AlphaFactorContribution> accepted,
130 const AlphaConfidenceParams &params, AlphaConfidenceScratch &scratch) {
131 AlphaConfidenceFactors out;
132 if (accepted.empty()) return out;
133
134 auto &b = scratch.b; b.clear();
135 auto &q = scratch.q; q.clear();
136 auto &resid = scratch.resid; resid.clear();
137 auto &art_v = scratch.art_v; art_v.clear();
138 auto &art_w = scratch.art_w; art_w.clear();
139 double b_total = 0.0, b_direct = 0.0;
140 for (const auto &c : accepted) {
141 if (!(c.b > 0.0)) continue;
142 b.push_back(c.b);
143 q.push_back(std::clamp(c.q_composite, 0.0, 1.0));
144 resid.push_back(c.residual_factor);
145 b_total += c.b;
146 if (c.is_direct) b_direct += c.b;
147 if (std::isfinite(c.artifact_conf)) {
148 art_v.push_back(std::clamp(c.artifact_conf, 0.0, 1.0));
149 art_w.push_back(c.b);
150 }
151 }
152 if (b.empty()) return out;
153
154 // A_separation. P1.6: q_p50/q_p90 share (q, b) -- one sort, one CDF walk.
155 const auto q_pcts = weighted_percentiles(q, b, std::array{0.50, 0.90});
156 const double q_p50 = q_pcts[0];
157 const double q_p90 = q_pcts[1];
158 out.a_separation = smoothstep(params.min_quality_separation,
159 params.full_quality_separation,
160 std::max(0.0, q_p90 - q_p50));
161
162 // A_artifact. Plan 14.4: fewer than min_artifact_contributors valid a_f
163 // => not applicable => 0 (not full confidence).
164 if (static_cast<int>(art_v.size()) >= params.min_artifact_contributors) {
165 const double a_p10 = weighted_percentile(art_v, art_w, 0.10);
166 out.a_artifact = smoothstep(params.artifact_lo, params.artifact_hi, a_p10);
167 out.artifact_applicable = true;
168 } else {
169 out.a_artifact = 0.0;
170 out.artifact_applicable = false;
171 }
172
173 // A_registration.
174 const double direct_fraction = b_total > 0.0 ? b_direct / b_total : 0.0;
175 const double residual_p20 = weighted_percentile(resid, b, 0.20);
176 out.a_registration =
177 std::min(smoothstep(params.direct_fraction_lo, params.direct_fraction_hi,
178 direct_fraction),
179 smoothstep(params.residual_p20_lo, params.residual_p20_hi,
180 residual_p20));
181 return out;
182}
