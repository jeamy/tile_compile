17float compute_correlation(const std::vector<float> &lhs,
18 const std::vector<float> &rhs) {
19 if (lhs.size() != rhs.size() || lhs.empty()) {
20 return 0.0f;
21 }
22 if (lhs.size() == 1u) {
23 return (std::fabs(lhs[0] - rhs[0]) <= 1.0e-6f) ? 1.0f : 0.0f;
24 }
25
26 const double lhs_mean =
27 std::accumulate(lhs.begin(), lhs.end(), 0.0) / static_cast<double>(lhs.size());
28 const double rhs_mean =
29 std::accumulate(rhs.begin(), rhs.end(), 0.0) / static_cast<double>(rhs.size());
30
31 double cov = 0.0;
32 double lhs_var = 0.0;
33 double rhs_var = 0.0;
34 double max_abs_delta = 0.0;
35 for (size_t i = 0; i < lhs.size(); ++i) {
36 const double dl = static_cast<double>(lhs[i]) - lhs_mean;
37 const double dr = static_cast<double>(rhs[i]) - rhs_mean;
38 cov += dl * dr;
39 lhs_var += dl * dl;
40 rhs_var += dr * dr;
41 max_abs_delta =
42 std::max(max_abs_delta, std::fabs(static_cast<double>(lhs[i] - rhs[i])));
43 }
44
45 constexpr double kVarEps = 1.0e-12;
46 if (lhs_var <= kVarEps && rhs_var <= kVarEps) {
47 return (max_abs_delta <= 1.0e-6) ? 1.0f : 0.0f;
48 }
49 if (lhs_var <= kVarEps || rhs_var <= kVarEps) {
50 return 0.0f;
51 }
52 return static_cast<float>(cov / std::sqrt(lhs_var * rhs_var));
53}
