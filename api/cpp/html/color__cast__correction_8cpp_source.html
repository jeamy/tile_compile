12double median_of(std::vector<double> &v) {
13 if (v.empty()) return 0.0;
14 const std::size_t mid = v.size() / 2;
15 std::nth_element(v.begin(), v.begin() + mid, v.end());
16 double m = v[mid];
17 if (v.size() % 2 == 0) {
18 const double lo = *std::max_element(v.begin(), v.begin() + mid);
19 m = 0.5 * (m + lo);
20 }
21 return m;
22}

32// 9x9 box blur (edge-clamped) via separable running sums.
33std::vector<float> box_blur9(const std::vector<float> &src, int rows, int cols) {
34 const int r = 4;
35 std::vector<float> tmp(src.size()), out(src.size());
36 for (int y = 0; y < rows; ++y) {
37 for (int x = 0; x < cols; ++x) {
38 double acc = 0.0;
39 for (int k = -r; k <= r; ++k) {
40 const int xx = std::clamp(x + k, 0, cols - 1);
41 acc += src[static_cast<std::size_t>(y) * cols + xx];
42 }
43 tmp[static_cast<std::size_t>(y) * cols + x] = static_cast<float>(acc / (2 * r + 1));
44 }
45 }
46 for (int y = 0; y < rows; ++y) {
47 for (int x = 0; x < cols; ++x) {
48 double acc = 0.0;
49 for (int k = -r; k <= r; ++k) {
50 const int yy = std::clamp(y + k, 0, rows - 1);
51 acc += tmp[static_cast<std::size_t>(yy) * cols + x];
52 }
53 out[static_cast<std::size_t>(y) * cols + x] = static_cast<float>(acc / (2 * r + 1));
54 }
55 }
56 return out;
57}

59// Grün-offset of the sky: G += (sky_r + sky_b) / 2 - sky_g on every valid pixel,
60// so the sky ends up neutral (G level = mean of R and B). Skipped when the
61// offset is implausibly large (> 25 % of the R/B sky level), which points to a
62// mis-measured sky rather than a tint. Values are kept >= 0.
63void apply_sky_offset(Matrix2Df &G, const std::vector<std::uint8_t> &valid,
64 const ColorCastCorrectionConfig &cfg,
65 ColorCastCorrectionResult &res) {
66 if (!cfg.neutralize_sky) return;
67 const double target = 0.5 * (res.sky_r + res.sky_b);
68 const double delta = target - res.sky_g;
69 if (!(std::fabs(delta) <= 0.25 * std::max(std::fabs(target), 1e-9))) return;
70 const int rows = static_cast<int>(G.rows()), cols = static_cast<int>(G.cols());
71 for (int y = 0; y < rows; ++y)
72 for (int x = 0; x < cols; ++x)
73 if (valid[static_cast<std::size_t>(y) * cols + x])
74 G(y, x) = static_cast<float>(std::max(0.0, static_cast<double>(G(y, x)) + delta));
75 res.sky_offset_g = delta;
76 res.sky_neutralized = true;
77}

82 Matrix2Df &R, Matrix2Df &G, Matrix2Df &B,
83 const ColorCastCorrectionConfig &cfg,
84 const std::vector<std::uint8_t> *valid_mask) {
85 ColorCastCorrectionResult res;
86 if (!cfg.enabled) return res;
87 const int rows = static_cast<int>(R.rows()), cols = static_cast<int>(R.cols());
88 if (rows <= 0 || cols <= 0 || G.rows() != rows || B.rows() != rows ||
89 G.cols() != cols || B.cols() != cols) {
90 res.status = "error";
91 res.error_message = "RGB dimensions mismatch";
92 return res;
93 }
94 if (!(cfg.max_amount > 0.0f && cfg.max_amount <= 1.0f) ||
95 !(cfg.target_ratio > 0.0f) || !(cfg.min_excess >= 1.0f) ||
96 !(cfg.object_sigma > 0.0f) || cfg.brightness_bins < 1 ||
97 cfg.brightness_bins > 32) {
98 res.status = "error";
99 res.error_message = "invalid color_cast_correction parameters";
100 return res;
101 }
102 const std::size_t n = static_cast<std::size_t>(rows) * cols;
103 const bool use_mask = valid_mask != nullptr && valid_mask->size() == n;
104
105 std::vector<std::uint8_t> valid(n, 0);
106 std::vector<float> lum(n, 0.0f);
107 std::size_t nvalid = 0;
108 for (int y = 0; y < rows; ++y) {
109 for (int x = 0; x < cols; ++x) {
110 const std::size_t i = static_cast<std::size_t>(y) * cols + x;
111 const float r = R(y, x), g = G(y, x), b = B(y, x);
112 bool ok = std::isfinite(r) && std::isfinite(g) && std::isfinite(b);
113 if (ok) ok = use_mask ? ((*valid_mask)[i] != 0) : !(r == 0.0f && g == 0.0f && b == 0.0f);
114 if (!ok) continue;
115 valid[i] = 1;
116 lum[i] = (r + g + b) / 3.0f;
117 ++nvalid;
118 }
119 }
120 if (nvalid < 1000) {
121 res.status = "too_few_object_pixels";
122 return res;
123 }
124 // Invalid pixels take the median luminance so they do not bias the blur.
125 {
126 std::vector<double> t;
127 const std::size_t step = std::max<std::size_t>(1, n / kMaxSamples);
128 for (std::size_t i = 0; i < n; i += step)
129 if (valid[i]) t.push_back(lum[i]);
130 const float fill = static_cast<float>(median_of(t));
131 for (std::size_t i = 0; i < n; ++i)
132 if (!valid[i]) lum[i] = fill;
133 }
134 // Blurred luminance WITHOUT the centre pixel: the object/sky decision must not
135 // depend on the pixel's own noise, otherwise the green excess of marginal
136 // pixels is biased (selection on a variable that contains G).
137 std::vector<float> lb = box_blur9(lum, rows, cols);
138 for (std::size_t i = 0; i < n; ++i) lb[i] = (81.0f * lb[i] - lum[i]) / 80.0f;
139
140 // Sky reference: the darkest 30 % of the blurred luminance.
141 const std::size_t step = std::max<std::size_t>(1, n / kMaxSamples);
142 std::vector<double> lb_samples;
143 for (std::size_t i = 0; i < n; i += step)
144 if (valid[i]) lb_samples.push_back(lb[i]);
145 const double dark_thr = percentile_of(lb_samples, 0.30);
146 std::vector<double> dark_l, dark_r, dark_g, dark_b;
147 for (std::size_t i = 0; i < n; i += step) {
148 if (!valid[i] || lb[i] > dark_thr) continue;
149 const int y = static_cast<int>(i / cols), x = static_cast<int>(i % cols);
150 dark_l.push_back(lb[i]);
151 dark_r.push_back(R(y, x));
152 dark_g.push_back(G(y, x));
153 dark_b.push_back(B(y, x));
154 }
155 if (dark_l.size() < 100) {
156 res.status = "too_few_object_pixels";
157 return res;
158 }
159 const double sky_l = median_of(dark_l);
160 {
161 // Noise of the blurred luminance: robust spread of differences between
162 // blurred values 9 px apart (disjoint windows), immune to the truncation of
163 // the dark set and to slowly varying structure.
164 std::vector<double> diffs;
165 for (int y = 0; y < rows; y += 3) {
166 for (int x = 0; x + 9 < cols; x += 3) {
167 const std::size_t i = static_cast<std::size_t>(y) * cols + x;
168 if (valid[i] && valid[i + 9]) diffs.push_back(static_cast<double>(lb[i]) - lb[i + 9]);
169 }
170 }
171 double sigma = 0.0;
172 if (diffs.size() >= 100) {
173 const double med = median_of(diffs);
174 for (double& d : diffs) d = std::fabs(d - med);
175 sigma = 1.4826 * median_of(diffs) / std::sqrt(2.0);
176 }
177 res.sky_r = median_of(dark_r);
178 res.sky_g = median_of(dark_g);
179 res.sky_b = median_of(dark_b);
180 // Object pixels: clearly above the sky, without the brightest 1 % (stars).
181 std::vector<double> excess;
182 for (std::size_t i = 0; i < n; i += step)
183 if (valid[i]) excess.push_back(lb[i] - sky_l);
184 const double upper = percentile_of(excess, 0.99);
185 const double lower = static_cast<double>(cfg.object_sigma) * std::max(sigma, 1e-9);
186 struct Sample { double e, xg, avg; };
187 std::vector<Sample> smp;
188 std::size_t n_obj = 0;
189 for (std::size_t i = 0; i < n; ++i) {
190 if (!valid[i]) continue;
191 const double e = lb[i] - sky_l;
192 if (e > lower && e <= upper) ++n_obj;
193 }
194 const std::size_t ostep = std::max<std::size_t>(1, n_obj / kMaxSamples);
195 std::size_t seen = 0;
196 for (std::size_t i = 0; i < n; ++i) {
197 if (!valid[i]) continue;
198 const double e = lb[i] - sky_l;
199 if (!(e > lower && e <= upper)) continue;
200 if ((seen++ % ostep) != 0) continue;
201 const int y = static_cast<int>(i / cols), x = static_cast<int>(i % cols);
202 smp.push_back({e, G(y, x) - res.sky_g,
203 0.5 * ((R(y, x) - res.sky_r) + (B(y, x) - res.sky_b))});
204 }
205 res.object_pixels = smp.size();
206 if (smp.size() < 1000) {
207 res.status = "too_few_object_pixels";
208 apply_sky_offset(G, valid, cfg, res);
209 res.applied = res.sky_neutralized;
210 return res;
211 }
212 std::sort(smp.begin(), smp.end(),
213 [](const Sample &a, const Sample &b) { return a.e < b.e; });
214 // Equal-count brightness classes (fewer when the object is small).
215 const int bins = std::max(1, std::min<int>(cfg.brightness_bins,
216 static_cast<int>(smp.size() / 1000)));
217 auto class_ratio = [&](std::size_t b0, std::size_t b1, double a) {
218 std::vector<double> g, m;
219 g.reserve(b1 - b0);
220 m.reserve(b1 - b0);
221 for (std::size_t k = b0; k < b1; ++k) {
222 const double ref = std::max(smp[k].avg, 0.0);
223 g.push_back(smp[k].xg - a * std::max(0.0, smp[k].xg - ref));
224 m.push_back(smp[k].avg);
225 }
226 const double mg = median_of(g), mm = median_of(m);
227 return mm > 0.0 ? mg / mm : 0.0;
228 };
229 std::vector<double> centers(static_cast<std::size_t>(bins)), amt(static_cast<std::size_t>(bins), 0.0);
230 std::vector<std::size_t> edge(static_cast<std::size_t>(bins) + 1);
231 for (int k = 0; k <= bins; ++k) edge[k] = smp.size() * static_cast<std::size_t>(k) / bins;
232 const double max_a = static_cast<double>(cfg.max_amount);
233 const double target = static_cast<double>(cfg.target_ratio);
234 for (int k = 0; k < bins; ++k) {
235 const std::size_t b0 = edge[k], b1 = edge[k + 1];
236 centers[k] = smp[(b0 + b1) / 2].e;
237 if (!(class_ratio(b0, b1, 0.0) > static_cast<double>(cfg.min_excess))) continue;
238 double a = max_a;
239 if (class_ratio(b0, b1, a) < target) {
240 double lo = 0.0, hi = a; // the ratio decreases with the amount
241 for (int it = 0; it < 24; ++it) {
242 const double mid = 0.5 * (lo + hi);
243 if (class_ratio(b0, b1, mid) > target) lo = mid;
244 else hi = mid;
245 }
246 a = 0.5 * (lo + hi);
247 }
248 amt[k] = a;
249 }
250 // The class amounts are used as they are: the classes hold many pixels, and
251 // the interpolation over the pixel brightness below already smooths the
252 // transition between them.
253 const std::vector<double> &sm = amt;
254 // Amount as a function of a pixel's own blurred excess: 0 at/below half the
255 // object threshold, rising to the first class amount at the threshold,
256 // linear between class centres, constant above the last centre.
257 auto amount_at = [&](double e) {
258 if (e <= 0.5 * lower) return 0.0;
259 if (e < lower) return sm[0] * (e - 0.5 * lower) / (0.5 * lower);
260 if (e <= centers.front() || bins == 1) return sm.front();
261 if (e >= centers.back()) return sm.back();
262 const auto it = std::upper_bound(centers.begin(), centers.end(), e);
263 const std::size_t j = static_cast<std::size_t>(it - centers.begin());
264 const double t = (e - centers[j - 1]) / std::max(centers[j] - centers[j - 1], 1e-12);
265 return sm[j - 1] + t * (sm[j] - sm[j - 1]);
266 };
267 // Overall ratios (median over all samples) before / after with per-pixel amounts.
268 {
269 std::vector<double> g0, g1, m;
270 for (const auto &q : smp) {
271 const double ref = std::max(q.avg, 0.0);
272 g0.push_back(q.xg);
273 g1.push_back(q.xg - amount_at(q.e) * std::max(0.0, q.xg - ref));
274 m.push_back(q.avg);
275 }
276 const double mm = median_of(m);
277 res.ratio_before = mm > 0.0 ? median_of(g0) / mm : 0.0;
278 res.ratio_after = mm > 0.0 ? median_of(g1) / mm : 0.0;
279 }
280 res.amounts.assign(sm.begin(), sm.end());
281 double asum = 0.0, amax = 0.0;
282 for (const auto &q : smp) {
283 const double v = amount_at(q.e);
284 asum += v;
285 amax = std::max(amax, v);
286 }
287 res.amount = static_cast<float>(asum / static_cast<double>(smp.size()));
288 if (!(amax > 1e-6)) {
289 res.status = "not_needed";
290 res.amount = 0.0f;
291 res.ratio_after = res.ratio_before;
292 apply_sky_offset(G, valid, cfg, res);
293 if (res.sky_neutralized) {
294 res.applied = true;
295 res.status = "sky_neutralized";
296 }
297 return res;
298 }
299 // Apply to every valid pixel with the amount of its own brightness.
300 for (int y = 0; y < rows; ++y) {
301 for (int x = 0; x < cols; ++x) {
302 const std::size_t i = static_cast<std::size_t>(y) * cols + x;
303 if (!valid[i]) continue;
304 const double a = amount_at(static_cast<double>(lb[i]) - sky_l);
305 if (a <= 0.0) continue;
306 const double xg = G(y, x) - res.sky_g;
307 const double ref = std::max(0.0, 0.5 * ((R(y, x) - res.sky_r) + (B(y, x) - res.sky_b)));
308 const double ex = std::max(0.0, xg - ref);
309 if (ex > 0.0) G(y, x) = static_cast<float>(G(y, x) - a * ex);
310 }
311 }
312 }
313 apply_sky_offset(G, valid, cfg, res);
314 res.applied = true;
315 res.status = "applied";
316 return res;
