135template <typename Fn>
136void parallel_for_indices(size_t count, int workers, Fn fn) {
137 if (count == 0) return;
138 workers = std::max(1, std::min<int>(workers, static_cast<int>(count)));
139 if (workers == 1) {
140 for (size_t i = 0; i < count; ++i) fn(i);
141 return;
142 }
143 std::atomic<size_t> next{0};
144 std::vector<std::thread> threads;
145 threads.reserve(static_cast<size_t>(workers));
146 for (int w = 0; w < workers; ++w) {
147 threads.emplace_back([&]() {
148 while (true) {
149 const size_t i = next.fetch_add(1);
150 if (i >= count) break;
151 fn(i);
152 }
153 });
154 }
155 for (auto& t : threads) {
156 if (t.joinable()) t.join();
157 }
158}
