73inline SyntheticWeightingDecision decide_synthetic_weighting(
74 const std::string &requested_weighting, int boundary_pair_count,
75 float boundary_pair_mean_abs_diff_p95,
76 float boundary_pair_scale_ratio_deviation_p95,
77 float boundary_post_background_delta_p95_abs,
78 float local_weight_mean_abs_delta_p95,
79 float local_weight_correlation_p05) {
80 SyntheticWeightingDecision out;
81 out.requested_weighting = requested_weighting;
82 out.effective_weighting = requested_weighting;
83 out.boundary_pair_count = boundary_pair_count;
84 out.boundary_pair_mean_abs_diff_p95 = boundary_pair_mean_abs_diff_p95;
85 out.boundary_pair_scale_ratio_deviation_p95 =
86 boundary_pair_scale_ratio_deviation_p95;
87 out.boundary_post_background_delta_p95_abs =
88 boundary_post_background_delta_p95_abs;
89 out.local_weight_mean_abs_delta_p95 = local_weight_mean_abs_delta_p95;
90 out.local_weight_correlation_p05 = local_weight_correlation_p05;
91
92 if (requested_weighting != "tile_weighted") {
93 return out;
94 }
95
96 constexpr int kMinObservedBoundaryPairs = 8;
97 // Guard only against clearly visible, severe seam regressions.
98 // Earlier thresholds were tight enough to reject historically good runs.
99 constexpr float kBoundaryMeanAbsDiffP95 = 0.25f;
100 constexpr float kBoundaryPostBackgroundDeltaP95 = 0.25f;
101
102 const bool enough_pairs = boundary_pair_count >= kMinObservedBoundaryPairs;
103 const bool severe_boundary_regression =
104 (std::isfinite(boundary_pair_mean_abs_diff_p95) &&
105 boundary_pair_mean_abs_diff_p95 > kBoundaryMeanAbsDiffP95);
106 const bool severe_background_step =
107 (std::isfinite(boundary_post_background_delta_p95_abs) &&
108 boundary_post_background_delta_p95_abs > kBoundaryPostBackgroundDeltaP95);
109
110 if (enough_pairs && severe_boundary_regression && severe_background_step) {
111 out.effective_weighting = "global";
112 out.tile_seam_guard_triggered = true;
113 }
114
115 return out;
116}

129inline void apply_common_overlap_to_tile_inplace(
130 Matrix2Df &tile, const Tile &t, const std::vector<uint8_t> &common_valid_mask,
131 int common_mask_width, int common_mask_height) {
132 if (tile.rows() != t.height || tile.cols() != t.width)
133 return;
134 if (common_mask_width <= 0 || common_mask_height <= 0 ||
135 common_valid_mask.empty()) {
136 return;
137 }
138
139 const int tile_cols = static_cast<int>(tile.cols());
140 const size_t mask_size = common_valid_mask.size();
141 float *tile_data = tile.data();
142 const float invalid = common_overlap_invalid_value();
143
144 for (int yy = 0; yy < t.height; ++yy) {
145 const int gy = t.y + yy;
146 const size_t tile_row_off =
147 static_cast<size_t>(yy) * static_cast<size_t>(tile_cols);
148 if (gy < 0 || gy >= common_mask_height) {
149 for (int xx = 0; xx < t.width; ++xx) {
150 tile_data[tile_row_off + static_cast<size_t>(xx)] = invalid;
151 }
152 continue;
153 }
154
155 const size_t row_off =
156 static_cast<size_t>(gy) * static_cast<size_t>(common_mask_width);
157
158 for (int xx = 0; xx < t.width; ++xx) {
159 const int gx = t.x + xx;
160 if (gx < 0 || gx >= common_mask_width) {
161 tile_data[tile_row_off + static_cast<size_t>(xx)] = invalid;
162 continue;
163 }
164 const size_t mask_idx = row_off + static_cast<size_t>(gx);
165 if (mask_idx >= mask_size || common_valid_mask[mask_idx] == 0) {
166 tile_data[tile_row_off + static_cast<size_t>(xx)] = invalid;
167 }
168 }
169 }
