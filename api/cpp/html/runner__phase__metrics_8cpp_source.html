33std::optional<double> extract_exposure_seconds(const io::FitsHeader &header) {
34 auto read_positive = [&](const char *key) -> std::optional<double> {
35 if (auto value = header.get_double(key);
36 value && std::isfinite(*value) && *value > 0.0) {
37 return value;
38 }
39 if (auto value = header.get_int(key); value && *value > 0) {
40 return static_cast<double>(*value);
41 }
42 if (auto value = header.get_string(key)) {
43 char *end = nullptr;
44 errno = 0;
45 const double d = std::strtod(value->c_str(), &end);
46 if (errno == 0 && end != value->c_str() && *end == '\0' &&
47 std::isfinite(d) && d > 0.0) {
48 return d;
49 }
50 }
51 return std::nullopt;
52 };
53
54 for (const char *key : {"EXPTIME", "EXPOSURE", "EXPOSURETIME",
55 "EXPOSURE_TIME", "EXP_TIME", "DURATION", "EXPOS"}) {
56 if (auto value = read_positive(key)) {
57 return value;
58 }
59 }
60 return std::nullopt;
61}

638 emitter.run_end(run_id, false, "error", log_file,
639 {{"message", std::string("Error during GLOBAL_METRICS: ") + (gm_error.empty() ? "unknown_error" : gm_error)}});
640 std::cerr << "Error during GLOBAL_METRICS: "
641 << (gm_error.empty() ? "unknown_error" : gm_error) << std::endl;
642 return false;
643 }
644
645 // Determine reference star count (max) and recompute wFWHM
646 for (const auto &sm : frame_star_metrics) {
647 if (sm.star_count > ref_star_count)
648 ref_star_count = sm.star_count;
649 }
650 if (ref_star_count > 0) {
651 for (auto &sm : frame_star_metrics) {
652 if (sm.star_count > 0 && sm.fwhm > 0) {
653 sm.wfwhm = sm.fwhm * static_cast<float>(ref_star_count) /
654 static_cast<float>(sm.star_count);
655 }
656 }
657 }
658
659 out.frame_star_metrics = frame_star_metrics;
660 out.global_weights = metrics::calculate_global_weights_with_stars(
661 frame_metrics, frame_star_metrics, cfg.global_metrics.weights.background,
662 cfg.global_metrics.weights.noise, cfg.global_metrics.weights.gradient,
663 cfg.global_metrics.weights.fwhm, cfg.global_metrics.weights.roundness,
664 cfg.global_metrics.weights.star_count, cfg.global_metrics.clamp[0],
665 cfg.global_metrics.clamp[1], cfg.global_metrics.adaptive_weights,
666 cfg.global_metrics.weight_exponent_scale);
667 auto &global_weights = out.global_weights;
668
669 {
670 core::json artifact;
671 artifact["metrics"] = core::json::array();
672 for (size_t i = 0; i < frame_metrics.size(); ++i) {
673 core::json m;
674 m["background"] = frame_metrics[i].background;
675 m["noise"] = frame_metrics[i].noise;
676 m["gradient_energy"] = frame_metrics[i].gradient_energy;
677 m["sky_gradient"] = frame_metrics[i].sky_gradient;
678 m["quality_score"] = frame_metrics[i].quality_score;
679 m["global_weight"] = (i < static_cast<size_t>(global_weights.size()))
680 ? global_weights[static_cast<int>(i)]
681 : 0.0f;
682 m["fwhm"] = frame_star_metrics[i].fwhm;
683 m["fwhm_x"] = frame_star_metrics[i].fwhm_x;
684 m["fwhm_y"] = frame_star_metrics[i].fwhm_y;
685 m["roundness"] = frame_star_metrics[i].roundness;
686 m["wfwhm"] = frame_star_metrics[i].wfwhm;
687 m["star_count"] = frame_star_metrics[i].star_count;
688 artifact["metrics"].push_back(m);
689 }
690
691 artifact["weights"] = {
692 {"background", cfg.global_metrics.weights.background},
693 {"noise", cfg.global_metrics.weights.noise},
694 {"gradient", cfg.global_metrics.weights.gradient},
695 {"fwhm", cfg.global_metrics.weights.fwhm},
696 {"roundness", cfg.global_metrics.weights.roundness},
697 {"star_count", cfg.global_metrics.weights.star_count}};
698 artifact["clamp"] = {cfg.global_metrics.clamp[0],
699 cfg.global_metrics.clamp[1]};
700 artifact["adaptive_weights"] = cfg.global_metrics.adaptive_weights;
701 artifact["adaptive_weighting_method"] =
702 "leave_one_out_positive_correlation_squared";
703 artifact["adaptive_weighting_target"] =
704 "For each metric, predict the leave-one-out consensus of the other two "
705 "higher-is-better normalized signals using static weights renormalized "
706 "over the remaining metrics.";
707 artifact["adaptive_weighting_tie_break"] =
708 "If utilities are degenerate or nearly tied, keep the static weights. "
709 "Otherwise clip adaptive weights to [0.1,0.7] and renormalize.";
710 core::write_text(run_dir / "artifacts" / "global_metrics.json",
711 artifact.dump(2));
712 }
713
714 return true;
715}
