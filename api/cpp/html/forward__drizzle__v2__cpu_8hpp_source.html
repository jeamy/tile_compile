3// Gate-10 host CPU port of the forward-drizzle v2 band kernel and the
4// backend-neutral boundary the production band driver runs against.
5//
6// The host port mirrors ForwardDrizzleV2CudaPrototypeKernel's contract
7// exactly: reserve() allocates the whole band workspace once, frames are
8// streamed through accumulate_frame()/accumulate_frame_local() (affine6 is
9// source->canvas in the caller's band-local target frame), and finalize()
10// emits one ForwardDrizzleV2PixelResult (+ optional
11// ForwardDrizzleV2ProfileResult) record per native pixel and channel. The
12// scatter, fold, reservoir keep predicate, sigma-clip, confidence and
13// profile/alpha semantics are the same mathematical contract the device
14// kernels implement --- verified against the CPU oracles and, where a device
15// exists, bit-tolerant against the CUDA output.

22// Backend-neutral kernel boundary for the gate-10 band driver. Both
23// implementations share the device kernel's call contract; the only
24// observable difference is stats() (no CUDA counters on the host) and
25// device_backend().
26class ForwardDrizzleV2Kernel {
27 public:
28 virtual ~ForwardDrizzleV2Kernel() = default;
29 virtual bool reserve(int native_cols, int native_rows, int source_w,
30 int source_h,
31 const ForwardDrizzleV2KernelConfig &cfg) = 0;
32 // Rebind the reserved workspace to the next band (active rows <= the
33 // reserved maximum) without allocating; clears all per-band state.
34 // Allowed right after reserve() (before any frame) or after a successful
35 // finalize(); only native_rows and band_origin_y_native may change.
36 virtual bool begin_band(
37 int native_rows, const ForwardDrizzleV2KernelConfig &cfg) = 0;
38 virtual bool accumulate_frame(
39 const double affine6[6], const float *source, const float *sigma2_or_null,
40 std::uint64_t frame_order,
41 const ForwardDrizzleV2FrameQuality *quality_or_null = nullptr,
42 const ForwardDrizzleV2FrameMeta *meta_or_null = nullptr) = 0;
43 virtual bool accumulate_frame_local(
44 const double affine6[6], const ForwardDrizzleV2LocalWarp &warp,
45 const float *source, const float *sigma2_or_null,
46 std::uint64_t frame_order,
47 const ForwardDrizzleV2FrameQuality *quality_or_null = nullptr,
48 const ForwardDrizzleV2FrameMeta *meta_or_null = nullptr) = 0;
49 // Window variants: `source` is a packed row-major buffer of
50 // window.width*window.height (absolute origin x_begin/y_begin, optional
51 // halo); only the ACTIVE rect (window.active_*, all-zero = whole buffer)
52 // is iterated. Float sigma2/Q planes are packed over the ACTIVE rect and
53 // indexed by the active-local tid; packed Q descriptors decode by absolute
54 // (sx, sy). sigma2_or_null and an ENABLED sigma2_model_or_null are
55 // mutually exclusive. The window buffer must be positive and contained in
56 // the reserved full source extent.
57 virtual bool accumulate_frame_window(
58 const double affine6[6], const ForwardDrizzleV2SourceWindow &window,
59 const float *source, const float *sigma2_or_null,
60 const ForwardDrizzleV2Sigma2FrameModel *sigma2_model_or_null,
61 std::uint64_t frame_order,
62 const ForwardDrizzleV2FrameQuality *quality_or_null = nullptr,
63 const ForwardDrizzleV2FrameMeta *meta_or_null = nullptr) = 0;
64 virtual bool accumulate_frame_local_window(
65 const double affine6[6], const ForwardDrizzleV2LocalWarp &warp,
66 const ForwardDrizzleV2SourceWindow &window, const float *source,
67 const float *sigma2_or_null,
68 const ForwardDrizzleV2Sigma2FrameModel *sigma2_model_or_null,
69 std::uint64_t frame_order,
70 const ForwardDrizzleV2FrameQuality *quality_or_null = nullptr,
71 const ForwardDrizzleV2FrameMeta *meta_or_null = nullptr) = 0;
72 // Geometry-cache variant (tranche 6): the frame's geometry is
73 // `leaf_count` committed cache leaves (raw internal canvas coordinates,
74 // canonical order); leaf source coordinates must lie inside the ACTIVE
75 // window rect. affine6 is validated but not applied. Requires
76 // leaf_count <= cfg.cached_leaf_capacity.
77 virtual bool accumulate_frame_cached_leaves(
78 const double affine6[6], const ForwardDrizzleV2SourceWindow &window,
79 const float *source, const float *sigma2_or_null,
80 const ForwardDrizzleV2Sigma2FrameModel *sigma2_model_or_null,
81 const ForwardDrizzleV2CachedLeaf *leaves, std::size_t leaf_count,
82 std::uint64_t unique_source_samples, std::uint64_t frame_order,
83 const ForwardDrizzleV2FrameQuality *quality_or_null = nullptr,
84 const ForwardDrizzleV2FrameMeta *meta_or_null = nullptr) = 0;
85 // Tranche 8 canonical affine path: one-shot full-target frame fed by the
86 // ragged source-sample list (canonical ascending (y,x) order, all
87 // coordinates in the reserved source extent, count <= source_w*source_h).
88 // sigma2_present=false reproduces the absent-sigma semantics. Quality is
89 // the aligned packed contract (presence bit => code array, null veto =>
90 // no veto; code 0 or veto decodes to NaN).
91 virtual bool accumulate_frame_affine_samples(
92 const double affine6[6], const ForwardDrizzleV2SourceSample *samples,
93 std::size_t sample_count, bool sigma2_present,
94 const ForwardDrizzleV2AlignedQuality *quality_or_null,
95 std::uint64_t frame_order,
96 const ForwardDrizzleV2FrameMeta *meta_or_null = nullptr) = 0;
97 // Affine target-tile piece lifecycle (tranche 7): one affine frame may be
98 // emitted as several target-x pieces; each piece scatters+folds only the
99 // internal x columns [target_x_begin*scale, (begin+cols)*scale) so
100 // overlapping source scan boxes never double-count. Pieces must be
101 // ordered by target x and non-overlapping, and the frame's quality
102 // stream presence must be identical on every piece. The local-warp and
103 // cached-geometry paths are always single-call and must not interleave
104 // with an open piece frame.
105 virtual bool begin_affine_frame(
106 std::uint64_t frame_order,
107 const ForwardDrizzleV2FrameMeta *meta_or_null = nullptr) = 0;
108 virtual bool accumulate_affine_piece(
109 const double affine6[6], int target_x_begin_native,
110 int target_cols_native,
111 const ForwardDrizzleV2SourceWindow &window, const float *source,
112 const float *sigma2_or_null,
113 const ForwardDrizzleV2Sigma2FrameModel *sigma2_model_or_null,
114 const ForwardDrizzleV2FrameQuality *quality_or_null = nullptr) = 0;
115 virtual bool finish_affine_frame(std::uint64_t frame_order) = 0;
116 // Frame whose source window is empty for this band: advances stream
117 // bookkeeping (and the meta row when profiles are enabled) without
118 // scatter/fold.
119 virtual bool skip_frame(
120 std::uint64_t frame_order,
121 const ForwardDrizzleV2FrameMeta *meta_or_null = nullptr) = 0;
122 // Full-frame estimator barrier: call once after the last pilot frame of a
123 // band (cfg.full_frame_estimator only). Freezes the clip bounds and folds
124 // the accepted pilot candidates into the full-frame sums.
125 virtual bool end_pilot() { return false; }
126 virtual bool finalize(ForwardDrizzleV2PixelResult *results,
127 ForwardDrizzleV2ProfileResult *profiles_or_null,
128 std::uint64_t *dense_overlap_count) = 0;
129 virtual const ForwardDrizzleV2PrototypeStats &stats() const = 0;
130 virtual bool device_backend() const = 0;
131 // Diagnosis hook: the CUDA adapter reports cudaGetErrorString() of the
132 // failing runtime call after a false return; the CPU kernel always
133 // returns "".
134 virtual std::string last_device_error() const { return {}; }
135};
