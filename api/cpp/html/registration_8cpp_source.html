115BoundingBox compute_warps_bounding_box(int frame_width, int frame_height,
116 const std::vector<WarpMatrix>& warps) {
117 if (warps.empty()) {
118 return BoundingBox{0, 0, frame_width, frame_height};
119 }
120
121 // Frame corners in input space
122 std::vector<std::pair<float, float>> corners = {
123 {0.0f, 0.0f},
124 {static_cast<float>(frame_width), 0.0f},
125 {0.0f, static_cast<float>(frame_height)},
126 {static_cast<float>(frame_width), static_cast<float>(frame_height)}
127 };
128
129 float min_x = 0.0f, min_y = 0.0f;
130 float max_x = static_cast<float>(frame_width);
131 float max_y = static_cast<float>(frame_height);
132
133 // Transform all corners of all frames
134 for (const auto& warp : warps) {
135 // Warp is R→M (reference to moving) with WARP_INVERSE_MAP
136 // To get output coordinates, we need the inverse: M→R
137 // For affine: [x', y'] = [a00 a01 tx] [x]
138 // [a10 a11 ty] [y]
139 // [1]
140
141 for (const auto& corner : corners) {
142 float x = corner.first;
143 float y = corner.second;
144
145 // Apply warp (this gives us the output position)
146 float x_out = warp(0, 0) * x + warp(0, 1) * y + warp(0, 2);
147 float y_out = warp(1, 0) * x + warp(1, 1) * y + warp(1, 2);
148
149 min_x = std::min(min_x, x_out);
150 min_y = std::min(min_y, y_out);
151 max_x = std::max(max_x, x_out);
152 max_y = std::max(max_y, y_out);
153 }
154 }
155
156 // Round to integers with some padding
157 BoundingBox bbox;
158 bbox.min_x = static_cast<int>(std::floor(min_x));
159 bbox.min_y = static_cast<int>(std::floor(min_y));
160 bbox.max_x = static_cast<int>(std::ceil(max_x));
161 bbox.max_y = static_cast<int>(std::ceil(max_y));
162
163 return bbox;
164}
